1//! Recover simple rational level crossings of low-degree Bernstein polynomials.
2use crate::constraint_system::Ratio;
3
4/// When the Bernstein sign sequence has one variation and both endpoints are
5/// nonzero, a single simple root lies inside (0,1). Floating point only suggests
6/// a parameter. Exact de Casteljau evaluation certifies each returned root.
7pub(crate) fn simple_zero(c: &[Ratio]) -> Option<Ratio> {
8    if !(2..=4).contains(&c.len()) || c[0].numerator() == 0 || c.last()?.numerator() == 0 {
9        return None;
10    }
11    let mut sign = c[0].numerator().signum();
12    let mut changes = 0;
13    for coefficient in c {
14        let next = coefficient.numerator().signum();
15        if next != 0 && next != sign {
16            changes += 1;
17            sign = next;
18        }
19    }
20    if changes != 1 {
21        return None;
22    }
23    let values: Vec<_> = c
24        .iter()
25        .map(|v| v.numerator() as f64 / v.denominator() as f64)
26        .collect();
27    let mut lo = 0.;
28    let mut hi = 1.;
29    for _ in 0..60 {
30        let mid = (lo + hi) / 2.;
31        let mut row = [0.; 4];
32        row[..c.len()].copy_from_slice(&values);
33        for n in (1..c.len()).rev() {
34            for i in 0..n {
35                row[i] = (1. - mid) * row[i] + mid * row[i + 1];
36            }
37        }
38        if row[0] == 0. {
39            lo = mid;
40            hi = mid;
41            break;
42        }
43        if row[0].is_sign_positive() == values[0].is_sign_positive() {
44            lo = mid;
45        } else {
46            hi = mid;
47        }
48    }
49    // Continued-fraction convergents of the isolated numerical estimate.
50    let mut x = (lo + hi) / 2.;
51    let (mut n_old, mut n) = (0_i128, 1_i128);
52    let (mut d_old, mut d) = (1_i128, 0_i128);
53    for _ in 0..32 {
54        let a = x.floor();
55        if !a.is_finite() || a > 1_000_000. {
56            break;
57        }
58        (n_old, n) = (n, a as i128 * n + n_old);
59        (d_old, d) = (d, a as i128 * d + d_old);
60        if d > 1_000_000 {
61            break;
62        }
63        if n > 0 && n < d {
64            let t = Ratio::new(n, d).ok()?;
65            if exact_value(c, t) == Some(Ratio::zero()) {
66                return Some(t);
67            }
68        }
69        let fraction = x - a;
70        if fraction == 0. {
71            break;
72        }
73        x = 1. / fraction;
74    }
75    None
76}
77
78fn exact_value(c: &[Ratio], t: Ratio) -> Option<Ratio> {
79    let complement = Ratio::one().checked_sub(t).ok()?;
80    let mut row = c.to_vec();
81    for n in (1..c.len()).rev() {
82        for i in 0..n {
83            row[i] = row[i]
84                .checked_mul(complement)
85                .ok()?
86                .checked_add(row[i + 1].checked_mul(t).ok()?)
87                .ok()?;
88        }
89    }
90    Some(row[0])
91}

93#[cfg(test)]
94mod tests {
95    use super::*;
96    fn integer(n: i128) -> Ratio {
97        Ratio::new(n, 1).unwrap()
98    }
99
100    #[test]
101    fn nondyadic_rational_event_survives_a_change_of_knot_interval() {
102        let polynomial = [-3, -1, 4, 12].map(integer);
103        // The degree-three Bernstein controls represent -3+6t+9t².
104        assert_eq!(simple_zero(&polynomial), Some(Ratio::new(1, 3).unwrap()));
105        assert_eq!(
106            simple_zero(&[-123, 874].map(integer)),
107            Some(Ratio::new(123, 997).unwrap())
108        );
109    }
110
111    #[test]
112    fn approximate_irrational_or_ambiguous_event_is_never_admitted() {
113        // 2t²-1 on the unit interval has an irrational zero.
114        assert_eq!(simple_zero(&[-1, -1, 1].map(integer)), None);
115        assert_eq!(simple_zero(&[1, -2, 1].map(integer)), None);
116        assert_eq!(simple_zero(&[0, 1].map(integer)), None);
117    }
