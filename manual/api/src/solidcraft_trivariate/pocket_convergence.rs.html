43fn integer(i: usize) -> Q {
44    Q::new(i as i128, 1).unwrap()

46fn number(q: Q) -> f64 {
47    q.numerator() as f64 / q.denominator() as f64

49fn endpoints(c: &Quad, a: usize, b: usize) -> [Point; 2] {
50    if a == 0 {
51        [c[2 * b], c[2 * b + 1]]
52    } else {
53        [c[b], c[2 + b]]
54    }

56impl Chart {
57    fn edge_controls(&self, axis: usize, endpoint: usize) -> [[Q; 2]; 3] {
58        std::array::from_fn(|k| {
59            let (i, j) = if axis == 0 {
60                (endpoint * 2, k)
61            } else {
62                (k, endpoint * 2)
63            };
64            self.xy[i * 3 + j]
65        })
66    }
67    fn edge_is_straight(&self, axis: usize, endpoint: usize) -> bool {
68        let points = self.edge_controls(axis, endpoint);
69        let dx = points[2][0].checked_sub(points[0][0]);
70        let dy = points[2][1].checked_sub(points[0][1]);
71        let (Ok(dx), Ok(dy)) = (dx, dy) else {
72            return false;
73        };
74        points[1][0]
75            .checked_sub(points[0][0])
76            .and_then(|x| x.checked_mul(dy))
77            .and_then(|x| {
78                points[1][1]
79                    .checked_sub(points[0][1])
80                    .and_then(|y| y.checked_mul(dx).and_then(|y| x.checked_sub(y)))
81            })
82            .is_ok_and(|x| x.numerator() == 0)
83    }
84    fn derivatives(&self, u: Q, v: Q) -> Result<[[Q; 2]; 2]> {
85        let (bu, du) = bernstein(2, u)?;
86        let (bv, dv) = bernstein(2, v)?;
87        let mut out = [[Q::zero(); 2]; 2];
88        for i in 0..3 {
89            for j in 0..3 {
90                let p = self.xy[i * 3 + j];
91                for k in 0..2 {
92                    out[0][k] =
93                        out[0][k].checked_add(du[i].checked_mul(bv[j])?.checked_mul(p[k])?)?;
94                    out[1][k] =
95                        out[1][k].checked_add(bu[i].checked_mul(dv[j])?.checked_mul(p[k])?)?;
96                }
97            }
98        }
99        Ok(out)
100    }
101    fn evaluate(&self, u: f64, v: f64) -> ([f64; 2], [[f64; 2]; 2]) {
102        let bu = [(1. - u).powi(2), 2. * u * (1. - u), u.powi(2)];
103        let bv = [(1. - v).powi(2), 2. * v * (1. - v), v.powi(2)];
104        let du = [-2. * (1. - u), 2. - 4. * u, 2. * u];
105        let dv = [-2. * (1. - v), 2. - 4. * v, 2. * v];
106        let mut position = [0.; 2];
107        let mut jac = [[0.; 2]; 2];
108        for i in 0..3 {
109            for j in 0..3 {
110                let p = self.xy[i * 3 + j].map(number);
111                for k in 0..2 {
112                    position[k] += bu[i] * bv[j] * p[k];
113                    jac[0][k] += du[i] * bv[j] * p[k];
114                    jac[1][k] += bu[i] * dv[j] * p[k];
115                }
116            }
117        }
118        (position, jac)
119    }
120}
121fn chart_from_block(b: &crate::bracket::BracketBlock) -> Result<Chart> {
122    if b.degrees != [1, 1, 1] && b.degrees != [2, 2, 1] {
123        return Err("unsupported curved pocket chart degree");
124    }
125    let expected = if b.degrees == [1, 1, 1] { 8 } else { 18 };
126    if b.controls.len() != expected {
127        return Err("invalid curved pocket control count");
128    }
129    let weight = b.controls[0][3];
130    if weight <= 0 || b.controls.iter().any(|h| h[3] != weight) {
131        return Err("curved pocket chart requires one positive common weight");
132    }
133    let point = |h: &[i64; 4]| -> Result<[Q; 2]> {
134        Ok([
135            Q::new(i128::from(h[0]), i128::from(h[3]))?,
136            Q::new(i128::from(h[1]), i128::from(h[3]))?,
137        ])
138    };
139    let mut xy = [[Q::zero(); 2]; 9];
140    if b.degrees == [2, 2, 1] {
141        for i in 0..3 {
142            for j in 0..3 {
143                let lo = &b.controls[(i * 3 + j) * 2];
144                let hi = &b.controls[(i * 3 + j) * 2 + 1];
145                if lo[0] != hi[0] || lo[1] != hi[1] || lo[3] != hi[3] {
146                    return Err("curved pocket XY chart changes along z");
147                }
148                xy[i * 3 + j] = point(lo)?;
149            }
150        }
151    } else {
152        let mut linear = [[Q::zero(); 2]; 4];
153        for i in 0..2 {
154            for j in 0..2 {
155                let lo = &b.controls[(i * 2 + j) * 2];
156                let hi = &b.controls[(i * 2 + j) * 2 + 1];
157                if lo[0] != hi[0] || lo[1] != hi[1] || lo[3] != hi[3] {
158                    return Err("bilinear pocket XY chart changes along z");
159                }
160                linear[i * 2 + j] = point(lo)?;
161            }
162        }
163        let elevate = [
164            [Q::one(), Q::zero()],
165            [Q::new(1, 2)?, Q::new(1, 2)?],
166            [Q::zero(), Q::one()],
167        ];
168        for i in 0..3 {
169            for j in 0..3 {
170                for k in 0..2 {
171                    for a in 0..2 {
172                        for d in 0..2 {
173                            xy[i * 3 + j][k] = xy[i * 3 + j][k].checked_add(
174                                elevate[i][a]
175                                    .checked_mul(elevate[j][d])?
176                                    .checked_mul(linear[a * 2 + d][k])?,
177                            )?;
178                        }
179                    }
180                }
181            }
182        }
183    }
184    let integer = |x: Q| -> Result<i64> {
185        if x.denominator() != 1 {
186            return Err("curved pocket corner is not an integer topology point");
187        }
188        i64::try_from(x.numerator()).map_err(|_| "curved pocket corner overflows")
189    };
190    let indices = [0, 2, 6, 8];
191    let corners = [
192        [integer(xy[indices[0]][0])?, integer(xy[indices[0]][1])?],
193        [integer(xy[indices[1]][0])?, integer(xy[indices[1]][1])?],
194        [integer(xy[indices[2]][0])?, integer(xy[indices[2]][1])?],
195        [integer(xy[indices[3]][0])?, integer(xy[indices[3]][1])?],
196    ];
197    Ok(Chart { corners, xy })
198}
199fn axis_q(rows: &[Row], t: Q) -> Result<BTreeMap<usize, [Q; 2]>> {
200    let (b, d) = bernstein(rows.len() - 1, t)?;
201    let mut out = BTreeMap::new();
202    for (k, row) in rows.iter().enumerate() {
203        for (&i, &c) in row {
204            let v = out.entry(i).or_insert([Q::zero(); 2]);
205            v[0] = v[0].checked_add(b[k].checked_mul(c)?)?;
206            v[1] = v[1].checked_add(d[k].checked_mul(c)?)?;
207        }
208    }
209    Ok(out)
210}
211struct Basis {
212    n: usize,
213    dim: usize,
214    charts: Vec<Chart>,
215    ids: Vec<Vec<(usize, Q)>>,
216    spans: Vec<Vec<Row>>,
217}
218impl Basis {
219    fn value_trace(&self, side: Side, k: usize, t: Q, sign: Q, row: &mut Row) -> Result<()> {
220        let (chart, axis, endpoint) = side;
221        let mut cells = [k; 2];
222        cells[axis] = endpoint * (self.n - 1);
223        let mut pt = [t; 2];
224        pt[axis] = integer(endpoint);
225        let bx = axis_q(&self.spans[cells[0]], pt[0])?;
226        let by = axis_q(&self.spans[cells[1]], pt[1])?;
227        for (i, [x, _]) in bx {
228            for (&j, &[y, _]) in &by {
229                add(
230                    row,
231                    self.ids[chart][i * self.dim + j].0,
232                    sign.checked_mul(x.checked_mul(y)?)?
233                        .checked_mul(self.ids[chart][i * self.dim + j].1)?,
234                )?;
235            }
236        }
237        Ok(())
238    }
239    fn jet(
240        &self,
241        side: Side,
242        k: usize,
243        t: Q,
244        x_component: bool,
245        sign: Q,
246        row: &mut Row,
247    ) -> Result<()> {
248        let (chart, a, b) = side;
249        let mut cells = [k; 2];
250        cells[a] = b * (self.n - 1);
251        let mut pt = [t; 2];
252        pt[a] = integer(b);
253        let parent = |j: usize| {
254            integer(cells[j])
255                .checked_add(pt[j])?
256                .checked_div(integer(self.n))
257        };
258        let source = self.charts[chart].derivatives(parent(0)?, parent(1)?)?;
259        let u = [source[0][0], source[0][1]];
260        let v = [source[1][0], source[1][1]];
261        let det = u[0]
262            .checked_mul(v[1])?
263            .checked_sub(v[0].checked_mul(u[1])?)?;
264        let scale = sign.checked_mul(integer(self.n))?.checked_div(det)?;
265        let minus = |x: Q| Q::zero().checked_sub(x);
266        let (du, dv) = if x_component {
267            (v[1], minus(u[1])?)
268        } else {
269            (minus(v[0])?, u[0])
270        };
271        let bx = axis_q(&self.spans[cells[0]], pt[0])?;
272        let by = axis_q(&self.spans[cells[1]], pt[1])?;
273        for (i, [x, dx]) in bx {
274            for (&j, &[y, dy]) in &by {
275                let value = du
276                    .checked_mul(dx)?
277                    .checked_mul(y)?
278                    .checked_add(dv.checked_mul(x)?.checked_mul(dy)?)?;
279                add(
280                    row,
281                    self.ids[chart][i * self.dim + j].0,
282                    scale
283                        .checked_mul(value)?
284                        .checked_mul(self.ids[chart][i * self.dim + j].1)?,
285                )?;
286            }
287        }
288        Ok(())
289    }
290}

292fn lift_residual(rows: &[Row], lift: &[BTreeMap<usize, f64>]) -> f64 {
293    rows.iter()
294        .map(|row| {
295            let mut sum = BTreeMap::<usize, f64>::new();
296            let mut magnitude = BTreeMap::<usize, f64>::new();
297            for (&id, &coefficient) in row {
298                for (&free, &value) in &lift[id] {
299                    let term = number(coefficient) * value;
300                    *sum.entry(free).or_default() += term;
301                    *magnitude.entry(free).or_default() += term.abs();
302                }
303            }
304            let scale = magnitude.values().copied().fold(0., f64::max);
305            if scale == 0. {
306                0.
307            } else {
308                sum.values().map(|x| x.abs() / scale).fold(0., f64::max)
309            }
310        })
311        .fold(0., f64::max)
312}

319fn axis_dofs(p: usize, n: usize) -> Vec<AxisDof> {
320    let mut out = Vec::new();
321    for knot in 0..=n {
322        out.push(AxisDof {
323            kind: AxisKind::Value,
324            knot,
325        });
326        out.push(AxisDof {
327            kind: AxisKind::Slope,
328            knot,
329        });
330        if p == 4 && knot < n {
331            out.push(AxisDof {
332                kind: AxisKind::Middle,
333                knot,
334            });
335        }
336    }
337    out
338}
339/// Exact local Bernstein controls from endpoint Hermite data. Quartic spans
340/// retain an independent middle control; adjacent spans share only knot jets.
341fn hermite_spans(p: usize, n: usize) -> Vec<Vec<Row>> {
342    let dofs = axis_dofs(p, n);
343    let find = |kind, knot| {
344        dofs.iter()
345            .position(|d| d.kind == kind && d.knot == knot)
346            .unwrap()
347    };
348    (0..n)
349        .map(|k| {
350            let mut rows = vec![Row::new(); p + 1];
351            rows[0].insert(find(AxisKind::Value, k), Q::one());
352            rows[1].insert(find(AxisKind::Value, k), Q::one());
353            rows[1].insert(find(AxisKind::Slope, k), Q::new(1, p as i128).unwrap());
354            if p == 3 {
355                rows[2].insert(find(AxisKind::Value, k + 1), Q::one());
356                rows[2].insert(find(AxisKind::Slope, k + 1), Q::new(-1, p as i128).unwrap());
357            } else {
358                rows[2].insert(find(AxisKind::Middle, k), Q::one());
359                rows[3].insert(find(AxisKind::Value, k + 1), Q::one());
360                rows[3].insert(find(AxisKind::Slope, k + 1), Q::new(-1, p as i128).unwrap());
361            }
362            rows[p].insert(find(AxisKind::Value, k + 1), Q::one());
363            rows
364        })
365        .collect()
366}
367fn edge_node(c: &Quad, axis: usize, endpoint: usize, tangential: AxisDof, n: usize) -> (Node, Q) {
368    let [start, end] = endpoints(c, axis, endpoint);
369    if start <= end {
370        (Node::Edge([start, end], tangential), Q::one())
371    } else {
372        // Canonical edge orientation mirrors knot and span indices; only an
373        // oriented tangential slope changes sign under that reversal.
374        let mapped = AxisDof {
375            kind: tangential.kind,
376            knot: match tangential.kind {
377                AxisKind::Value | AxisKind::Slope => n - tangential.knot,
378                AxisKind::Middle => n - 1 - tangential.knot,
379            },
380        };
381        let sign = if tangential.kind == AxisKind::Slope {
382            Q::new(-1, 1).unwrap()
383        } else {
384            Q::one()
385        };
386        (Node::Edge([end, start], mapped), sign)
387    }
388}
389fn node(c: &Quad, chart: usize, i: usize, j: usize, n: usize, dofs: &[AxisDof]) -> (Node, Q) {
390    let (u, v) = (dofs[i], dofs[j]);
391    let ub = (u.kind == AxisKind::Value && (u.knot == 0 || u.knot == n)).then_some(u.knot / n);
392    let vb = (v.kind == AxisKind::Value && (v.knot == 0 || v.knot == n)).then_some(v.knot / n);
393    match (ub, vb) {
394        (Some(a), Some(b)) => (Node::Vertex(c[2 * a + b]), Q::one()),
395        (Some(b), None) => edge_node(c, 0, b, v, n),
396        (None, Some(b)) => edge_node(c, 1, b, u, n),
397        (None, None) => (Node::Interior(chart, i, j), Q::one()),
398    }
399}
400struct Leaf {
401    c: Chart,
402    z: [f64; 2],
403    ij: [usize; 2],
404    dofs: Vec<usize>,
405    rows: Vec<Vec<(usize, f64)>>,
406}
407impl Leaf {
408    fn new(
409        b: &Basis,
410        col: usize,
411        z: [f64; 2],
412        ij: [usize; 2],
413        lift: &[BTreeMap<usize, f64>],
414    ) -> Result<Self> {
415        let mut rows = Vec::new();
416        let mut active = BTreeSet::new();
417        for a in &b.spans[ij[0]] {
418            for d in &b.spans[ij[1]] {
419                let mut row = BTreeMap::new();
420                for (&i, &ca) in a {
421                    for (&j, &cb) in d {
422                        let w = ca.checked_mul(cb)?;
423                        let (local_id, orientation) = b.ids[col][i * b.dim + j];
424                        for (&id, &c) in &lift[local_id] {
425                            *row.entry(id).or_insert(0.) += number(w.checked_mul(orientation)?) * c;
426                        }
427                    }
428                }
429                row.retain(|_, v| *v != 0.);
430                active.extend(row.keys().copied());
431                rows.push(row);
432            }
433        }
434        let dofs = Vec::from_iter(active);
435        let rows = rows
436            .into_iter()
437            .map(|r| {
438                r.into_iter()
439                    .map(|(id, v)| (dofs.binary_search(&id).unwrap(), v))
440                    .collect()
441            })
442            .collect();
443        Ok(Self {
444            c: b.charts[col].clone(),
445            z,
446            ij,
447            dofs,
448            rows,
449        })
450    }
451    fn sample(
452        &self,
453        n: usize,
454        p: usize,
455        t: [f64; 3],
456    ) -> Result<(ElementGeometrySample, ElementBasisSample)> {
457        let [u, v] = std::array::from_fn(|k| (self.ij[k] as f64 + t[k]) / n as f64);
458        let height = self.z[1] - self.z[0];
459        let z = self.z[0] + height * t[2];
460        let mut pos = [0., 0., z];
461        let mut jac = [[0.; 3]; 3];
462        jac[2][2] = height;
463        let (xy, dxy) = self.c.evaluate(u, v);
464        pos[..2].copy_from_slice(&xy);
465        jac[0][..2].copy_from_slice(&dxy[0].map(|x| x / n as f64));
466        jac[1][..2].copy_from_slice(&dxy[1].map(|x| x / n as f64));
467        let det = jac[0][0] * jac[1][1] - jac[0][1] * jac[1][0];
468        let axis = |x: f64| {
469            // Recurrence at both endpoints, including quartic end functions.
470            let mut b = [0.; 5];
471            b[0] = 1.;
472            let mut d = [0.; 5];
473            for k in 1..=p {
474                if k == p {
475                    for i in 0..k {
476                        d[i] -= k as f64 * b[i];
477                        d[i + 1] += k as f64 * b[i];
478                    }
479                }
480                for i in (1..=k).rev() {
481                    b[i] = (1. - x) * b[i] + x * b[i - 1];
482                }
483                b[0] *= 1. - x;
484            }
485            (b, d)
486        };
487        let (bx, dx) = axis(t[0]);
488        let (by, dy) = axis(t[1]);
489        let mut values = vec![0.; self.dofs.len()];
490        let mut gradients = vec![[0.; 3]; self.dofs.len()];
491        for i in 0..=p {
492            for j in 0..=p {
493                let val = bx[i] * by[j] / 20.;
494                let du = dx[i] * by[j];
495                let dv = bx[i] * dy[j];
496                let gx = z / 20. * (jac[1][1] * du - jac[0][1] * dv) / det;
497                let gy = z / 20. * (jac[0][0] * dv - jac[1][0] * du) / det;
498                for &(id, c) in &self.rows[i * (p + 1) + j] {
499                    values[id] += c * val * z;
500                    gradients[id][0] += c * gx;
501                    gradients[id][1] += c * gy;
502                    gradients[id][2] += c * val;
503                }
504            }
505        }
506        Ok((
507            ElementGeometrySample {
508                position: pos,
509                jacobian_columns: jac,
510                determinant: det * height,
511            },
512            ElementBasisSample {
513                dofs: self.dofs.clone(),
514                values,
515                physical_gradients: gradients,
516            },
517        ))
518    }
519}

525/// Preserve the original bilinear benchmark API and its source geometry.
526pub fn solve(p: usize, n: usize, c1: bool, order: usize) -> Result<Json> {
527    if ![8, 10].contains(&order) {
528        return Err("unsupported pocket study parameters");
529    }
530    let recipe = crate::json::parse(include_str!(
531        "../tests/references/blind_bilinear_pocket.recipe.json"
532    ))?;
533    let model = crate::interchange::build(&recipe)?;
534    let job = object([
535        ("schema", Json::String("section_quintic_mms_v1".into())),
536        ("quadrature", Json::Number(order.to_string())),
537        ("x_origin_mm", Json::Number("-10".into())),
538        ("x_scale_mm", Json::Number("70".into())),
539    ]);
540    solve_model(&model, p, n, c1, &job)
541}

566/// Solve with an explicit subset of physical C1 joins for diagnostic controls.
567pub fn solve_model_selected(
568    model: &Json,
569    p: usize,
570    n: usize,
571    joins: SectionJoins,
572    job: &Json,
573) -> Result<Json> {
574    if ![3, 4].contains(&p) || ![1, 2, 4, 8].contains(&n) {
575        return Err("unsupported pocket study parameters");
576    }
577    crate::interchange::audit(model)?;
578    let blocks = decode_blocks(model)?;
579    let mut z_bounds = [f64::INFINITY, f64::NEG_INFINITY];
580    for b in &blocks {
581        for h in &b.controls {
582            let z = h[2] as f64 / h[3] as f64;
583            z_bounds[0] = z_bounds[0].min(z);
584            z_bounds[1] = z_bounds[1].max(z);
585        }
586    }
587    if (z_bounds[0] - 0.).abs() > 1e-12 || (z_bounds[1] - 20.).abs() > 1e-12 {
588        return Err("curved pocket study requires z bounds [0,20] mm");
589    }
590    let mut charts = Vec::new();
591    let mut columns = Vec::new();
592    for b in &blocks {
593        let c = chart_from_block(b)?;
594        let col = charts.iter().position(|q| q == &c).unwrap_or_else(|| {
595            charts.push(c.clone());
596            charts.len() - 1
597        });
598        columns.push(col);
599    }
600    if charts.len() != 5 || blocks.len() != 9 {
601        return Err("unexpected scaffold topology");
602    }
603    let dofs = axis_dofs(p, n);
604    let dim = dofs.len();
605    let spans = hermite_spans(p, n);
606    let mut nodes = BTreeMap::new();
607    let mut ids = Vec::new();
608    let mut edges: BTreeMap<[Point; 2], Vec<Side>> = BTreeMap::new();
609    for (k, c) in charts.iter().enumerate() {
610        let mut local = Vec::new();
611        for i in 0..dim {
612            for j in 0..dim {
613                let next = nodes.len();
614                let (key, sign) = node(&c.corners, k, i, j, n, &dofs);
615                let id = *nodes.entry(key).or_insert(next);
616                local.push((id, sign));
617            }
618        }
619        ids.push(local);
620        for a in 0..2 {
621            for b in 0..2 {
622                let mut pair = endpoints(&c.corners, a, b);
623                pair.sort();
624                edges.entry(pair).or_default().push((k, a, b));
625            }
626        }
627    }
628    for sides in edges.values() {
629        if sides.len() > 2 {
630            return Err("nonmanifold chart edge");
631        }
632        if let [a, b] = sides[..] {
633            let first = charts[a.0].edge_controls(a.1, a.2);
634            let mut second = charts[b.0].edge_controls(b.1, b.2);
635            if endpoints(&charts[a.0].corners, a.1, a.2)
636                != endpoints(&charts[b.0].corners, b.1, b.2)
637            {
638                second.reverse();
639            }
640            if first != second {
641                return Err("shared pocket edge has mismatched full Bezier trace");
642            }
643        }
644    }
645    let count = nodes.len();
646    let basis = Basis {
647        n,
648        dim,
649        charts,
650        ids,
651        spans,
652    };
653    let mut equations = Vec::new();
654    let mut value_probe_rows = Vec::new();
655    let mut gradient_probe_rows = Vec::new();
656    if joins != SectionJoins::None {
657        for ([start, end], sides) in edges {
658            if sides.len() == 1 {
659                continue;
660            }
661            let [a, b] = sides[..] else {
662                return Err("nonmanifold chart edge");
663            };
664            let radial = basis.charts[a.0].edge_is_straight(a.1, a.2)
665                && basis.charts[b.0].edge_is_straight(b.1, b.2);
666            let selected = match joins {
667                SectionJoins::None => false,
668                SectionJoins::All => true,
669                SectionJoins::Radial => radial,
670                SectionJoins::Footprint => !radial,
671            };
672            if !selected {
673                continue;
674            }
675            let reverse = endpoints(&basis.charts[a.0].corners, a.1, a.2)
676                != endpoints(&basis.charts[b.0].corners, b.1, b.2);
677            // A physical coordinate direction transverse to the common tangent.
678            let x_component = start[1] != end[1];
679            for k in 0..n {
680                // Clearing the physical-map Jacobian denominators gives a
681                // univariate numerator of degree at most p+5 on each span.
682                // p+6 exact points therefore imply equality throughout it.
683                // Bilinear pairs have the sharper bound p+1. Keeping their
684                // original sample grid preserves the conditioned Hermite lift.
685                let jet_degree = if blocks.iter().all(|block| block.degrees == [1, 1, 1]) {
686                    p + 1
687                } else {
688                    p + 5
689                };
690                for r in 0..=jet_degree {
691                    let t = Q::new(r as i128, jet_degree as i128)?;
692                    let mut row = Row::new();
693                    basis.jet(a, k, t, x_component, Q::one(), &mut row)?;
694                    let (kb, tb) = if reverse {
695                        (n - 1 - k, Q::one().checked_sub(t)?)
696                    } else {
697                        (k, t)
698                    };
699                    basis.jet(b, kb, tb, x_component, Q::new(-1, 1)?, &mut row)?;
700                    equations.push(row);
701                }
702                for r in 1..=2 {
703                    let t = Q::new(r as i128, (p + 7) as i128)?;
704                    let (kb, tb) = if reverse {
705                        (n - 1 - k, Q::one().checked_sub(t)?)
706                    } else {
707                        (k, t)
708                    };
709                    let mut value = Row::new();
710                    basis.value_trace(a, k, t, Q::one(), &mut value)?;
711                    basis.value_trace(b, kb, tb, Q::new(-1, 1)?, &mut value)?;
712                    value_probe_rows.push(value);
713                    for x_component in [false, true] {
714                        let mut gradient = Row::new();
715                        basis.jet(a, k, t, x_component, Q::one(), &mut gradient)?;
716                        basis.jet(b, kb, tb, x_component, Q::new(-1, 1)?, &mut gradient)?;
717                        gradient_probe_rows.push(gradient);
718                    }
719                }
720            }
721        }
722    }
723    let (exact_lift, free) = jet_lift(count, &equations)?;
724    let (lift, interpolation_residual) =
725        if !equations.is_empty() && blocks.iter().any(|b| b.degrees == [2, 2, 1]) {
726            crate::spline_constraints::evaluation_kernel(count, &equations, free.len())?
727        } else {
728            (
729                exact_lift
730                    .iter()
731                    .map(|r| r.iter().map(|(&i, &q)| (i, number(q))).collect())
732                    .collect(),
733                0.,
734            )
735        };
736    let value_probe_residual = lift_residual(&value_probe_rows, &lift);
737    let gradient_probe_residual = lift_residual(&gradient_probe_rows, &lift);
738    if !equations.is_empty() && (value_probe_residual > 1e-8 || gradient_probe_residual > 1e-8) {
739        return Err("indeterminate: independent physical continuity probe residual");
740    }
741    let mut leaves = Vec::new();
742    let mut boundary = Vec::new();
743    let mut has_bottom = false;
744    let mut has_top = false;
745    let mut has_floor = false;
746    let exterior = model.field("topology")?.field("exterior")?.items()?;
747    for (k, b) in blocks.iter().enumerate() {
748        let mut sides = Vec::new();
749        for entry in exterior {
750            let e = entry.items()?;
751            if e[0].text()? == b.id {
752                sides.push(decode_side_name(e[1].text()?)?)
753            }
754        }
755        for &(axis, upper) in &sides {
756            if axis == 2 {
757                let h = &b.controls[usize::from(upper)];
758                let z = h[2] as f64 / h[3] as f64;
759                has_bottom |= z.abs() < 1e-12;
760                has_top |= (z - 20.).abs() < 1e-12;
761                has_floor |= (z - 14.).abs() < 1e-12;
762            }
763        }
764        for i in 0..n {
765            for j in 0..n {
766                let ij = [i, j];
767                for &(a, hi) in &sides {
768                    if a == 2 || ij[a] == if hi { n - 1 } else { 0 } {
769                        boundary.push((leaves.len(), a, hi));
770                    }
771                }
772                leaves.push(Leaf::new(
773                    &basis,
774                    columns[k],
775                    [
776                        b.controls[0][2] as f64 / b.controls[0][3] as f64,
777                        b.controls[1][2] as f64 / b.controls[1][3] as f64,
778                    ],
779                    ij,
780                    &lift,
781                )?);
782            }
783        }
784    }
785    if !(has_bottom && has_top && has_floor) {
786        return Err("curved pocket model lacks bottom, upper, or floor boundary");
787    }
788    let essential = resolve(free.len(), &[], &[])?;
789    let mut result = solve_sampled(
790        &IntegrationDomain {
791            scalar_dofs: free.len(),
792            cell_count: leaves.len(),
793            z_bounds,
794            boundary: &boundary,
795            essential: &essential,
796        },
797        job,
798        |i, t| leaves[i].sample(n, p, t),
799    )?;
800    if let Json::Object(fields) = &mut result {
801        let exact_degree_bound = p + 5;
802        fields.insert(
803            "chart_count".into(),
804            Json::Number(basis.charts.len().to_string()),
805        );
806        fields.insert("span_count_per_axis".into(), Json::Number(n.to_string()));
807        fields.insert(
808            "interpolation_jet_residual".into(),
809            Json::Number(interpolation_residual.to_string()),
810        );
811        fields.insert(
812            "independent_c0_probe_relative_residual".into(),
813            Json::Number(value_probe_residual.to_string()),
814        );
815        fields.insert(
816            "independent_c1_gradient_probe_relative_residual".into(),
817            Json::Number(gradient_probe_residual.to_string()),
818        );
819        fields.insert(
820            "c1_join_selection".into(),
821            Json::String(format!("{joins:?}")),
822        );
823        fields.insert(
824            "c1_constraint_rows".into(),
825            Json::Number(equations.len().to_string()),
826        );
827        fields.insert(
828            "c1_constraint_rank".into(),
829            Json::Number(count.saturating_sub(free.len()).to_string()),
830        );
831        fields.insert(
832            "c1_gradient_numerator_degree_bound".into(),
833            Json::Number(exact_degree_bound.to_string()),
834        );
835        fields.insert(
836            "c1_points_per_span".into(),
837            Json::Number((exact_degree_bound + 1).to_string()),
838        );
839    }
840    Ok(result)
841}
