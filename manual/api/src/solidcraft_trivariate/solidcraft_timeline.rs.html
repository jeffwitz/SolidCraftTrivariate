1//! Translate the verified solidcraft/66 bracket timeline subset.
2//!
3//! This module consumes the canonical history emitted by `solidcraft-cli script`.
4//! It never runs a user script in the host kernel. Before constructing anything,
5//! it resolves every feature, curve and generated face reference against the
6//! sketch's directed six-edge contour and verifies the geometric support.
7
8use crate::{
9    interchange,
10    json::{object, Json},
11    solidcraft_bridge,
12};

14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15enum Segment {
16    Bottom,
17    Right,
18    Top,
19    InnerVertical,
20    UpperLeft,
21    Left,
22}
23
24impl Segment {
25    fn directed_endpoints(self) -> ([i64; 2], [i64; 2]) {
26        match self {
27            Self::Bottom => ([0, 0], [60, 0]),
28            Self::Right => ([60, 0], [60, 8]),
29            Self::Top => ([60, 8], [8, 8]),
30            Self::InnerVertical => ([8, 8], [8, 50]),
31            Self::UpperLeft => ([8, 50], [0, 50]),
32            Self::Left => ([0, 50], [0, 0]),
33        }
34    }
35}

37fn id(value: &Json, expected: i64) -> Result<(), &'static str> {
38    if value.number::<i64>()? == expected {
39        Ok(())
40    } else {
41        Err("unexpected timeline feature identity")
42    }
43}

45fn text(value: &Json, expected: &str) -> Result<(), &'static str> {
46    if value.text()? == expected {
47        Ok(())
48    } else {
49        Err("unexpected timeline semantic value")
50    }
51}

53fn bool_value(value: &Json, expected: bool) -> Result<(), &'static str> {
54    if value == &Json::Bool(expected) {
55        Ok(())
56    } else {
57        Err("suppressed or altered timeline feature")
58    }
59}

61/// Parse a finite decimal token and multiply it by an integer scale exactly.
62/// A non-integral scaled result is rejected rather than rounded.
63fn scaled_decimal(value: &Json, scale: i128, allow_string: bool) -> Result<i128, &'static str> {
64    let token = match value {
65        Json::Number(s) => s.as_str(),
66        Json::String(s) if allow_string => s.as_str(),
67        _ => return Err("timeline value must be a literal number"),
68    };
69    let (mantissa, exponent) = match token.split_once(['e', 'E']) {
70        Some((m, e)) => (m, e.parse::<i32>().map_err(|_| "timeline numeric range")?),
71        None => (token, 0),
72    };
73    let (negative, mantissa) = if let Some(rest) = mantissa.strip_prefix('-') {
74        (true, rest)
75    } else {
76        (false, mantissa)
77    };
78    let (whole, fraction) = mantissa.split_once('.').unwrap_or((mantissa, ""));
79    if whole.is_empty()
80        || !whole.bytes().all(|b| b.is_ascii_digit())
81        || !fraction.bytes().all(|b| b.is_ascii_digit())
82    {
83        return Err("invalid timeline decimal literal");
84    }
85    let mut digits = format!("{whole}{fraction}");
86    let mut power = exponent
87        .checked_sub(i32::try_from(fraction.len()).map_err(|_| "timeline numeric range")?)
88        .ok_or("timeline numeric range")?;
89    while digits.ends_with('0') {
90        digits.pop();
91        power = power.checked_add(1).ok_or("timeline numeric range")?;
92    }
93    if digits.is_empty() {
94        return Ok(0);
95    }
96    let mut result = digits
97        .parse::<i128>()
98        .map_err(|_| "timeline numeric range")?
99        .checked_mul(scale)
100        .ok_or("timeline numeric range")?;
101    if power >= 0 {
102        let factor = 10_i128
103            .checked_pow(power as u32)
104            .ok_or("timeline numeric range")?;
105        result = result.checked_mul(factor).ok_or("timeline numeric range")?;
106    } else {
107        let divisor = 10_i128
108            .checked_pow(power.unsigned_abs())
109            .ok_or("timeline numeric range")?;
110        if result % divisor != 0 {
111            return Err("timeline value is outside the exact parameter lattice");
112        }
113        result /= divisor;
114    }
115    if negative {
116        result.checked_neg().ok_or("timeline numeric range")
117    } else {
118        Ok(result)
119    }
120}

122fn scalar(value: &Json, expected: i64) -> Result<(), &'static str> {
123    if scaled_decimal(value, 1, false)? == i128::from(expected) {
124        Ok(())
125    } else {
126        Err("unexpected sketch coordinate or direction")
127    }
128}

130fn vector<const N: usize>(value: &Json, expected: [i64; N]) -> Result<(), &'static str> {
131    let values = value.items()?;
132    if values.len() != N {
133        return Err("unexpected timeline vector length");
134    }
135    for (value, expected) in values.iter().zip(expected) {
136        scalar(value, expected)?;
137    }
138    Ok(())
139}

141fn feature<'a>(command: &'a Json, expected_type: &str) -> Result<&'a Json, &'static str> {
142    command.keys(&["command", "params"])?;
143    text(command.field("command")?, "timeline.add")?;
144    let params = command.field("params")?;
145    params.keys(&["feature"])?;
146    let feature = params.field("feature")?;
147    text(feature.field("type")?, expected_type)?;
148    Ok(feature)
149}

151fn validate_sketch(sketch_feature: &Json) -> Result<[String; 6], &'static str> {
152    sketch_feature.keys(&["id", "name", "plane", "sketch", "suppressed", "type"])?;
153    id(sketch_feature.field("id")?, 1)?;
154    text(sketch_feature.field("name")?, "Profile")?;
155    text(sketch_feature.field("type")?, "sketch")?;
156    bool_value(sketch_feature.field("suppressed")?, false)?;
157
158    let plane = sketch_feature.field("plane")?;
159    plane.keys(&["name", "type"])?;
160    text(plane.field("name")?, "XY")?;
161    text(plane.field("type")?, "origin")?;
162
163    let sketch = sketch_feature.field("sketch")?;
164    sketch.keys(&["constraints", "counters", "curves", "points"])?;
165    let counters = sketch.field("counters")?;
166    counters.keys(&["k", "l"])?;
167    id(counters.field("k")?, 1)?;
168    id(counters.field("l")?, 6)?;
169
170    let points = sketch.field("points")?.items()?;
171    if points.len() != 7 {
172        return Err("bracket sketch must contain the origin and six vertices");
173    }
174    let origin = &points[0];
175    origin.keys(&["fixed", "id", "pos"])?;
176    text(origin.field("id")?, "origin")?;
177    bool_value(origin.field("fixed")?, true)?;
178    vector(origin.field("pos")?, [0, 0])?;
179
180    let mut point_positions = [[0_i64; 2]; 7];
181    let mut point_ids = Vec::with_capacity(7);
182    for (index, point) in points.iter().enumerate() {
183        point.keys(&["fixed", "id", "pos"])?;
184        let point_id = point.field("id")?.text()?.to_owned();
185        if point_id.is_empty() || point_ids.contains(&point_id) {
186            return Err("sketch point ids must be nonempty and unique");
187        }
188        point_ids.push(point_id);
189        let is_origin = index == 0;
190        bool_value(point.field("fixed")?, is_origin)?;
191        let pos = point.field("pos")?.items()?;
192        if pos.len() != 2 {
193            return Err("sketch point must have two coordinates");
194        }
195        for axis in 0..2 {
196            point_positions[index][axis] = i64::try_from(scaled_decimal(&pos[axis], 1, false)?)
197                .map_err(|_| "sketch coordinate range")?;
198        }
199    }
200
201    let constraints = sketch.field("constraints")?.items()?;
202    if constraints.len() != 1 {
203        return Err("unsupported or missing sketch constraint");
204    }
205    let constraint = &constraints[0];
206    constraint.keys(&["id", "p", "q", "type"])?;
207    text(constraint.field("id")?, "k1")?;
208    text(constraint.field("type")?, "coincident")?;
209    id(constraint.field("p")?, 0)?;
210    let origin_vertex = (1..=6)
211        .find(|&index| point_positions[index] == [0, 0])
212        .ok_or("sketch has no origin vertex")?;
213    if (1..=6)
214        .filter(|&index| point_positions[index] == [0, 0])
215        .count()
216        != 1
217    {
218        return Err("sketch origin vertex is ambiguous");
219    }
220    id(
221        constraint.field("q")?,
222        i64::try_from(origin_vertex).map_err(|_| "sketch point index range")?,
223    )?;
224
225    let curves = sketch.field("curves")?.items()?;
226    if curves.len() != 6 {
227        return Err("bracket sketch must contain six line segments");
228    }
229    let expected_segments = [
230        Segment::Bottom,
231        Segment::Right,
232        Segment::Top,
233        Segment::InnerVertical,
234        Segment::UpperLeft,
235        Segment::Left,
236    ];
237    let mut semantic_ids: [Option<String>; 6] = std::array::from_fn(|_| None);
238    let mut seen_ids = Vec::<String>::new();
239    let mut outgoing = [0_u8; 7];
240    let mut incoming = [0_u8; 7];
241    for curve in curves {
242        curve.keys(&["a", "b", "construction", "id", "type"])?;
243        text(curve.field("type")?, "line")?;
244        bool_value(curve.field("construction")?, false)?;
245        let curve_id = curve.field("id")?.text()?.to_owned();
246        if curve_id.is_empty() || seen_ids.contains(&curve_id) {
247            return Err("sketch curve ids must be nonempty and unique");
248        }
249        seen_ids.push(curve_id.clone());
250        let a = usize::try_from(curve.field("a")?.number::<i64>()?)
251            .map_err(|_| "invalid sketch endpoint index")?;
252        let b = usize::try_from(curve.field("b")?.number::<i64>()?)
253            .map_err(|_| "invalid sketch endpoint index")?;
254        if !(1..=6).contains(&a) || !(1..=6).contains(&b) || a == b {
255            return Err("sketch line endpoint is outside the vertex graph");
256        }
257        outgoing[a] += 1;
258        incoming[b] += 1;
259        let pa = point_positions[a];
260        let pb = point_positions[b];
261        let Some((semantic, _)) = expected_segments
262            .iter()
263            .enumerate()
264            .find(|(_, segment)| segment.directed_endpoints() == (pa, pb))
265        else {
266            return Err("sketch lines do not form the supported directed L contour");
267        };
268        if semantic_ids[semantic].replace(curve_id).is_some() {
269            return Err("duplicate geometric segment in sketch contour");
270        }
271    }
272    for index in 1..=6 {
273        if outgoing[index] != 1 || incoming[index] != 1 {
274            return Err("sketch line graph is not one directed six-edge loop");
275        }
276    }
277    if semantic_ids.iter().any(Option::is_none) {
278        return Err("sketch line graph omits a contour segment");
279    }
280    Ok(semantic_ids.map(Option::unwrap))
281}

283fn validate_extrude(feature: &Json) -> Result<(), &'static str> {
284    feature.keys(&[
285        "body_names",
286        "extent",
287        "id",
288        "name",
289        "operation",
290        "param_names",
291        "profiles",
292        "sketch",
293        "suppressed",
294        "type",
295    ])?;
296    id(feature.field("id")?, 2)?;
297    text(feature.field("name")?, "Extrude1")?;
298    text(feature.field("type")?, "extrude")?;
299    bool_value(feature.field("suppressed")?, false)?;
300    text(feature.field("operation")?, "new_body")?;
301    id(feature.field("sketch")?, 1)?;
302    let body_names = feature.field("body_names")?.items()?;
303    if body_names.len() != 1 {
304        return Err("extrusion must create one bracket body");
305    }
306    text(&body_names[0], "Bracket")?;
307    let params = feature.field("param_names")?.items()?;
308    if params.len() != 1 {
309        return Err("extrusion parameter reference is unresolved");
310    }
311    text(&params[0], "d1")?;
312    let profiles = feature.field("profiles")?;
313    profiles.keys(&["type"])?;
314    text(profiles.field("type")?, "all")?;
315    let extent = feature.field("extent")?;
316    extent.keys(&["convention", "direction", "distance"])?;
317    text(extent.field("convention")?, "total_distance")?;
318    text(extent.field("direction")?, "positive")?;
319    if scaled_decimal(extent.field("distance")?, 1, true)? != 40 {
320        return Err("unsupported bracket extrusion distance");
321    }
322    Ok(())
323}

325fn side_name(value: &Json) -> Result<&str, &'static str> {
326    let name = value.text()?;
327    name.strip_prefix("F2:side:")
328        .ok_or("generated face or edge does not reference the extrusion")

331fn curve_index(name: &str, curve_ids: &[String; 6]) -> Result<usize, &'static str> {
332    curve_ids
333        .iter()
334        .position(|id| id == name)
335        .ok_or("timeline references an unknown sketch curve")

338fn validate_fillet(feature: &Json, curve_ids: &[String; 6]) -> Result<(), &'static str> {
339    feature.keys(&[
340        "edge_names",
341        "edge_refs",
342        "edges",
343        "id",
344        "name",
345        "param_names",
346        "radius",
347        "suppressed",
348        "type",
349    ])?;
350    id(feature.field("id")?, 3)?;
351    text(feature.field("name")?, "Fillet1")?;
352    text(feature.field("type")?, "fillet")?;
353    bool_value(feature.field("suppressed")?, false)?;
354    let params = feature.field("param_names")?.items()?;
355    if params.len() != 1 {
356        return Err("fillet parameter reference is unresolved");
357    }
358    text(&params[0], "d2")?;
359    if scaled_decimal(feature.field("radius")?, 8, true)? != 48 {
360        return Err("unsupported inside-corner fillet radius");
361    }
362    let edges = feature.field("edges")?.items()?;
363    if edges.len() != 1 {
364        return Err("fillet must select the single inside vertical edge");
365    }
366    vector(&edges[0], [8, 8, 20])?;
367    let edge_names = feature.field("edge_names")?.items()?;
368    if edge_names.len() != 1 {
369        return Err("fillet edge reference count mismatch");
370    }
371    let names = edge_names[0].text()?.split('|').collect::<Vec<_>>();
372    if names.len() != 2 {
373        return Err("fillet must reference the two contour sides at the inside corner");
374    }
375    let mut semantic = Vec::new();
376    for name in names {
377        let index = curve_index(side_name(&Json::String(name.to_owned()))?, curve_ids)?;
378        semantic.push(index);
379    }
380    semantic.sort_unstable();
381    if semantic != [2, 3] {
382        return Err("fillet reference is not the inside contour corner");
383    }
384    let refs = feature.field("edge_refs")?.items()?;
385    if refs.len() != 1 {
386        return Err("fillet analytic edge reference count mismatch");
387    }
388    let edge = &refs[0];
389    edge.keys(&["dir", "max", "min"])?;
390    vector(edge.field("dir")?, [0, 0, 1])?;
391    vector(edge.field("min")?, [0, 0, 0])?;
392    vector(edge.field("max")?, [60, 50, 40])?;
393    Ok(())
394}

402fn validate_hole(
403    feature: &Json,
404    expected_id: i64,
405    expected_param: &str,
406    curve_ids: &[String; 6],
407) -> Result<Hole, &'static str> {
408    feature.keys(&[
409        "diameter",
410        "direction",
411        "face_names",
412        "hole",
413        "id",
414        "name",
415        "param_names",
416        "position",
417        "suppressed",
418        "type",
419    ])?;
420    id(feature.field("id")?, expected_id)?;
421    let expected_name = if expected_id == 4 { "Hole1" } else { "Hole2" };
422    text(feature.field("name")?, expected_name)?;
423    text(feature.field("type")?, "hole")?;
424    bool_value(feature.field("suppressed")?, false)?;
425    let params = feature.field("param_names")?.items()?;
426    if params.len() != 1 {
427        return Err("hole parameter reference is unresolved");
428    }
429    text(&params[0], expected_param)?;
430    let hole_type = feature.field("hole")?;
431    hole_type.keys(&["type"])?;
432    text(hole_type.field("type")?, "simple")?;
433
434    let faces = feature.field("face_names")?.items()?;
435    if faces.len() != 1 {
436        return Err("hole must select one generated side face");
437    }
438    let face = side_name(&faces[0])?;
439    let line_index = curve_index(face, curve_ids)?;
440    let diameter_eighths = scaled_decimal(feature.field("diameter")?, 8, true)?;
441    if !(40..=80).contains(&diameter_eighths) {
442        return Err("hole diameter is outside the SC1 qualified lattice");
443    }
444
445    let support = if line_index == 2 {
446        vector(feature.field("position")?, [40, 8, 20])?;
447        vector(feature.field("direction")?, [0, -1, 0])?;
448        Segment::Top
449    } else if line_index == 3 {
450        vector(feature.field("position")?, [8, 32, 20])?;
451        vector(feature.field("direction")?, [-1, 0, 0])?;
452        Segment::InnerVertical
453    } else {
454        return Err("hole support is not one of the two qualified bracket faces");
455    };
456    Ok(Hole {
457        support,
458        diameter_eighths,
459    })
460}

462fn decimal_eighths(ticks: i128) -> String {
463    let whole = ticks.div_euclid(8);
464    match ticks.rem_euclid(8) {
465        0 => whole.to_string(),
466        1 => format!("{whole}.125"),
467        2 => format!("{whole}.25"),
468        3 => format!("{whole}.375"),
469        4 => format!("{whole}.5"),
470        5 => format!("{whole}.625"),
471        6 => format!("{whole}.75"),
472        _ => format!("{whole}.875"),
473    }
474}

485fn direct_script(curve_ids: &[String; 6], holes: [Hole; 2]) -> Result<Json, &'static str> {
486    let _ = curve_ids;
487    let mut ordered = holes;
488    ordered.sort_by_key(|hole| match hole.support {
489        Segment::Top => 0,
490        Segment::InnerVertical => 1,
491        _ => 2,
492    });
493    if ordered[0].support != Segment::Top || ordered[1].support != Segment::InnerVertical {
494        return Err("timeline must contain one hole on each qualified face");
495    }
496    let command = |name: &str, params: Json| {
497        object([
498            ("command", Json::String(name.to_owned())),
499            ("params", params),
500        ])
501    };
502    Ok(object([(
503        "commands",
504        Json::Array(vec![
505            command(
506                "sketch.create",
507                object([
508                    ("plane", Json::String("XY".into())),
509                    ("name", Json::String("Profile".into())),
510                ]),
511            ),
512            command(
513                "sketch.line",
514                object([
515                    (
516                        "points",
517                        Json::Array(
518                            [[0, 0], [60, 0], [60, 8], [8, 8], [8, 50], [0, 50]]
519                                .iter()
520                                .map(|values| point(values))
521                                .collect(),
522                        ),
523                    ),
524                    ("closed", Json::Bool(true)),
525                ]),
526            ),
527            command("sketch.finish", object([])),
528            command(
529                "solid.extrude",
530                object([
531                    ("distance", Json::Number("40".into())),
532                    ("body_name", Json::String("Bracket".into())),
533                ]),
534            ),
535            command(
536                "solid.fillet",
537                object([
538                    ("edges", Json::Array(vec![point(&[8, 8, 20])])),
539                    ("radius", Json::Number("6".into())),
540                ]),
541            ),
542            command(
543                "solid.hole",
544                object([
545                    ("position", point(&[40, 8, 20])),
546                    (
547                        "diameter",
548                        Json::Number(decimal_eighths(ordered[0].diameter_eighths)),
549                    ),
550                ]),
551            ),
552            command(
553                "solid.hole",
554                object([
555                    ("position", point(&[8, 32, 20])),
556                    (
557                        "diameter",
558                        Json::Number(decimal_eighths(ordered[1].diameter_eighths)),
559                    ),
560                ]),
561            ),
562        ]),
563    )]))
564}

566/// Translate a validated canonical solidcraft/66 bracket timeline into the
567/// native audited parameterized bracket JSON model.
568pub fn translate_timeline(script: &Json) -> Result<Json, &'static str> {
569    script.keys(&["about", "commands"])?;
570    let about = script.field("about")?;
571    about.keys(&["design", "format", "replay", "version"])?;
572    let design = about.field("design")?.text()?;
573    if design != "editable" && design != "original" {
574        return Err("timeline is not an editable or original SolidCraft design");
575    }
576    text(about.field("format")?, "solidcraft/66")?;
577    if about.field("replay")?.text()?.is_empty() || about.field("version")?.text()?.is_empty() {
578        return Err("incomplete SolidCraft timeline metadata");
579    }
580    let commands = script.field("commands")?.items()?;
581    if commands.len() != 5 {
582        return Err("SolidCraft bracket timeline must contain five active features");
583    }
584    let sketch = feature(&commands[0], "sketch")?;
585    let curve_ids = validate_sketch(sketch)?;
586    let extrude = feature(&commands[1], "extrude")?;
587    validate_extrude(extrude)?;
588    let fillet = feature(&commands[2], "fillet")?;
589    validate_fillet(fillet, &curve_ids)?;
590    let first = feature(&commands[3], "hole")?;
591    let first = validate_hole(first, 4, "d3", &curve_ids)?;
592    let second = feature(&commands[4], "hole")?;
593    let second = validate_hole(second, 5, "d4", &curve_ids)?;
594    if first.support == second.support {
595        return Err("timeline duplicates a bracket hole support");
596    }
597    let direct = direct_script(&curve_ids, [first, second])?;
598    let recipe = solidcraft_bridge::bracket_recipe(&direct)?;
599    let model = interchange::build(&recipe)?;
600    let audit = interchange::audit(&model)?;
601    if audit.field("status")?.text()? != "valid" {
602        return Err("translated timeline failed the native model audit");
603    }
604    Ok(model)
605}

627    fn feature_mut(script: &mut Json, command_index: usize) -> &mut BTreeMap<String, Json> {
628        let Json::Object(root) = script else {
629            unreachable!()
630        };
631        let Json::Array(commands) = root.get_mut("commands").unwrap() else {
632            unreachable!()
633        };
634        let Json::Object(command) = &mut commands[command_index] else {
635            unreachable!()
636        };
637        let Json::Object(params) = command.get_mut("params").unwrap() else {
638            unreachable!()
639        };
640        let Json::Object(feature) = params.get_mut("feature").unwrap() else {
641            unreachable!()
642        };
643        feature
644    }
