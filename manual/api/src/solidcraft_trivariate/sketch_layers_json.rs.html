10pub(crate) const FAMILY: &str = "sketch_layer_partition_v1";
11pub(crate) const BOOLEAN_FAMILY: &str = "sketch_boolean_layers_v1";
12pub(crate) const METHOD: &str = "certified_sketch_layer_partition";
13fn n<T: ToString>(x: T) -> Json {
14    Json::Number(x.to_string())
15}
16fn fraction(v: Ratio) -> Json {
17    Json::Array(vec![n(v.numerator()), n(v.denominator())])
18}
19fn curve(j: &Json) -> Result<SketchLoop, &'static str> {
20    Ok(SketchLoop {
21        id: j.field("id")?.text()?.into(),
22        vertices: j
23            .field("vertices")?
24            .items()?
25            .iter()
26            .map(pair)
27            .collect::<Result<_, _>>()?,
28    })
29}
30fn parse(j: &Json, model: bool) -> Result<SketchLayers, &'static str> {
31    let boolean = j.field(if model { "schema" } else { "family" })?.text()? == BOOLEAN_FAMILY;
32    let features = if boolean { "operations" } else { "cuts" };
33    let mut operations = boolean.then(Vec::new);
34    let mut keys = vec!["outer", features, "height_mm"];
35    if j.field("construction").is_ok() {
36        keys.push("construction");
37    }
38    if j.field("top_shift_mm").is_ok() {
39        keys.push("top_shift_mm");
40    }
41    let strategy = if let Ok(v) = j.field("partition_strategy") {
42        keys.push("partition_strategy");
43        match v.text()? {
44            "automatic" => PlannerStrategy::Automatic,
45            "triangulate" => PlannerStrategy::Triangulate,
46            "ring_planar_rerank" => PlannerStrategy::RingPlanarRerank,
47            "ring_planar_lookahead" => PlannerStrategy::RingPlanarLookahead,
48            _ => return Err("unknown_sketch_partition_strategy"),
49        }
50    } else {
51        PlannerStrategy::Automatic
52    };
53    if nurbs_mode(j)?.is_some() {
54        keys.push("nurbs_partition");
55        if model {
56            keys.push("nurbs_patches");
57        }
58    }
59    if model {
60        keys.extend([
61            "schema",
62            "blocks",
63            "design_faces",
64            "topology",
65            "partition_plan",
66        ]);
67    } else {
68        keys.push("family");
69    }
70    j.keys(&keys)?;
71    let outer = j.field("outer")?;
72    outer.keys(&["id", "vertices"])?;
73    let mut cuts = Vec::new();
74    for c in j.field(features)?.items()? {
75        if let Some(ops) = &mut operations {
76            c.keys(&["id", "vertices", "z_interval_mm", "operation"])?;
77            ops.push(match c.field("operation")?.text()? {
78                "add" => true,
79                "remove" => false,
80                _ => return Err("invalid_sketch_material_operation"),
81            });
82        } else {
83            c.keys(&["id", "vertices", "z_interval_mm"])?;
84        }
85        cuts.push((curve(c)?, pair(c.field("z_interval_mm")?)?));
86    }
87    let mut guides = Vec::new();
88    for g in j
89        .field("construction")
90        .ok()
91        .map(Json::items)
92        .transpose()?
93        .unwrap_or(&[])
94    {
95        g.keys(&["id", "a", "b"])?;
96        guides.push(Segment {
97            id: g.field("id")?.text()?.into(),
98            a: pair(g.field("a")?)?,
99            b: pair(g.field("b")?)?,
100            role: Role::Construction,
101        });
102    }
103    Ok(SketchLayers {
104        outer: curve(outer)?,
105        force_triangulation: strategy == PlannerStrategy::Triangulate,
106        operations,
107        cuts,
108        guides,
109        height: j.field("height_mm")?.number()?,
110        shift: j
111            .field("top_shift_mm")
112            .ok()
113            .map(pair)
114            .transpose()?
115            .unwrap_or([0, 0]),
116        strategy,
117    })
118}
119fn plan(p: &Partition) -> Json {
120    let g = &p.graph;
121    let vertices = g
122        .vertices
123        .iter()
124        .map(|pt| {
125            Json::Array(
126                pt.iter()
127                    .copied()
128                    .map(|v| {
129                        fraction(
130                            v.checked_mul(Ratio::new(1, 2).unwrap())
131                                .expect("bounded sketch coordinate"),
132                        )
133                    })
134                    .collect(),
135            )
136        })
137        .collect();
138    let edges = g
139        .edges
140        .iter()
141        .map(|e| {
142            object([
143                ("vertices", Json::Array(e.vertices.iter().map(n).collect())),
144                (
145                    "sources",
146                    Json::Array(
147                        e.provenance
148                            .iter()
149                            .map(|s| {
150                                object([
151                                    ("id", Json::String(s.segment_id.clone())),
152                                    ("forward", Json::Bool(s.forward)),
153                                    (
154                                        "role",
155                                        Json::String(
156                                            match s.role {
157                                                Role::Boundary => "boundary",
158                                                Role::Construction => "construction",
159                                            }
160                                            .into(),
161                                        ),
162                                    ),
163                                ])
164                            })
165                            .collect(),
166                    ),
167                ),
168            ])
169        })
170        .collect();
171    let faces = g
172        .faces
173        .iter()
174        .map(|f| Json::Array(f.vertices.iter().map(n).collect()))
175        .collect();
176    let layers = p
177        .layer_counts
178        .iter()
179        .map(|(z, b)| {
180            object([
181                ("interval_mm", Json::Array(z.iter().map(n).collect())),
182                ("material_blocks", n(b)),
183            ])
184        })
185        .collect();
186    let mut result = object([
187        ("method", Json::String(p.method.into())),
188        ("vertices_mm", Json::Array(vertices)),
189        ("edges", Json::Array(edges)),
190        ("bounded_faces", Json::Array(faces)),
191        (
192            "generated_horizontal_levels_twice_mm",
193            Json::Array(p.generated_levels.iter().map(n).collect()),
194        ),
195        ("layers", Json::Array(layers)),
196    ]);
197    if let Some((baseline_valid, baseline_blocks, guided_valid, guided_blocks)) =
198        p.candidate_comparison
199    {
200        if let Json::Object(fields) = &mut result {
201            fields.insert(
202                "candidate_comparison".into(),
203                object([
204                    (
205                        "baseline",
206                        object([
207                            ("valid_certificate", Json::Bool(baseline_valid)),
208                            ("blocks", n(baseline_blocks)),
209                        ]),
210                    ),
211                    (
212                        "guided",
213                        object([
214                            ("valid_certificate", Json::Bool(guided_valid)),
215                            ("blocks", n(guided_blocks)),
216                        ]),
217                    ),
218                ]),
219            );
220        }
221    }
222    result
223}

230fn nurbs_mode(j: &Json) -> Result<Option<NurbsMode>, &'static str> {
231    let Ok(mode) = j.field("nurbs_partition") else {
232        return Ok(None);
233    };
234    match mode.text()? {
235        "assemble" => Ok(Some(NurbsMode::Assemble)),
236        "optimize" => Ok(Some(NurbsMode::Optimize)),
237        _ => Err("unknown_nurbs_partition_mode"),
238    }
239}

241fn select(job: &SketchLayers, mode: Option<NurbsMode>) -> Result<Partition, &'static str> {
242    if mode == Some(NurbsMode::Optimize) {
243        crate::sketch_layers::partition_nurbs_aware(job)
244    } else {
245        partition(job)
246    }
247}

249/// Recover the orientation and verify the entire extracted homogeneous net up to
250/// one positive common factor. Four rotations preserve the extrusion axis and sign.
251fn source_rotation(span: &[[[i64; 4]; 4]], source: &[[i64; 4]; 8]) -> Result<usize, &'static str> {
252    for rotation in 0..4 {
253        let source_at = |s: usize, r: usize, z: usize| {
254            let (u, v) = match rotation {
255                0 => (s, r),
256                1 => (1 - r, s),
257                2 => (1 - s, 1 - r),
258                _ => (r, 1 - s),
259            };
260            source[4 * u + 2 * v + z]
261        };
262        let w_src = source_at(0, 0, 0)[3];
263        let w_span = span[0][0][3];
264        if w_src <= 0 || w_span <= 0 {
265            return Err("nonpositive_sketch_spline_weight");
266        }
267        let matches = span.iter().enumerate().all(|(s, section)| {
268            section.iter().enumerate().all(|(j, point)| {
269                let other = source_at(s, j / 2, j % 2);
270                point.iter().zip(other).all(|(a, b)| {
271                    i128::from(*a) * i128::from(w_src) == i128::from(b) * i128::from(w_span)
272                })
273            })
274        });
275        if matches {
276            return Ok(rotation);
277        }
278    }
279    Err("sketch_spline_extraction_mismatch")
280}

282fn patches(p: &Partition) -> Result<Json, &'static str> {
283    let cells = p
284        .blocks
285        .iter()
286        .map(|b| {
287            if b.degrees != [1, 1, 1] {
288                return Err("unsupported_sketch_spline_degree");
289            }
290            b.controls
291                .as_slice()
292                .try_into()
293                .map_err(|_| "invalid_sketch_spline_control_count")
294        })
295        .collect::<Result<Vec<[[i64; 4]; 8]>, _>>()?;
296    let assembled = crate::sketch_nurbs::assemble(&cells);
297    let mut covered = vec![false; cells.len()];
298    let list = assembled
299        .iter()
300        .map(|patch| {
301            let spans = patch.cells.len();
302            if spans == 0 || patch.sections.len() != spans + 1 {
303                return Err("invalid_sketch_spline_shape");
304            }
305            let mut rotations = Vec::with_capacity(spans);
306            for (span, &index) in patch.sections.windows(2).zip(&patch.cells) {
307                let used = covered.get_mut(index).ok_or("unknown_sketch_spline_cell")?;
308                if *used {
309                    return Err("repeated_sketch_spline_cell");
310                }
311                *used = true;
312                rotations.push(n(source_rotation(span, &cells[index])?));
313            }
314            let knots = |count| {
315                Json::Array(
316                    std::iter::once(0)
317                        .chain(0..=count)
318                        .chain(std::iter::once(count))
319                        .map(n)
320                        .collect(),
321                )
322            };
323            let sections = patch
324                .sections
325                .iter()
326                .map(|row| {
327                    Json::Array(
328                        row.iter()
329                            .map(|point| Json::Array(point.iter().map(n).collect()))
330                            .collect(),
331                    )
332                })
333                .collect();
334            Ok(object([
335                ("source_xy_rotations", Json::Array(rotations)),
336                (
337                    "degrees",
338                    Json::Array([1, 1, 1].into_iter().map(n).collect()),
339                ),
340                ("knots", Json::Array(vec![knots(spans), knots(1), knots(1)])),
341                ("sections", Json::Array(sections)),
342                (
343                    "cells",
344                    Json::Array(
345                        patch
346                            .cells
347                            .iter()
348                            .map(|&i| Json::String(p.blocks[i].id.clone()))
349                            .collect(),
350                    ),
351                ),
352            ]))
353        })
354        .collect::<Result<Vec<_>, _>>()?;
355    if covered.contains(&false) {
356        return Err("missing_sketch_spline_cell");
357    }
358    Ok(Json::Array(list))
359}

361fn ledger(p: &Partition, patch_list: Option<&Json>) -> Result<Json, &'static str> {
362    let mut result = plan(p);
363    if let (Some(list), Json::Object(fields)) = (patch_list, &mut result) {
364        fields.insert("nurbs_patch_count".into(), n(list.items()?.len()));
365        fields.insert("bezier_span_count".into(), n(p.blocks.len()));
366    }
367    Ok(result)
368}

370fn representation(p: &Partition, mode: Option<NurbsMode>) -> Result<Option<Json>, &'static str> {
371    mode.map(|_| patches(p)).transpose()
372}

374pub(crate) fn plan_sketch(recipe: &Json) -> Result<Json, &'static str> {
375    let mode = nurbs_mode(recipe)?;
376    let p = select(&parse(recipe, false)?, mode)?;
377    ledger(&p, representation(&p, mode)?.as_ref())
378}

380pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
381    let mode = nurbs_mode(recipe)?;
382    let p = select(&parse(recipe, false)?, mode)?;
383    let patch_list = representation(&p, mode)?;
384    let ledger = ledger(&p, patch_list.as_ref())?;
385    let mut model = encode(recipe.field("family")?.text()?, p.blocks, &p.certificate);
386    if let Json::Object(fields) = &mut model {
387        for k in [
388            "outer",
389            "cuts",
390            "operations",
391            "construction",
392            "height_mm",
393            "top_shift_mm",
394            "partition_strategy",
395            "nurbs_partition",
396        ] {
397            if let Ok(v) = recipe.field(k) {
398                fields.insert(k.into(), v.clone());
399            }
400        }
401        fields.insert("partition_plan".into(), ledger);
402        if let Some(list) = patch_list {
403            fields.insert("nurbs_patches".into(), list);
404        }
405    }
406    Ok(model)
407}
408pub(crate) fn audit(model: &Json) -> Result<Json, &'static str> {
409    let job = parse(model, true)?;
410    let mode = nurbs_mode(model)?;
411    let blocks = decode_blocks(model)?;
412    let replay = |p: &Partition| -> Result<bool, &'static str> {
413        let patch_list = representation(p, mode)?;
414        Ok(model.field("nurbs_patches").ok() == patch_list.as_ref()
415            && blocks == p.blocks
416            && model.field("topology")? == &topology(&p.certificate.incidence)
417            && model.field("design_faces")? == &design(&p.certificate)
418            && model.field("partition_plan")? == &ledger(p, patch_list.as_ref())?)
419    };
420    // Older files can store a fully valid choice displaced by a newer proposal.
421    // Replay all coefficients, design labels, incidence and ledger together.
422    let planners = [
423        partition,
424        crate::sketch_layers::partition_before_rings,
425        crate::sketch_layers::partition_rectangular_hosts,
426        crate::sketch_layers::partition_classic,
427        crate::sketch_layers::partition_legacy,
428    ];
429    let mut found = None;
430    if mode.is_some() {
431        let p = select(&job, mode)?;
432        if replay(&p)? {
433            found = Some(p);
434        }
435    } else {
436        for planner in planners {
437            if let Ok(part) = planner(&job) {
438                if replay(&part)? {
439                    found = Some(part);
440                    break;
441                }
442            }
443        }
444    }
445    let p = found.ok_or("sketch_partition_replay_mismatch")?;
446    let mut report = object([
447        ("status", Json::String("valid".into())),
448        ("method", Json::String(METHOD.into())),
449        ("blocks", n(p.blocks.len())),
450        ("shared_faces", n(p.certificate.incidence.interfaces.len())),
451        ("exterior_sides", n(p.certificate.incidence.exterior.len())),
452        (
453            "determinant_lower_bound_mm3",
454            n(p.certificate.determinant_lower_bound_mm3),
455        ),
456        ("volume_exact_mm3", fraction(p.volume)),
457        ("bounded_sketch_faces", n(p.planar_faces)),
458        ("cavities", n(p.cavities)),
459        ("hausdorff_error_upper_bound_mm", n(0)),
460    ]);
461    if let (Some(_), Json::Object(fields)) = (mode, &mut report) {
462        fields.insert(
463            "nurbs_patch_count".into(),
464            n(model.field("nurbs_patches")?.items()?.len()),
465        );
466        fields.insert("bezier_span_count".into(), n(p.blocks.len()));
467    }
468    Ok(report)
469}

475    #[test]
476    fn octagon_replays_four_patches_and_rejects_modified_knots() {
477        let mut recipe = crate::json::parse(include_str!("../docs/site/data/bench-poly-through_octagon_high_aspect/bench_poly_through_octagon_high_aspect.recipe.json")).unwrap();
478        let Json::Object(fields) = &mut recipe else {
479            unreachable!()
480        };
481        fields.insert("nurbs_partition".into(), Json::String("assemble".into()));
482        let mut model = build(&recipe).unwrap();
483        let report = plan_sketch(&recipe).unwrap();
484        assert_eq!(report.field("nurbs_patch_count").unwrap(), &n(4));
485        assert_eq!(report.field("bezier_span_count").unwrap(), &n(8));
486        assert_eq!(model.field("partition_plan").unwrap(), &report);
487        audit(&model).unwrap();
488        let Json::Object(fields) = &mut model else {
489            unreachable!()
490        };
491        let Some(Json::Array(list)) = fields.get_mut("nurbs_patches") else {
492            unreachable!()
493        };
494        let Json::Object(first) = &mut list[0] else {
495            unreachable!()
496        };
497        first.insert("knots".into(), Json::Array(vec![]));
498        assert_eq!(audit(&model), Err("sketch_partition_replay_mismatch"));
499    }
