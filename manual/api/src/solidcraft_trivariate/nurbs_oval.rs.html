4type Pair = [Vec<Point>; 2];

6fn ordinate(p: Point, d: usize) -> Result<R, Error> {
7    div(p[d], p[2])
8}

10fn meridians(x: &NurbsCurve2, z: &NurbsCurve2) -> Result<Vec<Pair>, Error> {
11    let curves = [
12        extract_bezier(x.degree, &x.knots, &x.controls)?,
13        extract_bezier(z.degree, &z.knots, &z.controls)?,
14    ];
15    let mut breaks = BTreeSet::new();
16    for curve in &curves {
17        for span in curve {
18            breaks.extend(span.interval);
19        }
20    }
21    let breaks: Vec<_> = breaks.into_iter().collect();
22    let mut out = Vec::new();
23    for t in breaks.windows(2) {
24        let mut pair: Pair = Default::default();
25        for (net, curve) in pair.iter_mut().zip(&curves) {
26            let span = curve
27                .iter()
28                .find(|s| s.interval[0] <= t[0] && s.interval[1] >= t[1])
29                .ok_or("meridians need common parameter support")?;
30            *net = restricted(span, [t[0], t[1]], x.degree.max(z.degree))?;
31        }
32        if pair[0].iter().zip(&pair[1]).any(|(a, b)| a[1..] != b[1..]) {
33            return Err("unsynchronized meridian ordinate or weight");
34        }
35        out.push(pair);
36    }
37    if out.len() == 1 {
38        let [x, z] = out.pop().unwrap();
39        let (a, b) = split(&x, R::new(1, 2)?)?;
40        let (c, d) = split(&z, R::new(1, 2)?)?;
41        out.extend([[a, c], [b, d]]);
42    }
43    if out.len() < 2 {
44        return Err("empty meridian");
45    }
46    for d in 0..2 {
47        let first = &out[0][d];
48        let last = &out.last().unwrap()[d];
49        let n = last.len() - 1;
50        if ordinate(first[0], 1)? != int(0)
51            || ordinate(last[n], 0)? != int(0)
52            || ordinate(first[0], 0)? != ordinate(first[1], 0)?
53            || ordinate(last[n], 1)? != ordinate(last[n - 1], 1)?
54        {
55            return Err("meridian requires regular reflected endpoints");
56        }
57        for (i, pair) in out.iter().enumerate() {
58            let net = &pair[d];
59            let n = net.len() - 1;
60            for w in net.windows(2) {
61                if less(ordinate(w[0], 0)?, ordinate(w[1], 0)?)?
62                    || less(ordinate(w[1], 1)?, ordinate(w[0], 1)?)?
63                {
64                    return Err("nonmonotone meridian controls");
65                }
66            }
67            for k in [0, n - 1] {
68                let strict = if i == 0 {
69                    less(ordinate(net[k], 1)?, ordinate(net[k + 1], 1)?)?
70                } else {
71                    less(ordinate(net[k + 1], 0)?, ordinate(net[k], 0)?)?
72                };
73                if !strict {
74                    return Err("uncertified regular meridian chart");
75                }
76            }
77        }
78    }
79    Ok(out)
80}

82fn world(p: [R; 3], w: R, sign: [i64; 3], center: [i64; 3]) -> Result<[R; 4], Error> {
83    let mut out = [w; 4];
84    for i in 0..3 {
85        out[i] = p[i]
86            .checked_mul(int(sign[i]))?
87            .checked_add(w.checked_mul(int(center[i]))?)?;
88    }
89    Ok(out)
90}

92/// Construct the material outside two intersecting extrusions of symmetric,
93/// smooth-ended NURBS sections in a rectangular stock.
94pub fn build_oval(
95    bounds: [[i64; 2]; 3],
96    center: [i64; 3],
97    x: &NurbsCurve2,
98    z: &NurbsCurve2,
99) -> Result<Vec<BracketBlock>, Error> {
100    let nets = meridians(x, z)?;
101    let tip = nets[0][0].len() - 1;
102    let seam = [
103        ordinate(nets[0][0][tip], 0)?,
104        ordinate(nets[0][0][tip], 1)?,
105        ordinate(nets[0][1][tip], 0)?,
106    ];
107    if seam.iter().any(|a| a.numerator() <= 0) {
108        return Err("degenerate meridian seam");
109    }
110    let size = [
111        ordinate(nets[0][0][0], 0)?,
112        ordinate(*nets.last().unwrap()[0].last().unwrap(), 1)?,
113        ordinate(nets[0][1][0], 0)?,
114    ];
115    let mut blocks = Vec::new();
116    for oct in 0..8 {
117        let mut sign = [1; 3];
118        let mut h = [int(0); 3];
119        for i in 0..3 {
120            let side = (oct >> i) & 1;
121            sign[i] = 2 * side as i64 - 1;
122            h[i] = int(bounds[i][side])
123                .checked_sub(int(center[i]))?
124                .checked_mul(int(sign[i]))?;
125            if !less(size[i], h[i])? {
126                return Err("meridian must lie strictly inside stock");
127            }
128        }
129        let gx = div(h[0].checked_add(size[0])?, int(2))?;
130        let gz = div(h[2].checked_add(size[2])?, int(2))?;
131        let orientation = sign.iter().product::<i64>();
132        for (i, pair) in nets.iter().enumerate() {
133            for dominant in [0, 2] {
134                let forward = (if dominant == 0 { 1 } else { -1 }) * orientation > 0;
135                let mut net = Vec::new();
136                for (a, b) in pair[0].iter().zip(&pair[1]) {
137                    let w = a[2];
138                    let inner = [a[0], a[1], b[0]];
139                    let outer = if i == 0 {
140                        [
141                            gx.checked_mul(w)?,
142                            div(h[1].checked_mul(a[1])?, seam[1])?,
143                            gz.checked_mul(w)?,
144                        ]
145                    } else {
146                        [
147                            div(gx.checked_mul(a[0])?, seam[0])?,
148                            h[1].checked_mul(w)?,
149                            div(gz.checked_mul(b[0])?, seam[2])?,
150                        ]
151                    };
152                    for p in [inner, outer] {
153                        for upper in [false, true] {
154                            let mut q = p;
155                            if upper == forward {
156                                q[dominant] = h[dominant].checked_mul(w)?;
157                            }
158                            net.push(world(q, w, sign, center)?);
159                        }
160                    }
161                }
162                blocks.push(BracketBlock {
163                    id: format!("oval_{oct}_{i}_{dominant}"),
164                    degrees: [pair[0].len() - 1, 1, 1],
165                    controls: integer_net(&net)?,
166                });
167            }
168        }
169        let mut net = Vec::new();
170        for xx in if orientation > 0 {
171            [gx, h[0]]
172        } else {
173            [h[0], gx]
174        } {
175            for a in &nets[0][0] {
176                let w = a[2];
177                for zz in [gz, h[2]] {
178                    net.push(world(
179                        [
180                            xx.checked_mul(w)?,
181                            div(h[1].checked_mul(a[1])?, seam[1])?,
182                            zz.checked_mul(w)?,
183                        ],
184                        w,
185                        sign,
186                        center,
187                    )?);
188                }
189            }
190        }
191        blocks.push(BracketBlock {
192            id: format!("oval_corner_{oct}"),
193            degrees: [1, tip, 1],
194            controls: integer_net(&net)?,
195        });
196    }
197    Ok(blocks)
198}
