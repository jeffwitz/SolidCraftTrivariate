1//! Edit the gallery annulus in its four original charts. The new axial knots
2//! define three integration spans per volume. All nets are homogeneous in mm.
3use crate::{bracket::BracketBlock, sections::section_blocks};
4
5const HEIGHTS: [i64; 4] = [0, 1, 31, 32]; // quarter millimetres
6
7fn control_net(source: &BracketBlock, chamfer: bool) -> Vec<[i64; 4]> {
8    assert_eq!(source.degrees, [2, 1, 1]);
9    let mut net = Vec::with_capacity(24);
10    for angular in 0..3 {
11        for radial in 0..2 {
12            let [x, y, _, w] = source.controls[4 * angular + 2 * radial];
13            for z in HEIGHTS {
14                // Outer radius 4 -> 3.75 and inner radius 2 -> 2.25.
15                let scale = if chamfer && (z == 0 || z == 32) {
16                    [15, 18][radial]
17                } else {
18                    16
19                };
20                net.push([x * scale, y * scale, z * 4 * w, 16 * w]);
21            }
22        }
23    }
24    net
25}

27use super::{numbers, Json};
28pub(super) fn document(chamfer: bool) -> Result<Json, &'static str> {
29    let blocks = section_blocks("full_annulus_v1")?;
30    Ok(super::original_document(
31        [2, 1, 1],
32        [
33            numbers([0, 0, 0, 1, 1, 1]),
34            numbers([0, 0, 1, 1]),
35            super::axial_knots(1),
36        ],
37        blocks.iter().map(|b| (&b.id, control_net(b, chamfer))),
38    ))
39}

41mod tests {

43    use crate::spatial_certificate::{certify_spatial, CertificateStatus};

45    #[test]
46    fn three_closed_spans_per_original_volume_are_regular_and_match_their_neighbors() {
47        let sources = section_blocks("full_annulus_v1").unwrap();
48        assert_eq!(sources.len(), 4);
49        for chamfer in [false, true] {
50            let nets: Vec<_> = sources.iter().map(|s| control_net(s, chamfer)).collect();
51            let mut lower = f64::INFINITY;
52            for (i, net) in nets.iter().enumerate() {
53                assert_eq!(net[16..24], nets[(i + 1) % 4][0..8]);
54                for layer in 0..3 {
55                    let span: Vec<_> = net
56                        .as_chunks::<4>()
57                        .0
58                        .iter()
59                        .flat_map(|row| row[layer..=layer + 1].iter().copied())
60                        .collect();
61                    let result = certify_spatial([2, 1, 1], &span, Default::default());
62                    assert_eq!(result.status, CertificateStatus::Valid);
63                    lower = lower.min(result.determinant_lower_bound_mm3.unwrap());
64                }
65                if !chamfer {
66                    for (old, new) in sources[i]
67                        .controls
68                        .as_chunks::<2>()
69                        .0
70                        .iter()
71                        .zip(net.as_chunks::<4>().0.iter())
72                    {
73                        assert_eq!(new[0], old[0].map(|x| 16 * x));
74                        assert_eq!(new[3], old[1].map(|x| 16 * x));
75                    }
76                }
77            }
78            println!("chamfer={chamfer}, closed span determinant >= {lower:.12} mm³");
79        }
80    }

82    #[test]
83    fn axial_stations_integrate_to_the_reference_step_volume() {
84        let sources = section_blocks("full_annulus_v1").unwrap();
85        for (chamfer, reference) in [
86            (false, 96.0 * std::f64::consts::PI),
87            (true, 299.2367002544223),
88        ] {
89            let net = control_net(&sources[0], chamfer);
90            let areas = HEIGHTS.map(|h| {
91                let k = HEIGHTS.iter().position(|&z| z == h).unwrap();
92                let [outer, inner] =
93                    [net[k], net[4 + k]].map(|[x, y, _, w]| (x as f64).hypot(y as f64) / w as f64);
94                assert!((outer + inner - 6.).abs() < 1e-13);
95                std::f64::consts::PI * (outer - inner) * (outer + inner)
96            });
97            // The sum of the radii is constant, so the area is affine on each span.
98            let volume: f64 = (0..3)
99                .map(|k| (areas[k] + areas[k + 1]) * (HEIGHTS[k + 1] - HEIGHTS[k]) as f64 / 8.)
100                .sum();
101            assert!((volume - reference).abs() < 1e-10);
102        }
103    }
