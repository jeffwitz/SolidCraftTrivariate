1//! Exchange and closed-domain certification of the bounded oblique pocket experiment.
2use crate::{
3    bracket_audit::BracketCertificate,
4    bracket_json::{design, encode_blocks, topology},
5    incidence::enumerate_faces,
6    json::{object, parse, Json},
7    oblique_pockets::build_oblique_blind_pockets,
8    spatial_certificate::{certify_spatial, CertificateBudget},
9};
10
11pub(crate) const FAMILY: &str = "oblique_blind_pockets_v1";
12pub(crate) const METHOD: &str = "certified_oblique_blind_pockets";
13const FIXTURE: &str = include_str!("../tests/references/oblique_blind_pockets.recipe.json");
14
15pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
16    if recipe != &parse(FIXTURE)? {
17        return Err("oblique pocket profiles or sweep parameters outside the qualified fixture");
18    }
19    let blocks = build_oblique_blind_pockets()?;
20    let incidence = enumerate_faces(&blocks)?;
21    let mut lower = f64::INFINITY;
22    for block in &blocks {
23        let bound = certify_spatial(block.degrees, &block.controls, CertificateBudget::default())
24            .determinant_lower_bound_mm3
25            .ok_or("oblique pocket chart lacks a strict Jacobian certificate")?;
26        lower = lower.min(bound);
27    }
28    let design_faces = incidence
29        .exterior
30        .iter()
31        .map(|side| {
32            (
33                side.clone(),
34                format!(
35                    "{}:{}{}",
36                    side.block,
37                    ["u", "v", "w"][side.axis],
38                    u8::from(side.upper)
39                ),
40            )
41        })
42        .collect();
43    let cert = BracketCertificate {
44        incidence,
45        determinant_lower_bound_mm3: lower,
46        radial_error_upper_mm: 0.,
47        design_faces,
48    };
49    Ok(object([
50        ("schema", Json::String(FAMILY.into())),
51        ("recipe", recipe.clone()),
52        ("blocks", encode_blocks(blocks)),
53        ("topology", topology(&cert.incidence)),
54        ("design_faces", design(&cert)),
55        (
56            "determinant_lower_bound_mm3",
57            Json::Number(lower.to_string()),
58        ),
59    ]))
60}
61
62pub(crate) fn audit(model: &Json) -> Result<Json, &'static str> {
63    if build(model.field("recipe")?)? != *model {
64        return Err("oblique pocket model differs from its certified construction");
65    }
66    let count =
67        |j: &Json| -> Result<Json, &'static str> { Ok(Json::Number(j.items()?.len().to_string())) };
68    let topology = model.field("topology")?;
69    Ok(object([
70        ("method", Json::String(METHOD.into())),
71        ("status", Json::String("valid".into())),
72        ("blocks", count(model.field("blocks")?)?),
73        ("shared_faces", count(topology.field("interfaces")?)?),
74        ("exterior_sides", count(topology.field("exterior")?)?),
75        (
76            "determinant_lower_bound_mm3",
77            model.field("determinant_lower_bound_mm3")?.clone(),
78        ),
79    ]))
80}
