1//! Divide a rectangular wing into disjoint hosts for bores of different axes.
2//! A separating slab boundary must clear every tool over its entire length.
3use crate::compact_hosts::{holed_box, Bounds, HoledBox};
4use crate::orthogonal_sketch::Bore;
5
6pub(crate) fn build(
7    bounds: Bounds,
8    bores: &[&Bore],
9    prefix: &str,
10) -> Result<Option<HoledBox>, &'static str> {
11    let Some(first) = bores.first() else {
12        return Ok(None);
13    };
14    if bores.iter().all(|b| b.axis == first.axis) {
15        return holed_box(first.axis, bounds, bores, prefix, false);
16    }
17    for axis in 0..3 {
18        if bores.iter().any(|b| b.axis == axis) {
19            continue;
20        }
21        let mut ordered = bores.to_vec();
22        ordered.sort_by_key(|b| (b.center[axis], &b.id));
23        for split in 1..ordered.len() {
24            let (lower, upper) = ordered.split_at(split);
25            let end = lower
26                .iter()
27                .map(|b| b.center[axis] + b.radius)
28                .max()
29                .unwrap();
30            let start = upper
31                .iter()
32                .map(|b| b.center[axis] - b.radius)
33                .min()
34                .unwrap();
35            let cut = end + (start - end) / 2;
36            if !(end < cut && cut < start && bounds[axis][0] < cut && cut < bounds[axis][1]) {
37                continue;
38            }
39            let mut a = bounds;
40            let mut b = bounds;
41            a[axis][1] = cut;
42            b[axis][0] = cut;
43            let Some(mut left) = build(a, lower, &format!("{prefix}_a{axis}_{cut}_lo"))? else {
44                continue;
45            };
46            let Some(mut right) = build(b, upper, &format!("{prefix}_a{axis}_{cut}_hi"))? else {
47                continue;
48            };
49            left.blocks.append(&mut right.blocks);
50            left.physical.append(&mut right.physical);
51            left.guards.append(&mut right.guards);
52            return Ok(Some(left));
53        }
54    }
55    Ok(None)
56}

58#[cfg(test)]
59mod tests {
60    use super::*;
61    #[test]
62    fn perpendicular_tools_require_a_separating_interval_in_the_common_cross_axis() {
63        let z = Bore {
64            id: "z_bore".into(),
65            axis: 2,
66            center: [25, 4, 20],
67            radius: 2,
68        };
69        let mut y = Bore {
70            id: "y_bore".into(),
71            axis: 1,
72            center: [40, 4, 20],
73            radius: 4,
74        };
75        let bounds = [[14, 60], [0, 8], [0, 40]];
76        // Allow clearance around the side-drilled bore within the host's XZ section.
77        let candidate = build(bounds, &[&z, &y], "arm").unwrap().unwrap();
78        assert_eq!(candidate.guards.len(), 2);
79        y.center[0] = 25;
80        assert!(build(bounds, &[&z, &y], "arm").unwrap().is_none());
81    }
