1//! Fixed two-circle overlap bore in a square plate.
2//!
3//! The family is intentionally limited to two radius-2 mm circles centered
4//! at (+/-1, 0) mm in [-10, 10]^2 x [0, 8] mm. The conic arcs use a frozen
5//! rational approximation of sqrt(3); all emitted homogeneous coefficients
6//! are exact integers and each radial generator shares its endpoint weights.
7
8use crate::bracket::BracketBlock;
9use std::convert::TryFrom;
10
11const S_NUM: i128 = 1_732_050_807_568_877;
12const S_DEN: i128 = 1_000_000_000_000_000;
13const SCALE: i128 = S_DEN;
14const HEIGHT_MM: i128 = 8;
15const SECTORS: [&str; 2] = ["right", "left"];
16// t intervals [-1,-1/3], [-1/3,0], [0,1/3], [1/3,1], as (p,r)/3.
17const INTERVALS: [(i128, i128); 4] = [(-3, 2), (-1, 1), (0, 1), (1, 2)];
18
19fn add(a: i128, b: i128) -> Result<i128, &'static str> {
20    a.checked_add(b).ok_or("overlap bore coefficient overflow")
21}
22
23fn sub(a: i128, b: i128) -> Result<i128, &'static str> {
24    a.checked_sub(b).ok_or("overlap bore coefficient overflow")
25}
26
27fn mul(a: i128, b: i128) -> Result<i128, &'static str> {
28    a.checked_mul(b).ok_or("overlap bore coefficient overflow")
29}
30
31fn output(value: i128) -> Result<i64, &'static str> {
32    i64::try_from(value).map_err(|_| "overlap bore control exceeds i64")
33}

35/// Exact homogeneous controls for one restricted rational conic arc.
36///
37/// Each triple is `(wx, wy, w)`, and the local curve parameter runs from zero
38/// to one over the corresponding fixed interval in the global conic parameter.
39fn arc_controls(interval: usize) -> Result<[[i128; 3]; 3], &'static str> {
40    let (p, r) = INTERVALS[interval];
41    let l = SCALE;
42
43    // Substitute t=(p+r*u)/3 into
44    //   X = 1 + 2(1-3t^2)/(1+3t^2),
45    //   Y = 4*s*t/(1+3t^2),
46    //   W = 1.
47    // Convert each power-basis quadratic to Bernstein controls. The middle
48    // control is c0+c1/2; c1 is always even for these integer parameters.
49    let p2 = mul(p, p)?;
50    let pr2 = mul(2, mul(p, r)?)?;
51    let r2 = mul(r, r)?;
52    let x_coeff = [mul(l, sub(9, p2)?)?, mul(l, -pr2)?, mul(l, -r2)?];
53    let y_coeff = [mul(4, mul(S_NUM, p)?)?, mul(4, mul(S_NUM, r)?)?, 0];
54    let w_coeff = [mul(l, add(3, p2)?)?, mul(l, pr2)?, mul(l, r2)?];
55    let coefficients = [x_coeff, y_coeff, w_coeff];
56    let mut controls = [[0_i128; 3]; 3];
57    for component in 0..3 {
58        let [c0, c1, c2] = coefficients[component];
59        if c1 % 2 != 0 {
60            return Err("nonintegral overlap bore Bernstein control");
61        }
62        controls[0][component] = c0;
63        controls[1][component] = add(c0, c1 / 2)?;
64        controls[2][component] = add(add(c0, c1)?, c2)?;
65    }
66    Ok(controls)
67}

69fn outer_polyline() -> [[i128; 2]; 5] {
70    [[0, -10], [10, -10], [10, 0], [10, 10], [0, 10]]
71}

73fn rotated_point(point: [i128; 2], half: usize) -> Result<[i128; 2], &'static str> {
74    if half == 0 {
75        Ok(point)
76    } else {
77        Ok([
78            point[0]
79                .checked_neg()
80                .ok_or("overlap bore coefficient overflow")?,
81            point[1]
82                .checked_neg()
83                .ok_or("overlap bore coefficient overflow")?,
84        ])
85    }
86}

88fn build_sector(half: usize, sector: usize) -> Result<BracketBlock, &'static str> {
89    let arc = arc_controls(sector)?;
90    let path = outer_polyline();
91    let [path_start, path_end] = [path[sector], path[sector + 1]];
92    let outer = [
93        path_start,
94        [
95            add(path_start[0], path_end[0])? / 2,
96            add(path_start[1], path_end[1])? / 2,
97        ],
98        path_end,
99    ];
100    let mut controls = Vec::with_capacity(12);
101    for angle in 0..3 {
102        let [x_inner, y_inner, weight] = arc[angle];
103        let [outer_x, outer_y] = rotated_point(outer[angle], half)?;
104        let (x_inner, y_inner) = if half == 0 {
105            (x_inner, y_inner)
106        } else {
107            (
108                x_inner
109                    .checked_neg()
110                    .ok_or("overlap bore coefficient overflow")?,
111                y_inner
112                    .checked_neg()
113                    .ok_or("overlap bore coefficient overflow")?,
114            )
115        };
116        let radial = [
117            [mul(outer_x, weight)?, mul(outer_y, weight)?, weight],
118            [x_inner, y_inner, weight],
119        ];
120        for [x, y, w] in radial {
121            for z in [0_i128, HEIGHT_MM] {
122                controls.push([output(x)?, output(y)?, output(mul(z, w)?)?, output(w)?]);
123            }
124        }
125    }
126    Ok(BracketBlock {
127        id: format!("overlap_bore_{}_sector_{sector:02}_z0_8", SECTORS[half]),
128        degrees: [2, 1, 1],
129        controls,
130    })
131}

133/// Build the fixed two-overlapping-circle bore as eight rational sectors.
134///
135/// The right half uses increasing conic parameter from the lower intersection
136/// to the upper intersection. The left half is its direct pi rotation. For
137/// each half, sector index 00..03 follows the outer polyline from the lower
138/// square seam, across the bottom/right/top edges, to the upper square seam.
139/// Controls are flattened in `(angle, radial, z)` order, with z fastest;
140/// radial index 0 is the square-side curve and radial index 1 the bore arc.
141///
142/// The approximation `S_NUM/S_DEN` substitutes for sqrt(3); construction of
143/// the represented rational model is exact, while the tiny boundary mismatch
144/// to the ideal intersection is certified independently by the audit.
145pub fn overlap_blocks() -> Result<Vec<BracketBlock>, &'static str> {
146    if S_DEN != SCALE || S_NUM <= 0 || S_NUM >= 2 * SCALE {
147        return Err("invalid frozen overlap bore constants");
148    }
149    let mut blocks = Vec::with_capacity(8);
150    for half in 0..2 {
151        for sector in 0..4 {
152            blocks.push(build_sector(half, sector)?);
153        }
154    }
155    if blocks.len() != 8
156        || blocks.iter().any(|block| {
157            block.degrees != [2, 1, 1]
158                || block.controls.len() != 12
159                || block.controls.iter().any(|control| control[3] <= 0)
160        })
161    {
162        return Err("invalid overlap bore control net");
163    }
164    Ok(blocks)
165}
