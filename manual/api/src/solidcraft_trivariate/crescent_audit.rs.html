11const P: i128 = 1_732_050_807_568_877;
12const Q: i128 = 1_000_000_000_000_000;

14/// Restore the planar section to its reference extrusion after checking all
15/// stored lower and upper controls against the physical half-mm slice planes.
16fn reference_extrusion(
17    block: &BracketBlock,
18    bounds: [i32; 2],
19) -> Result<BracketBlock, &'static str> {
20    if !matches!(block.degrees, [2, 1, 1] | [2, 2, 1])
21        || block.controls.len() != 6 * (block.degrees[1] + 1)
22    {
23        return Err("crescent_control_layout");
24    }
25    let mut restored = block.clone();
26    for pair in restored.controls.as_chunks_mut::<2>().0 {
27        if pair[0][..2] != pair[1][..2] || pair[0][3] != pair[1][3] || pair[0][3] <= 0 {
28            return Err("crescent_extrusion");
29        }
30        for (k, p) in pair.iter_mut().enumerate() {
31            if 2 * i128::from(p[2]) != i128::from(bounds[k]) * i128::from(p[3]) {
32                return Err("crescent_layer_height");
33            }
34            p[2] = (8 * k as i64)
35                .checked_mul(p[3])
36                .ok_or("crescent_height_range")?;
37        }
38    }
39    Ok(restored)
40}

42fn conic_edge(
43    nodes: [[i128; 3]; 3],
44    start: i128,
45    end: i128,
46    reflected_x: bool,
47) -> Result<(), &'static str> {
48    // t varies from start/6 to end/6. Five exact nodes prove the complete
49    // degree-four identity after clearing the two quadratic denominators.
50    for n in 0..=4i128 {
51        let t = (4 - n) * start + n * end;
52        let d = 24;
53        let w = d * d + 3 * t * t;
54        let sigma = [(4 - n).pow(2), 2 * n * (4 - n), n * n];
55        let [x, y, weight]: [i128; 3] =
56            std::array::from_fn(|k| (0..3).map(|i| sigma[i] * nodes[i][k]).sum());
57        // Left branch is the pi-rotation of the reference right arc.
58        // The hidden right branch is the reflection in x=1 of its visible branch.
59        let right = w + 2 * (d * d - 3 * t * t);
60        let target_x = if reflected_x { 2 * w - right } else { -right };
61        let y_sign = if reflected_x { 1 } else { -1 };
62        if x * w != weight * target_x || ratio(y, weight) != ratio(y_sign * 4 * t * d * P, w * Q) {
63            return Err("crescent_boundary_conic");
64        }
65    }
66    Ok(())
67}

69fn ratio(n: i128, d: i128) -> (i128, i128) {
70    let (mut a, mut b) = (n.abs(), d);
71    while b != 0 {
72        (a, b) = (b, a % b);
73    }
74    (n / a, d / a)
75}

77fn linear(edge: [[i128; 3]; 3]) -> Result<(), &'static str> {
78    if (0..3).any(|k| 2 * edge[1][k] != edge[0][k] + edge[2][k]) {
79        return Err("crescent_noncanonical_straight_seam");
80    }
81    Ok(())
82}

84fn certify_crescent(block: &BracketBlock, lower: bool, patch: &str) -> Result<f64, &'static str> {
85    if block.degrees != [2, 2, 1] {
86        return Err("crescent_biquadratic_layout");
87    }
88    let profile: Vec<_> = block
89        .controls
90        .as_chunks::<2>()
91        .0
92        .iter()
93        .map(|p| [p[0][0], p[0][1], p[0][3]])
94        .collect();
95    let h = |u: usize, v: usize| {
96        let [x, y, w] = profile[3 * (if lower { 2 - u } else { u }) + v].map(i128::from);
97        [x, if lower { -y } else { y }, w]
98    };
99    let left = std::array::from_fn(|v| h(0, v));
100    let right = std::array::from_fn(|v| h(2, v));
101    let bottom = std::array::from_fn(|u| h(u, 0));
102    let top = std::array::from_fn(|u| h(u, 2));
103    match patch {
104        "a" => {
105            conic_edge(left, 0, -2, false)?;
106            linear(right)?;
107            linear(top)?;
108        }
109        "b" => {
110            conic_edge(right, 0, 1, true)?;
111            linear(left)?;
112            linear(top)?;
113        }
114        "c" => {
115            conic_edge(right, 1, 2, true)?;
116            conic_edge(top, -2, -6, false)?;
117            linear(left)?;
118            linear(bottom)?;
119        }
120        _ => return Err("crescent_patch_identity"),
121    }
122    if patch != "c" {
123        let e = if patch == "a" { bottom[2] } else { bottom[0] };
124        if e[0] != -2 * e[2] {
125            return Err("crescent_baseline_midpoint");
126        }
127        if bottom.iter().any(|p| p[1] != 0)
128            || bottom
129                .windows(2)
130                .any(|p| p[0][0] * p[1][2] >= p[1][0] * p[0][2])
131            || bottom[0][0] < -3 * bottom[0][2]
132            || bottom[2][0] > -bottom[2][2]
133        {
134            return Err("crescent_symmetry_edge");
135        }
136    }
137    // D,E,F meet the physical boundary, while O lies strictly inside the
138    // triangle of the three cuts. The three straight incident seams have
139    // different directions and meet only at O. With positive chart
140    // orientation, cancellation of matching seams gives winding number
141    // one in this half-crescent and zero outside.
142    let interior = match patch {
143        "a" => h(2, 2),
144        "b" => h(0, 2),
145        _ => h(0, 0),
146    };
147    let [x, y, w] = interior;
148    // Three exact oriented half-plane tests for D=(-2,s), E=(-2,0),
149    // F=(-9/13,8s/13) with s=P/Q.
150    if x <= -2 * w || 17 * Q * y <= 8 * P * (x + 2 * w) || 17 * Q * y >= P * (7 * w - 5 * x) {
151        return Err("crescent_center_outside_cut_triangle");
152    }
153    certify_planar_refined([2, 2], &profile).map_err(|e| match e {
154        PlanarError::Invalid => "crescent_nonpositive_jacobian",
155        PlanarError::Indeterminate => "crescent_indeterminate_bernstein",
156    })
157}

159pub fn audit_crescent_pocket(
160    depth_half_mm: i32,
161    blocks: &[BracketBlock],
162) -> Result<BracketCertificate, &'static str> {
163    if !(0..=16).contains(&depth_half_mm) {
164        return Err("crescent_depth_range");
165    }
166    let cut = 16 - depth_half_mm;
167    let mut lower_ring = Vec::new();
168    let mut upper_ring = Vec::new();
169    let mut crescent_count = 0;
170    let mut seen = BTreeSet::new();
171    let mut minimum = f64::INFINITY;
172    let mut features = BTreeMap::new();
173    for block in blocks {
174        if !seen.insert(&block.id) {
175            return Err("crescent_duplicate_identity");
176        }
177        if let Some(name) = block.id.strip_prefix("g10_crescent_") {
178            let (half, patch) = name.split_once('_').ok_or("crescent_identity")?;
179            if !matches!(half, "upper" | "lower") || cut == 0 {
180                return Err("crescent_unexpected_half");
181            }
182            let lower = half == "lower";
183            let source = reference_extrusion(block, [0, cut])?;
184            minimum =
185                minimum.min(certify_crescent(&source, lower, patch)? * (f64::from(cut) * 0.5));
186            if patch != "a" {
187                features.insert((block.id.clone(), 0, !lower), "bore_right".to_owned());
188            }
189            features.insert((block.id.clone(), 2, false), "z0".into());
190            features.insert(
191                (block.id.clone(), 2, true),
192                if depth_half_mm == 0 {
193                    "z8"
194                } else {
195                    "pocket_floor"
196                }
197                .into(),
198            );
199            crescent_count += 1;
200            continue;
201        }
202        let (base, upper) = if let Some(s) = block.id.strip_prefix("g10_ring_upper_") {
203            (s.strip_suffix("_split_8").ok_or("crescent_layer_id")?, true)
204        } else if let Some(s) = block.id.strip_prefix("g10_ring_lower_") {
205            (
206                s.strip_suffix("_z0_split").ok_or("crescent_layer_id")?,
207                false,
208            )
209        } else {
210            return Err("crescent_unknown_block");
211        };
212        let z = if upper { [cut, 16] } else { [0, cut] };
213        if z[0] == z[1] {
214            return Err("crescent_empty_layer");
215        }
216        let mut source = reference_extrusion(block, z)?;
217        source.id = format!("overlap_bore_{base}_z0_8");
218        if upper {
219            upper_ring.push(source);
220        } else {
221            lower_ring.push(source);
222        }
223    }
224    if crescent_count != if cut > 0 { 6 } else { 0 } {
225        return Err("crescent_inventory");
226    }
227    for (ring, upper, bounds) in [
228        (&upper_ring, true, [cut, 16]),
229        (&lower_ring, false, [0, cut]),
230    ] {
231        if bounds[0] == bounds[1] {
232            continue;
233        }
234        let cert = audit_overlap(ring)?;
235        minimum = minimum
236            .min(cert.determinant_lower_bound_mm3 * (f64::from(bounds[1] - bounds[0]) / 16.));
237        for (side, tag) in cert.design_faces {
238            if side.axis == 2
239                && ((side.upper && bounds[1] != 16) || (!side.upper && bounds[0] != 0))
240            {
241                continue;
242            }
243            if tag == "bore_left" && !upper {
244                continue;
245            }
246            let base = side
247                .block
248                .strip_prefix("overlap_bore_")
249                .and_then(|s| s.strip_suffix("_z0_8"))
250                .ok_or("crescent_source_id")?;
251            let layer = if upper { "upper" } else { "lower" };
252            let feature = if tag == "bore_left" {
253                "pocket_wall".to_owned()
254            } else {
255                tag
256            };
257            features.insert(
258                (
259                    format!(
260                        "g10_ring_{layer}_{base}_{}",
261                        if upper { "split_8" } else { "z0_split" }
262                    ),
263                    side.axis,
264                    side.upper,
265                ),
266                feature,
267            );
268        }
269    }
270    let incidence = enumerate_faces(blocks)?;
271    let expected = match depth_half_mm {
272        0 => 20,
273        16 => 8,
274        _ => 36,
275    };
276    if incidence.interfaces.len() != expected || features.len() != incidence.exterior.len() {
277        return Err("crescent_incidence");
278    }
279    let mut design_faces = Vec::new();
280    for side in &incidence.exterior {
281        let tag = features
282            .remove(&(side.block.clone(), side.axis, side.upper))
283            .ok_or("crescent_unexpected_side")?;
284        design_faces.push((side.clone(), tag));
285    }
286    Ok(BracketCertificate {
287        incidence,
288        design_faces,
289        determinant_lower_bound_mm3: minimum.next_down(),
290        radial_error_upper_mm: 7e-16,
291    })
