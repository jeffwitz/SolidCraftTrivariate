22/// Shared exact coverage proof for constrained triangle meshes, independent
23/// of the triangulation algorithm and reusable by perforated profiles.
24pub(crate) fn validate_triangle_coverage(
25    triangles:&[Tri],vertices:&[Vertex],expected_count:usize,expected_area2:i64,boundary:&DirectedBoundary,
26)->Result<(),&'static str>{
27    if triangles.len()!=expected_count{return Err("polygon_section_coverage")}
28    let mut area_sum=0_i64;
29    let mut edge_uses=BTreeMap::<(usize,usize),Vec<(usize,usize)>>::new();
30    for tri in triangles{
31        let [a,b,c]=tri.ids;
32        let area=cross(vertices[a].xy_mm,vertices[b].xy_mm,vertices[c].xy_mm);
33        if area<=0{return Err("polygon_triangle_orientation")}
34        area_sum=area_sum.checked_add(area).ok_or("polygon_area_overflow")?;
35        for i in 0..3{let a=tri.ids[i];let b=tri.ids[(i+1)%3];edge_uses.entry(edge_key(a,b)).or_default().push((a,b));}
36    }
37    if area_sum!=expected_area2{return Err("polygon_section_coverage")}
38    validate_triangle_intersections(triangles,vertices)?;
39    for (edge,uses) in &edge_uses{
40        if let Some((directed,_))=boundary.get(edge){
41            if uses.len()!=1||uses[0]!=*directed{return Err("polygon_boundary_incidence")}
42        }else if uses.len()!=2||uses[0].0!=uses[1].1||uses[0].1!=uses[1].0{
43            return Err("polygon_internal_incidence")
44        }
45    }
46    if boundary.keys().any(|edge|!edge_uses.contains_key(edge)){return Err("polygon_section_coverage")}
47    Ok(())
48}

50pub(crate) fn cross(a:[i32;2],b:[i32;2],c:[i32;2])->i64{
51    i64::from(b[0]-a[0])*i64::from(c[1]-a[1])-i64::from(b[1]-a[1])*i64::from(c[0]-a[0])
52}
53pub(crate) fn on_segment(a:[i32;2],b:[i32;2],p:[i32;2])->bool{
54    cross(a,b,p)==0&&(0..2).all(|d|(a[d].min(b[d])..=a[d].max(b[d])).contains(&p[d]))
55}
56pub(crate) fn segments_contact(a:[i32;2],b:[i32;2],c:[i32;2],d:[i32;2])->bool{
57    let (x,y,z,w)=(cross(a,b,c),cross(a,b,d),cross(c,d,a),cross(c,d,b));
58    (x.signum()!=y.signum()&&z.signum()!=w.signum())||on_segment(a,b,c)||on_segment(a,b,d)||on_segment(c,d,a)||on_segment(c,d,b)
59}
60pub(crate) fn inside_triangle(p:[i32;2],t:[[i32;2];3])->bool{cross(t[0],t[1],p)>0&&cross(t[1],t[2],p)>0&&cross(t[2],t[0],p)>0}
61fn validate_contour(recipe:&PolygonPrism)->Result<(Vec<usize>,i64),&'static str>{
62    let n=recipe.vertices.len();
63    if !(3..=24).contains(&n)||!(1..=64).contains(&recipe.height_mm)||recipe.top_shift_mm.iter().any(|x|!(-32..=32).contains(x)){return Err("polygon_prism_range")}
64    let mut ids=BTreeMap::new();let mut positions=BTreeSet::new();
65    for (i,v) in recipe.vertices.iter().enumerate(){
66        if v.id.is_empty()||v.id.len()>24||!v.id.bytes().all(|b|b.is_ascii_alphanumeric()||b==b'_')||ids.insert(v.id.as_str(),i).is_some(){return Err("polygon_vertex_id")}
67        if v.xy_mm.iter().any(|x|!(-64..=64).contains(x))||!positions.insert(v.xy_mm){return Err("polygon_vertex_coordinates")}
68    }
69    let mut area2=0_i64;
70    for i in 0..n{
71        let a=recipe.vertices[i].xy_mm;let b=recipe.vertices[(i+1)%n].xy_mm;let c=recipe.vertices[(i+2)%n].xy_mm;
72        if cross(a,b,c)==0{return Err("polygon_collinear_adjacent_edges")}
73        area2+=i64::from(a[0])*i64::from(b[1])-i64::from(a[1])*i64::from(b[0]);
74        for j in i+1..n{
75            if j==i+1||(i==0&&j==n-1){continue}
76            let c=recipe.vertices[j].xy_mm;let d=recipe.vertices[(j+1)%n].xy_mm;
77            if segments_contact(a,b,c,d){return Err("polygon_nonadjacent_edge_contact")}
78        }
79    }
80    if area2==0{return Err("polygon_zero_area")}
81    let mut order=(0..n).collect::<Vec<_>>();if area2<0{order.reverse();area2 = -area2;}
82    let first=order.iter().enumerate().min_by_key(|(_,i)|&recipe.vertices[**i].id).unwrap().0;
83    order.rotate_left(first);
84    Ok((order,area2))
85}
86pub(crate) fn parse_triangle_id(id:&str,ids:&BTreeMap<&str,usize>)->Result<([usize;3],usize),&'static str>{
87    let fields=id.split('/').collect::<Vec<_>>();if fields.len()!=5||fields[0]!="tri"{return Err("polygon_block_id")}
88    let a=*ids.get(fields[1]).ok_or("polygon_block_vertex")?;let b=*ids.get(fields[2]).ok_or("polygon_block_vertex")?;let c=*ids.get(fields[3]).ok_or("polygon_block_vertex")?;let at=*ids.get(fields[4]).ok_or("polygon_block_vertex")?;
89    if a==b||b==c||c==a{return Err("polygon_block_id")}
90    Ok(([a,b,c],at))
91}
92pub(crate) fn fields_for_index<'a>(ids:&'a BTreeMap<&'a str,usize>,index:usize)->&'a str{ids.iter().find(|(_,i)|**i==index).unwrap().0}
93pub(crate) fn expected_controls(tri:[usize;3],at:usize,vertices:&[Vertex],height:i32,shift:[i32;2])->Option<Vec<[i64;4]>>{
94    let place=tri.iter().position(|x|*x==at)?;let v=tri[place];let prev=tri[(place+2)%3];let next=tri[(place+1)%3];
95    let a=vertices[v].xy_mm;let p=vertices[prev].xy_mm;let n=vertices[next].xy_mm;
96    let centroid=[2_i64*(i64::from(a[0])+i64::from(p[0])+i64::from(n[0])),2_i64*(i64::from(a[1])+i64::from(p[1])+i64::from(n[1]))];
97    let points=[
98        [6*i64::from(a[0]),6*i64::from(a[1])],
99        [3*i64::from(a[0]+p[0]),3*i64::from(a[1]+p[1])],
100        [3*i64::from(a[0]+n[0]),3*i64::from(a[1]+n[1])],
101        centroid,
102    ];
103    Some(points.into_iter().flat_map(|xy|{
104        let bottom=[xy[0],xy[1],0,6];
105        let top=[xy[0]+6*i64::from(shift[0]),xy[1]+6*i64::from(shift[1]),6*i64::from(height),6];
106        [bottom,top]
107    }).collect())
108}
109fn block_corner_xy(block:&BracketBlock,index:usize)->[i64;2]{[block.controls[2*index][0],block.controls[2*index][1]]}
110pub(crate) fn check_quad_jacobian(block:&BracketBlock)->Result<i128,&'static str>{
111    let p00=block_corner_xy(block,0);let p01=block_corner_xy(block,1);let p10=block_corner_xy(block,2);let p11=block_corner_xy(block,3);
112    let a=[p10[0]-p00[0],p10[1]-p00[1]];let b=[p01[0]-p00[0],p01[1]-p00[1]];
113    let c=[p11[0]-p10[0]-p01[0]+p00[0],p11[1]-p10[1]-p01[1]+p00[1]];
114    let det=|x:[i64;2],y:[i64;2]|i128::from(x[0])*i128::from(y[1])-i128::from(x[1])*i128::from(y[0]);
115    let values=[det(a,b),det(a,[b[0]+c[0],b[1]+c[1]]),det([a[0]+c[0],a[1]+c[1]],b),det([a[0]+c[0],a[1]+c[1]],[b[0]+c[0],b[1]+c[1]])];
116    let min=*values.iter().min().unwrap();if min<=0{return Err("polygon_nonpositive_jacobian")};Ok(min)
117}
118pub(crate) fn edge_key(a:usize,b:usize)->(usize,usize){if a<b{(a,b)}else{(b,a)}}
119pub(crate) fn validate_triangle_intersections(tris:&[Tri],vertices:&[Vertex])->Result<(),&'static str>{
120    for i in 0..tris.len(){for j in i+1..tris.len(){
121        let a=tris[i].ids;let b=tris[j].ids;
122        for x in 0..3{for y in 0..3{
123            let (ai,an)=(a[x],a[(x+1)%3]);let (bi,bn)=(b[y],b[(y+1)%3]);
124            if edge_key(ai,an)==edge_key(bi,bn){continue}
125            let (pa,pan,pb,pbn)=(vertices[ai].xy_mm,vertices[an].xy_mm,vertices[bi].xy_mm,vertices[bn].xy_mm);
126            let proper=cross(pa,pan,pb).signum()*cross(pa,pan,pbn).signum()<0
127                &&cross(pb,pbn,pa).signum()*cross(pb,pbn,pan).signum()<0;
128            if proper{return Err("polygon_triangle_overlap")}
129            for (point, endpoints) in [(pa,(pb,pbn)),(pan,(pb,pbn)),(pb,(pa,pan)),(pbn,(pa,pan))]{
130                if on_segment(endpoints.0,endpoints.1,point)&&point!=endpoints.0&&point!=endpoints.1{
131                    return Err("polygon_triangle_t_junction")
132                }
133            }
134        }}
135        for &v in &a{if inside_triangle(vertices[v].xy_mm,b.map(|x|vertices[x].xy_mm)){return Err("polygon_triangle_overlap")}}
136        for &v in &b{if inside_triangle(vertices[v].xy_mm,a.map(|x|vertices[x].xy_mm)){return Err("polygon_triangle_overlap")}}
137    }}
138    Ok(())
139}

141/// Independently reconstruct triangle coverage and audit all quadrilateral prism blocks.
142pub fn audit_polygon_prism(recipe:&PolygonPrism,blocks:&[BracketBlock])->Result<PolygonCertificate,&'static str>{
143    let (contour,area2)=validate_contour(recipe)?;
144    let idmap=recipe.vertices.iter().enumerate().map(|(i,v)|(v.id.as_str(),i)).collect::<BTreeMap<_,_>>();
145    if blocks.len()!=3*(recipe.vertices.len()-2){return Err("polygon_block_count")}
146    let mut unique=BTreeSet::new();let mut groups=BTreeMap::<[usize;3],BTreeMap<usize,&BracketBlock>>::new();let mut min_det=i128::MAX;
147    for block in blocks{
148        if !unique.insert(&block.id){return Err("polygon_duplicate_block")}
149        if block.degrees!=[1,1,1]||block.controls.len()!=8{return Err("polygon_altered_block")}
150        let (triangle,at)=parse_triangle_id(&block.id,&idmap)?;
151        if block.id!=format!("tri/{}/{}/{}/{}",fields_for_index(&idmap,triangle[0]),fields_for_index(&idmap,triangle[1]),fields_for_index(&idmap,triangle[2]),fields_for_index(&idmap,at))
152            ||triangle.iter().map(|i|fields_for_index(&idmap,*i)).min().unwrap()!=fields_for_index(&idmap,triangle[0])
153        {return Err("polygon_block_id")}
154        let a=recipe.vertices[triangle[0]].xy_mm;let b=recipe.vertices[triangle[1]].xy_mm;let c=recipe.vertices[triangle[2]].xy_mm;
155        let area_tri=cross(a,b,c);if area_tri<=0{return Err("polygon_triangle_orientation")}
156        let expected=expected_controls(triangle,at,&recipe.vertices,recipe.height_mm,recipe.top_shift_mm).ok_or("polygon_block_id")?;
157        if block.controls!=expected{return Err("polygon_altered_block")}
158        min_det=min_det.min(check_quad_jacobian(block)?);
159        if groups.entry(triangle).or_default().insert(at,block).is_some(){return Err("polygon_duplicate_corner_patch")}
160    }
161    let mut triangles=Vec::new();
162    for (ids,patches) in groups{
163        if patches.len()!=3||ids.iter().any(|id|!patches.contains_key(id)){return Err("polygon_missing_corner_patch")}
164        triangles.push(Tri{ids});
165    }
166    let mut boundary=DirectedBoundary::new();
167    for i in 0..contour.len(){
168        let a=contour[i];let b=contour[(i+1)%contour.len()];
169        let (lo,hi)=if recipe.vertices[a].id<recipe.vertices[b].id{(&recipe.vertices[a].id,&recipe.vertices[b].id)}else{(&recipe.vertices[b].id,&recipe.vertices[a].id)};
170        boundary.insert(edge_key(a,b),((a,b),format!("edge/{lo}/{hi}")));
171    }
172    validate_triangle_coverage(&triangles,&recipe.vertices,recipe.vertices.len()-2,area2,&boundary)?;
173    let incidence:Incidence=enumerate_faces(blocks)?;
174    let mut design_faces=Vec::with_capacity(incidence.exterior.len());
175    for side in &incidence.exterior{
176        let parts=side.block.split('/').collect::<Vec<_>>();if parts.len()!=5||parts[0]!="tri"{return Err("polygon_boundary_provenance")}
177        let tri=[*idmap.get(parts[1]).ok_or("polygon_boundary_provenance")?,*idmap.get(parts[2]).ok_or("polygon_boundary_provenance")?,*idmap.get(parts[3]).ok_or("polygon_boundary_provenance")?];
178        let at=*idmap.get(parts[4]).ok_or("polygon_boundary_provenance")?;let place=tri.iter().position(|x|*x==at).ok_or("polygon_boundary_provenance")?;
179        let prev=tri[(place+2)%3];let next=tri[(place+1)%3];
180        let tag=match (side.axis,side.upper){
181            (2,false)=>"start".to_owned(),(2,true)=>"end".to_owned(),
182            (0,false)=>boundary.get(&edge_key(prev,at)).map(|x|x.1.clone()).ok_or("polygon_internal_face_exterior")?,
183            (1,false)=>boundary.get(&edge_key(at,next)).map(|x|x.1.clone()).ok_or("polygon_internal_face_exterior")?,
184            _=>return Err("polygon_internal_face_exterior"),
185        };
186        design_faces.push((side.clone(),tag));
187    }
188    design_faces.sort_by(|a,b|a.0.cmp(&b.0));
189    let volume=area2.checked_mul(3).and_then(|x|x.checked_mul(i64::from(recipe.height_mm))).ok_or("polygon_volume_overflow")?;
190    let det_num=min_det.checked_mul(i128::from(recipe.height_mm)).ok_or("polygon_jacobian_overflow")?;
191    let det_bound=next_down(det_num as f64/36.);
192    let cert=BracketCertificate{incidence,determinant_lower_bound_mm3:det_bound,radial_error_upper_mm:0.,design_faces};
193    Ok(PolygonCertificate{certificate:cert,volume_sixths_mm3:volume,projected_area_twice_mm2:area2})
194}

196pub(crate) fn next_down(value:f64)->f64{
197    if value.is_nan()||value==f64::NEG_INFINITY{return value}
198    if value==0.0{return -f64::from_bits(1)}
199    let bits=value.to_bits();
200    f64::from_bits(if value>0.0{bits-1}else{bits+1})
