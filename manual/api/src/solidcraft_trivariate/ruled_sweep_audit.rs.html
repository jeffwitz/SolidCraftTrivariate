11/// Certificate for the exact ruled-sweep family.
12#[derive(Debug, Clone, PartialEq)]
13pub struct RuledSweepCertificate {
14    pub certificate: BracketCertificate,
15    pub ideal_volume_mm3: f64,
16    /// Exact minimum of the squared planar scale along the sweep segment.
17    pub minimum_squared_scale: [i64; 2],
18}

20fn gcd(mut a: i128, mut b: i128) -> i128 {
21    a = a.abs();
22    b = b.abs();
23    while b != 0 {
24        (a, b) = (b, a % b);
25    }
26    a
27}

29fn canonical_net(controls: &[[i64; 4]]) -> Result<Vec<[i128; 4]>, &'static str> {
30    let mut divisor = 0_i128;
31    for point in controls {
32        for coefficient in point {
33            divisor = gcd(divisor, i128::from(*coefficient));
34        }
35    }
36    if divisor <= 0 {
37        return Err("ruled_sweep_zero_net");
38    }
39    if controls.iter().any(|point| point[3] <= 0) {
40        return Err("ruled_sweep_nonpositive_weight");
41    }
42    Ok(controls
43        .iter()
44        .map(|point| point.map(|value| i128::from(value) / divisor))
45        .collect())
46}
47
48fn canonical_wide_net(controls: &[[i128; 4]]) -> Result<Vec<[i128; 4]>, &'static str> {
49    let mut divisor = 0_i128;
50    for point in controls {
51        for coefficient in point {
52            divisor = gcd(divisor, *coefficient);
53        }
54    }
55    if divisor <= 0 {
56        return Err("ruled_sweep_zero_expected_net");
57    }
58    Ok(controls
59        .iter()
60        .map(|point| point.map(|value| value / divisor))
61        .collect())
62}

64fn expected_net(block: &BracketBlock, recipe: SweepRecipe) -> Result<Vec<[i128; 4]>, &'static str> {
65    if !matches!(block.degrees, [1, 1, 1] | [2, 1, 1])
66        || block.controls.len() != (block.degrees[0] + 1) * 4
67    {
68        return Err("ruled_sweep_invalid_source_block");
69    }
70    let [a, b] = recipe.top_multiplier_eighths.map(i128::from);
71    let [tx, ty] = recipe.top_translation_eighths_mm.map(i128::from);
72    let mut expected = Vec::with_capacity(block.controls.len());
73    for pair in block.controls.as_chunks::<2>().0 {
74        let [x, y, _, weight] = pair[0].map(i128::from);
75        if weight <= 0 {
76            return Err("ruled_sweep_nonpositive_source_weight");
77        }
78        expected.push([8 * x, 8 * y, 0, 8 * weight]);
79        expected.push([
80            a * x - b * y + tx * weight,
81            b * x + a * y + ty * weight,
82            8 * i128::from(recipe.height_mm) * weight,
83            8 * weight,
84        ]);
85    }
86    canonical_wide_net(&expected)
87}

89fn minimum_squared_scale(recipe: SweepRecipe) -> [i64; 2] {
90    let [a, b] = recipe.top_multiplier_eighths.map(i64::from);
91    let dx = a - 8;
92    let denominator = dx * dx + b * b;
93    if denominator == 0 {
94        return [1, 1];
95    }
96    let stationary_numerator = 8 * (8 - a);
97    if stationary_numerator <= 0 {
98        [1, 1]
99    } else if stationary_numerator >= denominator {
100        [a * a + b * b, 64]
101    } else {
102        [b * b, denominator]
103    }
104}

106/// Audit exact control relations, incidence, selections, volume, and the
107/// continuous planar scale bound for a ruled sweep of a certified section.
108pub fn audit_ruled_sweep(
109    recipe: SweepRecipe,
110    blocks: &[BracketBlock],
111) -> Result<RuledSweepCertificate, &'static str> {
112    recipe.validate()?;
113    let family = recipe.section.family();
114    let source = section_blocks(family)?;
115    let section = audit_section(family, &source)?;
116    if blocks.len() != source.len() {
117        return Err("ruled_sweep_block_count");
118    }
119    let mut candidate_by_id = BTreeMap::new();
120    for block in blocks {
121        if candidate_by_id.insert(block.id.as_str(), block).is_some() {
122            return Err("ruled_sweep_duplicate_block_id");
123        }
124    }
125    for base in &source {
126        let candidate = candidate_by_id
127            .get(base.id.as_str())
128            .ok_or("ruled_sweep_missing_block")?;
129        if candidate.degrees != base.degrees {
130            return Err("ruled_sweep_degree_mismatch");
131        }
132        let expected = expected_net(base, recipe)?;
133        let actual = canonical_net(&candidate.controls)?;
134        if actual != expected {
135            return Err("ruled_sweep_control_relation_mismatch");
136        }
137    }
138
139    let incidence = enumerate_faces(blocks)?;
140    if incidence != section.incidence {
141        return Err("ruled_sweep_incidence_mismatch");
142    }
143    let minimum_squared_scale = minimum_squared_scale(recipe);
144    if minimum_squared_scale[0] <= 0 || minimum_squared_scale[1] <= 0 {
145        return Err("ruled_sweep_collapsed_section");
146    }
147    // Each binary64 operation in a reported bound is rounded outward.
148    let scale_min = (minimum_squared_scale[0] as f64 / minimum_squared_scale[1] as f64).next_down();
149    let [a, b] = recipe.top_multiplier_eighths.map(f64::from);
150    let scale_max = (1.0_f64).max((a * a + b * b) / 64.0).sqrt();
151    let height_ratio = f64::from(recipe.height_mm) / 8.0;
152    let determinant_lower_bound_mm3 =
153        ((section.determinant_lower_bound_mm3 * height_ratio).next_down() * scale_min).next_down();
154    if !determinant_lower_bound_mm3.is_finite() || determinant_lower_bound_mm3 <= 0.0 {
155        return Err("ruled_sweep_determinant_bound");
156    }
157    let ideal_volume_mm3 =
158        section.volume_mm3 * height_ratio * (1.0 + a / 8.0 + (a * a + b * b) / 64.0) / 3.0;
159    if !ideal_volume_mm3.is_finite() || ideal_volume_mm3 <= 0.0 {
160        return Err("ruled_sweep_volume_overflow");
161    }
162    let radial_error_upper_mm = if section.radial_error_upper_mm == 0. {
163        0.
164    } else {
165        (section.radial_error_upper_mm * scale_max.next_up()).next_up()
166    };
167    Ok(RuledSweepCertificate {
168        certificate: BracketCertificate {
169            incidence,
170            determinant_lower_bound_mm3,
171            radial_error_upper_mm,
172            design_faces: section.design_faces,
173        },
174        ideal_volume_mm3,
175        minimum_squared_scale,
176    })
177}
