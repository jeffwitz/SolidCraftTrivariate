15type Point = [i64; 2];
16type Quad = [Point; 4];
17type Side = (usize, usize, usize);
18#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
19enum AxisKind {
20    Value,
21    Slope,
22    Middle,
23}
24#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
25struct AxisDof {
26    kind: AxisKind,
27    knot: usize,
28}
29fn integer(i: usize) -> Q {
30    Q::new(i as i128, 1).unwrap()
31}
32fn number(q: Q) -> f64 {
33    q.numerator() as f64 / q.denominator() as f64
34}
35fn endpoints(c: &Quad, a: usize, b: usize) -> [Point; 2] {
36    if a == 0 {
37        [c[2 * b], c[2 * b + 1]]
38    } else {
39        [c[b], c[2 + b]]
40    }
41}
42fn axis_q(rows: &[Row], t: Q) -> Result<BTreeMap<usize, [Q; 2]>> {
43    let (b, d) = bernstein(rows.len() - 1, t)?;
44    let mut out = BTreeMap::new();
45    for (k, row) in rows.iter().enumerate() {
46        for (&i, &c) in row {
47            let v = out.entry(i).or_insert([Q::zero(); 2]);
48            v[0] = v[0].checked_add(b[k].checked_mul(c)?)?;
49            v[1] = v[1].checked_add(d[k].checked_mul(c)?)?;
50        }
51    }
52    Ok(out)
53}
54/// D(u,v) of the parent chart, with integer vertex differences.
55fn tangent(c: &Quad, a: usize, t: Q) -> Result<[Q; 2]> {
56    let lo = endpoints(c, 1 - a, 0);
57    let hi = endpoints(c, 1 - a, 1);
58    let mut v = [Q::zero(); 2];
59    for j in 0..2 {
60        let l = i128::from(lo[1][j]) - i128::from(lo[0][j]);
61        let h = i128::from(hi[1][j]) - i128::from(hi[0][j]);
62        v[j] = Q::new(l, 1)?.checked_add(t.checked_mul(Q::new(h - l, 1)?)?)?;
63    }
64    Ok(v)
65}
66struct Basis {
67    n: usize,
68    dim: usize,
69    charts: Vec<Quad>,
70    ids: Vec<Vec<(usize, Q)>>,
71    spans: Vec<Vec<Row>>,
72}
73impl Basis {
74    fn jet(
75        &self,
76        side: Side,
77        k: usize,
78        t: Q,
79        x_component: bool,
80        sign: Q,
81        row: &mut Row,
82    ) -> Result<()> {
83        let (chart, a, b) = side;
84        let mut cells = [k; 2];
85        cells[a] = b * (self.n - 1);
86        let mut pt = [t; 2];
87        pt[a] = integer(b);
88        let parent = |j: usize| {
89            integer(cells[j])
90                .checked_add(pt[j])?
91                .checked_div(integer(self.n))
92        };
93        let u = tangent(&self.charts[chart], 0, parent(1)?)?;
94        let v = tangent(&self.charts[chart], 1, parent(0)?)?;
95        let det = u[0]
96            .checked_mul(v[1])?
97            .checked_sub(v[0].checked_mul(u[1])?)?;
98        let scale = sign.checked_mul(integer(self.n))?.checked_div(det)?;
99        let minus = |x: Q| Q::zero().checked_sub(x);
100        let (du, dv) = if x_component {
101            (v[1], minus(u[1])?)
102        } else {
103            (minus(v[0])?, u[0])
104        };
105        let bx = axis_q(&self.spans[cells[0]], pt[0])?;
106        let by = axis_q(&self.spans[cells[1]], pt[1])?;
107        for (i, [x, dx]) in bx {
108            for (&j, &[y, dy]) in &by {
109                let value = du
110                    .checked_mul(dx)?
111                    .checked_mul(y)?
112                    .checked_add(dv.checked_mul(x)?.checked_mul(dy)?)?;
113                add(
114                    row,
115                    self.ids[chart][i * self.dim + j].0,
116                    scale
117                        .checked_mul(value)?
118                        .checked_mul(self.ids[chart][i * self.dim + j].1)?,
119                )?;
120            }
121        }
122        Ok(())
123    }
124}
125#[derive(PartialEq, Eq, PartialOrd, Ord)]
126enum Node {
127    Vertex(Point),
128    Edge([Point; 2], AxisDof),
129    Interior(usize, usize, usize),
130}
131fn axis_dofs(p: usize, n: usize) -> Vec<AxisDof> {
132    let mut out = Vec::new();
133    for knot in 0..=n {
134        out.push(AxisDof {
135            kind: AxisKind::Value,
136            knot,
137        });
138        out.push(AxisDof {
139            kind: AxisKind::Slope,
140            knot,
141        });
142        if p == 4 && knot < n {
143            out.push(AxisDof {
144                kind: AxisKind::Middle,
145                knot,
146            });
147        }
148    }
149    out
150}
151/// Exact local Bernstein controls from endpoint Hermite data. Quartic spans
152/// retain an independent middle control; adjacent spans share only knot jets.
153fn hermite_spans(p: usize, n: usize) -> Vec<Vec<Row>> {
154    let dofs = axis_dofs(p, n);
155    let find = |kind, knot| {
156        dofs.iter()
157            .position(|d| d.kind == kind && d.knot == knot)
158            .unwrap()
159    };
160    (0..n)
161        .map(|k| {
162            let mut rows = vec![Row::new(); p + 1];
163            rows[0].insert(find(AxisKind::Value, k), Q::one());
164            rows[1].insert(find(AxisKind::Value, k), Q::one());
165            rows[1].insert(find(AxisKind::Slope, k), Q::new(1, p as i128).unwrap());
166            if p == 3 {
167                rows[2].insert(find(AxisKind::Value, k + 1), Q::one());
168                rows[2].insert(find(AxisKind::Slope, k + 1), Q::new(-1, p as i128).unwrap());
169            } else {
170                rows[2].insert(find(AxisKind::Middle, k), Q::one());
171                rows[3].insert(find(AxisKind::Value, k + 1), Q::one());
172                rows[3].insert(find(AxisKind::Slope, k + 1), Q::new(-1, p as i128).unwrap());
173            }
174            rows[p].insert(find(AxisKind::Value, k + 1), Q::one());
175            rows
176        })
177        .collect()
178}
179fn edge_node(c: &Quad, axis: usize, endpoint: usize, tangential: AxisDof, n: usize) -> (Node, Q) {
180    let [start, end] = endpoints(c, axis, endpoint);
181    if start <= end {
182        (Node::Edge([start, end], tangential), Q::one())
183    } else {
184        // Canonical edge orientation mirrors knot and span indices; only an
185        // oriented tangential slope changes sign under that reversal.
186        let mapped = AxisDof {
187            kind: tangential.kind,
188            knot: match tangential.kind {
189                AxisKind::Value | AxisKind::Slope => n - tangential.knot,
190                AxisKind::Middle => n - 1 - tangential.knot,
191            },
192        };
193        let sign = if tangential.kind == AxisKind::Slope {
194            Q::new(-1, 1).unwrap()
195        } else {
196            Q::one()
197        };
198        (Node::Edge([end, start], mapped), sign)
199    }
200}
201fn node(c: &Quad, chart: usize, i: usize, j: usize, n: usize, dofs: &[AxisDof]) -> (Node, Q) {
202    let (u, v) = (dofs[i], dofs[j]);
203    let ub = (u.kind == AxisKind::Value && (u.knot == 0 || u.knot == n)).then_some(u.knot / n);
204    let vb = (v.kind == AxisKind::Value && (v.knot == 0 || v.knot == n)).then_some(v.knot / n);
205    match (ub, vb) {
206        (Some(a), Some(b)) => (Node::Vertex(c[2 * a + b]), Q::one()),
207        (Some(b), None) => edge_node(c, 0, b, v, n),
208        (None, Some(b)) => edge_node(c, 1, b, u, n),
209        (None, None) => (Node::Interior(chart, i, j), Q::one()),
210    }
211}
212struct Leaf {
213    c: Quad,
214    z: [f64; 2],
215    ij: [usize; 2],
216    dofs: Vec<usize>,
217    rows: Vec<Vec<(usize, f64)>>,
218}
219impl Leaf {
220    fn new(b: &Basis, col: usize, z: [f64; 2], ij: [usize; 2], lift: &[Row]) -> Result<Self> {
221        let mut rows = Vec::new();
222        let mut active = BTreeSet::new();
223        for a in &b.spans[ij[0]] {
224            for d in &b.spans[ij[1]] {
225                let mut row = Row::new();
226                for (&i, &ca) in a {
227                    for (&j, &cb) in d {
228                        let w = ca.checked_mul(cb)?;
229                        let (local_id, orientation) = b.ids[col][i * b.dim + j];
230                        for (&id, &c) in &lift[local_id] {
231                            add(&mut row, id, w.checked_mul(c)?.checked_mul(orientation)?)?
232                        }
233                    }
234                }
235                active.extend(row.keys().copied());
236                rows.push(row);
237            }
238        }
239        let dofs = Vec::from_iter(active);
240        let rows = rows
241            .into_iter()
242            .map(|r| {
243                r.into_iter()
244                    .map(|(id, v)| (dofs.binary_search(&id).unwrap(), number(v)))
245                    .collect()
246            })
247            .collect();
248        Ok(Self {
249            c: b.charts[col],
250            z,
251            ij,
252            dofs,
253            rows,
254        })
255    }
256    fn sample(
257        &self,
258        n: usize,
259        p: usize,
260        t: [f64; 3],
261    ) -> Result<(ElementGeometrySample, ElementBasisSample)> {
262        let [u, v] = std::array::from_fn(|k| (self.ij[k] as f64 + t[k]) / n as f64);
263        let height = self.z[1] - self.z[0];
264        let z = self.z[0] + height * t[2];
265        let mut pos = [0., 0., z];
266        let mut jac = [[0.; 3]; 3];
267        jac[2][2] = height;
268        for k in 0..2 {
269            let [a, b, c, d] = self.c.map(|xy| xy[k] as f64);
270            pos[k] = (1. - u) * ((1. - v) * a + v * b) + u * ((1. - v) * c + v * d);
271            jac[0][k] = ((1. - v) * (c - a) + v * (d - b)) / n as f64;
272            jac[1][k] = ((1. - u) * (b - a) + u * (d - c)) / n as f64;
273        }
274        let det = jac[0][0] * jac[1][1] - jac[0][1] * jac[1][0];
275        let axis = |x: f64| {
276            // Recurrence at both endpoints, including quartic end functions.
277            let mut b = [0.; 5];
278            b[0] = 1.;
279            let mut d = [0.; 5];
280            for k in 1..=p {
281                if k == p {
282                    for i in 0..k {
283                        d[i] -= k as f64 * b[i];
284                        d[i + 1] += k as f64 * b[i];
285                    }
286                }
287                for i in (1..=k).rev() {
288                    b[i] = (1. - x) * b[i] + x * b[i - 1];
289                }
290                b[0] *= 1. - x;
291            }
292            (b, d)
293        };
294        let (bx, dx) = axis(t[0]);
295        let (by, dy) = axis(t[1]);
296        let mut values = vec![0.; self.dofs.len()];
297        let mut gradients = vec![[0.; 3]; self.dofs.len()];
298        for i in 0..=p {
299            for j in 0..=p {
300                let val = bx[i] * by[j] / 20.;
301                let du = dx[i] * by[j];
302                let dv = bx[i] * dy[j];
303                let gx = z / 20. * (jac[1][1] * du - jac[0][1] * dv) / det;
304                let gy = z / 20. * (jac[0][0] * dv - jac[1][0] * du) / det;
305                for &(id, c) in &self.rows[i * (p + 1) + j] {
306                    values[id] += c * val * z;
307                    gradients[id][0] += c * gx;
308                    gradients[id][1] += c * gy;
309                    gradients[id][2] += c * val;
310                }
311            }
312        }
313        Ok((
314            ElementGeometrySample {
315                position: pos,
316                jacobian_columns: jac,
317                determinant: det * height,
318            },
319            ElementBasisSample {
320                dofs: self.dofs.clone(),
321                values,
322                physical_gradients: gradients,
323            },
324        ))
325    }
326}
327/// Keep the exact kernel in its conditioned Hermite interpolation coordinates.
328fn jet_lift(count: usize, equations: &[Row]) -> Result<(Vec<Row>, Vec<usize>)> {
329    conditioned_kernel(count, equations)
330}

332/// Run the pre-registered conforming vector benchmark from its original sketch.
333pub fn solve(p: usize, n: usize, c1: bool, order: usize) -> Result<Json> {
334    if ![3, 4].contains(&p) || ![1, 2, 4, 8].contains(&n) || ![8, 10].contains(&order) {
335        return Err("unsupported pocket study parameters");
336    }
337    let recipe = crate::json::parse(include_str!(
338        "../tests/references/blind_bilinear_pocket.recipe.json"
339    ))?;
340    let model = crate::interchange::build(&recipe)?;
341    let blocks = decode_blocks(&model)?;
342    let mut charts = Vec::new();
343    let mut columns = Vec::new();
344    for b in &blocks {
345        if b.degrees != [1; 3] || b.controls.iter().any(|h| h[3] != 1) {
346            return Err("expected trilinear unit-weight scaffold");
347        }
348        let c = std::array::from_fn(|k| [b.controls[2 * k][0], b.controls[2 * k][1]]);
349        let col = charts.iter().position(|q| q == &c).unwrap_or_else(|| {
350            charts.push(c);
351            charts.len() - 1
352        });
353        columns.push(col);
354    }
355    if charts.len() != 5 || blocks.len() != 9 {
356        return Err("unexpected scaffold topology");
357    }
358    let dofs = axis_dofs(p, n);
359    let dim = dofs.len();
360    let spans = hermite_spans(p, n);
361    let mut nodes = BTreeMap::new();
362    let mut ids = Vec::new();
363    let mut edges: BTreeMap<[Point; 2], Vec<Side>> = BTreeMap::new();
364    for (k, c) in charts.iter().enumerate() {
365        let mut local = Vec::new();
366        for i in 0..dim {
367            for j in 0..dim {
368                let next = nodes.len();
369                let (key, sign) = node(c, k, i, j, n, &dofs);
370                let id = *nodes.entry(key).or_insert(next);
371                local.push((id, sign));
372            }
373        }
374        ids.push(local);
375        for a in 0..2 {
376            for b in 0..2 {
377                let mut pair = endpoints(c, a, b);
378                pair.sort();
379                edges.entry(pair).or_default().push((k, a, b));
380            }
381        }
382    }
383    let count = nodes.len();
384    let basis = Basis {
385        n,
386        dim,
387        charts,
388        ids,
389        spans,
390    };
391    let mut equations = Vec::new();
392    if c1 {
393        for ([start, end], sides) in edges {
394            if sides.len() == 1 {
395                continue;
396            }
397            let [a, b] = sides[..] else {
398                return Err("nonmanifold chart edge");
399            };
400            let reverse =
401                endpoints(&basis.charts[a.0], a.1, a.2) != endpoints(&basis.charts[b.0], b.1, b.2);
402            // A physical coordinate direction transverse to the common tangent.
403            let x_component = start[1] != end[1];
404            for k in 0..n {
405                for r in 0..=p + 1 {
406                    let t = Q::new(r as i128, (p + 1) as i128)?;
407                    let mut row = Row::new();
408                    basis.jet(a, k, t, x_component, Q::one(), &mut row)?;
409                    let (kb, tb) = if reverse {
410                        (n - 1 - k, Q::one().checked_sub(t)?)
411                    } else {
412                        (k, t)
413                    };
414                    basis.jet(b, kb, tb, x_component, Q::new(-1, 1)?, &mut row)?;
415                    equations.push(row);
416                }
417            }
418        }
419    }
420    let (lift, free) = jet_lift(count, &equations)?;
421    let mut leaves = Vec::new();
422    let mut boundary = Vec::new();
423    let exterior = model.field("topology")?.field("exterior")?.items()?;
424    for (k, b) in blocks.iter().enumerate() {
425        let mut sides = Vec::new();
426        for entry in exterior {
427            let e = entry.items()?;
428            if e[0].text()? == b.id {
429                sides.push(decode_side_name(e[1].text()?)?)
430            }
431        }
432        for i in 0..n {
433            for j in 0..n {
434                let ij = [i, j];
435                for &(a, hi) in &sides {
436                    if a == 2 || ij[a] == if hi { n - 1 } else { 0 } {
437                        boundary.push((leaves.len(), a, hi));
438                    }
439                }
440                leaves.push(Leaf::new(
441                    &basis,
442                    columns[k],
443                    [b.controls[0][2] as f64, b.controls[1][2] as f64],
444                    ij,
445                    &lift,
446                )?);
447            }
448        }
449    }
450    let essential = resolve(free.len(), &[], &[])?;
451    let job = object([
452        ("schema", Json::String("section_quintic_mms_v1".into())),
453        ("quadrature", Json::Number(order.to_string())),
454        ("x_origin_mm", Json::Number("-10".into())),
455        ("x_scale_mm", Json::Number("70".into())),
456    ]);
457    solve_sampled(
458        &IntegrationDomain {
459            scalar_dofs: free.len(),
460            cell_count: leaves.len(),
461            z_bounds: [0., 20.],
462            boundary: &boundary,
463            essential: &essential,
464        },
465        &job,
466        |i, t| leaves[i].sample(n, p, t),
467    )
