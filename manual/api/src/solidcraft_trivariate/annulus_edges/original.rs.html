1//! Three edge states on the four charts of full_annulus_v1.
2//! The fillet loses rank at the four circular rails on the end faces.
3use crate::{bracket::BracketBlock, sections::section_blocks};

5use super::EdgeProfile as Profile;

7fn net(source: &BracketBlock, state: Profile) -> Vec<[i64; 4]> {
8    assert_eq!(source.degrees, [2, 1, 1]);
9    let mut controls = Vec::with_capacity(63);
10    for angle in 0..3 {
11        let [x, y, _, w] = source.controls[4 * angle]; // outer angular row, nominal radius 4
12        for radial in 0..3_i64 {
13            for layer in 0..7 {
14                let stage = layer.min(6 - layer);
15                let weight = [1, 1, 2, 2][stage];
16                // Homogeneous setback in eighths of a millimetre.
17                let setback = match state {
18                    Profile::Sharp => 0,
19                    Profile::Chamfer => [2, 1, 0, 0][stage],
20                    Profile::Fillet => [2, 0, 0, 0][stage],
21                };
22                let radius = 8 * (4 - radial) * weight + (radial - 1) * setback;
23                let mut height = if state == Profile::Fillet {
24                    // The interior radial pole lifts the cap parametrization away
25                    // from the edge; its coefficient vanishes on both walls.
26                    [0, i64::from(radial == 1), 4, 64][stage]
27                } else {
28                    [0, 1, 4, 64][stage]
29                };
30                if layer > 3 {
31                    height = 64 * weight - height;
32                }
33                controls.push([x * radius, y * radius, 4 * w * height, 32 * w * weight]);
34            }
35        }
36    }
37    controls
38}

40use super::{numbers, Json};
41pub(super) fn document(state: Profile) -> Result<Json, &'static str> {
42    let blocks = section_blocks("full_annulus_v1")?;
43    Ok(super::original_document(
44        [2, 2, 2],
45        [
46            numbers([0, 0, 0, 1, 1, 1]),
47            numbers([0, 0, 0, 1, 1, 1]),
48            super::axial_knots(2),
49        ],
50        blocks.iter().map(|b| (&b.id, net(b, state))),
51    ))
52}
53#[cfg(test)]
54mod tests {
55    use super::*;
56    #[test]
57    fn all_three_states_preserve_the_same_weights_and_all_four_exact_interfaces() {
58        let sectors = section_blocks("full_annulus_v1").unwrap();
59        assert_eq!(sectors.len(), 4);
60        for state in [Profile::Sharp, Profile::Chamfer, Profile::Fillet] {
61            let nets: Vec<_> = sectors.iter().map(|s| net(s, state)).collect();
62            for (i, poles) in nets.iter().enumerate() {
63                assert_eq!(poles.len(), 63);
64                assert_eq!(poles[42..], nets[(i + 1) % 4][..21]);
65                assert!(poles
66                    .iter()
67                    .zip(net(&sectors[i], Profile::Sharp))
68                    .all(|(a, b)| a[3] == b[3]));
69            }
70        }
71    }
