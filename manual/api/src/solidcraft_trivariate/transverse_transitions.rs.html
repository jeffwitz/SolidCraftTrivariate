11trait CheckedRatioExt {
12    fn add(self, other: Self) -> Option<Self>
13    where
14        Self: Sized;
15    fn sub(self, other: Self) -> Option<Self>
16    where
17        Self: Sized;
18    fn mul(self, other: Self) -> Option<Self>
19    where
20        Self: Sized;
21    fn div(self, other: Self) -> Option<Self>
22    where
23        Self: Sized;
24}

26impl CheckedRatioExt for Ratio {
27    fn add(self, other: Self) -> Option<Self> {
28        self.checked_add(other).ok()
29    }
30    fn sub(self, other: Self) -> Option<Self> {
31        self.checked_sub(other).ok()
32    }
33    fn mul(self, other: Self) -> Option<Self> {
34        self.checked_mul(other).ok()
35    }
36    fn div(self, other: Self) -> Option<Self> {
37        self.checked_div(other).ok()
38    }
39}

41fn qnew(numerator: i128, denominator: i128) -> Option<Q> {
42    Q::new(numerator, denominator).ok()
43}

45fn qtext(value: Q) -> String {
46    if value.denominator() == 1 {
47        value.numerator().to_string()
48    } else {
49        format!("{}/{}", value.numerator(), value.denominator())
50    }
51}

53type P = Vec<Q>;
54fn trim(mut p: P) -> P {
55    while p.last().is_some_and(|value| value.numerator() == 0) {
56        p.pop();
57    }
58    p
59}
60fn padd(a: &P, b: &P, scale: i128) -> Option<P> {
61    let mut c = vec![qnew(0, 1)?; a.len().max(b.len())];
62    for (i, x) in a.iter().enumerate() {
63        c[i] = c[i].add(*x)?
64    }
65    for (i, x) in b.iter().enumerate() {
66        c[i] = c[i].add(x.mul(qnew(scale, 1)?)?)?
67    }
68    Some(trim(c))
69}
70fn pmul(a: &P, b: &P) -> Option<P> {
71    if a.is_empty() || b.is_empty() {
72        return Some(vec![]);
73    }
74    let mut c = vec![qnew(0, 1)?; a.len() + b.len() - 1];
75    for (i, x) in a.iter().enumerate() {
76        for (j, y) in b.iter().enumerate() {
77            c[i + j] = c[i + j].add(x.mul(*y)?)?
78        }
79    }
80    Some(trim(c))
81}
82fn deriv(a: &P) -> Option<P> {
83    let mut out = Vec::new();
84    for (i, x) in a.iter().enumerate().skip(1) {
85        out.push(x.mul(qnew(i as i128, 1)?)?)
86    }
87    Some(trim(out))
88}
89fn cross(a: &[P; 2], b: &[P; 2]) -> Option<P> {
90    padd(&pmul(&a[0], &b[1])?, &pmul(&a[1], &b[0])?, -1)
91}
92fn bernstein(row: &[Q]) -> Option<P> {
93    if row.is_empty() {
94        return Some(vec![]);
95    }
96    let d = row.len() - 1;
97    let mut out = vec![qnew(0, 1)?; d + 1];
98    for (i, v) in row.iter().enumerate() {
99        let choose = binom(d, i)?;
100        for j in 0..=d - i {
101            let term = v.mul(qnew(
102                choose
103                    .checked_mul(binom(d - i, j)?)?
104                    .checked_mul(if j % 2 == 0 { 1 } else { -1 })?,
105                1,
106            )?)?;
107            out[i + j] = out[i + j].add(term)?;
108        }
109    }
110    Some(trim(out))
111}
112fn binom(n: usize, k: usize) -> Option<i128> {
113    let k = k.min(n - k);
114    let mut r = 1i128;
115    for i in 0..k {
116        r = r
117            .checked_mul((n - i) as i128)?
118            .checked_div((i + 1) as i128)?;
119    }
120    Some(r)
121}
122fn quotient_affine(a: &P, b: &P) -> Option<Option<(Q, Q)>> {
123    if b.is_empty() || b[0].numerator() == 0 {
124        return None;
125    }
126    let c0 = if a.is_empty() {
127        qnew(0, 1)?
128    } else {
129        a[0].div(b[0])?
130    };
131    let a1 = if a.len() > 1 { a[1] } else { qnew(0, 1)? };
132    let b1 = if b.len() > 1 { b[1] } else { qnew(0, 1)? };
133    let c1 = a1.sub(c0.mul(b1)?)?.div(b[0])?;
134    let product = pmul(&vec![c0, c1], b)?;
135    let residual = padd(a, &product, -1)?;
136    Some(if residual.is_empty() {
137        Some((c0, c1))
138    } else {
139        None
140    })
141}

143fn parse_face(v: &Json) -> Result<(&str, &str), &'static str> {
144    let a = v.items()?;
145    if a.len() != 2 {
146        return Err("expected block id and side");
147    };
148    Ok((a[0].text()?, a[1].text()?))
149}
150fn block_jet(block: &Json, side: &str, reverse: bool) -> Result<([P; 2], [P; 2], P), &'static str> {
151    let deg = block.field("degrees")?.items()?;
152    if deg.len() != 3 {
153        return Err("expected three degrees");
154    };
155    let du = deg[0].number::<usize>()?;
156    let dv = deg[1].number::<usize>()?;
157    let dw = deg[2].number::<usize>()?;
158    if du == 0 || dv == 0 || du > 64 || dv > 64 || dw != 1 {
159        return Err("unsupported transition degree");
160    };
161    let controls = block.field("controls")?.items()?;
162    if controls.len() != 2 * (du + 1) * (dv + 1)
163        || controls
164            .iter()
165            .any(|point| point.items().map_or(true, |values| values.len() != 4))
166    {
167        return Err("invalid transverse control net");
168    }
169    let common_weight = controls
170        .first()
171        .and_then(|c| c.items().ok())
172        .and_then(|c| c.get(3))
173        .and_then(|w| w.number::<i128>().ok())
174        .filter(|w| *w > 0)
175        .filter(|w| {
176            controls.iter().all(|c| {
177                c.items()
178                    .ok()
179                    .and_then(|v| v.get(3))
180                    .and_then(|w| w.number::<i128>().ok())
181                    == Some(*w)
182            })
183        });
184    let mut grid = vec![vec![Vec::<Q>::new(); dv + 1]; du + 1];
185    for i in 0..=du {
186        for j in 0..=dv {
187            let v = &controls[2 * (i * (dv + 1) + j)];
188            let mut c = Vec::new();
189            for x in v.items()? {
190                let value = x.number::<i128>()?;
191                c.push(if let Some(w) = common_weight {
192                    qnew(value, w).ok_or("rational overflow")?
193                } else {
194                    qnew(value, 1).ok_or("rational overflow")?
195                })
196            }
197            grid[i][j] = c;
198        }
199    }
200    let axis = match side.as_bytes() {
201        [b'u', b'0'] => 0,
202        [b'u', b'1'] => 0,
203        [b'v', b'0'] => 1,
204        [b'v', b'1'] => 1,
205        _ => return Err("unsupported side"),
206    };
207    let end = side.as_bytes()[1] == b'1';
208    let d = if axis == 0 { du } else { dv };
209    let e = if end { d } else { 0 };
210    let n = if end {
211        d.saturating_sub(1)
212    } else {
213        (d + 1).min(d)
214    };
215    let len = if axis == 0 { dv + 1 } else { du + 1 };
216    let mut h: [Vec<Q>; 4] = std::array::from_fn(|_| Vec::new());
217    let mut near: [Vec<Q>; 4] = std::array::from_fn(|_| Vec::new());
218    for t in 0..len {
219        let (i, j, ni, nj) = if axis == 0 {
220            (e, t, n, t)
221        } else {
222            (t, e, t, n)
223        };
224        for c in 0..4 {
225            h[c].push(grid[i][j][c]);
226            near[c].push(grid[ni][nj][c]);
227        }
228    }
229    // Parameter reversal changes every polynomial's argument. Bernstein reversal
230    // is exact by reversing the control row before conversion.
231    if reverse {
232        for c in 0..4 {
233            h[c].reverse();
234            near[c].reverse();
235        }
236    }
237    let hp: [P; 4] = std::array::from_fn(|c| bernstein(&h[c]).unwrap_or_default());
238    let np: [P; 4] = std::array::from_fn(|c| bernstein(&near[c]).unwrap_or_default());
239    let weight = hp[3].clone();
240    let wt = deriv(&weight).ok_or("rational overflow")?;
241    let weight_normal = padd(&np[3], &hp[3], -1)
242        .ok_or("rational overflow")?
243        .into_iter()
244        .map(|x| x.mul(qnew(d as i128, 1).unwrap()).unwrap())
245        .collect::<P>();
246    let mut derivs: [[P; 2]; 2] = std::array::from_fn(|_| std::array::from_fn(|_| Vec::new()));
247    for c in 0..2 {
248        let tangent = deriv(&hp[c]).ok_or("rational overflow")?;
249        let normal = padd(&np[c], &hp[c], -1).ok_or("rational overflow")?;
250        let normal = normal
251            .into_iter()
252            .map(|x| x.mul(qnew(d as i128, 1).unwrap()).unwrap())
253            .collect::<P>();
254        let dt = padd(
255            &pmul(&tangent, &weight).ok_or("rational overflow")?,
256            &pmul(&hp[c], &wt).ok_or("rational overflow")?,
257            -1,
258        )
259        .ok_or("rational overflow")?;
260        let dn = padd(
261            &pmul(&normal, &weight).ok_or("rational overflow")?,
262            &pmul(&hp[c], &weight_normal).ok_or("rational overflow")?,
263            -1,
264        )
265        .ok_or("rational overflow")?;
266        derivs[0][c] = dt;
267        derivs[1][c] = dn;
268    }
269    let weight2 = pmul(&weight, &weight).ok_or("rational overflow")?;
270    Ok((derivs[0].clone(), derivs[1].clone(), weight2))
271}

277fn directional(a: &Json, sa: &str, b: &Json, sb: &str, reverse: bool) -> DirectionalResult {
278    let (ta, na, wa) = block_jet(a, sa, false)?;
279    let (_, nb, wb) = block_jet(b, sb, reverse)?;
280    let ca = cross(&na, &ta).ok_or("rational overflow")?;
281    let cb = cross(&nb, &ta).ok_or("rational overflow")?;
282    let den = pmul(&ca, &wb).ok_or("rational overflow")?;
283    let nalpha = pmul(&cb, &wa).ok_or("rational overflow")?;
284    let beta_num =
285        pmul(&cross(&na, &nb).ok_or("rational overflow")?, &wa).ok_or("rational overflow")?;
286    let Some(alpha) = quotient_affine(&nalpha, &den).ok_or("rational overflow")? else {
287        return Ok(("outside_affine_transition_test", None, None));
288    };
289    let Some(beta) = quotient_affine(&beta_num, &den).ok_or("rational overflow")? else {
290        return Ok(("outside_affine_transition_test", None, None));
291    };
292    if alpha
293        .0
294        .mul(alpha.0.add(alpha.1).ok_or("rational overflow")?)
295        .ok_or("rational overflow")?
296        .numerator()
297        <= 0
298    {
299        return Ok(("transition_singular_on_face", None, None));
300    }
301    let kind = if alpha.1.numerator() != 0 || beta.1.numerator() != 0 {
302        "affine_transition"
303    } else {
304        "constant_transition"
305    };
306    Ok((kind, Some(alpha), Some(beta)))
307}
308fn coeffs(alpha: Option<(Q, Q)>, beta: Option<(Q, Q)>) -> Json {
309    let pair = |p: Option<(Q, Q)>| {
310        p.map_or(Json::Array(vec![]), |(a, b)| {
311            Json::Array(vec![Json::String(qtext(a)), Json::String(qtext(b))])
312        })
313    };
314    if alpha.is_none() || beta.is_none() {
315        object([])
316    } else {
317        object([("alpha", pair(alpha)), ("beta", pair(beta))])
318    }
319}

321/// Classify every interface in one model, returning the same per-link schema as
322/// `scripts/census_transverse_transitions.py`.
323pub fn classify(model: &Json) -> Result<Json, &'static str> {
324    let blocks = model.field("blocks")?.items()?;
325    let mut by_id = std::collections::BTreeMap::new();
326    for b in blocks {
327        by_id.insert(b.field("id")?.text()?, b);
328    }
329    let mut records = Vec::new();
330    for link in model.field("topology")?.field("interfaces")?.items()? {
331        let first = link.field("first")?;
332        let second = link.field("second")?;
333        let (a, sa) = parse_face(first)?;
334        let (b, sb) = parse_face(second)?;
335        let orient = link.field("orientation")?;
336        let swap = orient.field("swap")?.bool()?;
337        let ff = orient.field("flip_first")?.bool()?;
338        let fs = orient.field("flip_second")?.bool()?;
339        let side_axial = sa.starts_with('w') || sb.starts_with('w');
340        let aa = by_id.get(a).ok_or("unknown block")?;
341        let bb = by_id.get(b).ok_or("unknown block")?;
342        let mut coeff = object([]);
343        let mut reverse_coeff = object([]);
344        let mut reversed = None;
345        let status;
346        if side_axial {
347            status = "axial_or_nonlateral"
348        } else {
349            let shift = |x: &Json| -> Result<Option<[Q; 3]>, &'static str> {
350                let dg = x.field("degrees")?.items()?;
351                if dg[2].number::<usize>()? != 1 {
352                    return Ok(None);
353                }
354                let c = x.field("controls")?.items()?;
355                let mut got = None;
356                let (pairs, _) = c.as_chunks::<2>();
357                for pair in pairs {
358                    let u = pair[0].items()?;
359                    let v = pair[1].items()?;
360                    let mut s = [Q::zero(); 3];
361                    for i in 0..3 {
362                        let wu = u[3].number::<i128>()?;
363                        let wv = v[3].number::<i128>()?;
364                        if wu != wv || wu <= 0 {
365                            return Ok(None);
366                        }
367                        s[i] = qnew(
368                            v[i].number::<i128>()?
369                                .checked_sub(u[i].number::<i128>()?)
370                                .ok_or("rational overflow")?,
371                            wu,
372                        )
373                        .ok_or("rational overflow")?;
374                    }
375                    if got.is_some_and(|old| old != s) {
376                        return Ok(None);
377                    }
378                    got = Some(s);
379                }
380                Ok(got)
381            };
382            let x = shift(aa)?;
383            let y = shift(bb)?;
384            if swap || fs || x.is_none() || x != y {
385                status = "unsupported_transition_frame"
386            } else {
387                let (s, c1, c2) = directional(aa, sa, bb, sb, ff)?;
388                status = s;
389                coeff = coeffs(c1, c2);
390                let (s, c1, c2) = directional(bb, sb, aa, sa, ff)?;
391                reversed = Some(s);
392                reverse_coeff = coeffs(c1, c2);
393            }
394        }
395        let face = |f: &Json| Json::Array(f.items().unwrap().to_vec());
396        let mut fields = std::collections::BTreeMap::new();
397        fields.insert("faces".into(), Json::Array(vec![face(first), face(second)]));
398        fields.insert("status".into(), Json::String(status.into()));
399        if let Json::Object(values) = coeff {
400            fields.extend(values);
401        }
402        if let Some(reversed) = reversed {
403            fields.insert("reversed_status".into(), Json::String(reversed.into()));
404            fields.insert("reversed_coefficients".into(), reverse_coeff);
405        }
406        records.push(Json::Object(fields));
407    }
408    Ok(Json::Array(records))
