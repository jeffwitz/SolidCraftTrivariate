1//! A rational fan for one convex footprint inside convex stock.
2//! Rays through both sets of corners keep every prescribed segment on the mesh.
3use crate::constraint_system::Ratio;

7use crate::sketch_layers::{edges, orient, SketchLayers, SketchLoop};

9use std::collections::BTreeMap;

11type Point = [Ratio; 2];
12type Vector = [i128; 2];
13type Quad = [Point; 4];
14type Cell = (Quad, FaceClass);
15
16fn det(a: Vector, b: Vector) -> i128 {
17    a[0] * b[1] - a[1] * b[0]
18}
19fn turn_sign(l: &SketchLoop) -> Option<i128> {
20    let v = &l.vertices;
21    let n = v.len();
22    let s = orient(v[0], v[1], v[2]).signum();
23    (s != 0 && (0..n).all(|i| orient(v[i], v[(i + 1) % n], v[(i + 2) % n]).signum() == s))
24        .then_some(i128::from(s))
25}
26fn exit(l: &SketchLoop, sign: i128, c: Vector, n: i128, d: Vector) -> Result<Point, &'static str> {
27    let mut limit: Option<Ratio> = None;
28    for (a, b) in edges(l) {
29        let e = [
30            i128::from(b[0]) - i128::from(a[0]),
31            i128::from(b[1]) - i128::from(a[1]),
32        ];
33        let dist = sign
34            * det(
35                e,
36                [c[0] - n * i128::from(a[0]), c[1] - n * i128::from(a[1])],
37            );
38        if dist <= 0 {
39            return Err("radial_origin_outside");
40        }
41        let velocity = -sign * det(e, d);
42        if velocity > 0 {
43            let t = Ratio::new(dist, velocity)?;
44            if limit.is_none_or(|x| t.cmp_value(&x).is_lt()) {
45                limit = Some(t);
46            }
47        }
48    }
49    let t = limit.ok_or("open_radial_loop")?;
50    let coordinate = |k: usize| {
51        Ratio::new(c[k], 1)?
52            .checked_add(t.checked_mul(Ratio::new(d[k], 1)?)?)?
53            .checked_mul(Ratio::new(1, n)?)
54    };
55    Ok([coordinate(0)?, coordinate(1)?])
56}
57fn convex(q: Quad) -> Result<bool, &'static str> {
58    for i in 0..4 {
59        if cross(q[i], q[(i + 1) % 4], q[(i + 2) % 4])?.numerator() <= 0 {
60            return Ok(false);
61        }
62    }
63    Ok(true)
64}
65fn cells(
66    mut rays: Vec<(Point, Point)>,
67    center: Point,
68    through: bool,
69) -> Result<Vec<Cell>, &'static str> {
70    use FaceClass::{Material, Void};
71    if !through && rays.len() % 2 == 1 {
72        let (a, b) = rays[0];
73        let (c, d) = rays[1];
74        rays.insert(1, (midpoint(a, c)?, midpoint(b, d)?));
75    }
76    let n = rays.len();
77    let ring: Vec<_> = (0..n)
78        .map(|i| {
79            let (a, b) = rays[i];
80            let (d, c) = rays[(i + 1) % n];
81            ([a, b, c, d], Material)
82        })
83        .collect();
84    if through {
85        return Ok(ring);
86    }
87    for offset in 0..2 {
88        let mut out = ring.clone();
89        for i in (offset..n).step_by(2) {
90            let q = [center, rays[i].0, rays[(i + 1) % n].0, rays[(i + 2) % n].0];
91            if !convex(q)? {
92                break;
93            }
94            out.push((q, Void));
95        }
96        if out.len() == n + n / 2 {
97            return Ok(out);
98        }
99    }
100    let mut out = Vec::new();
101    for ([a, b, c, d], _) in ring {
102        let m = midpoint(a, d)?;
103        let k = midpoint(b, c)?;
104        let ca = midpoint(center, a)?;
105        let cd = midpoint(center, d)?;
106        let g = centroid([center, a, d])?;
107        out.extend([
108            ([a, b, k, m], Material),
109            ([m, k, c, d], Material),
110            ([center, ca, g, cd], Void),
111            ([a, m, g, ca], Void),
112            ([d, cd, g, m], Void),
113        ]);
114    }
115    Ok(out)
116}
117pub(crate) fn graph(
118    job: &SketchLayers,
119    cells: Vec<Cell>,
120    guide: &str,
121) -> Result<Arrangement, &'static str> {
122    let mut result = Arrangement {
123        vertices: vec![],
124        edges: vec![],
125        faces: vec![],
126    };
127    let mut points = BTreeMap::new();
128    let mut edge_ids = BTreeMap::new();
129    let mut uses: Vec<[usize; 2]> = vec![];
130    for (quad, class) in cells {
131        if !convex(quad)? {
132            return Err("nonconvex_radial_cell");
133        }
134        let mut vertices = Vec::new();
135        for p in quad {
136            let id = *points.entry(p).or_insert_with(|| {
137                result.vertices.push(p);
138                result.vertices.len() - 1
139            });
140            vertices.push(id);
141        }
142        let mut sides = Vec::new();
143        for i in 0..4 {
144            let (a, b) = (vertices[i], vertices[(i + 1) % 4]);
145            let key = [a.min(b), a.max(b)];
146            let id = if let Some(id) = edge_ids.get(&key) {
147                *id
148            } else {
149                let [p, q] = key.map(|v| result.vertices[v]);
150                let mut source = None;
151                for contour in std::iter::once(&job.outer).chain(job.cuts.iter().map(|(l, _)| l)) {
152                    for (j, (u, v)) in edges(contour).enumerate() {
153                        let [u, v] = [u, v].map(|p| p.map(|x| Ratio::new(x.into(), 1).unwrap()));
154                        if cross(u, v, p)?.numerator() == 0 && cross(u, v, q)?.numerator() == 0 {
155                            let k = usize::from(u[0] == v[0]);
156                            let lo = if u[k].cmp_value(&v[k]).is_lt() {
157                                u[k]
158                            } else {
159                                v[k]
160                            };
161                            let hi = if lo == u[k] { v[k] } else { u[k] };
162                            if [p[k], q[k]]
163                                .into_iter()
164                                .any(|x| x.cmp_value(&lo).is_lt() || x.cmp_value(&hi).is_gt())
165                            {
166                                continue;
167                            }
168                            source = Some(Provenance {
169                                segment_id: format!("boundary/{}/{j}", contour.id),
170                                role: Role::Boundary,
171                                forward: p[k].cmp_value(&q[k]) == u[k].cmp_value(&v[k]),
172                            });
173                            break;
174                        }
175                    }
176                    if source.is_some() {
177                        break;
178                    }
179                }
180                let id = result.edges.len();
181                result.edges.push(ArrangementEdge {
182                    vertices: key,
183                    provenance: vec![source.unwrap_or_else(|| Provenance {
184                        segment_id: format!("guide/{guide}/{id}"),
185                        role: Role::Construction,
186                        forward: true,
187                    })],
188                });
189                uses.push([0; 2]);
190                edge_ids.insert(key, id);
191                id
192            };
193            let forward = a < b;
194            uses[id][usize::from(forward)] += 1;
195            sides.push(FaceEdge { edge: id, forward });
196        }
197        result.faces.push(Face {
198            vertices,
199            edges: sides,
200            class,
201        });
202    }
203    let stock = format!("boundary/{}/", job.outer.id);
204    let through: Vec<_> = job
205        .cuts
206        .iter()
207        .filter(|(_, z)| *z == [0, job.height])
208        .map(|(l, _)| format!("boundary/{}/", l.id))
209        .collect();
210    for (edge, directions) in result.edges.iter().zip(uses) {
211        let exterior = edge.provenance[0].segment_id.starts_with(&stock)
212            || through
213                .iter()
214                .any(|prefix| edge.provenance[0].segment_id.starts_with(prefix));
215        if (exterior && directions.iter().sum::<usize>() != 1)
216            || (!exterior && directions != [1, 1])
217        {
218            return Err("radial_edge_incidence");
219        }
220    }
221    let two = Ratio::new(2, 1)?;
222    for p in &mut result.vertices {
223        for x in p {
224            *x = x.checked_mul(two)?
225        }
226    }
227    Ok(result)
228}
229pub(crate) fn footprint(job: &SketchLayers) -> Result<Option<Arrangement>, &'static str> {
230    if job.force_triangulation
231        || !job.guides.is_empty()
232        || job.operations.as_ref().is_some_and(|ops| ops != &[false])
233        || job.cuts.len() != 1
234    {
235        return Ok(None);
236    }
237    let inner = &job.cuts[0].0;
238    if inner.vertices.len() < 3
239        || job.outer.vertices.len() < 3
240        || inner.vertices.len() + job.outer.vertices.len() > 48
241    {
242        return Ok(None);
243    }
244    let (Some(si), Some(so)) = (turn_sign(inner), turn_sign(&job.outer)) else {
245        return Ok(None);
246    };
247    let n = inner.vertices.len() as i128;
248    let c = inner.vertices.iter().fold([0; 2], |v, p| {
249        [v[0] + i128::from(p[0]), v[1] + i128::from(p[1])]
250    });
251    let mut dirs: Vec<_> = inner
252        .vertices
253        .iter()
254        .chain(&job.outer.vertices)
255        .map(|p| [n * i128::from(p[0]) - c[0], n * i128::from(p[1]) - c[1]])
256        .collect();
257    let half = |v: Vector| v[1] < 0 || (v[1] == 0 && v[0] < 0);
258    dirs.sort_by(|&a, &b| half(a).cmp(&half(b)).then_with(|| det(b, a).cmp(&0)));
259    dirs.dedup_by(|a, b| half(*a) == half(*b) && det(*a, *b) == 0);
260    let mut rays = vec![];
261    for d in dirs {
262        rays.push((exit(inner, si, c, n, d)?, exit(&job.outer, so, c, n, d)?));
263    }
264    let center = [Ratio::new(c[0], n)?, Ratio::new(c[1], n)?];
265    graph(
266        job,
267        cells(rays, center, job.cuts[0].1 == [0, job.height])?,
268        "radial",
269    )
270    .map(Some)
271}

273#[cfg(test)]
274mod tests {
275    use super::*;
276
277    #[test]
278    fn rational_exits_keep_both_contours_and_cap_a_blind_odd_loop() {
279        let mut job = SketchLayers {
280            outer: SketchLoop {
281                id: "stock".into(),
282                vertices: vec![[0, 0], [42, 1], [46, 24], [30, 43], [1, 31]],
283            },
284            cuts: vec![(
285                SketchLoop {
286                    id: "cut".into(),
287                    vertices: vec![[10, 12], [14, 22], [25, 15]],
288                },
289                [0, 10],
290            )],
291            height: 10,
292            guides: vec![],
293            operations: None,
294            shift: [0, 0],
295            strategy: crate::sketch_layers::PlannerStrategy::Automatic,
296            force_triangulation: false,
297        };
298        let ring = footprint(&job).unwrap().unwrap();
299        assert!(ring.vertices.iter().flatten().any(|v| v.denominator() > 1));
300        assert_eq!(ring.faces.len(), 8);
301        for (contour, n) in [("stock", 5), ("cut", 3)] {
302            for i in 0..n {
303                assert!(ring
304                    .edges
305                    .iter()
306                    .flat_map(|e| &e.provenance)
307                    .any(|p| p.segment_id == format!("boundary/{contour}/{i}")));
308            }
309        }
310        job.cuts[0].1 = [3, 10];
311        let cap = footprint(&job).unwrap().unwrap();
312        assert!(cap.faces.iter().any(|f| f.class == FaceClass::Void));
313        job.outer.vertices.reverse();
314        assert_eq!(footprint(&job).unwrap().unwrap().vertices, cap.vertices);
315    }
