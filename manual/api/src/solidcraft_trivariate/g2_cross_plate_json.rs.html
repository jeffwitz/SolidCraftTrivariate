1//! Certified persistence of solids built from explicit planar sketch plans.

9use std::collections::BTreeSet;

11pub(crate) const FAMILY: &str = "g2_cross_plate_v1";
12pub(crate) const FACE_POCKET: &str = "s3_face_pocket_v1";
13pub(crate) const FACE_POCKET_METHOD: &str = "face_attached_rectangular_pocket_partition";
14pub(crate) const PRIMITIVE: &str = "primitive_sweep_v1";
15pub(crate) const PRIMITIVE_METHOD: &str = "native_primitive_sweep_template";
16pub(crate) const METHOD: &str = "periodic_cubic_transverse_bore_partition";

18fn candidate(recipe: &Json) -> Result<(Json, f64), &'static str> {
19    let family = recipe.field("family")?.text()?;
20    let (plan, (blocks, mut faces)) = match family {
21        FAMILY => {
22            let (parameters, plan) = crate::sketch_partition_plan::plan(recipe)?;
23            (plan, g2_cross_plate::construct_partition(&parameters)?)
24        }
25        FACE_POCKET => {
26            let parent = recipe.field("parent_model")?;
27            if parent.field("schema")?.text()? != FAMILY
28                || parent.field("source_recipe")?.field("family")?.text()? != FAMILY
29            {
30                return Err("face pocket requires a transverse plate parent");
31            }
32            let plan = object([
33                ("parent_sketch_plan", parent.field("sketch_plan")?.clone()),
34                ("operation", recipe.field("operation")?.clone()),
35            ]);
36            (plan, crate::g2_face_pocket::construct(recipe)?)
37        }
38        PRIMITIVE => (
39            crate::primitive_template_json::plan(recipe)?,
40            crate::primitive_template_json::construct(recipe)?,
41        ),
42        _ => return Err("unknown swept sketch family"),
43    };
44    let mut lower = f64::INFINITY;
45    for block in &blocks {
46        let cert = crate::spatial_certificate::certify_spatial(
47            block.degrees,
48            &block.controls,
49            Default::default(),
50        );
51        let bound = cert
52            .determinant_lower_bound_mm3
53            .ok_or("uncertified transverse plate block")?;
54        lower = lower.min(bound);
55    }
56    if !lower.is_finite() || lower <= 0.0 {
57        return Err("unresolved periodic plate regularity");
58    }
59    let incidence = enumerate_faces(&blocks)?;
60    faces.sort_by(|a, b| a.0.cmp(&b.0));
61    let labels: BTreeSet<_> = faces.iter().map(|(side, _)| side).collect();
62    if labels.len() != faces.len() || labels != incidence.exterior.iter().collect() {
63        return Err("incomplete periodic plate boundary provenance");
64    }
65    let cert = BracketCertificate {
66        incidence,
67        design_faces: faces,
68        determinant_lower_bound_mm3: lower,
69        radial_error_upper_mm: 0.0,
70    };
71    let mut model = encode(family, blocks, &cert);
72    if let Json::Object(fields) = &mut model {
73        fields.insert("source_recipe".into(), recipe.clone());
74        fields.insert("sketch_plan".into(), plan);
75    }
76    Ok((model, lower))
77}

79pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
80    candidate(recipe).map(|(model, _)| model)
81}

83pub(crate) fn audit(model: &Json) -> Result<Json, &'static str> {
84    let (expected, lower) = candidate(model.field("source_recipe")?)?;
85    if *model != expected {
86        return Err("periodic_plate_stored_partition_mismatch");
87    }
88    let count =
89        |value: &Json| Ok::<_, &'static str>(Json::Number(value.items()?.len().to_string()));
90    let topology = model.field("topology")?;
91    let source = model.field("source_recipe")?;
92    let (method, bores) = match source.field("family")?.text()? {
93        FAMILY => (METHOD, 2),
94        FACE_POCKET => (FACE_POCKET_METHOD, 2),
95        PRIMITIVE => (
96            PRIMITIVE_METHOD,
97            usize::from(source.field("operation")?.field("kind")?.text()? == "revolve"),
98        ),
99        _ => return Err("unknown swept sketch family"),
100    };
101    Ok(object([
102        ("status", Json::String("valid".into())),
103        ("method", Json::String(method.into())),
104        ("blocks", count(model.field("blocks")?)?),
105        ("shared_faces", count(topology.field("interfaces")?)?),
106        ("exterior_sides", count(topology.field("exterior")?)?),
107        (
108            "determinant_lower_bound_mm3",
109            Json::Number(lower.to_string()),
110        ),
111        ("through_bores", Json::Number(bores.to_string())),
112        ("cavities", Json::Number("0".into())),
113        ("hausdorff_error_upper_bound_mm", Json::Number("0".into())),
114    ]))
115}
