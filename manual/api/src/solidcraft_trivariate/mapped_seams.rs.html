1//! Discover hanging rectangular traces and pull their edge coordinates back
2//! to the adjacent tensor chart. Other face shapes remain subject to the final
3//! exact whole-face incidence and exterior-label checks.
4use crate::bracket::BracketBlock;
5use crate::constraint_system::Ratio;
6use crate::incidence::{enumerate_faces, BlockSide};
7use crate::mapped_cut_propagation::{CutParameter, CutRequest, FaceMap};
8use std::collections::{BTreeMap, BTreeSet};
9
10fn tangents(normal: usize) -> [usize; 2] {
11    match normal {
12        0 => [1, 2],
13        1 => [0, 2],
14        _ => [0, 1],
15    }
16}
17struct Rectangle<'a> {
18    side: &'a BlockSide,
19    block: usize,
20    normal: usize,
21    sign: i8,
22    plane: Ratio,
23    world: [usize; 2],
24    starts: [Ratio; 2],
25    ends: [Ratio; 2],
26}
27impl Rectangle<'_> {
28    fn interval(&self, axis: usize) -> [Ratio; 2] {
29        let i = usize::from(self.world[1] == axis);
30        let (a, b) = (self.starts[i], self.ends[i]);
31        if a.cmp_value(&b).is_lt() {
32            [a, b]
33        } else {
34            [b, a]
35        }
36    }
37}
38fn rectangle<'a>(
39    b: &BracketBlock,
40    side: &'a BlockSide,
41    index: usize,
42) -> Result<Option<Rectangle<'a>>, &'static str> {
43    let [s, t] = tangents(side.axis);
44    let get = |i: usize, j: usize| {
45        let mut v = [0; 3];
46        v[side.axis] = if side.upper { b.degrees[side.axis] } else { 0 };
47        v[s] = i;
48        v[t] = j;
49        b.controls[(v[0] * (b.degrees[1] + 1) + v[1]) * (b.degrees[2] + 1) + v[2]]
50    };
51    let p = get(0, 0);
52    let endpoints = [get(b.degrees[s], 0), get(0, b.degrees[t])];
53    if p[3] <= 0 || endpoints.iter().any(|q| q[3] != p[3]) {
54        return Ok(None);
55    }
56    let mut world = [0; 2];
57    let mut inc = [0i128; 2];
58    for a in 0..2 {
59        let changed: Vec<_> = (0..3).filter(|&k| p[k] != endpoints[a][k]).collect();
60        if changed.len() != 1 {
61            return Ok(None);
62        }
63        world[a] = changed[0];
64        inc[a] = i128::from(endpoints[a][world[a]]) - i128::from(p[world[a]]);
65    }
66    if world[0] == world[1] {
67        return Ok(None);
68    }
69    let n = 3 - world[0] - world[1];
70    for i in 0..=b.degrees[s] {
71        for j in 0..=b.degrees[t] {
72            let q = get(i, j);
73            if q[3] != p[3] || q[n] != p[n] {
74                return Ok(None);
75            }
76            for (a, k, deg) in [(0, i, b.degrees[s]), (1, j, b.degrees[t])] {
77                let delta = i128::from(q[world[a]]) - i128::from(p[world[a]]);
78                if deg as i128 * delta != k as i128 * inc[a] {
79                    return Ok(None);
80                }
81            }
82        }
83    }
84    let cyclic = (world[0] + 1) % 3 == world[1];
85    let sign = if cyclic ^ (side.axis == 1) ^ !side.upper ^ (inc[0] < 0) ^ (inc[1] < 0) {
86        1
87    } else {
88        -1
89    };
90    let frac = |v: i64| Ratio::new(v.into(), p[3].into());
91    Ok(Some(Rectangle {
92        side,
93        block: index,
94        normal: n,
95        sign,
96        plane: frac(p[n])?,
97        world,
98        starts: [frac(p[world[0]])?, frac(p[world[1]])?],
99        ends: [frac(endpoints[0][world[0]])?, frac(endpoints[1][world[1]])?],
100    }))
101}
102fn request(
103    receiver: &Rectangle,
104    other: &Rectangle,
105    seen: &mut BTreeSet<(usize, usize, CutParameter, String)>,
106) -> Result<(), &'static str> {
107    for (i, world) in receiver.world.into_iter().enumerate() {
108        let [lo, hi] = receiver.interval(world);
109        for x in other.interval(world) {
110            if !(lo.cmp_value(&x).is_lt() && x.cmp_value(&hi).is_lt()) {
111                continue;
112            }
113            let length = receiver.ends[i].checked_sub(receiver.starts[i])?;
114            let u = x
115                .checked_sub(receiver.starts[i])?
116                .checked_mul(Ratio::new(length.denominator(), length.numerator())?)?;
117            let side = other.side;
118            seen.insert((
119                receiver.block,
120                tangents(receiver.side.axis)[i],
121                CutParameter::new(u.numerator(), u.denominator())?,
122                format!(
123                    "face/{}/{}/{}",
124                    side.block,
125                    side.axis,
126                    usize::from(side.upper)
127                ),
128            ));
129        }
130    }
131    Ok(())
132}
133pub(crate) fn discover(
134    blocks: &[BracketBlock],
135) -> Result<(Vec<FaceMap>, Vec<CutRequest>), &'static str> {
136    let incidence = enumerate_faces(blocks)?;
137    let index: BTreeMap<_, _> = blocks
138        .iter()
139        .enumerate()
140        .map(|(i, b)| (b.id.as_str(), i))
141        .collect();
142    let mut maps = vec![];
143    for interface in incidence.interfaces {
144        let a = interface.first;
145        let b = interface.second;
146        let d = interface.orientation;
147        let to = tangents(b.axis);
148        maps.push(FaceMap {
149            a_block: index[a.block.as_str()],
150            a_normal_axis: a.axis,
151            a_upper: a.upper,
152            b_block: index[b.block.as_str()],
153            b_normal_axis: b.axis,
154            b_upper: b.upper,
155            a_to_b: [
156                (to[usize::from(d.swap)], d.flip_first),
157                (to[usize::from(!d.swap)], d.flip_second),
158            ],
159            isoparametric_admitted: true,
160            provenance: format!(
161                "{}:{}:{}/{}:{}:{}",
162                a.block, a.axis, a.upper, b.block, b.axis, b.upper
163            ),
164        });
165    }
166    let mut planes: BTreeMap<(usize, Ratio), Vec<Rectangle>> = BTreeMap::new();
167    let mut seen = BTreeSet::new();
168    for face in &incidence.exterior {
169        let id = index[face.block.as_str()];
170        if let Some(r) = rectangle(&blocks[id], face, id)? {
171            let group = planes.entry((r.normal, r.plane)).or_default();
172            for q in group.iter().filter(|q| q.sign != r.sign && q.block != id) {
173                if r.world.iter().all(|&w| {
174                    let a = r.interval(w);
175                    let b = q.interval(w);
176                    a[0].cmp_value(&b[1]).is_lt() && b[0].cmp_value(&a[1]).is_lt()
177                }) {
178                    request(&r, q, &mut seen)?;
179                    request(q, &r, &mut seen)?;
180                }
181            }
182            group.push(r);
183        }
184    }
185    Ok((
186        maps,
187        seen.into_iter()
188            .map(|(block, axis, parameter, provenance)| CutRequest {
189                block,
190                axis,
191                parameter,
192                provenance,
193            })
194            .collect(),
195    ))
196}

198#[cfg(test)]
199mod tests {
200    use super::*;
201    use crate::compact_hosts::plain_box;
202    use crate::mapped_cut_propagation::propagate_exact_cuts;
203
204    #[test]
205    fn hanging_thirds_close_through_a_previously_conforming_neighbor() {
206        let blocks = vec![
207            plain_box("back".into(), [[-2, 0], [0, 3], [0, 2]]),
208            plain_box("front".into(), [[0, 2], [0, 3], [0, 2]]),
209            plain_box("low".into(), [[2, 4], [0, 1], [0, 2]]),
210            plain_box("high".into(), [[2, 4], [1, 3], [0, 2]]),
211        ];
212        let (faces, cuts) = discover(&blocks).unwrap();
213        assert_eq!(cuts.len(), 2);
214        assert!(cuts.iter().all(|s| s.block == 1
215            && s.axis == 1
216            && s.parameter == CutParameter::new(1, 3).unwrap()));
217        let fine = propagate_exact_cuts(&blocks, &faces, &cuts, 6).unwrap();
218        assert_eq!(fine.blocks.len(), 6);
219        assert!(discover(&fine.blocks).unwrap().1.is_empty());
220        assert_eq!(enumerate_faces(&fine.blocks).unwrap().interfaces.len(), 7);
221    }
222}
