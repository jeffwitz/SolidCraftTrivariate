1//! Sparse exact elimination of homogeneous jet equations on scalar field coefficients.
2use crate::constraint_system::Ratio;
3use std::collections::BTreeMap;

5pub(crate) type Row = BTreeMap<usize, Ratio>;
6pub(crate) type Result<T> = std::result::Result<T, &'static str>;
7
8pub(crate) fn add(row: &mut Row, i: usize, v: Ratio) -> Result<()> {
9    let value = row
10        .get(&i)
11        .copied()
12        .unwrap_or(Ratio::zero())
13        .checked_add(v)?;
14    if value == Ratio::zero() {
15        row.remove(&i);
16    } else {
17        row.insert(i, value);
18    }
19    if row.len() > 4096 {
20        return Err("indeterminate: jet row budget");
21    }
22    Ok(())
23}
24fn axpy(row: &mut Row, k: Ratio, other: &Row) -> Result<()> {
25    for (&i, &v) in other {
26        add(row, i, k.checked_mul(v)?)?;
27    }
28    Ok(())
29}

31/// All row indices here are pivot priorities (lower first). Retain the original
32/// equations for a separate exact residual check after backward substitution.
33pub(crate) fn nullspace(n: usize, equations: &[Row]) -> Result<(Vec<Row>, Vec<usize>)> {
34    match direct_nullspace(n, equations) {
35        Err(reason) if reason.contains("arithmetic budget") || reason.contains("overflow") => {
36            modular_nullspace(n, equations)
37        }
38        result => result,
39    }
40}
41fn direct_nullspace(n: usize, equations: &[Row]) -> Result<(Vec<Row>, Vec<usize>)> {
42    let mut pivots: BTreeMap<usize, Row> = BTreeMap::new();
43    for original in equations {
44        let mut row = original.clone();
45        while let Some((&i, &c)) = row.first_key_value() {
46            if i >= n {
47                return Err("constraint index outside field");
48            }
49            if let Some(other) = pivots.get(&i) {
50                axpy(&mut row, Ratio::zero().checked_sub(c)?, other)
51                    .map_err(|_| "indeterminate: jet forward elimination arithmetic budget")?;
52            } else {
53                let inv = Ratio::new(c.denominator(), c.numerator())?;
54                for v in row.values_mut() {
55                    *v = v
56                        .checked_mul(inv)
57                        .map_err(|_| "indeterminate: jet pivot normalization arithmetic budget")?;
58                }
59                pivots.insert(i, row);
60                break;
61            }
62        }
63    }
64    let free: Vec<_> = (0..n).filter(|i| !pivots.contains_key(i)).collect();
65    let mut lift = vec![Row::new(); n];
66    for (j, &i) in free.iter().enumerate() {
67        lift[i].insert(j, Ratio::one());
68    }
69    for (i, row) in pivots.iter().rev() {
70        let mut terms = Row::new();
71        for (&j, &v) in row.range(i + 1..) {
72            axpy(&mut terms, Ratio::zero().checked_sub(v)?, &lift[j])?;
73        }
74        lift[*i] = terms;
75    }
76    for equation in equations {
77        let mut residual = Row::new();
78        for (&i, &v) in equation {
79            axpy(&mut residual, v, &lift[i])?;
80        }
81        if !residual.is_empty() {
82            return Err("nonzero exact jet residual");
83        }
84    }
85    Ok((lift, free))
86}

88// A modular rank is a lower bound on the characteristic-zero rank when the
89// input denominators are units. If reconstruction gives a verified rational
90// kernel of complementary dimension with an identity free submatrix, both
91// bounds are sharp. Modular agreement alone is never used to admit a field.
92const P1: u64 = (1u64 << 61) - 1;
93const P2: u64 = 2_305_843_009_213_693_921;
94fn product<const P: u64>(a: u64, b: u64) -> u64 {
95    (u128::from(a) * u128::from(b) % u128::from(P)) as u64
96}
97fn inverse<const P: u64>(mut a: u64) -> u64 {
98    let mut power = P - 2;
99    let mut result = 1;
100    while power > 0 {
101        if power & 1 == 1 {
102            result = product::<P>(result, a);
103        }
104        a = product::<P>(a, a);
105        power >>= 1;
106    }
107    result
108}
109fn residue<const P: u64>(x: Ratio) -> Result<u64> {
110    let den = x.denominator().rem_euclid(i128::from(P)) as u64;
111    if den == 0 {
112        return Err("indeterminate: modular denominator");
113    }
114    Ok(product::<P>(
115        x.numerator().rem_euclid(i128::from(P)) as u64,
116        inverse::<P>(den),
117    ))
118}
119type ModularRow = BTreeMap<usize, u64>;
120fn mod_sub<const P: u64>(row: &mut ModularRow, scale: u64, other: &ModularRow) -> Result<()> {
121    for (&j, &v) in other {
122        let old = row.get(&j).copied().unwrap_or(0);
123        let value = (old + P - product::<P>(scale, v)) % P;
124        if value == 0 {
125            row.remove(&j);
126        } else {
127            row.insert(j, value);
128        }
129    }
130    if row.len() > 4096 {
131        return Err("indeterminate: modular jet row budget");
132    }
133    Ok(())
134}
135fn modular_basis<const P: u64>(
136    n: usize,
137    equations: &[Row],
138) -> Result<(Vec<ModularRow>, Vec<usize>)> {
139    let mut pivots: BTreeMap<usize, ModularRow> = BTreeMap::new();
140    for e in equations {
141        let mut r = ModularRow::new();
142        for (&i, &c) in e {
143            if i >= n {
144                return Err("invalid jet index");
145            }
146            let v = residue::<P>(c)?;
147            if v != 0 {
148                r.insert(i, v);
149            }
150        }
151        while let Some((&i, &v)) = r.first_key_value() {
152            if let Some(p) = pivots.get(&i) {
153                mod_sub::<P>(&mut r, v, p)?;
154            } else {
155                let inv = inverse::<P>(v);
156                for c in r.values_mut() {
157                    *c = product::<P>(*c, inv);
158                }
159                pivots.insert(i, r);
160                break;
161            }
162        }
163    }
164    let free: Vec<_> = (0..n).filter(|i| !pivots.contains_key(i)).collect();
165    let mut reduced = vec![ModularRow::new(); n];
166    for (j, &i) in free.iter().enumerate() {
167        reduced[i].insert(j, 1);
168    }
169    for (&i, row) in pivots.iter().rev() {
170        let mut expansion = ModularRow::new();
171        for (&j, &v) in row.range(i + 1..) {
172            mod_sub::<P>(&mut expansion, v, &reduced[j])?;
173        }
174        reduced[i] = expansion;
175    }
176    Ok((reduced, free))
177}

179fn reconstruct(a: u64, b: u64) -> Result<Ratio> {
180    // CRT and Euclidean rational reconstruction. Both primes and their
181    // product fit signed i128; every Euclidean coefficient is bounded by p.
182    let modulus = i128::from(P1) * i128::from(P2);
183    let delta = (b + P2 - a % P2) % P2;
184    let x =
185        i128::from(a) + i128::from(P1) * i128::from(product::<P2>(delta, inverse::<P2>(P1 % P2)));
186    let bound = (modulus / 2).isqrt();
187    let (mut m, mut r, mut t, mut u) = (modulus, x, 0i128, 1i128);
188    while r > bound {
189        let q = m / r;
190        (m, r) = (r, m - q * r);
191        (t, u) = (u, t - q * u);
192    }
193    if u == 0 || u.abs() > bound {
194        return Err("indeterminate: two-prime rational reconstruction bound");
195    }
196    let result = Ratio::new(r, u)?;
197    if residue::<P1>(result)? != a || residue::<P2>(result)? != b {
198        return Err("indeterminate: rational reconstruction residues");
199    }
200    Ok(result)
201}
202fn modular_nullspace(n: usize, equations: &[Row]) -> Result<(Vec<Row>, Vec<usize>)> {
203    let (a, free) = modular_basis::<P1>(n, equations)?;
204    let (b, other) = modular_basis::<P2>(n, equations)?;
205    if free != other {
206        return Err("indeterminate: incompatible modular pivot profiles");
207    }
208    let (c, fc) = modular_basis::<2_305_843_009_213_693_907>(n, equations)?;
209    let (d, fd) = modular_basis::<2_305_843_009_213_693_723>(n, equations)?;
210    if fc != free || fd != free {
211        return Err("indeterminate: incompatible four-prime pivot profiles");
212    }
213    let mut cache = BTreeMap::new();
214    let mut lift = Vec::with_capacity(n);
215    for (row, (a, b)) in a.iter().zip(&b).enumerate() {
216        let mut values = Row::new();
217        // Columns vanishing at both primes are checked by the exact kernel
218        // identity below. Both modular pivot profiles give the rank lower bound.
219        let keys: std::collections::BTreeSet<_> = a
220            .keys()
221            .chain(b.keys())
222            .chain(c[row].keys())
223            .chain(d[row].keys())
224            .copied()
225            .collect();
226        for i in keys {
227            let x = *a.get(&i).unwrap_or(&0);
228            let y = *b.get(&i).unwrap_or(&0);
229            let residues = [
230                x,
231                y,
232                *c[row].get(&i).unwrap_or(&0),
233                *d[row].get(&i).unwrap_or(&0),
234            ];
235            let value = if let Some(&v) = cache.get(&residues) {
236                v
237            } else {
238                let small = reconstruct(x, y).ok().filter(|&r| {
239                    residue::<2_305_843_009_213_693_907>(r) == Ok(residues[2])
240                        && residue::<2_305_843_009_213_693_723>(r) == Ok(residues[3])
241                });
242                let r = match small {
243                    Some(v) => v,
244                    None => crate::spline_reconstruction::recover(residues)?,
245                };
246                cache.insert(residues, r);
247                r
248            };
249            values.insert(i, value);
250        }
251        lift.push(values);
252    }
253    for equation in equations {
254        let mut residual = Row::new();
255        for (&i, &v) in equation {
256            axpy(&mut residual, v, &lift[i])?;
257        }
258        if !residual.is_empty() {
259            return Err("indeterminate: reconstructed jet kernel has nonzero exact residual");
260        }
261    }
262    Ok((lift, free))
263}

265/// Choose an interpolatory coordinate system for coupled transverse jets. Floating
266/// arithmetic proposes a column permutation only; rank, identities and acceptance
267/// still come entirely from exact elimination of the original equations.
268pub(crate) fn conditioned_kernel(n: usize, equations: &[Row]) -> Result<(Vec<Row>, Vec<usize>)> {
269    let mut pivots: Vec<(usize, Vec<(usize, f64)>)> = Vec::new();
270    let mut chosen = vec![false; n];
271    for row in equations {
272        let mut r = vec![0.; n];
273        let mut norm = 0_f64;
274        for (&i, &v) in row {
275            r[i] = v.numerator() as f64 / v.denominator() as f64;
276            norm = norm.max(r[i].abs());
277        }
278        if norm == 0. {
279            continue;
280        }
281        for (i, other) in &pivots {
282            let v = r[*i];
283            if v != 0. {
284                for &(j, x) in other {
285                    r[j] -= v * x;
286                }
287                r[*i] = 0.;
288            }
289        }
290        let Some((i, &v)) = r
291            .iter()
292            .enumerate()
293            .filter(|(i, _)| !chosen[*i])
294            .max_by(|(_, a), (_, b)| a.abs().total_cmp(&b.abs()))
295        else {
296            continue;
297        };
298        if !v.is_finite() || v.abs() < 1e-10 * norm {
299            continue;
300        }
301        chosen[i] = true;
302        pivots.push((
303            i,
304            r.iter()
305                .enumerate()
306                .filter_map(|(j, &x)| if x != 0. { Some((j, x / v)) } else { None })
307                .collect(),
308        ));
309    }
310    let order: Vec<_> = pivots
311        .iter()
312        .map(|(i, _)| *i)
313        .chain((0..n).filter(|&i| !chosen[i]))
314        .collect();
315    let mut ids = vec![0; n];
316    for (j, &i) in order.iter().enumerate() {
317        ids[i] = j;
318    }
319    let transformed = equations
320        .iter()
321        .map(|r| r.iter().map(|(&i, &v)| (ids[i], v)).collect())
322        .collect::<Vec<_>>();
323    let (lift, free) = match nullspace(n, &transformed) {
324        Ok(result) => result,
325        Err(_) => {
326            if let Ok(basis) = nullspace(n, equations) {
327                return Ok(basis);
328            }
329            // An interpolatory basis can have large denominators even when
330            // the same space has a small basis in other free coordinates.
331            // Alternate reproducible pivot orders, retaining exact checks.
332            for seed in 1..=8 {
333                let mut perm: Vec<_> = (0..n).collect();
334                let mut state = seed as u64;
335                for i in (1..n).rev() {
336                    state = state.wrapping_mul(6364136223846793005).wrapping_add(1);
337                    perm.swap(i, ((state >> 32) as usize) % (i + 1));
338                }
339                let mut index = vec![0; n];
340                for (j, &i) in perm.iter().enumerate() {
341                    index[i] = j;
342                }
343                let matrix = equations
344                    .iter()
345                    .map(|r| r.iter().map(|(&i, &v)| (index[i], v)).collect())
346                    .collect::<Vec<_>>();
347                if let Ok((basis, roots)) = nullspace(n, &matrix) {
348                    return Ok((
349                        index.iter().map(|&i| basis[i].clone()).collect(),
350                        roots.into_iter().map(|j| perm[j]).collect(),
351                    ));
352                }
353            }
354            return Err("indeterminate: exact jet kernel reconstruction budget");
355        }
356    };
357    let free = free.into_iter().map(|j| order[j]).collect();
358    Ok((ids.into_iter().map(|i| lift[i].clone()).collect(), free))
359}

361/// Eliminate inherited axial conditions first. Each transverse row is then
362/// restricted exactly to those roots. Prefer the short section equations
363/// over their redundant axial interpolants to retain a well-scaled basis.
364pub(crate) fn joint_kernel(
365    n: usize,
366    axial: &[Row],
367    transverse: &[Row],
368) -> Result<(Vec<Row>, Vec<usize>)> {
369    let (first, roots) = nullspace(n, axial)?;
370    if transverse.is_empty() {
371        return Ok((first, roots));
372    }
373    if roots.len() > 12000 {
374        return Err("indeterminate: transverse elimination budget");
375    }
376    let restrict = |row: &Row, lift: &[Row]| -> Result<Row> {
377        let mut output = Row::new();
378        for (&j, &v) in row {
379            axpy(&mut output, v, &lift[j])?;
380        }
381        Ok(output)
382    };
383    let mut projected = transverse
384        .iter()
385        .map(|r| restrict(r, &first))
386        .collect::<Result<Vec<_>>>()?;
387    projected.sort_by_key(|r| {
388        (
389            r.len(),
390            r.values().map(|v| v.denominator()).max().unwrap_or(1),
391        )
392    });
393    let (second, independent) = conditioned_kernel(roots.len(), &projected)?;
394    let lift = first
395        .iter()
396        .map(|r| restrict(r, &second))
397        .collect::<Result<Vec<_>>>()?;
398    for r in axial.iter().chain(transverse) {
399        if !restrict(r, &lift)?.is_empty() {
400            return Err("nonzero composed jet residual");
401        }
402    }
403    Ok((lift, independent.into_iter().map(|i| roots[i]).collect()))
404}
