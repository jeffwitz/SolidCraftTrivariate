1//! Exact rational material charts for intersecting X/Z through-bores of equal radius.
2//!
3//! In each octant the two cylinder inequalities reduce to
4//! min(|x|,|z|)^2 + y^2 >= r^2. Sweep the meridional section in the
5//! dominant coordinate from the symmetry plane to the outside of the stock.
6//! Coordinates in this description are relative to the axis intersection.
7
8use crate::bracket::BracketBlock;
9
10struct Octant {
11    center: [i64; 3],
12    sign: [i64; 3],
13    extent: [i64; 3],
14    radius: i64,
15    guard: i64,
16}
17
18impl Octant {
19    fn world(&self, local: [i64; 3], weight: i64) -> [i64; 4] {
20        let p: [i64; 3] =
21            std::array::from_fn(|i| self.center[i] * weight + self.sign[i] * local[i]);
22        [p[0], p[1], p[2], weight]
23    }
24
25    fn sector(&self, dominant: usize, arc: i64, id: String) -> BracketBlock {
26        let mut controls = Vec::with_capacity(12);
27        let handedness = if dominant == 0 { 1 } else { -1 };
28        let forward = handedness * self.sign.iter().product::<i64>() > 0;
29        // Stereographic q runs through [0, 1/2] then [1/2, 1].
30        // Use the blossom at (p/2,t/2) to form exact quadratic controls.
31        for (i, (p, t)) in [(arc, arc), (arc, arc + 1), (arc + 1, arc + 1)]
32            .into_iter()
33            .enumerate()
34        {
35            let i = i as i64;
36            let circle = [
37                self.radius * (4 - p * t),
38                self.radius * 2 * (p + t),
39                4 + p * t,
40            ];
41            // The outer edge is linear with constant weight. Its midpoint
42            // control preserves the degree-two trace on the neighboring box.
43            let outer = if arc == 0 {
44                [4 * self.guard, 2 * i * self.extent[1], 4]
45            } else {
46                [2 * (2 - i) * self.guard, 4 * self.extent[1], 4]
47            };
48            for [minor, height, weight] in [circle, outer] {
49                for upper in [false, true] {
50                    let mut point = [minor, height, minor];
51                    if upper == forward {
52                        point[dominant] = self.extent[dominant] * weight;
53                    }
54                    controls.push(self.world(point, weight));
55                }
56            }
57        }
58        BracketBlock {
59            id,
60            degrees: [2, 1, 1],
61            controls,
62        }
63    }
64
65    fn corner(&self, id: String) -> BracketBlock {
66        let intervals: [[i64; 2]; 3] = std::array::from_fn(|i| {
67            let start = if i == 1 { 0 } else { self.guard };
68            let mut ends = [start, self.extent[i]].map(|v| self.center[i] + self.sign[i] * v);
69            ends.sort();
70            ends
71        });
72        let mut controls = Vec::with_capacity(8);
73        for x in intervals[0] {
74            for y in intervals[1] {
75                for z in intervals[2] {
76                    controls.push([x, y, z, 1]);
77                }
78            }
79        }
80        BracketBlock {
81            id,
82            degrees: [1; 3],
83            controls,
84        }
85    }
86}

88/// Partition the material of an axis-aligned stock after both through-cuts.
89/// Each sign octant has four swept circular sectors and one outer brick.
90/// The guard is an internal partition coordinate; changing it preserves the part.
91pub fn steinmetz_box(
92    bounds: [[i32; 2]; 3],
93    center: [i32; 3],
94    radius: i32,
95    guard: i32,
96    id: &str,
97) -> Result<Vec<BracketBlock>, &'static str> {
98    if radius <= 0 || guard <= radius || id.is_empty() {
99        return Err("invalid_steinmetz_parameters");
100    }
101    let center = center.map(i64::from);
102    let radius = i64::from(radius);
103    let guard = i64::from(guard);
104    let mut blocks = Vec::with_capacity(40);
105    for octant in 0..8 {
106        let sign = std::array::from_fn(|i| if octant & (1 << i) == 0 { -1 } else { 1 });
107        let extent = std::array::from_fn(|i| {
108            let end = bounds[i][usize::from(sign[i] > 0)];
109            sign[i] * (i64::from(end) - center[i])
110        });
111        if extent[0] <= guard || extent[2] <= guard || extent[1] <= radius {
112            return Err("steinmetz_partition_clearance");
113        }
114        let chart = Octant {
115            center,
116            sign,
117            extent,
118            radius,
119            guard,
120        };
121        for dominant in [0, 2] {
122            for arc in [0, 1] {
123                blocks.push(chart.sector(
124                    dominant,
125                    arc,
126                    format!("{id}_o{octant}_axis{dominant}_arc{arc}"),
127                ));
128            }
129        }
130        blocks.push(chart.corner(format!("{id}_o{octant}_corner")));
131    }
132    Ok(blocks)
133}
