17fn string(s: &str) -> Json {
18    Json::String(s.to_owned())
19}
20fn num(n: impl ToString) -> Json {
21    Json::Number(n.to_string())
22}
23fn array<T>(s: impl IntoIterator<Item = T>, f: impl FnMut(T) -> Json) -> Json {
24    Json::Array(s.into_iter().map(f).collect())
25}
26fn dims<const N: usize, T: std::str::FromStr>(v: &Json) -> Result<[T; N], &'static str> {
27    v.items()?
28        .iter()
29        .map(Json::number)
30        .collect::<Result<Vec<_>, _>>()?
31        .try_into()
32        .map_err(|_| "incorrect array length")
33}
34fn id(s: &Json, values: &[&'static str]) -> Result<&'static str, &'static str> {
35    let s = s.text()?;
36    values
37        .iter()
38        .copied()
39        .find(|v| s == *v)
40        .ok_or("unknown identifier")
41}

43fn annular_json(m: AnnularPair) -> Json {
44    object([
45        ("schema", string("annular_pair_v1")),
46        ("heights_mm", array(m.heights_mm, num)),
47        (
48            "blocks",
49            array(m.blocks, |b| {
50                object([
51                    ("id", string(b.id)),
52                    ("degrees", array(b.degrees, num)),
53                    ("controls", array(b.controls, |p| array(p, num))),
54                ])
55            }),
56        ),
57        ("interface", array(m.interface, string)),
58    ])
59}

61fn decode_annular(model: &Json) -> Result<AnnularPair, &'static str> {
62    model.keys(&["schema", "heights_mm", "blocks", "interface"])?;
63    let heights_mm = dims(model.field("heights_mm")?)?;
64    let mut blocks = vec![];
65    for b in model.field("blocks")?.items()? {
66        b.keys(&["id", "degrees", "controls"])?;
67        let controls = b
68            .field("controls")?
69            .items()?
70            .iter()
71            .map(dims::<4, f64>)
72            .collect::<Result<_, _>>()?;
73        blocks.push(AnnularBlock {
74            id: id(b.field("id")?, &["lower", "upper"])?,
75            degrees: dims(b.field("degrees")?)?,
76            controls,
77        });
78    }
79    let interface = model
80        .field("interface")?
81        .items()?
82        .iter()
83        .map(|v| id(v, &["lower.w1", "upper.w0", "identity_uv"]))
84        .collect::<Result<Vec<_>, _>>()?
85        .try_into()
86        .map_err(|_| "incorrect incidence count")?;
87    Ok(AnnularPair {
88        heights_mm,
89        blocks,
90        interface,
91    })
92}

94// The historical affine family admits arbitrary positive integer widths.
95// Its only arithmetic operation is exact addition of two nonnegative integers.
96fn natural(v: &Json) -> Result<&str, &'static str> {
97    match v {
98        Json::Number(n)
99            if n.as_bytes()
100                .first()
101                .is_some_and(|b| matches!(b, b'1'..=b'9'))
102                && n.bytes().all(|b| b.is_ascii_digit()) =>
103        {
104            Ok(n)
105        }
106        _ => Err("expected a positive integer width"),
107    }
108}
109fn widths(v: &Json) -> Result<[&str; 2], &'static str> {
110    let [a, b] = v.items()? else {
111        return Err("expected two widths");
112    };
113    Ok([natural(a)?, natural(b)?])
114}
115fn add_decimal(a: &str, b: &str) -> String {
116    let mut x = a.bytes().rev();
117    let mut y = b.bytes().rev();
118    let mut digits = vec![];
119    let mut carry = 0;
120    loop {
121        let p = x.next();
122        let q = y.next();
123        if p.is_none() && q.is_none() && carry == 0 {
124            break;
125        }
126        let v = p.map_or(0, |c| c - b'0') + q.map_or(0, |c| c - b'0') + carry;
127        digits.push(b'0' + v % 10);
128        carry = v / 10;
129    }
130    digits.reverse();
131    String::from_utf8(digits).expect("decimal digits")
132}
133fn affine_model(w: [&str; 2]) -> Json {
134    let total = add_decimal(w[0], w[1]);
135    let blocks = [("left", "0", w[0]), ("right", w[0], total.as_str())];
136    object([
137        ("schema", string("affine_pair_v1")),
138        ("widths_mm", array(w, num)),
139        (
140            "blocks",
141            array(blocks, |(name, x0, x1)| {
142                let mut points = vec![];
143                for x in [x0, x1] {
144                    for y in 0..2 {
145                        for z in 0..2 {
146                            points.push(Json::Array(vec![num(x), num(y), num(z), num(1)]));
147                        }
148                    }
149                }
150                object([("id", string(name)), ("controls", Json::Array(points))])
151            }),
152        ),
153        (
154            "interface",
155            array(["left.u1", "right.u0", "identity_vw"], string),
156        ),
157    ])
158}
159fn audit_affine(m: &Json) -> Result<Json, &'static str> {
160    m.keys(&["schema", "widths_mm", "blocks", "interface"])?;
161    let w = widths(m.field("widths_mm")?)?;
162    let sum = add_decimal(w[0], w[1]);
163    let blocks = m.field("blocks")?.items()?;
164    if blocks.len() != 2
165        || *m.field("interface")? != array(["left.u1", "right.u0", "identity_vw"], string)
166    {
167        return Err("affine incidence");
168    }
169    for (index, b) in blocks.iter().enumerate() {
170        b.keys(&["id", "controls"])?;
171        if b.field("id")?.text()? != ["left", "right"][index] {
172            return Err("affine block identity");
173        }
174        let pts = b.field("controls")?.items()?;
175        if pts.len() != 8 {
176            return Err("affine control count");
177        }
178        for (n, p) in pts.iter().enumerate() {
179            let xs = ["0", w[0], sum.as_str()];
180            if *p
181                != Json::Array(vec![
182                    num(xs[index + n / 4]),
183                    num((n / 2) % 2),
184                    num(n % 2),
185                    num(1),
186                ])
187            {
188                return Err("affine coefficient mismatch");
189            }
190        }
191    }
192    let min = w
193        .into_iter()
194        .min_by(|a, b| a.len().cmp(&b.len()).then(a.cmp(b)))
195        .unwrap();
196    Ok(object([
197        ("status", string("valid")),
198        ("method", string("exact_integer_affine_coefficients")),
199        ("min_det_mm3", num(min)),
200        ("volume_mm3", num(sum)),
201        ("shared_faces", num(1)),
202        ("exterior_sides", num(10)),
203    ]))
204}

206pub fn build(recipe: &Json) -> Result<Json, &'static str> {
207    match recipe.field("family")?.text()? {
208        crate::curved_sketch_audit::FAMILY => crate::curved_sketch_audit::build(recipe),
209        super::affine_image::FAMILY => super::affine_image::build(recipe),
210        "ruled_section_sweep_v1" => super::ruled_json::build(recipe),
211        super::coaxial_json::FAMILY => super::coaxial_json::build(recipe),
212        super::circular_bend_json::FAMILY => super::circular_bend_json::build(recipe),
213        super::orthogonal_json::FAMILY => super::orthogonal_json::build(recipe),
214        super::rational_sector_json::FAMILY => super::rational_sector_json::build(recipe),
215        super::nurbs_cross_json::FAMILY => super::nurbs_cross_json::build(recipe),
216        super::nurbs_cross_json::BOUNDED_FAMILY => super::nurbs_cross_json::build(recipe),
217        super::nurbs_cross_json::OVAL_FAMILY => super::nurbs_cross_json::build(recipe),
218        super::counterbores_json::FAMILY => super::counterbores_json::build(recipe),
219        super::countersinks_json::FAMILY => super::countersinks_json::build(recipe),
220        super::sketch_layers_json::FAMILY | super::sketch_layers_json::BOOLEAN_FAMILY => {
221            super::sketch_layers_json::build(recipe)
222        }
223        super::g2_cross_plate_json::FAMILY | super::g2_cross_plate_json::PRIMITIVE | super::g2_cross_plate_json::FACE_POCKET => super::g2_cross_plate_json::build(recipe),
224        super::nurbs_multiloop_json::FAMILY | super::nurbs_multiloop_json::EDGE_FAMILY => {
225            super::nurbs_multiloop_json::build(recipe)
226        }
227        super::nurbs_sketch_json::FAMILY => super::nurbs_sketch_json::build(recipe),
228        super::orthogonal_sketch_json::FAMILY => super::orthogonal_sketch_json::build(recipe),
229        super::box_cylinder_boolean_json::FAMILY => super::box_cylinder_boolean_json::build(recipe),
230        super::box_cylinder_boolean::SUITE_FAMILY => super::box_cylinder_boolean_json::build_suite(recipe),
231        super::interval_bores_json::FAMILY => super::interval_bores_json::build(recipe),
232        super::circular_polygon_json::FAMILY => super::circular_polygon_json::build(recipe),
233        super::polygon_json::FAMILY | super::polygon_json::PERFORATED => {
234            super::polygon_json::build(recipe)
235        }
236        "axial_boolean_v1" => super::axial_json::build(recipe),
237        family
238            if family == "filleted_bracket_v1"
239                || SECTION_FAMILIES.contains(&family)
240                || super::bracket_json::PARAMETRIC_FAMILIES.contains(&family) =>
241        {
242            super::bracket_json::build(recipe)
243        }
244        "affine_pair_v1" => {
245            recipe.keys(&["family", "widths_mm"])?;
246            Ok(affine_model(widths(recipe.field("widths_mm")?)?))
247        }
248        "annular_pair_v1" => {
249            recipe.keys(&["family", "heights_mm"])?;
250            Ok(annular_json(build_annular(dims(
251                recipe.field("heights_mm")?,
252            )?)?))
253        }
254        _ => Err("unsupported family"),
255    }

258/// Report planar junctions and transverse cuts before creating any volume blocks.
259pub fn plan_sketch(recipe: &Json) -> Result<Json, &'static str> {
260    if matches!(
261        recipe.field("family")?.text()?,
262        crate::sketch_layers_json::FAMILY | crate::sketch_layers_json::BOOLEAN_FAMILY
263    ) {
264        return crate::sketch_layers_json::plan_sketch(recipe);
265    }
266    if recipe.field("family")?.text()? == "primitive_sweep_v1" {
267        return crate::primitive_template_json::plan(recipe);
268    }
269    crate::sketch_partition_plan::plan(recipe).map(|(_, report)| report)
270}
