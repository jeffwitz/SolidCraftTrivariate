10type Point = [Ratio; 2];
11type Quad = [Point; 4];
12type HostPlan = Vec<Vec<Point>>;

15fn ccw(contour: &SketchLoop) -> Option<Vec<Point>> {
16    let v = &contour.vertices;
17    let n = v.len();
18    let sign = orient(v[0], v[1], v[2]).signum();
19    if sign == 0 || !(0..n).all(|i| orient(v[i], v[(i + 1) % n], v[(i + 2) % n]).signum() == sign) {
20        return None;
21    }
22    let mut points: Vec<_> = v.iter().map(|p| p.map(ratio)).collect();
23    if sign < 0 {
24        points.reverse()
25    }
26    let first = points.iter().enumerate().min_by_key(|(_, p)| *p)?.0;
27    points.rotate_left(first);
28    Some(points)
29}

31// Choose a strictly convex ladder across the footprint of a blind cut.
32// Rank rotations by the largest corner Jacobian distortion.
33fn cap(poly: &[Point]) -> Option<Vec<Quad>> {
34    let n = poly.len();
35    let mut best: Option<(f64, Vec<Quad>)> = None;
36    for start in 0..n / 2 {
37        let p = |i: usize| poly[(start + i) % n];
38        let quads: Vec<Quad> = (0..n / 2 - 1)
39            .map(|i| [p(i), p(i + 1), p(n - 2 - i), p(n - 1 - i)])
40            .collect();
41        let mut worst = 0.0_f64;
42        for q in &quads {
43            for i in 0..4 {
44                let [a, b, d] = [q[i], q[(i + 1) % 4], q[(i + 3) % 4]];
45                let det = cross(a, b, d).ok()?;
46                if det.numerator() <= 0 {
47                    worst = f64::INFINITY;
48                    break;
49                }
50                let [a, b, d] =
51                    [a, b, d].map(|p| p.map(|v| v.numerator() as f64 / v.denominator() as f64));
52                let norm: f64 = (0..2)
53                    .map(|k| (b[k] - a[k]).powi(2) + (d[k] - a[k]).powi(2))
54                    .sum();
55                worst = worst.max(norm / (det.numerator() as f64 / det.denominator() as f64));
56            }
57        }
58        if worst.is_finite() && best.as_ref().is_none_or(|(score, _)| worst < *score) {
59            best = Some((worst, quads))
60        }
61    }
62    best.map(|(_, quads)| quads)
63}

77fn visit_with(
78    job: &SketchLayers,
79    host_candidates: HostPlanner,
80    consider: &mut impl FnMut(Arrangement),
81) {
82    if job.force_triangulation
83        || !job.guides.is_empty()
84        || job.operations.is_some()
85        || job.cuts.is_empty()
86    {
87        return;
88    }
89    let Some(stock) = ccw(&job.outer) else { return };
90    let Some(mut holes) = job
91        .cuts
92        .iter()
93        .map(|(l, _)| ccw(l))
94        .collect::<Option<Vec<_>>>()
95    else {
96        return;
97    };
98    let mut caps = Vec::new();
99    for (hole, (_, interval)) in holes.iter_mut().zip(&job.cuts) {
100        if *interval == [0, job.height] {
101            continue;
102        }
103        // A triangular floor needs three genuine corners per boundary cycle
104        // and cannot be one regular quadrilateral. Use the same edge midpoints
105        // for the three floor cells and for the adjacent ring.
106        if hole.len() == 3 {
107            let [a, b, c] = [hole[0], hole[1], hole[2]];
108            let (Ok(ab), Ok(bc), Ok(ca), Ok(g)) = (
109                midpoint(a, b),
110                midpoint(b, c),
111                midpoint(c, a),
112                centroid([a, b, c]),
113            ) else {
114                return;
115            };
116            caps.extend(
117                [[a, ab, g, ca], [b, bc, g, ab], [c, ca, g, bc]].map(|q| (q, FaceClass::Void)),
118            );
119            *hole = vec![a, ab, b, bc, c, ca];
120            continue;
121        }
122        if hole.len() % 2 == 1 {
123            let n = hole.len();
124            let edge = (0..n)
125                .max_by_key(|&i| {
126                    let [a, b] = [hole[i], hole[(i + 1) % n]];
127                    (0..2)
128                        .map(|j| (a[j].numerator() - b[j].numerator()).pow(2))
129                        .sum::<i128>()
130                })
131                .unwrap();
132            let Ok(m) = midpoint(hole[edge], hole[(edge + 1) % n]) else {
133                return;
134            };
135            hole.insert(edge + 1, m);
136        }
137        let Some(quads) = cap(hole) else { return };
138        caps.extend(quads.into_iter().map(|q| (q, FaceClass::Void)));
139    }
140    for hosts in host_candidates(&stock, &holes) {
141        let mut submit = |rings: Vec<Quad>| {
142            let mut cells: Vec<_> = rings
143                .into_iter()
144                .map(|q| (q, FaceClass::Material))
145                .collect();
146            cells.extend_from_slice(&caps);
147            if let Ok(graph) = crate::sketch_radial::graph(job, cells, "convex_ring") {
148                consider(graph)
149            }
150        };
151        if let Some(rings) = plan_convex_rings(&hosts, &holes) {
152            submit(rings)
153        }
154        let mut order: Vec<_> = (0..holes.len()).collect();
155        if let Some(rings) = crate::sketch_optional_ring::rings(&hosts, &holes, &order) {
156            submit(rings)
157        }
158        if order.len() > 1 {
159            order.reverse();
160            if let Some(rings) = crate::sketch_optional_ring::rings(&hosts, &holes, &order) {
161                submit(rings)
162            }
163        }
164    }
165}
