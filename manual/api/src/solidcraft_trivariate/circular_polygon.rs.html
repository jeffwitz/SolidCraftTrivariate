27fn valid_bore_id(id: &str) -> bool {
28    !id.is_empty() && id.len() <= 18 && id.bytes().all(|b| b.is_ascii_alphanumeric() || b == b'_')
29}

31fn checked_control(value: i128) -> Result<i64, &'static str> {
32    i64::try_from(value).map_err(|_| "circular polygon control exceeds i64")
33}

35fn transformed_relative(point: [i128; 3], quadrant: usize) -> [i128; 3] {
36    let [x, y, w] = point;
37    match quadrant {
38        0 => [x, y, w],
39        1 => [-x, y, w],
40        2 => [-x, -y, w],
41        _ => [x, -y, w],
42    }
43}
44
45fn source_index(i: usize, quadrant: usize) -> usize {
46    if quadrant % 2 == 1 {
47        2 - i
48    } else {
49        i
50    }
51}

53fn quarter_net(
54    bore: &CircularPolygonBore,
55    height: i32,
56    shift: [i32; 2],
57    quadrant: usize,
58    half: usize,
59) -> Result<BracketBlock, &'static str> {
60    if quadrant > 3 || half > 1 {
61        return Err("invalid circular polygon sector");
62    }
63    // The uncut sector is represented at scale two so the diamond-edge
64    // midpoint is integral. All other coordinates and weights are integral.
65    let r = i128::from(bore.radius_mm);
66    let g = i128::from(bore.guard_mm);
67    let center = bore.center_mm.map(i128::from);
68    let outer = [[2 * g, 0, 2], [g, g, 2], [0, 2 * g, 2]];
69    let circle = [[2 * r, 0, 2], [2 * r, 2 * r, 2], [0, 4 * r, 4]];
70    let mut base = [[[0_i128; 4]; 2]; 3]; // [u][v][x,y,z,w], before extrusion layer
71    for (i, row) in base.iter_mut().enumerate() {
72        let si = source_index(i, quadrant);
73        for (v, source) in [(0, outer[si]), (1, circle[si])] {
74            let [rx, ry, w] = transformed_relative(source, quadrant);
75            row[v] = [center[0] * w + rx, center[1] * w + ry, 0, w];
76        }
77    }
78    // Add the two extrusion layers and their common in-plane translation.
79    let mut layered = [[[[0_i128; 4]; 2]; 2]; 3];
80    for (i, row) in layered.iter_mut().enumerate() {
81        for (v, layers) in row.iter_mut().enumerate() {
82            let [x, y, _, w] = base[i][v];
83            for (layer, z) in [0_i128, i128::from(height)].into_iter().enumerate() {
84                layers[layer] = [
85                    x + i128::from(shift[0]) * w * i128::from(layer > 0),
86                    y + i128::from(shift[1]) * w * i128::from(layer > 0),
87                    z * w,
88                    w,
89                ];
90            }
91        }
92    }
93    // The complete profile coefficients are 2*H. Restrict exactly to either
94    // half interval and serialize all homogeneous controls at gauge eight.
95    let mut controls = Vec::with_capacity(12);
96    let restricted: [[[[i128; 4]; 2]; 2]; 3] = std::array::from_fn(|i| {
97        std::array::from_fn(|v| {
98            std::array::from_fn(|w| {
99                let b0 = layered[0][v][w];
100                let b1 = layered[1][v][w];
101                let b2 = layered[2][v][w];
102                match (half, i) {
103                    (0, 0) => b0.map(|x| x * 4),
104                    (0, 1) => std::array::from_fn(|k| 2 * (b0[k] + b1[k])),
105                    (0, 2) => std::array::from_fn(|k| b0[k] + 2 * b1[k] + b2[k]),
106                    (1, 0) => std::array::from_fn(|k| b0[k] + 2 * b1[k] + b2[k]),
107                    (1, 1) => std::array::from_fn(|k| 2 * (b1[k] + b2[k])),
108                    (1, 2) => b2.map(|x| x * 4),
109                    _ => unreachable!("validated circular polygon sector indices"),
110                }
111            })
112        })
113    });
114    for u in &restricted {
115        for v in u {
116            for numerator in v {
117                controls.push([
118                    checked_control(numerator[0])?,
119                    checked_control(numerator[1])?,
120                    checked_control(numerator[2])?,
121                    checked_control(numerator[3])?,
122                ]);
123            }
124        }
125    }
126    // The construction above uses the base scale two and the exact split
127    // factors. RING_GAUGE documents the physical homogeneous scale.
128    debug_assert_eq!(RING_GAUGE, 8);
129    if controls.iter().any(|p| p[3] <= 0) {
130        return Err("nonpositive circular polygon weight");
131    }
132    let sector = ["east", "north", "west", "south"][quadrant];
133    Ok(BracketBlock {
134        id: format!(
135            "bore_{}_sector_{}_half_{}_z0_{}",
136            bore.id, sector, half, height
137        ),
138        degrees: [2, 1, 1],
139        controls,
140    })
141}

143/// Construct an audited-domain candidate made of a G18 exterior and exact
144/// rational annuli around one to four circular through-bores.
145pub fn build_circular_polygon_prism(
146    recipe: &CircularPolygonPrism,
147) -> Result<Vec<BracketBlock>, &'static str> {
148    if !(1..=4).contains(&recipe.bores.len())
149        || recipe.outer.vertices.len() + 4 * recipe.bores.len() > 32
150    {
151        return Err("circular polygon bore count or vertex budget out of range");
152    }
153    let mut bore_ids = BTreeSet::new();
154    let mut all_ids = BTreeSet::new();
155    if recipe.outer.id.is_empty() || !all_ids.insert(recipe.outer.id.clone()) {
156        return Err("invalid circular polygon outer loop id");
157    }
158    for vertex in &recipe.outer.vertices {
159        if vertex.id.is_empty() || !all_ids.insert(vertex.id.clone()) {
160            return Err("duplicate circular polygon vertex identity");
161        }
162    }
163    let mut bores = recipe.bores.iter().collect::<Vec<_>>();
164    bores.sort_by(|a, b| a.id.cmp(&b.id));
165    let mut guard_loops = Vec::with_capacity(bores.len());
166    for bore in &bores {
167        if !valid_bore_id(&bore.id) || !bore_ids.insert(bore.id.clone()) {
168            return Err("invalid or duplicate circular polygon bore id");
169        }
170        if !all_ids.insert(bore.id.clone()) {
171            return Err("circular polygon bore id collides with a profile identity");
172        }
173        if bore.radius_mm < 1 || i64::from(bore.guard_mm) < 2 * i64::from(bore.radius_mm) {
174            return Err("circular polygon guard must be at least twice the positive radius");
175        }
176        let loop_id = format!("guard_{}", bore.id);
177        if !all_ids.insert(loop_id.clone()) {
178            return Err("circular polygon guard loop id collision");
179        }
180        let c = bore.center_mm.map(i64::from);
181        let g = i64::from(bore.guard_mm);
182        let points = [
183            [c[0] + g, c[1]],
184            [c[0], c[1] + g],
185            [c[0] - g, c[1]],
186            [c[0], c[1] - g],
187        ];
188        let suffixes = ["e", "n", "w", "s"];
189        let mut vertices = Vec::with_capacity(4);
190        for (point, suffix) in points.into_iter().zip(suffixes) {
191            if point.iter().any(|x| !(-64..=64).contains(x)) {
192                return Err("circular polygon guard vertex outside native domain");
193            }
194            let id = format!("g_{}_{}", bore.id, suffix);
195            if !all_ids.insert(id.clone()) {
196                return Err("circular polygon guard vertex id collision");
197            }
198            vertices.push(Vertex {
199                id,
200                xy_mm: [point[0] as i32, point[1] as i32],
201            });
202        }
203        guard_loops.push(LoopProfile {
204            id: loop_id,
205            vertices,
206        });
207    }
208    let mut blocks = build_perforated_prism(&PerforatedPrism {
209        outer: recipe.outer.clone(),
210        holes: guard_loops,
211        height_mm: recipe.height_mm,
212        top_shift_mm: recipe.top_shift_mm,
213    })?;
214    for bore in bores {
215        for quadrant in 0..4 {
216            for half in 0..2 {
217                blocks.push(quarter_net(
218                    bore,
219                    recipe.height_mm,
220                    recipe.top_shift_mm,
221                    quadrant,
222                    half,
223                )?);
224            }
225        }
226    }
227    blocks.sort_by(|a, b| a.id.cmp(&b.id));
228    Ok(blocks)
229}
