1//! Canonical interchange for certified axial-product operations on the fixed z axis.
2use crate::axial_boolean::{
3    build_axial_boolean, AxialAxis, AxialBooleanOperation as Op, AxialOperand,
4    AxialSectionFamily as Family, OperandLabel,
5};
6use crate::axial_boolean_audit::audit_axial_boolean;
7use crate::bracket_json::{encode_blocks, side_json, topology};
8use crate::json::{object, Json};
9
10fn string(s: &str) -> Json {
11    Json::String(s.to_owned())
12}
13fn number(x: impl ToString) -> Json {
14    Json::Number(x.to_string())
15}
16fn label(l: OperandLabel) -> Json {
17    string(match l {
18        OperandLabel::A => "A",
19        OperandLabel::B => "B",
20    })
21}
22
23fn canonical(recipe: &Json) -> Result<(Json, Json), &'static str> {
24    recipe.keys(&["family", "section", "operation", "a_z_mm", "b_z_mm"])?;
25    if recipe.field("family")?.text()? != "axial_boolean_v1" {
26        return Err("axial_boolean_recipe");
27    }
28    let section = match recipe.field("section")?.text()? {
29        "reference_rectangle_v1" => Family::ReferenceRectangleV1,
30        "pierced_square_v1" => Family::PiercedSquareV1,
31        _ => return Err("axial_boolean_unsupported_section"),
32    };
33    let op = match recipe.field("operation")?.text()? {
34        "union" => Op::Union,
35        "intersection" => Op::Intersection,
36        "difference" => Op::Difference,
37        _ => return Err("axial_boolean_operation"),
38    };
39    let operand = |field| -> Result<AxialOperand, &'static str> {
40        let [a, b] = recipe.field(field)?.items()? else {
41            return Err("axial_boolean_interval");
42        };
43        Ok(AxialOperand {
44            section,
45            axis: AxialAxis::Z,
46            interval_mm: [a.number()?, b.number()?],
47        })
48    };
49    let (a, b) = (operand("a_z_mm")?, operand("b_z_mm")?);
50    let m = build_axial_boolean(op, a, b).map_err(|e| match e {
51        crate::axial_boolean::AxialBooleanError::NullInterval => "axial_boolean_zero_height",
52        crate::axial_boolean::AxialBooleanError::InvertedInterval => {
53            "axial_boolean_inverted_interval"
54        }
55        crate::axial_boolean::AxialBooleanError::IntervalOutOfRange => {
56            "axial_boolean_interval_out_of_range"
57        }
58        _ => "axial_boolean_failed",
59    })?;
60    audit_axial_boolean(op, a, b, &m)?;
61    let report = object([
62        ("status", string("valid")),
63        ("method", string("certified_axial_product_partition")),
64        ("components", number(m.component_count)),
65        ("blocks", number(m.blocks.len())),
66        ("shared_faces", number(m.incidence.interfaces.len())),
67        ("exterior_sides", number(m.incidence.exterior.len())),
68        ("ideal_volume_mm3", number(m.volume_mm3)),
69    ]);
70    let intervals = m
71        .selected_intervals
72        .iter()
73        .map(|s| {
74            object([
75                ("z_mm", Json::Array(s.bounds_mm.map(number).into())),
76                (
77                    "owners",
78                    Json::Array(s.operands.iter().copied().map(label).collect()),
79                ),
80            ])
81        })
82        .collect();
83    let faces = m
84        .faces
85        .iter()
86        .map(|f| {
87            object([
88                ("side", side_json(&f.side)),
89                (
90                    "sources",
91                    Json::Array(
92                        f.provenance
93                            .iter()
94                            .map(|s| {
95                                object([
96                                    ("operand", label(s.operand)),
97                                    ("feature", string(&s.source_face)),
98                                    ("orientation_reversed", Json::Bool(s.orientation_reversed)),
99                                ])
100                            })
101                            .collect(),
102                    ),
103                ),
104            ])
105        })
106        .collect();
107    let model = object([
108        ("schema", string("axial_boolean_v1")),
109        ("recipe", recipe.clone()),
110        ("intervals", Json::Array(intervals)),
111        ("topology", topology(&m.incidence)),
112        ("design_faces", Json::Array(faces)),
113        ("blocks", encode_blocks(m.blocks)),
114    ]);
115    Ok((model, report))
116}
117
118pub(super) fn build(recipe: &Json) -> Result<Json, &'static str> {
119    Ok(canonical(recipe)?.0)
120}
121
122pub(super) fn audit(model: &Json) -> Result<Json, &'static str> {
123    let (expected, report) = canonical(model.field("recipe")?)?;
124    if model != &expected {
125        return Err("axial_boolean_canonical_model_mismatch");
126    }
127    Ok(report)
128}
