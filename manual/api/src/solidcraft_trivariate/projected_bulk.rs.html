1//! Q2 elasticity with the volumetric strain projected onto a discontinuous
2//! trilinear pressure space. Local pressure variables are statically condensed.
3//! Positive recovered mean stress denotes tension.
4use crate::{
5    analysis_space::AnalysisSpace,
6    elasticity::{assemble_lame, basis_geometry, gauss, Stiffness},
7};
8use std::collections::BTreeMap;
9
10const N: usize = 81;
11struct ElementProjection {
12    nodes: [usize; 27],
13    /// (M^-1 B) in local component-interleaved displacement order.
14    recovery: Box<[[f64; N]; 8]>,
15}
16pub struct ProjectedElasticity {
17    pub stiffness: Stiffness,
18    bulk_modulus: f64,
19    elements: BTreeMap<String, ElementProjection>,
20}
21
22fn pressure_basis([u, v, w]: [f64; 3]) -> [f64; 8] {
23    let basis = [[1. - u, u], [1. - v, v], [1. - w, w]];
24    std::array::from_fn(|i| basis[0][i / 4] * basis[1][i / 2 % 2] * basis[2][i % 2])
25}
26
27impl ProjectedElasticity {
28    /// Q1 coefficients of the projected mean stress κ div u on this element.
29    pub fn mean_stress_coefficients(&self, id: &str, u: &[f64]) -> Result<[f64; 8], &'static str> {
30        let elem = self
31            .elements
32            .get(id)
33            .ok_or("unknown projected stress block")?;
34        if u.len() != self.stiffness.size() {
35            return Err("invalid projected stress displacement");
36        }
37        let local: [f64; N] = std::array::from_fn(|i| u[3 * elem.nodes[i / 3] + i % 3]);
38        if local.iter().any(|x| !x.is_finite()) {
39            return Err("nonfinite projected stress displacement");
40        }
41        let values = std::array::from_fn(|j| {
42            self.bulk_modulus
43                * elem.recovery[j]
44                    .iter()
45                    .zip(local)
46                    .map(|(a, b)| a * b)
47                    .sum::<f64>()
48        });
49        if values.iter().any(|v| !v.is_finite()) {
50            return Err("nonfinite recovered mean stress");
51        }
52        Ok(values)
53    }
54    /// Evaluate reconstructed hydrostatic stress at a point in the closed chart.
55    pub fn mean_stress_at(coefficients: [f64; 8], uvw: [f64; 3]) -> Result<f64, &'static str> {
56        if uvw.iter().any(|v| !(0.0..=1.0).contains(v))
57            || coefficients.iter().any(|p| !p.is_finite())
58        {
59            return Err("invalid pressure evaluation");
60        }
61        Ok(pressure_basis(uvw)
62            .iter()
63            .zip(coefficients)
64            .map(|(a, b)| a * b)
65            .sum())
66    }
67}

69/// Assemble fully integrated deviatoric stiffness and the positive local Schur complement.
70/// Quadrature order is shared by the pressure mass, divergence and deviatoric terms.
71pub fn assemble_projected_bulk(
72    space: &AnalysisSpace,
73    young: f64,
74    poisson: f64,
75    order: usize,
76) -> Result<ProjectedElasticity, &'static str> {
77    if !young.is_finite() || young <= 0. || !(-1.0 < poisson && poisson < 0.5) {
78        return Err("invalid projected isotropic material");
79    }
80    let mu = young / (2. * (1. + poisson));
81    let bulk = young / (3. * (1. - 2. * poisson));
82    if !bulk.is_finite() || !mu.is_finite() {
83        return Err("nonfinite projected moduli");
84    }
85    let quad = gauss(order)?;
86    if quad.len() < 2 {
87        return Err("pressure mass requires at least two nodes per axis");
88    }
89    let mut stiffness = assemble_lame(space, -2. * mu / 3., mu, order)?;
90    let mut elements = BTreeMap::new();
91    for b in &space.blocks {
92        let mut mass = [[0.; 8]; 8];
93        let mut coupling = Box::new([[0.; N]; 8]);
94        for &(u, wu) in &quad {
95            for &(v, wv) in &quad {
96                for &(w, ww) in &quad {
97                    let t = [u, v, w];
98                    let s = basis_geometry(b, t)?;
99                    let p = pressure_basis(t);
100                    let dv = wu * wv * ww * s.determinant;
101                    for (i, pi) in p.iter().enumerate() {
102                        let weight = dv * pi;
103                        for (j, pj) in p.iter().enumerate().take(i + 1) {
104                            mass[i][j] += weight * pj;
105                        }
106                        for (node, g) in s.gradients.iter().enumerate() {
107                            for (c, gc) in g.iter().enumerate() {
108                                coupling[i][3 * node + c] += weight * gc;
109                            }
110                        }
111                    }
112                }
113            }
114        }
115        // Lower Cholesky M=LLᵀ and forward solution F=L^-1 B.
116        for i in 0..8 {
117            for j in 0..=i {
118                let mut v = mass[i][j];
119                for (&a, &b) in mass[i][..j].iter().zip(&mass[j][..j]) {
120                    v -= a * b;
121                }
122                mass[i][j] = if i == j {
123                    if !v.is_finite() || v <= 0. {
124                        return Err("singular local Q1 pressure mass");
125                    }
126                    v.sqrt()
127                } else {
128                    v / mass[j][j]
129                };
130            }
131            for col in 0..N {
132                let mut v = coupling[i][col];
133                for j in 0..i {
134                    v -= mass[i][j] * coupling[j][col];
135                }
136                coupling[i][col] = v / mass[i][i];
137            }
138        }
139        for (i, &ii) in b.dofs.iter().enumerate() {
140            let row = stiffness.offsets[ii]..stiffness.offsets[ii + 1];
141            for (j, &jj) in b.dofs.iter().enumerate() {
142                let pos = row.start
143                    + stiffness.columns[row.clone()]
144                        .binary_search(&jj)
145                        .map_err(|_| "missing deviatoric CSR block")?;
146                for a in 0..3 {
147                    for d in 0..3 {
148                        let gram = coupling
149                            .iter()
150                            .map(|f| f[3 * i + a] * f[3 * j + d])
151                            .sum::<f64>();
152                        stiffness.values[pos][a][d] += bulk * gram;
153                    }
154                }
155            }
156        }
157        // Backward solution Lᵀ T=F for pressure recovery after displacement solve.
158        for i in (0..8).rev() {
159            for col in 0..N {
160                let mut v = coupling[i][col];
161                for j in i + 1..8 {
162                    v -= mass[j][i] * coupling[j][col];
163                }
164                coupling[i][col] = v / mass[i][i];
165            }
166        }
167        let element = ElementProjection {
168            nodes: b
169                .dofs
170                .as_slice()
171                .try_into()
172                .map_err(|_| "invalid projected DOF count")?,
173            recovery: coupling,
174        };
175        if elements.insert(b.id.clone(), element).is_some() {
176            return Err("duplicate projected elasticity block");
177        }
178    }
179    stiffness.validate()?;
180    Ok(ProjectedElasticity {
181        stiffness,
182        bulk_modulus: bulk,
183        elements,
184    })
185}
