1//! Budgeted adaptive quadrature for surface tractions on certified block faces.
2//!
3//! The panel estimator compares a parent Gauss panel with the sum over its four
4//! dyadic children. It checks area, the complete local nodal load vector,
5//! resultant, and moment on each selected patch. This is an a-posteriori
6//! convergence diagnostic; it is not a mathematical error bound.
7
8use crate::{
9    analysis_space::{AnalysisBlock, AnalysisSpace},
10    boundary_traction::{LoadPatch, SurfaceLoad},
11    elasticity::{basis_geometry, gauss, Vector},
12    incidence::BlockSide,
13};
14use std::collections::{BTreeMap, BTreeSet};

16/// Explicit work and convergence limits for surface quadrature.
17#[derive(Clone, Copy, Debug, PartialEq)]
18pub struct SurfaceQuadrature {
19    pub relative_tolerance: f64,
20    pub max_depth: u8,
21    /// Counts every evaluated panel, including coarse panels used by the estimator.
22    pub max_panels: usize,
23}
24
25impl SurfaceQuadrature {
26    /// Check the limits independently of a mesh, including for kinematic-only jobs.
27    pub fn validate(&self) -> Result<(), &'static str> {
28        if !self.relative_tolerance.is_finite()
29            || self.relative_tolerance <= 0.0
30            || self.relative_tolerance > 1.0
31            || self.max_depth == 0
32            || self.max_depth > 30
33            || self.max_panels == 0
34        {
35            return Err("invalid adaptive surface quadrature configuration");
36        }
37        Ok(())
38    }
39}

41/// Adaptively integrated traction and convergence diagnostics.
42pub struct AdaptiveSurfaceLoad {
43    pub load: SurfaceLoad,
44    /// Number of panel integrals actually evaluated, including estimator panels.
45    pub panels: usize,
46    /// Accepted dyadic depths relative to each requested patch in input order.
47    pub patch_depths: Vec<[u8; 2]>,
48    /// Maximum dyadic subdivision depth of accepted patches.
49    pub deepest: u8,
50    /// Largest final parent-vs-children estimate across selected patches/metrics.
51    pub maximum_estimated_relative_error: f64,
52}

56#[derive(Clone, Default)]
57struct PanelIntegral {
58    nodal: [[f64; 3]; 27],
59    area: f64,
60    resultant: Vector,
61    moment: Vector,
62    absolute_force_scale: f64,
63    absolute_moment_scale: f64,
64}

66fn cross(a: Vector, b: Vector) -> Vector {
67    std::array::from_fn(|k| a[(k + 1) % 3] * b[(k + 2) % 3] - a[(k + 2) % 3] * b[(k + 1) % 3])
68}
69
70fn norm(vector: Vector) -> f64 {
71    vector[0].hypot(vector[1]).hypot(vector[2])
72}
73
74fn patch_rect(patch: &LoadPatch) -> Result<Rect, &'static str> {
75    if patch.depth > 30 {
76        return Err("surface patch depth exceeds the dyadic coordinate range");
77    }
78    let scale = 1_u32 << patch.depth;
79    if patch
80        .intervals
81        .iter()
82        .any(|&[lo, hi]| lo >= hi || hi > scale)
83    {
84        return Err("empty or invalid surface patch rectangle");
85    }
86    Ok(patch
87        .intervals
88        .map(|span| span.map(|x| f64::from(x) / f64::from(scale))))
89}

91fn split(rect: Rect) -> [Rect; 4] {
92    let mid = rect.map(|[lo, hi]| (lo + hi) * 0.5);
93    [
94        [[rect[0][0], mid[0]], [rect[1][0], mid[1]]],
95        [[mid[0], rect[0][1]], [rect[1][0], mid[1]]],
96        [[rect[0][0], mid[0]], [mid[1], rect[1][1]]],
97        [[mid[0], rect[0][1]], [mid[1], rect[1][1]]],
98    ]
99}

101fn integrate_panel<F: Fn(Vector, Vector) -> Vector>(
102    block: &AnalysisBlock,
103    face: &BlockSide,
104    rect: Rect,
105    quadrature: &[(f64, f64)],
106    traction: &F,
107) -> Result<PanelIntegral, &'static str> {
108    let mut result = PanelIntegral {
109        nodal: [[0.0; 3]; 27],
110        area: 0.0,
111        resultant: [0.0; 3],
112        moment: [0.0; 3],
113        absolute_force_scale: 0.0,
114        absolute_moment_scale: 0.0,
115    };
116    let axes = match face.axis {
117        0 => [1, 2],
118        1 => [0, 2],
119        2 => [0, 1],
120        _ => return Err("invalid traction face axis"),
121    };
122    let sign = if face.upper ^ (face.axis == 1) {
123        1.0
124    } else {
125        -1.0
126    };
127    let width = rect.map(|[lo, hi]| hi - lo);
128    for &(u, wu) in quadrature {
129        for &(v, wv) in quadrature {
130            let mut parameters = [0.0; 3];
131            parameters[face.axis] = f64::from(u8::from(face.upper));
132            parameters[axes[0]] = rect[0][0] + width[0] * u;
133            parameters[axes[1]] = rect[1][0] + width[1] * v;
134            let sample = basis_geometry(block, parameters)?;
135            let area_vector = cross(
136                sample.jacobian_columns[axes[0]],
137                sample.jacobian_columns[axes[1]],
138            );
139            let jacobian = norm(area_vector);
140            if !jacobian.is_finite() || jacobian <= 0.0 {
141                return Err("invalid surface Jacobian");
142            }
143            let measure = jacobian * wu * wv * width[0] * width[1];
144            let normal = area_vector.map(|x| sign * x / jacobian);
145            let density = traction(sample.position, normal);
146            if density.iter().any(|x| !x.is_finite()) {
147                return Err("nonfinite integrated traction");
148            }
149            let force = density.map(|x| x * measure);
150            let moment = cross(sample.position, force);
151            result.area += measure;
152            result.absolute_force_scale += norm(density) * measure;
153            result.absolute_moment_scale += norm(cross(sample.position, density)) * measure;
154            for (c, (&force_component, &moment_component)) in force.iter().zip(&moment).enumerate()
155            {
156                result.resultant[c] += force_component;
157                result.moment[c] += moment_component;
158            }
159            for n in 0..27 {
160                for (c, &component) in force.iter().enumerate() {
161                    result.nodal[n][c] += sample.values[n] * component;
162                }
163            }
164        }
165    }
166    if !result.area.is_finite()
167        || !result.absolute_force_scale.is_finite()
168        || !result.absolute_moment_scale.is_finite()
169        || result
170            .resultant
171            .iter()
172            .chain(&result.moment)
173            .any(|x| !x.is_finite())
174        || result.nodal.iter().flatten().any(|x| !x.is_finite())
175    {
176        return Err("nonfinite adaptive surface panel integral");
177    }
178    Ok(result)
179}

181fn add(a: &mut PanelIntegral, b: &PanelIntegral) {
182    for n in 0..27 {
183        for c in 0..3 {
184            a.nodal[n][c] += b.nodal[n][c];
185        }
186    }
187    a.area += b.area;
188    for c in 0..3 {
189        a.resultant[c] += b.resultant[c];
190        a.moment[c] += b.moment[c];
191    }
192    a.absolute_force_scale += b.absolute_force_scale;
193    a.absolute_moment_scale += b.absolute_moment_scale;
194}

196fn sum_panels(panels: &[(Rect, PanelIntegral)]) -> PanelIntegral {
197    let mut total = PanelIntegral {
198        nodal: [[0.0; 3]; 27],
199        area: 0.0,
200        resultant: [0.0; 3],
201        moment: [0.0; 3],
202        absolute_force_scale: 0.0,
203        absolute_moment_scale: 0.0,
204    };
205    for (_, panel) in panels {
206        add(&mut total, panel);
207    }
208    total
209}

211fn relative_difference(difference: f64, magnitude: f64, physical_scale: f64) -> f64 {
212    let denominator = if physical_scale == 0.0 || magnitude > physical_scale * 1e-12 {
213        magnitude
214    } else {
215        physical_scale
216    };
217    if denominator == 0.0 {
218        if difference == 0.0 {
219            0.0
220        } else {
221            f64::INFINITY
222        }
223    } else {
224        difference / denominator
225    }
226}

228fn estimate_error(coarse: &PanelIntegral, fine: &PanelIntegral) -> Result<f64, &'static str> {
229    let area_error = relative_difference(
230        (fine.area - coarse.area).abs(),
231        fine.area.abs(),
232        fine.area.abs(),
233    );
234    let mut nodal_diff_squared = 0.0;
235    let mut nodal_norm_squared = 0.0;
236    for n in 0..27 {
237        for c in 0..3 {
238            nodal_diff_squared += (fine.nodal[n][c] - coarse.nodal[n][c]).powi(2);
239            nodal_norm_squared += fine.nodal[n][c].powi(2);
240        }
241    }
242    let nodal_error = relative_difference(
243        nodal_diff_squared.sqrt(),
244        nodal_norm_squared.sqrt(),
245        fine.absolute_force_scale,
246    );
247    let force_error = relative_difference(
248        norm(std::array::from_fn(|c| {
249            fine.resultant[c] - coarse.resultant[c]
250        })),
251        norm(fine.resultant),
252        fine.absolute_force_scale,
253    );
254    let moment_error = relative_difference(
255        norm(std::array::from_fn(|c| fine.moment[c] - coarse.moment[c])),
256        norm(fine.moment),
257        fine.absolute_moment_scale,
258    );
259    let maximum = [area_error, nodal_error, force_error, moment_error]
260        .into_iter()
261        .fold(0.0_f64, f64::max);
262    if maximum.is_finite() {
263        Ok(maximum)
264    } else {
265        Err("adaptive surface estimate has zero local scale")
266    }
267}

276fn composite_grid(
277    rect: Rect,
278    depths: [u8; 2],
279    limit: usize,
280    evaluated: &mut usize,
281    sample: impl Fn(Rect) -> Result<PanelIntegral, &'static str>,
282) -> Result<PanelIntegral, &'static str> {
283    let [nu, nv] = depths.map(|depth| 1_usize << depth);
284    let count = nu.checked_mul(nv).ok_or("surface panel count overflow")?;
285    if count > limit.saturating_sub(*evaluated) {
286        return Err("adaptive surface quadrature panel budget exceeded");
287    }
288    let span = |axis: usize, i: usize, n: usize| {
289        let [lo, hi] = rect[axis];
290        [i, i + 1].map(|j| lo + (hi - lo) * (j as f64 / n as f64))
291    };
292    let mut sum = PanelIntegral::default();
293    for i in 0..nu {
294        let u = span(0, i, nu);
295        for j in 0..nv {
296            add(&mut sum, &sample([u, span(1, j, nv)])?);
297        }
298    }
299    *evaluated += count;
300    Ok(sum)
301}

303/// Reuse the accepted grid from the immediately preceding primal integration.
304/// Job loads select whole certified faces; patch ordering is shared with the primal.
305pub(crate) fn replay_traction_rate(
306    space: &AnalysisSpace,
307    patches: &[LoadPatch],
308    depths: &[[u8; 2]],
309    order: usize,
310    budget: usize,
311    traction: impl Fn(Vector, Vector) -> Vector,
312) -> Result<(Vec<f64>, usize), &'static str> {
313    if patches.len() != depths.len() {
314        return Err("frozen surface plan does not match load patches");
315    }
316    let blocks: BTreeMap<_, _> = space.blocks.iter().map(|b| (b.id.as_str(), b)).collect();
317    let quadrature = gauss(order)?;
318    let mut count = 0;
319    let mut nodal = vec![
320        0.;
321        space
322            .ndof
323            .checked_mul(3)
324            .ok_or("force vector dimension overflow")?
325    ];
326    for (patch, &depth) in patches.iter().zip(depths) {
327        if depth.into_iter().any(|level| level > 30) {
328            return Err("invalid frozen surface plan depth");
329        }
330        let block = blocks
331            .get(patch.side.block.as_str())
332            .ok_or("unknown traction block")?;
333        let integral = composite_grid(patch_rect(patch)?, depth, budget, &mut count, |r| {
334            integrate_panel(block, &patch.side, r, &quadrature, &traction)
335        })?;
336        for (&id, force) in block.dofs.iter().zip(&integral.nodal) {
337            for (component, value) in force.iter().enumerate() {
338                nodal[3 * id + component] += value;
339            }
340        }
341    }
342    if nodal.iter().any(|value| !value.is_finite()) {
343        return Err("nonfinite frozen surface load rate");
344    }
345    Ok((nodal, count))
346}

348fn directional_patch(
349    rect: Rect,
350    first: PanelIntegral,
351    config: SurfaceQuadrature,
352    evaluated: &mut usize,
353    sample: impl Fn(Rect) -> Result<PanelIntegral, &'static str>,
354) -> Result<(PanelIntegral, [u8; 2], f64), &'static str> {
355    let mut depths = [0_u8; 2];
356    let mut coarse = first;
357    let mut composite =
358        |depths| composite_grid(rect, depths, config.max_panels, evaluated, &sample);
359    loop {
360        let [du, dv] = depths;
361        if du >= config.max_depth || dv >= config.max_depth {
362            return Err("adaptive surface quadrature tolerance not met within depth budget");
363        }
364        let u = composite([du + 1, dv])?;
365        let v = composite([du, dv + 1])?;
366        let uv = composite([du + 1, dv + 1])?;
367        let eu = estimate_error(&coarse, &u)?.max(estimate_error(&v, &uv)?);
368        let ev = estimate_error(&coarse, &v)?.max(estimate_error(&u, &uv)?);
369        let error = eu.max(ev).max(estimate_error(&coarse, &uv)?);
370        if error <= config.relative_tolerance {
371            return Ok((uv, [du + 1, dv + 1], error));
372        }
373        if eu > 2.0 * ev {
374            depths[0] += 1;
375            coarse = u;
376        } else if ev > 2.0 * eu {
377            depths[1] += 1;
378            coarse = v;
379        } else {
380            depths = [du + 1, dv + 1];
381            coarse = uv;
382        }
383    }
384}

386/// Integrate surface tractions by face-wise dyadic composite Gauss refinement.
387///
388/// The returned discrepancy is an estimator based on adjacent composite levels,
389/// not a certified error bound. Pressure/resultant cancellation is normalized
390/// against the integrated absolute traction magnitude when the resultant is
391/// smaller than `1e-12` of that physical scale.
392pub fn integrate_adaptive_traction<F: Fn(Vector, Vector) -> Vector>(
393    space: &AnalysisSpace,
394    exterior: &BTreeSet<BlockSide>,
395    patches: &[LoadPatch],
396    order: usize,
397    config: SurfaceQuadrature,
398    traction: F,
399) -> Result<AdaptiveSurfaceLoad, &'static str> {
400    integrate_surface_traction(
401        space,
402        exterior,
403        patches,
404        order,
405        (config, SurfaceSplit::Isotropic),
406        traction,
407    )
408}

410/// Integrate using the selected empirical refinement strategy.
411pub fn integrate_surface_traction<F: Fn(Vector, Vector) -> Vector>(
412    space: &AnalysisSpace,
413    exterior: &BTreeSet<BlockSide>,
414    patches: &[LoadPatch],
415    order: usize,
416    options: (SurfaceQuadrature, SurfaceSplit),
417    traction: F,
418) -> Result<AdaptiveSurfaceLoad, &'static str> {
419    let (config, strategy) = options;
420    config.validate()?;
421    let quadrature = gauss(order)?;
422    if patches.is_empty() {
423        return Err("empty traction selection");
424    }
425    let mut known = BTreeMap::<&str, &AnalysisBlock>::new();
426    for block in &space.blocks {
427        if known.insert(&block.id, block).is_some() {
428            return Err("duplicate analysis block");
429        }
430    }
431    let force_len = space
432        .ndof
433        .checked_mul(3)
434        .ok_or("force vector dimension overflow")?;
435    let mut result = SurfaceLoad {
436        nodal_force_n: vec![0.0; force_len],
437        area_mm2: 0.0,
438        resultant_n: [0.0; 3],
439        moment_n_mm: [0.0; 3],
440    };
441    let mut used = BTreeMap::<BlockSide, Vec<Rect>>::new();
442    let mut validated_patches = Vec::with_capacity(patches.len());
443    for patch in patches {
444        let face = &patch.side;
445        if face.axis >= 3 || !exterior.contains(face) {
446            return Err("traction requires a certified exterior side");
447        }
448        let block = known
449            .get(face.block.as_str())
450            .ok_or("unknown traction block")?;
451        if block.dofs.len() != 27 || block.dofs.iter().any(|&d| d >= space.ndof) {
452            return Err("invalid traction block DOFs");
453        }
454        let rect = patch_rect(patch)?;
455        let old_rects = used.entry(face.clone()).or_default();
456        if old_rects
457            .iter()
458            .any(|old| (0..2).all(|i| rect[i][0] < old[i][1] && old[i][0] < rect[i][1]))
459        {
460            return Err("overlapping traction patches");
461        }
462        old_rects.push(rect);
463        validated_patches.push((face.clone(), *block, rect));
464    }
465    let mut panel_evaluations = 0usize;
466    let mut deepest = 0u8;
467    let mut patch_depths = Vec::with_capacity(patches.len());
468    let mut maximum_error = 0.0_f64;
469
470    for (face, block, rect) in validated_patches {
471        if panel_evaluations >= config.max_panels {
472            return Err("adaptive surface quadrature panel budget exceeded");
473        }
474        let first = integrate_panel(block, &face, rect, &quadrature, &traction)?;
475        panel_evaluations = panel_evaluations
476            .checked_add(1)
477            .ok_or("surface panel count overflow")?;
478        let (accepted, depth, error) = if strategy == SurfaceSplit::Directional {
479            directional_patch(rect, first, config, &mut panel_evaluations, |r| {
480                integrate_panel(block, &face, r, &quadrature, &traction)
481            })?
482        } else {
483            let mut current = vec![(rect, first)];
484            let mut current_depth = 0u8;
485            loop {
486                let next_len = current
487                    .len()
488                    .checked_mul(4)
489                    .ok_or("surface panel count overflow")?;
490                if next_len > config.max_panels.saturating_sub(panel_evaluations) {
491                    return Err("adaptive surface quadrature panel budget exceeded");
492                }
493                let mut fine_panels = Vec::with_capacity(next_len);
494                for (parent_rect, _) in &current {
495                    for child in split(*parent_rect) {
496                        let integral =
497                            integrate_panel(block, &face, child, &quadrature, &traction)?;
498                        fine_panels.push((child, integral));
499                    }
500                }
501                panel_evaluations = panel_evaluations
502                    .checked_add(next_len)
503                    .ok_or("surface panel count overflow")?;
504                let coarse_sum = sum_panels(&current);
505                let fine_sum = sum_panels(&fine_panels);
506                let error = estimate_error(&coarse_sum, &fine_sum)?;
507                if error <= config.relative_tolerance {
508                    break (fine_sum, [current_depth + 1; 2], error);
509                }
510                if current_depth + 1 >= config.max_depth {
511                    return Err(
512                        "adaptive surface quadrature tolerance not met within depth budget",
513                    );
514                }
515                current = fine_panels;
516                current_depth += 1;
517            }
518        };
519        deepest = deepest.max(depth[0]).max(depth[1]);
520        patch_depths.push(depth);
521        maximum_error = maximum_error.max(error);
522        result.area_mm2 += accepted.area;
523        for c in 0..3 {
524            result.resultant_n[c] += accepted.resultant[c];
525            result.moment_n_mm[c] += accepted.moment[c];
526        }
527        for (&dof, force) in block.dofs.iter().zip(&accepted.nodal) {
528            for (c, &component) in force.iter().enumerate() {
529                result.nodal_force_n[3 * dof + c] += component;
530            }
531        }
532    }
533    if !result.area_mm2.is_finite()
534        || result
535            .nodal_force_n
536            .iter()
537            .chain(&result.resultant_n)
538            .chain(&result.moment_n_mm)
539            .any(|x| !x.is_finite())
540    {
541        return Err("nonfinite assembled adaptive surface load");
542    }
543    Ok(AdaptiveSurfaceLoad {
544        load: result,
545        panels: panel_evaluations,
546        patch_depths,
547        deepest,
548        maximum_estimated_relative_error: maximum_error,
549    })
