1//! Exact closure of isoparametric cuts over oriented tensor-product interfaces.
2//!
3//! A cut is represented by a rational parameter on one block axis.  On every
4//! incident complete face, its parameter is transported through the declared
5//! axis permutation and reversal.  The module refuses a crossing that is not
6//! declared isoparametric, and checks the full child count before allocation.
7

11type HomogeneousPoint = [i128; 4];
12type SplitNet = (Vec<HomogeneousPoint>, Vec<HomogeneousPoint>);

14#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
15pub struct CutParameter {
16    pub numerator: i128,
17    pub denominator: i128,
18}
19
20impl CutParameter {
21    pub fn new(numerator: i128, denominator: i128) -> Result<Self, &'static str> {
22        if denominator <= 0 || numerator <= 0 || numerator >= denominator {
23            return Err("mapped_cut_parameter_outside_open_unit_interval");
24        }
25        let g = gcd(numerator as u128, denominator as u128) as i128;
26        Ok(Self {
27            numerator: numerator / g,
28            denominator: denominator / g,
29        })
30    }
31
32    fn reversed(self) -> Result<Self, &'static str> {
33        Self::new(
34            self.denominator
35                .checked_sub(self.numerator)
36                .ok_or("mapped_cut_integer_overflow")?,
37            self.denominator,
38        )
39    }
40}

42fn gcd(mut a: u128, mut b: u128) -> u128 {
43    while b != 0 {
44        let r = a % b;
45        a = b;
46        b = r;
47    }
48    a.max(1)
49}

51/// A complete-face incidence. `a_to_b[t]` gives the destination block axis
52/// corresponding to the `t`th ascending tangential axis of face A, and whether
53/// that coordinate is reversed. The face normal axes must be omitted.
54///
55/// This topology-level primitive trusts `isoparametric_admitted`; production
56/// callers should populate it from exact incidence and recertify the assembled
57/// children and interfaces after closure.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub struct FaceMap {
60    pub a_block: usize,
61    pub a_normal_axis: usize,
62    pub a_upper: bool,
63    pub b_block: usize,
64    pub b_normal_axis: usize,
65    pub b_upper: bool,
66    pub a_to_b: [(usize, bool); 2],
67    /// Set only after exact homogeneous traces and orientation were admitted.
68    pub isoparametric_admitted: bool,
69    pub provenance: String,
70}

115fn checked_mul(a: i128, b: i128) -> Result<i128, &'static str> {
116    a.checked_mul(b).ok_or("mapped_cut_integer_overflow")
117}

119fn blend(a: [i128; 4], b: [i128; 4], n: i128, d: i128) -> Result<[i128; 4], &'static str> {
120    let mut out = [0; 4];
121    for c in 0..4 {
122        out[c] = checked_mul(d - n, a[c])?
123            .checked_add(checked_mul(n, b[c])?)
124            .ok_or("mapped_cut_integer_overflow")?;
125    }
126    Ok(out)
127}

129/// Split one homogeneous Bernstein net at `t`, preserving a common integer
130/// scale on each child. This is exact homogeneous de Casteljau subdivision.
131fn split_curve(net: &[[i64; 4]], t: CutParameter) -> Result<SplitNet, &'static str> {
132    let degree = net.len().checked_sub(1).ok_or("mapped_cut_empty_curve")?;
133    if degree == 0 || degree > 8 || net.iter().any(|p| p[3] <= 0) {
134        return Err("mapped_cut_unsupported_curve_degree_or_weight");
135    }
136    let d = t.denominator;
137    let p = t.numerator;
138    let mut rows = vec![net.iter().map(|x| x.map(i128::from)).collect::<Vec<_>>()];
139    for level in 1..=degree {
140        let previous = &rows[level - 1];
141        let mut row = Vec::with_capacity(degree + 1 - level);
142        for i in 0..(degree + 1 - level) {
143            row.push(blend(previous[i], previous[i + 1], p, d)?);
144        }
145        rows.push(row);
146    }
147    let mut left = Vec::with_capacity(degree + 1);
148    let mut right = Vec::with_capacity(degree + 1);
149    for k in 0..=degree {
150        left.push(scale_point(rows[k][0], pow(d, degree - k)?)?);
151        right.push(scale_point(rows[degree - k][k], pow(d, k)?)?);
152    }
153    Ok((left, right))
154}

156fn pow(x: i128, n: usize) -> Result<i128, &'static str> {
157    let mut out = 1_i128;
158    for _ in 0..n {
159        out = checked_mul(out, x)?;
160    }
161    Ok(out)
162}

164fn scale_point(p: [i128; 4], s: i128) -> Result<[i128; 4], &'static str> {
165    let mut q = [0; 4];
166    for c in 0..4 {
167        q[c] = checked_mul(p[c], s)?;
168    }
169    Ok(q)
170}

172fn to_i64_net(mut net: Vec<[i128; 4]>) -> Result<Vec<[i64; 4]>, &'static str> {
173    let mut g = 0_u128;
174    for p in &net {
175        for x in p {
176            g = gcd(g, x.unsigned_abs());
177        }
178    }
179    let g = i128::try_from(g.max(1)).map_err(|_| "mapped_cut_integer_overflow")?;
180    net.drain(..)
181        .map(|p| {
182            Ok([
183                i64::try_from(p[0] / g).map_err(|_| "mapped_cut_child_control_exceeds_i64")?,
184                i64::try_from(p[1] / g).map_err(|_| "mapped_cut_child_control_exceeds_i64")?,
185                i64::try_from(p[2] / g).map_err(|_| "mapped_cut_child_control_exceeds_i64")?,
186                i64::try_from(p[3] / g).map_err(|_| "mapped_cut_child_control_exceeds_i64")?,
187            ])
188        })
189        .collect()
190}

192fn shape(block: &BracketBlock) -> Result<[usize; 3], &'static str> {
193    if block.degrees.iter().any(|d| *d == 0 || *d > 8) {
194        return Err("mapped_cut_invalid_block_shape");
195    }
196    let s = block
197        .degrees
198        .map(|d| d.checked_add(1).ok_or("mapped_cut_invalid_block_shape"));
199    let s = [s[0]?, s[1]?, s[2]?];
200    if block.controls.len() != s.iter().product::<usize>() {
201        return Err("mapped_cut_invalid_block_shape");
202    }
203    if block.controls.iter().any(|p| p[3] <= 0) {
204        return Err("mapped_cut_nonpositive_weight");
205    }
206    Ok(s)
207}
208
209fn idx(s: [usize; 3], p: [usize; 3]) -> usize {
210    (p[0] * s[1] + p[1]) * s[2] + p[2]
211}

213fn split_block(
214    block: &BracketBlock,
215    axis: usize,
216    t: CutParameter,
217) -> Result<(BracketBlock, BracketBlock), &'static str> {
218    let s = shape(block)?;
219    if axis > 2 {
220        return Err("mapped_cut_invalid_axis");
221    }
222    let mut left = vec![[0_i128; 4]; block.controls.len()];
223    let mut right = vec![[0_i128; 4]; block.controls.len()];
224    let other: Vec<usize> = (0..3).filter(|&a| a != axis).collect();
225    for i in 0..s[other[0]] {
226        for j in 0..s[other[1]] {
227            let mut line = Vec::with_capacity(s[axis]);
228            for k in 0..s[axis] {
229                let mut p = [0; 3];
230                p[other[0]] = i;
231                p[other[1]] = j;
232                p[axis] = k;
233                line.push(block.controls[idx(s, p)]);
234            }
235            let (l, r) = split_curve(&line, t)?;
236            for k in 0..s[axis] {
237                let mut p = [0; 3];
238                p[other[0]] = i;
239                p[other[1]] = j;
240                p[axis] = k;
241                left[idx(s, p)] = l[k];
242                right[idx(s, p)] = r[k];
243            }
244        }
245    }
246    let child = |suffix: &str, controls| BracketBlock {
247        id: format!("{}{}", block.id, suffix),
248        degrees: block.degrees,
249        controls,
250    };
251    Ok((
252        child("_cut_lo", to_i64_net(left)?),
253        child("_cut_hi", to_i64_net(right)?),
254    ))
255}

257fn sort_cuts(cuts: &mut Vec<CutParameter>) -> Result<(), &'static str> {
258    cuts.sort_by(|a, b| {
259        crate::constraint_system::Ratio::new(a.numerator, a.denominator)
260            .expect("validated cut parameter")
261            .cmp_value(
262                &crate::constraint_system::Ratio::new(b.numerator, b.denominator)
263                    .expect("validated cut parameter"),
264            )
265    });
266    cuts.dedup();
267    Ok(())
268}

270fn validate_map(m: &FaceMap, blocks: usize) -> Result<(), &'static str> {
271    if m.a_block >= blocks
272        || m.b_block >= blocks
273        || m.a_block == m.b_block
274        || m.a_normal_axis > 2
275        || m.b_normal_axis > 2
276    {
277        return Err("mapped_cut_invalid_face_map");
278    }
279    let a_tan: Vec<_> = (0..3).filter(|&x| x != m.a_normal_axis).collect();
280    let b_tan: BTreeSet<_> = (0..3).filter(|&x| x != m.b_normal_axis).collect();
281    let mapped: BTreeSet<_> = m.a_to_b.iter().map(|x| x.0).collect();
282    if mapped != b_tan
283        || m.a_to_b
284            .iter()
285            .enumerate()
286            .any(|(i, x)| x.0 == m.b_normal_axis || a_tan[i] == m.a_normal_axis)
287    {
288        return Err("mapped_cut_invalid_axis_permutation");
289    }
290    Ok(())
291}

293/// Close cuts over the supplied complete-face graph, then subdivide all
294/// affected blocks. The caller supplies only faces whose traces were already
295/// admitted by exact homogeneous comparison. Positive Jacobian certificates
296/// transfer to each child by positive affine reparameterization.
297pub fn propagate_exact_cuts(
298    blocks: &[BracketBlock],
299    faces: &[FaceMap],
300    requests: &[CutRequest],
301    budget: usize,
302) -> Result<PropagatedPlan, &'static str> {
303    if blocks.is_empty() || requests.is_empty() || budget == 0 {
304        return Err("mapped_cut_empty_input_or_budget");
305    }
306    for m in faces {
307        validate_map(m, blocks.len())?;
308    }
309    for block in blocks {
310        shape(block)?;
311    }
312    let mut queue = VecDeque::new();
313    let mut seen = BTreeSet::new();
314    let mut provenance: BTreeMap<Key, BTreeSet<String>> = BTreeMap::new();
315    for r in requests {
316        if r.block >= blocks.len() || r.axis > 2 {
317            return Err("mapped_cut_request_out_of_range");
318        }
319        shape(&blocks[r.block])?;
320        if CutParameter::new(r.parameter.numerator, r.parameter.denominator)? != r.parameter {
321            return Err("mapped_cut_noncanonical_parameter");
322        }
323        queue.push_back((
324            Key {
325                block: r.block,
326                axis: r.axis,
327                parameter: r.parameter,
328            },
329            r.provenance.clone(),
330        ));
331    }
332    while let Some((key, source)) = queue.pop_front() {
333        provenance.entry(key).or_default().insert(source.clone());
334        if !seen.insert((key, source.clone())) {
335            continue;
336        }
337        for face in faces {
338            let (source_block, source_normal, destination_block, destination_normal, mapping) =
339                if face.a_block == key.block {
340                    (
341                        face.a_block,
342                        face.a_normal_axis,
343                        face.b_block,
344                        face.b_normal_axis,
345                        face.a_to_b,
346                    )
347                } else if face.b_block == key.block {
348                    let mut inverse = [(0, false); 2];
349                    for (a_index, &(b_axis, reversed)) in face.a_to_b.iter().enumerate() {
350                        let b_index = (0..3)
351                            .filter(|&a| a != face.b_normal_axis)
352                            .position(|a| a == b_axis)
353                            .ok_or("mapped_cut_invalid_axis_permutation")?;
354                        inverse[b_index] = (x_axis_for(face.a_normal_axis, a_index), reversed);
355                    }
356                    (
357                        face.b_block,
358                        face.b_normal_axis,
359                        face.a_block,
360                        face.a_normal_axis,
361                        inverse,
362                    )
363                } else {
364                    continue;
365                };
366            let _ = (source_block, destination_block);
367            if key.axis == source_normal {
368                continue;
369            }
370            if !face.isoparametric_admitted {
371                return Err("mapped_cut_incompatible_trace");
372            }
373            let source_tangent_index = (0..3)
374                .filter(|&a| a != source_normal)
375                .position(|a| a == key.axis)
376                .ok_or("mapped_cut_invalid_axis_permutation")?;
377            let (dest_axis, reversed) = mapping[source_tangent_index];
378            if dest_axis == destination_normal {
379                return Err("mapped_cut_invalid_axis_permutation");
380            }
381            let parameter = if reversed {
382                key.parameter.reversed()?
383            } else {
384                key.parameter
385            };
386            queue.push_back((
387                Key {
388                    block: destination_block,
389                    axis: dest_axis,
390                    parameter,
391                },
392                source.clone(),
393            ));
394        }
395    }
396    let mut per_block: Vec<[Vec<CutParameter>; 3]> = (0..blocks.len())
397        .map(|_| std::array::from_fn(|_| Vec::new()))
398        .collect();
399    for key in provenance.keys() {
400        per_block[key.block][key.axis].push(key.parameter);
401    }
402    let mut predicted = 0_usize;
403    for cuts in &mut per_block {
404        for axis_cuts in cuts.iter_mut() {
405            sort_cuts(axis_cuts)?;
406        }
407        let n = cuts
408            .iter()
409            .try_fold(1_usize, |n, c| n.checked_mul(c.len() + 1))
410            .ok_or("mapped_cut_block_budget")?;
411        predicted = predicted.checked_add(n).ok_or("mapped_cut_block_budget")?;
412    }
413    if predicted > budget {
414        return Err("mapped_cut_block_budget");
415    }
416
417    let propagated_cut_count = provenance.len();
418    let mut output = Vec::with_capacity(predicted);
419    let mut source_to_children = Vec::with_capacity(blocks.len());
420    let mut boundary_children = Vec::new();
421    for (source_index, block) in blocks.iter().enumerate() {
422        let mut pieces = vec![(block.clone(), [[true, true]; 3])];
423        for (axis, axis_cuts) in per_block[source_index].iter().enumerate() {
424            let mut cuts = axis_cuts.clone();
425            sort_cuts(&mut cuts)?;
426            if cuts.is_empty() {
427                continue;
428            }
429            let mut next = Vec::new();
430            for (mut piece, mut piece_boundary) in pieces {
431                let mut previous = CutParameter {
432                    numerator: 0,
433                    denominator: 1,
434                };
435                for cut in &cuts {
436                    let local_num = cut
437                        .numerator
438                        .checked_mul(previous.denominator)
439                        .and_then(|x| {
440                            x.checked_sub(previous.numerator.checked_mul(cut.denominator)?)
441                        })
442                        .ok_or("mapped_cut_integer_overflow")?;
443                    let local_den = cut
444                        .denominator
445                        .checked_mul(
446                            previous
447                                .denominator
448                                .checked_sub(previous.numerator)
449                                .ok_or("mapped_cut_integer_overflow")?,
450                        )
451                        .ok_or("mapped_cut_integer_overflow")?;
452                    let local = CutParameter::new(local_num, local_den)?;
453                    let prefix = piece.id.clone();
454                    let (mut left, right) = split_block(&piece, axis, local)?;
455                    left.id = format!(
456                        "{}_a{}_{}of{}",
457                        prefix, axis, cut.numerator, cut.denominator
458                    );
459                    let mut left_boundary = piece_boundary;
460                    left_boundary[axis][1] = false;
461                    next.push((left, left_boundary));
462                    piece_boundary[axis][0] = false;
463                    piece = right;
464                    piece.id = format!(
465                        "{}_a{}_after{}of{}",
466                        prefix, axis, cut.numerator, cut.denominator
467                    );
468                    previous = *cut;
469                }
470                next.push((piece, piece_boundary));
471            }
472            pieces = next;
473        }
474        let ids = pieces.iter().map(|(b, _)| b.id.clone()).collect::<Vec<_>>();
475        for (child, boundary) in pieces {
476            let child_index = output.len();
477            for (axis, sides) in boundary.iter().enumerate() {
478                for upper in [false, true] {
479                    if sides[usize::from(upper)] {
480                        boundary_children.push(BoundaryChild {
481                            source_block: source_index,
482                            child_block: child_index,
483                            axis,
484                            upper,
485                        });
486                    }
487                }
488            }
489            output.push(child);
490        }
491        source_to_children.push((block.id.clone(), ids));
492    }
493    let cuts = provenance
494        .into_iter()
495        .map(|(key, values)| PropagatedCut {
496            block: key.block,
497            axis: key.axis,
498            parameter: key.parameter,
499            provenance: values.into_iter().collect(),
500        })
501        .collect::<Vec<_>>();
502    Ok(PropagatedPlan {
503        blocks: output,
504        cuts,
505        source_to_children,
506        boundary_children,
507        requested_cut_count: requests.len(),
508        propagated_cut_count,
509        child_count: predicted,
510        budget,
511    })
512}

514fn x_axis_for(normal: usize, tangent_index: usize) -> usize {
515    (0..3)
516        .filter(|&a| a != normal)
517        .nth(tangent_index)
518        .unwrap_or(0)
