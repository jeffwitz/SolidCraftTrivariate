11fn add(a: i128, b: i128) -> Result<i128, &'static str> {
12    a.checked_add(b).ok_or("crescent coefficient overflow")
13}
14fn sub(a: i128, b: i128) -> Result<i128, &'static str> {
15    a.checked_sub(b).ok_or("crescent coefficient overflow")
16}
17fn mul(a: i128, b: i128) -> Result<i128, &'static str> {
18    a.checked_mul(b).ok_or("crescent coefficient overflow")
19}
20fn out(value: i128) -> Result<i64, &'static str> {
21    i64::try_from(value).map_err(|_| "crescent control exceeds i64")
22}

29fn certify_patch_planarity(controls: &[[i64; 4]]) -> Result<(), &'static str> {
30    if controls.len() != 18 {
31        return Err("invalid crescent patch control count");
32    }
33    let (pairs, remainder) = controls.as_chunks::<2>();
34    if !remainder.is_empty() {
35        return Err("invalid crescent patch control count");
36    }
37    let planar = pairs
38        .iter()
39        .map(|pair| [pair[0][0], pair[0][1], pair[0][3]])
40        .collect::<Vec<_>>();
41    crate::planar::certify_planar([2, 2], &planar)
42        .map(|_| ())
43        .map_err(|error| match error {
44            crate::planar::PlanarError::Invalid => "crescent patch has nonpositive Jacobian",
45            crate::planar::PlanarError::Indeterminate => "crescent patch regularity indeterminate",
46        })
47}

49fn source_arc(blocks: &[BracketBlock], id: &str) -> Result<[[i128; 4]; 3], &'static str> {
50    let block = blocks
51        .iter()
52        .find(|b| b.id == id)
53        .ok_or("missing G7 source sector")?;
54    if block.degrees != [2, 1, 1] || block.controls.len() != 12 {
55        return Err("invalid G7 source sector");
56    }
57    let mut arc = [[0_i128; 4]; 3];
58    for (i, control) in arc.iter_mut().enumerate() {
59        let p = block.controls[(i * 2 + 1) * 2];
60        if p[3] <= 0 {
61            return Err("nonpositive G7 source weight");
62        }
63        *control = p.map(i128::from);
64    }
65    Ok(arc)
66}

73fn right_circle_to_bc(mut arc: [[i128; 4]; 3]) -> Result<[[i128; 4]; 3], &'static str> {
74    // The source arc runs from (3,0) to (2,s). Reflect about x=1 to get B->C.
75    for p in &mut arc {
76        p[0] = sub(mul(2, p[3])?, p[0])?;
77        for component in p.iter_mut() {
78            *component = mul(3, *component)?;
79        }
80    }
81    Ok(arc)
82}

84type Curve = [[i128; 4]; 3];

87fn scale_curve(curve: Curve, scale: i128) -> Result<Curve, &'static str> {
88    let mut result = curve;
89    for p in &mut result {
90        for component in p.iter_mut() {
91            *component = mul(*component, scale)?;
92        }
93    }
94    Ok(result)
95}

97fn split_half(curve: Curve, scale: i128) -> Result<(Curve, Curve), &'static str> {
98    // The de Casteljau midpoint has denominator four for quadratic Bernstein data.
99    // `scale` must be divisible by four; both child nets use one common multiplier.
100    if scale % 4 != 0 {
101        return Err("split scale must clear the quadratic midpoint denominator");
102    }
103    let factor = scale / 4;
104    let mut left = [[0_i128; 4]; 3];
105    let mut right = [[0_i128; 4]; 3];
106    for c in 0..4 {
107        left[0][c] = mul(scale, curve[0][c])?;
108        left[1][c] = mul(mul(2, factor)?, add(curve[0][c], curve[1][c])?)?;
109        let midpoint = add(add(curve[0][c], mul(2, curve[1][c])?)?, curve[2][c])?;
110        left[2][c] = mul(factor, midpoint)?;
111        right[0][c] = left[2][c];
112        right[1][c] = mul(mul(2, factor)?, add(curve[1][c], curve[2][c])?)?;
113        right[2][c] = mul(scale, curve[2][c])?;
114    }
115    Ok((left, right))
116}

118fn line_curve(start: Vertex, end: Vertex) -> Result<Curve, &'static str> {
119    let mut result = [[0_i128; 4]; 3];
120    result[0] = start;
121    result[2] = end;
122    for c in 0..4 {
123        let sum = add(start[c], end[c])?;
124        if sum % 2 != 0 {
125            return Err("line elevation is not integral at common scale");
126        }
127        result[1][c] = sum / 2;
128    }
129    Ok(result)
130}

132fn patch_controls(
133    bottom: Curve,
134    right: Curve,
135    top: Curve,
136    left: Curve,
137) -> Result<Vec<[i64; 4]>, &'static str> {
138    if bottom[0] != left[0] || bottom[2] != right[0] || right[2] != top[2] || top[0] != left[2] {
139        return Err("crescent patch corner traces disagree");
140    }
141    let mut net = [[[0_i128; 4]; 3]; 3];
142    for i in 0..3 {
143        net[i][0] = bottom[i];
144        net[i][2] = top[i];
145    }
146    net[0].copy_from_slice(&left);
147    net[2].copy_from_slice(&right);
148    // Homogeneous Coons center for a biquadratic patch with four quadratic edges.
149    let mut center = [0_i128; 4];
150    for (c, component) in center.iter_mut().enumerate() {
151        let edge_sum = add(
152            add(net[1][0][c], net[1][2][c])?,
153            add(net[0][1][c], net[2][1][c])?,
154        )?;
155        let corner_sum = add(
156            add(net[0][0][c], net[0][2][c])?,
157            add(net[2][0][c], net[2][2][c])?,
158        )?;
159        *component = sub(mul(2, edge_sum)?, corner_sum)?;
160    }
161    net[1][1] = center;
162    if net.iter().flatten().any(|p| p[3] <= 0) {
163        return Err("nonpositive crescent patch weight");
164    }
165    let mut controls = Vec::with_capacity(18);
166    for row in net {
167        for point in row {
168            for z in [0_i128, 8] {
169                let mut p = point;
170                p[2] = mul(z, p[3])?;
171                controls.push(
172                    p.map(out)
173                        .into_iter()
174                        .collect::<Result<Vec<_>, _>>()?
175                        .try_into()
176                        .map_err(|_| "invalid crescent point")?,
177                );
178            }
179        }
180    }
181    Ok(controls)
182}

184fn upper_crescent_patches(g7: &[BracketBlock]) -> Result<[BracketBlock; 3], &'static str> {
185    // Pull the two exposed left-circle arcs and transform the right-circle arc.
186    // The boundary curves receive one common scale per source edge; all seam
187    // traces below are degree-elevated from the same homogeneous vertices.
188    let arc_ad = scale_curve(
189        reversed(source_arc(g7, "overlap_bore_left_sector_01_z0_8")?),
190        8,
191    )?; // A->D
192    let arc_dc = scale_curve(
193        reversed(source_arc(g7, "overlap_bore_left_sector_00_z0_8")?),
194        8,
195    )?; // D->C
196    let arc_bc = right_circle_to_bc(source_arc(g7, "overlap_bore_right_sector_02_z0_8")?)?; // B->C
197    let (arc_bf, arc_fc) = split_half(arc_bc, 8)?;
198    let [a, d] = [arc_ad[0], arc_ad[2]];
199    let [d2, c] = [arc_dc[0], arc_dc[2]];
200    let [b, f] = [arc_bf[0], arc_bf[2]];
201    let [f2, c2] = [arc_fc[0], arc_fc[2]];
202    if d != d2 || f != f2 || c != c2 {
203        return Err("crescent arc endpoints disagree");
204    }
205    let base = a[3] / 24;
206    if base <= 0 || mul(24, base)? != a[3] {
207        return Err("invalid G7 scale for crescent vertices");
208    }
209    if b[3] != mul(72, base)? || c[3] != mul(96, base)? || f[3] != mul(78, base)? {
210        return Err("crescent source arc weights changed");
211    }
212    let e = [mul(-2, mul(48, base)?)?, 0, 0, mul(48, base)?];
213    let o_weight = mul(64, base)?;
214    let ox = mul(-7, o_weight)?;
215    let oy = mul(7, o_weight)?;
216    if ox % 4 != 0 || oy % 8 != 0 {
217        return Err("interior crescent vertex is not integral");
218    }
219    let o = [ox / 4, oy / 8, 0, o_weight];
220    let line = |p, q| line_curve(p, q);
221    let upper = [
222        BracketBlock {
223            id: "g10_crescent_upper_a".into(),
224            degrees: [2, 2, 1],
225            controls: patch_controls(line(a, e)?, line(e, o)?, line(d, o)?, arc_ad)?,
226        },
227        BracketBlock {
228            id: "g10_crescent_upper_b".into(),
229            degrees: [2, 2, 1],
230            controls: patch_controls(line(e, b)?, arc_bf, line(o, f)?, line(e, o)?)?,
231        },
232        BracketBlock {
233            id: "g10_crescent_upper_c".into(),
234            degrees: [2, 2, 1],
235            controls: patch_controls(line(o, f)?, arc_fc, arc_dc, line(o, d)?)?,
236        },
237    ];
238    for patch in &upper {
239        if patch.controls.iter().any(|p| p[3] <= 0) {
240            return Err("nonpositive crescent patch weight");
241        }
242    }
243    Ok(upper)
244}

246fn lower_crescent_patches(upper: &[BracketBlock; 3]) -> Result<[BracketBlock; 3], &'static str> {
247    let mut lower: [BracketBlock; 3] = std::array::from_fn(|i| BracketBlock {
248        id: format!("g10_crescent_lower_{}", ["a", "b", "c"][i]),
249        degrees: [2, 2, 1],
250        controls: vec![[0; 4]; 18],
251    });
252    // Reflection y -> -y followed by reversing u preserves positive orientation.
253    for (source, target) in upper.iter().zip(lower.iter_mut()) {
254        let (source_rows, source_tail) = source.controls.as_chunks::<6>();
255        let (target_rows, target_tail) = target.controls.as_chunks_mut::<6>();
256        if !source_tail.is_empty() || !target_tail.is_empty() {
257            return Err("invalid crescent patch controls");
258        }
259        for (source_row, target_row) in source_rows.iter().rev().zip(target_rows) {
260            for (src, dst) in source_row.iter().zip(target_row.iter_mut()) {
261                *dst = *src;
262                dst[1] = dst[1]
263                    .checked_neg()
264                    .ok_or("crescent coefficient overflow")?;
265            }
266        }
267    }
268    Ok(lower)
269}

271fn extrude_layer(
272    block: &BracketBlock,
273    bottom_tick: i32,
274    top_tick: i32,
275    id: String,
276) -> Result<BracketBlock, &'static str> {
277    if bottom_tick >= top_tick {
278        return Err("zero height crescent layer");
279    }
280    let mut result = block.clone();
281    result.id = id;
282    for (control_index, p) in result.controls.iter_mut().enumerate() {
283        let w = i128::from(p[3]);
284        let upper = control_index % 2 == 1;
285        let z_tick = if upper { top_tick } else { bottom_tick };
286        let n = mul(i128::from(z_tick), w)?;
287        if n % 2 != 0 {
288            return Err("half-mm layer is not exactly representable");
289        }
290        p[2] = out(n / 2)?;
291    }
292    Ok(result)
293}

299fn source_sector_id(id: &str) -> Result<(&'static str, &str), &'static str> {
300    let side = if id.contains("_right_sector_") {
301        "right"
302    } else if id.contains("_left_sector_") {
303        "left"
304    } else {
305        return Err("invalid G7 sector id");
306    };
307    let (_, tail) = id.split_once("_sector_").ok_or("invalid G7 sector id")?;
308    let (sector, _) = tail.split_once("_z0_8").ok_or("invalid G7 sector id")?;
309    Ok((side, sector))
310}

312/// Build the plate with a through bore and an intersecting blind pocket.
313/// `depth_half_mm` is a half-millimetre integer in the closed range 0..=16.
314/// The returned collection contains only positive-height layers.
315pub fn crescent_pocket_blocks(depth_half_mm: i32) -> Result<Vec<BracketBlock>, &'static str> {
316    if !(0..=MAX_DEPTH_TICKS).contains(&depth_half_mm) {
317        return Err("invalid crescent pocket depth");
318    }
319    let g7 = overlap_blocks()?;
320    if g7.len() != 8 {
321        return Err("invalid G7 sector count");
322    }
323    let split_tick = MAX_DEPTH_TICKS - depth_half_mm;
324    let crescent_patches = if split_tick > 0 {
325        let upper = upper_crescent_patches(&g7)?;
326        let lower = lower_crescent_patches(&upper)?;
327        Some((upper, lower))
328    } else {
329        None
330    };
331    let mut result = Vec::with_capacity(22);
332    if split_tick > 0 {
333        for source in &g7 {
334            let mut renamed = source.clone();
335            let (side, sector) = source_sector_id(&source.id)?;
336            renamed.id = layer_id(&format!("lower_{side}_sector_{sector}"), "z0_split");
337            result.push(extrude_layer(&renamed, 0, split_tick, renamed.id.clone())?);
338        }
339        let (upper, lower) = crescent_patches.unwrap();
340        for patch in upper.into_iter().chain(lower) {
341            result.push(extrude_layer(&patch, 0, split_tick, patch.id.clone())?);
342        }
343    }
344    if split_tick < MAX_DEPTH_TICKS {
345        for source in &g7 {
346            let mut renamed = source.clone();
347            let (side, sector) = source_sector_id(&source.id)?;
348            renamed.id = layer_id(&format!("upper_{side}_sector_{sector}"), "split_8");
349            result.push(extrude_layer(
350                &renamed,
351                split_tick,
352                MAX_DEPTH_TICKS,
353                renamed.id.clone(),
354            )?);
355        }
356    }
357    Ok(result)
358}

364    #[test]
365    fn g10_depth_regimes_omit_zero_height_layers() {
366        let zero = crescent_pocket_blocks(0).unwrap();
367        let full = crescent_pocket_blocks(16).unwrap();
368        let blind = crescent_pocket_blocks(10).unwrap();
369        assert_eq!(zero.len(), 14);
370        assert_eq!(full.len(), 8);
371        assert_eq!(blind.len(), 22);
372        assert!(zero.iter().all(|b| !b.id.contains("upper_right")));
373        assert!(full.iter().all(|b| !b.id.contains("lower")));
374        assert!(blind
375            .iter()
376            .all(|b| b.controls.len() == b.degrees.iter().map(|d| d + 1).product::<usize>()));
377    }

379    #[test]
380    fn six_crescent_patches_have_positive_weights_and_exact_shared_seams() {
381        let g7 = overlap_blocks().unwrap();
382        let patches = upper_crescent_patches(&g7).unwrap();
383        assert_eq!(
384            patches.iter().map(|b| b.id.as_str()).collect::<Vec<_>>(),
385            vec![
386                "g10_crescent_upper_a",
387                "g10_crescent_upper_b",
388                "g10_crescent_upper_c"
389            ]
390        );
391        assert!(patches.iter().all(|p| p.degrees == [2, 2, 1]
392            && p.controls.len() == 18
393            && p.controls.iter().all(|h| h[3] > 0)));
394        for patch in &patches {
395            certify_patch_planarity(&patch.controls).unwrap();
396        }
397        // The baseline split uses the corner ratios A:B:C:D=1:3:4:4/3.
398        let (a, b, c) = (&patches[0], &patches[1], &patches[2]);
399        let wa = a.controls[index(0, 0, 0)][3];
400        let wb = b.controls[index(2, 0, 0)][3];
401        let wc = c.controls[index(2, 2, 0)][3];
402        let wd = a.controls[index(0, 2, 0)][3];
403        assert_eq!(i128::from(wb), i128::from(wa) * 3);
404        assert_eq!(i128::from(wc), i128::from(wa) * 4);
405        assert_eq!(i128::from(wd) * 3, i128::from(wa) * 4);
406        // A-E-O and O-F seams are exactly the same elevated homogeneous lines.
407        for q in 0..3 {
408            assert_eq!(a.controls[index(2, q, 0)], b.controls[index(0, q, 0)]);
409            assert_eq!(b.controls[index(q, 2, 0)], c.controls[index(q, 0, 0)]);
410        }
411        let lower = lower_crescent_patches(&patches).unwrap();
412        for (up, down) in patches.iter().zip(&lower) {
413            assert_eq!(down.degrees, up.degrees);
414            certify_patch_planarity(&down.controls).unwrap();
415            let (lower_rows, _) = down.controls.as_chunks::<6>();
416            let (upper_rows, _) = up.controls.as_chunks::<6>();
417            for (lower_row, upper_row) in lower_rows.iter().zip(upper_rows.iter().rev()) {
418                let (lower_edges, _) = lower_row.as_chunks::<2>();
419                let (upper_edges, _) = upper_row.as_chunks::<2>();
420                for (u, v) in lower_edges.iter().zip(upper_edges.iter()) {
421                    assert_eq!(u[0][0], v[0][0]);
422                    assert_eq!(u[0][1], -v[0][1]);
423                    assert_eq!(u[0][3], v[0][3]);
424                }
425            }
426        }
427    }
