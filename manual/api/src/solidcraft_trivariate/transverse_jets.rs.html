1//! Strong physical gradient matching for independent straight extrusions.
2//! On a vertical face, at fixed axial Bernstein index, all cleared gradient
3//! equations are polynomials in one parameter. Exact unisolvent evaluation
4//! generates their coefficient row space, including the endpoints.
5use crate::{
6    constraint_system::Ratio as Q,
7    incidence::{BlockSide, Interface},
8    inherited_space::{ratio_div, Section},
9    spline_constraints::{add, Result, Row},
10};

12fn mul(a: Q, b: Q) -> Result<Q> {
13    a.checked_mul(b)
14}
15fn diff_product(a: Q, b: Q, c: Q, d: Q) -> Result<Q> {
16    mul(a, b)?.checked_sub(mul(c, d)?)
17}
18pub(crate) fn bernstein(p: usize, t: Q) -> Result<(Vec<Q>, Vec<Q>)> {
19    let s = Q::one().checked_sub(t)?;
20    let mut prev = vec![Q::one()];
21    for level in 1..=p {
22        let mut next = vec![Q::zero(); level + 1];
23        for (i, &x) in prev.iter().enumerate() {
24            next[i] = next[i].checked_add(mul(s, x)?)?;
25            next[i + 1] = next[i + 1].checked_add(mul(t, x)?)?;
26        }
27        if level == p {
28            let mut der = vec![Q::zero(); p + 1];
29            let deg = Q::new(p as i128, 1)?;
30            for (i, &x) in prev.iter().enumerate() {
31                let v = mul(x, deg)?;
32                der[i] = der[i].checked_sub(v)?;
33                der[i + 1] = der[i + 1].checked_add(v)?;
34            }
35            return Ok((next, der));
36        }
37        prev = next;
38    }
39    Err("degree zero transverse direction")
40}

42fn section_gradients(s: &Section, side: &BlockSide, t: Q) -> Result<Vec<[Q; 2]>> {
43    let t_fixed = Q::new(i128::from(side.upper), 1)?;
44    let u = if side.axis == 0 { t_fixed } else { t };
45    let v = if side.axis == 1 { t_fixed } else { t };
46    let (bu, du) = bernstein(s.d[0], u)?;
47    let (bv, dv) = bernstein(s.d[1], v)?;
48    let mut h = [[Q::zero(); 3]; 3]; // point, u derivative, v derivative (X,Y,W)
49    let mut funcs = Vec::new();
50    for i in 0..=s.d[0] {
51        for j in 0..=s.d[1] {
52            let p = s.block.controls[2 * (i * (s.d[1] + 1) + j)];
53            let b = [mul(bu[i], bv[j])?, mul(du[i], bv[j])?, mul(bu[i], dv[j])?];
54            let mut weighted = [Q::zero(); 3];
55            for (k, &f) in b.iter().enumerate() {
56                for (c, &x) in [p[0], p[1], p[3]].iter().enumerate() {
57                    h[k][c] = h[k][c].checked_add(mul(f, Q::new(x.into(), 1)?)?)?;
58                }
59                weighted[k] = mul(f, Q::new(p[3].into(), 1)?)?;
60            }
61            funcs.push(weighted);
62        }
63    }
64    let w = h[0][2];
65    let xu = diff_product(h[1][0], w, h[0][0], h[1][2])?;
66    let xv = diff_product(h[2][0], w, h[0][0], h[2][2])?;
67    let yu = diff_product(h[1][1], w, h[0][1], h[1][2])?;
68    let yv = diff_product(h[2][1], w, h[0][1], h[2][2])?;
69    let det = diff_product(xu, yv, xv, yu)?;
70    if det.numerator() <= 0 {
71        return Err("source section does not have positive orientation");
72    }
73    funcs
74        .into_iter()
75        .map(|b| {
76            let nu = diff_product(b[1], w, b[0], h[1][2])?;
77            let nv = diff_product(b[2], w, b[0], h[2][2])?;
78            Ok([
79                ratio_div(diff_product(nu, yv, nv, yu)?, det)?,
80                ratio_div(diff_product(nv, xu, nu, xv)?, det)?,
81            ])
82        })
83        .collect()
84}

86pub(crate) fn append(
87    link: &Interface,
88    a: &Section,
89    b: &Section,
90    ids: &[usize],
91    rows: &mut Vec<Row>,
92) -> Result<()> {
93    let o = link.orientation;
94    if o.swap || o.flip_second || a.shear != b.shear || a.z != b.z || a.d[2] != b.d[2] {
95        return Err("unsupported transverse axial orientation or interval");
96    }
97    let pa = a.d[1 - link.first.axis];
98    let pb = b.d[1 - link.second.axis];
99    // Each rational map's cleared Jacobian entries have degree <= 2p.
100    // Numerator and determinant of a physical gradient each have degree <= 4p.
101    let bound = 4 * (pa + pb);
102    // Probe endpoints and low-height rationals first. They define the same
103    // polynomial row space and avoid gratuitous denominator growth in elimination.
104    let mut probes = vec![Q::zero(), Q::one()];
105    for d in 2..=bound + 1 {
106        for k in 1..d {
107            let t = Q::new(k as i128, d as i128)?;
108            if t.denominator() == d as i128 {
109                probes.push(t);
110            }
111        }
112        if probes.len() > bound {
113            break;
114        }
115    }
116    let mut constraints = Vec::new();
117    for t in probes.into_iter().take(bound + 1) {
118        let a_grad = section_gradients(a, &link.first, t)?;
119        let b_grad = section_gradients(
120            b,
121            &link.second,
122            if o.flip_first {
123                Q::one().checked_sub(t)?
124            } else {
125                t
126            },
127        )?;
128        for w in 0..=a.d[2] {
129            for c in 0..2 {
130                let mut row = Row::new();
131                for (s, grad, sign) in [(a, &a_grad, Q::one()), (b, &b_grad, Q::new(-1, 1)?)] {
132                    for (i, g) in grad.iter().enumerate() {
133                        let at = [i / (s.d[1] + 1), i % (s.d[1] + 1), w];
134                        add(&mut row, ids[s.flat(at)], mul(sign, g[c])?)?;
135                    }
136                }
137                if !row.is_empty() {
138                    constraints.push(row);
139                }
140            }
141        }
142    }
143    // Reduce this one interface before intersecting constraints at edges and
144    // vertices. The local relations typically have far smaller numerators
145    // and supports than the redundant physical point equations.
146    let mut local_ids: Vec<_> = constraints.iter().flat_map(|r| r.keys().copied()).collect();
147    local_ids.sort_unstable();
148    local_ids.dedup();
149    let local = constraints
150        .iter()
151        .map(|r| {
152            r.iter()
153                .map(|(&i, &c)| (local_ids.binary_search(&i).unwrap(), c))
154                .collect()
155        })
156        .collect::<Vec<Row>>();
157    let (lift, free) = crate::spline_constraints::conditioned_kernel(local_ids.len(), &local)?;
158    for (i, r) in lift.iter().enumerate() {
159        if free.contains(&i) {
160            continue;
161        }
162        let mut equation = Row::from([(local_ids[i], Q::one())]);
163        for (&j, &c) in r {
164            add(&mut equation, local_ids[free[j]], Q::zero().checked_sub(c)?)?;
165        }
166        rows.push(equation);
167    }
168    Ok(())
169}
