108struct UnionFind {
109    parent: Vec<usize>,
110}
111
112impl UnionFind {
113    fn new() -> Self {
114        Self { parent: Vec::new() }
115    }
116    fn add(&mut self) -> usize {
117        let id = self.parent.len();
118        self.parent.push(id);
119        id
120    }
121    fn find(&mut self, value: usize) -> usize {
122        if self.parent[value] != value {
123            self.parent[value] = self.find(self.parent[value]);
124        }
125        self.parent[value]
126    }
127    fn join(&mut self, left: usize, right: usize) {
128        let (left, right) = (self.find(left), self.find(right));
129        if left != right {
130            let (low, high) = if left < right {
131                (left, right)
132            } else {
133                (right, left)
134            };
135            self.parent[high] = low;
136        }
137    }
138}

140fn tangent_axes(axis: usize) -> [usize; 2] {
141    match axis {
142        0 => [1, 2],
143        1 => [0, 2],
144        _ => [0, 1],
145    }

148fn side_point(side: &LeafSide, uv: [u32; 2], denominator: u32) -> ChartPoint {
149    let [a, b] = tangent_axes(side.root_side.axis);
150    let mut coordinates = [0; 3];
151    let width = denominator >> side.leaf.level;
152    coordinates[side.root_side.axis] =
153        side.leaf.index[side.root_side.axis] * width + u32::from(side.root_side.upper) * width;
154    coordinates[a] = uv[0];
155    coordinates[b] = uv[1];
156    ChartPoint {
157        root: side.leaf.root.clone(),
158        coordinates,
159    }
160}

162fn corners(rect: FaceRect) -> [[u32; 2]; 4] {
163    let [[u0, u1], [v0, v1]] = rect.ranges;
164    [[u0, v0], [u1, v0], [u1, v1], [u0, v1]]
165}

167fn flip(value: u32, denominator: u32, reverse: bool) -> u32 {
168    if reverse {
169        denominator - value
170    } else {
171        value
172    }
173}

175fn first_to_second(point: [u32; 2], denominator: u32, orientation: Orientation) -> [u32; 2] {
176    let a = flip(point[0], denominator, orientation.flip_first);
177    let b = flip(point[1], denominator, orientation.flip_second);
178    if orientation.swap {
179        [b, a]
180    } else {
181        [a, b]
182    }
183}
184
185fn second_to_first(point: [u32; 2], denominator: u32, orientation: Orientation) -> [u32; 2] {
186    if orientation.swap {
187        [
188            flip(point[1], denominator, orientation.flip_first),
189            flip(point[0], denominator, orientation.flip_second),
190        ]
191    } else {
192        [
193            flip(point[0], denominator, orientation.flip_first),
194            flip(point[1], denominator, orientation.flip_second),
195        ]
196    }
197}

199fn side_orientation(side: &BlockSide) -> i8 {
200    let axis_sign = if side.axis.is_multiple_of(2) { 1 } else { -1 };
201    axis_sign * if side.upper { 1 } else { -1 }

218fn make_raw_faces(
219    adaptive: &AdaptiveSpace,
220    partition: &AdaptivePartition,
221) -> Result<Vec<RawFace>, &'static str> {
222    let report = partition.snapshot()?;
223    let mut feature_by_leaf_side = BTreeMap::<LeafSide, Vec<String>>::new();
224    for (feature, sides) in &adaptive.boundary_patches {
225        for side in sides {
226            feature_by_leaf_side
227                .entry(side.clone())
228                .or_default()
229                .push(feature.clone());
230        }
231    }
232    for features in feature_by_leaf_side.values_mut() {
233        features.sort();
234        features.dedup();
235    }
236    let mut result = Vec::with_capacity(report.interfaces.len() + report.exterior.len());
237    for patch in report.interfaces {
238        result.push(RawFace {
239            first: patch.first.clone(),
240            first_rect: patch.first_subrect,
241            second: Some((
242                patch.second.clone(),
243                patch.second_subrect,
244                patch.orientation,
245            )),
246            source: FaceSource::Interface {
247                first: patch.first,
248                second: patch.second,
249                first_rect: patch.first_subrect,
250                second_rect: patch.second_subrect,
251                orientation: patch.orientation,
252            },
253        });
254    }
255    for patch in report.exterior {
256        let features = feature_by_leaf_side
257            .get(&patch.side)
258            .cloned()
259            .unwrap_or_default();
260        result.push(RawFace {
261            first: patch.side.clone(),
262            first_rect: patch.root_rect,
263            second: None,
264            source: FaceSource::Exterior {
265                side: patch.side,
266                rect: patch.root_rect,
267                features,
268            },
269        });
270    }
271    Ok(result)
272}

274fn edge_vertices(
275    start: [u32; 2],
276    end: [u32; 2],
277    side: &LeafSide,
278    points: &BTreeMap<ChartPoint, usize>,
279    global: &BTreeMap<ChartPoint, usize>,
280    denominator: u32,
281) -> Result<Vec<(usize, [u32; 3])>, &'static str> {
282    let varying = if start[0] != end[0] { 0 } else { 1 };
283    if (start[0] == end[0]) == (start[1] == end[1]) {
284        return Err("degenerate or non-axis-aligned face edge");
285    }
286    let fixed = 1 - varying;
287    let low = start[varying].min(end[varying]);
288    let high = start[varying].max(end[varying]);
289    let mut candidates = points
290        .keys()
291        .filter_map(|point| {
292            if point.root != side.leaf.root {
293                return None;
294            }
295            let [a, b] = tangent_axes(side.root_side.axis);
296            let uv = [point.coordinates[a], point.coordinates[b]];
297            let width = denominator >> side.leaf.level;
298            let plane = side.leaf.index[side.root_side.axis] * width
299                + u32::from(side.root_side.upper) * width;
300            let normal_matches = point.coordinates[side.root_side.axis] == plane;
301            (normal_matches && uv[fixed] == start[fixed] && (low..=high).contains(&uv[varying]))
302                .then(|| (uv[varying], global[point], point.coordinates))
303        })
304        .collect::<Vec<_>>();
305    candidates.sort_by_key(|entry| entry.0);
306    candidates.dedup_by_key(|entry| entry.0);
307    if start[varying] > end[varying] {
308        candidates.reverse();
309    }
310    let ids = candidates
311        .into_iter()
312        .map(|entry| (entry.1, entry.2))
313        .collect::<Vec<_>>();
314    if ids.len() < 2 || ids.first().map(|entry| entry.0) == ids.last().map(|entry| entry.0) {
315        return Err("open or collapsed face boundary segment");
316    }
317    Ok(ids)
318}

320fn face_cycle(
321    side: &LeafSide,
322    rect: FaceRect,
323    points: &BTreeMap<ChartPoint, usize>,
324    global: &BTreeMap<ChartPoint, usize>,
325    edge_ids: &mut BTreeMap<(usize, usize), RawEdge>,
326    denominator: u32,
327) -> Result<Vec<EdgeUse>, &'static str> {
328    let direction = side_orientation(&side.root_side);
329    let mut boundary = corners(rect);
330    if direction < 0 {
331        boundary = [boundary[0], boundary[3], boundary[2], boundary[1]];
332    }
333    let mut cycle = Vec::new();
334    for index in 0..4 {
335        let start = boundary[index];
336        let end = boundary[(index + 1) % 4];
337        let chain = edge_vertices(start, end, side, points, global, denominator)?;
338        for pair in chain.windows(2) {
339            if pair[0].0 == pair[1].0 {
340                return Err("collapsed oriented edge");
341            }
342            let key = (pair[0].0.min(pair[1].0), pair[0].0.max(pair[1].0));
343            let sign = if pair[0].0 < pair[1].0 { 1 } else { -1 };
344            let segment = ChartSegment {
345                root: side.leaf.root.clone(),
346                start: if pair[0].1 <= pair[1].1 {
347                    pair[0].1
348                } else {
349                    pair[1].1
350                },
351                end: if pair[0].1 <= pair[1].1 {
352                    pair[1].1
353                } else {
354                    pair[0].1
355                },
356            };
357            edge_ids
358                .entry(key)
359                .or_insert_with(|| RawEdge {
360                    provenance: BTreeSet::new(),
361                })
362                .provenance
363                .insert(segment.clone());
364            cycle.push((key, segment, sign));
365        }
366    }
367    Ok(cycle)
368}

370fn cell_id_map(
371    report_leaves: &[LeafKey],
372    adaptive: &AdaptiveSpace,
373) -> Result<(Vec<ComplexCell>, BTreeMap<LeafKey, usize>), &'static str> {
374    let mut cells = Vec::with_capacity(report_leaves.len());
375    let mut ids = BTreeMap::new();
376    for leaf in report_leaves {
377        let id = cells.len();
378        let block_id = adaptive
379            .leaf_blocks
380            .get(leaf)
381            .ok_or("leaf complex cell has no analysis block")?
382            .clone();
383        if ids.insert(leaf.clone(), id).is_some() {
384            return Err("duplicate adaptive leaf in cell complex");
385        }
386        cells.push(ComplexCell {
387            id,
388            leaf: leaf.clone(),
389            block_id,
390            boundary: Vec::new(),
391        });
392    }
393    Ok((cells, ids))
394}

396fn validate_chains(complex: &OrientedComplex) -> Result<(), &'static str> {
397    for face in &complex.faces {
398        let mut vertex_sum = BTreeMap::<usize, i32>::new();
399        for incidence in &face.boundary {
400            let edge = complex
401                .edges
402                .get(incidence.id)
403                .ok_or("face references missing edge")?;
404            *vertex_sum.entry(edge.vertices[0]).or_default() -= i32::from(incidence.sign);
405            *vertex_sum.entry(edge.vertices[1]).or_default() += i32::from(incidence.sign);
406        }
407        if vertex_sum.values().any(|&value| value != 0) {
408            return Err("boundary of a 2-cell is not a closed cycle");
409        }
410    }
411    for cell in &complex.cells {
412        let mut edge_sum = BTreeMap::<usize, i32>::new();
413        for incidence in &cell.boundary {
414            let face = complex
415                .faces
416                .get(incidence.id)
417                .ok_or("cell references missing face")?;
418            for edge in &face.boundary {
419                *edge_sum.entry(edge.id).or_default() +=
420                    i32::from(incidence.sign) * i32::from(edge.sign);
421            }
422        }
423        if edge_sum.values().any(|&value| value != 0) {
424            return Err("boundary of a 3-cell is not a closed 2-chain");
425        }
426    }
427    let mut face_usage = BTreeMap::<usize, Vec<i8>>::new();
428    for cell in &complex.cells {
429        for incidence in &cell.boundary {
430            face_usage
431                .entry(incidence.id)
432                .or_default()
433                .push(incidence.sign);
434        }
435    }
436    for face in &complex.faces {
437        let uses = face_usage.get(&face.id).ok_or("oriented face is unused")?;
438        let expected = if complex.boundary_faces.contains(&face.id) {
439            1
440        } else {
441            2
442        };
443        if uses.len() != expected
444            || (expected == 1 && uses[0] != 1)
445            || (expected == 2
446                && (uses[0] == uses[1] || uses.iter().map(|x| i16::from(*x)).sum::<i16>() != 0))
447        {
448            return Err("oriented face has invalid cell incidence");
449        }
450    }
451    let mut boundary_edge_usage = BTreeMap::<usize, Vec<i8>>::new();
452    for &face_id in &complex.boundary_faces {
453        for edge in &complex.faces[face_id].boundary {
454            boundary_edge_usage
455                .entry(edge.id)
456                .or_default()
457                .push(edge.sign);
458        }
459    }
460    if boundary_edge_usage.values().any(|uses| {
461        uses.len() != 2 || uses[0] == uses[1] || i16::from(uses[0]) + i16::from(uses[1]) != 0
462    }) {
463        return Err("exterior boundary is not a closed oriented surface");
464    }
465    Ok(())
466}

468fn segment_provenance_connected(
469    provenance: &BTreeSet<ChartSegment>,
470    glued_segments: &BTreeSet<(ChartSegment, ChartSegment)>,
471) -> bool {
472    if provenance.is_empty() {
473        return false;
474    }
475    let mut reached = BTreeSet::from([provenance.iter().next().unwrap().clone()]);
476    loop {
477        let before = reached.len();
478        for (left, right) in glued_segments {
479            if provenance.contains(left) && provenance.contains(right) {
480                if reached.contains(left) {
481                    reached.insert(right.clone());
482                }
483                if reached.contains(right) {
484                    reached.insert(left.clone());
485                }
486            }
487        }
488        if reached.len() == before {
489            return reached.len() == provenance.len();
490        }
491    }
492}

494/// Construct and verify the signed cellular boundary operators through C₃.
495pub fn build_oriented_complex(
496    adaptive: &AdaptiveSpace,
497    partition: &AdaptivePartition,
498) -> Result<OrientedComplex, &'static str> {
499    let report = partition.snapshot()?;
500    if adaptive.space.blocks.len() != report.leaves.len()
501        || adaptive.leaf_blocks.len() != report.leaves.len()
502    {
503        return Err("adaptive cells and leaf partition disagree");
504    }
505    let raw_faces = make_raw_faces(adaptive, partition)?;
506    let denominator = partition.grid_denominator();
507    let mut chart_indices = BTreeMap::<ChartPoint, usize>::new();
508    let mut union = UnionFind::new();
509    for face in &raw_faces {
510        for uv in corners(face.first_rect) {
511            add_chart_point(
512                side_point(&face.first, uv, denominator),
513                &mut chart_indices,
514                &mut union,
515            );
516        }
517        if let Some((second, rect, _)) = &face.second {
518            for uv in corners(*rect) {
519                add_chart_point(
520                    side_point(second, uv, denominator),
521                    &mut chart_indices,
522                    &mut union,
523                );
524            }
525        }
526    }
527    for face in &raw_faces {
528        let Some((second, second_rect, orientation)) = &face.second else {
529            continue;
530        };
531        let [a, b] = tangent_axes(face.first.root_side.axis);
532        let width = denominator >> face.first.leaf.level;
533        let normal = face.first.leaf.index[face.first.root_side.axis] * width
534            + u32::from(face.first.root_side.upper) * width;
535        let first_nodes = chart_indices
536            .keys()
537            .filter(|point| {
538                if point.root != face.first.leaf.root
539                    || point.coordinates[face.first.root_side.axis] != normal
540                {
541                    return false;
542                }
543                let uv = [point.coordinates[a], point.coordinates[b]];
544                (0..2).all(|axis| {
545                    face.first_rect.ranges[axis][0] <= uv[axis]
546                        && uv[axis] <= face.first_rect.ranges[axis][1]
547                })
548            })
549            .cloned()
550            .collect::<Vec<_>>();
551        let [sa, sb] = tangent_axes(second.root_side.axis);
552        let second_width = denominator >> second.leaf.level;
553        let second_normal = second.leaf.index[second.root_side.axis] * second_width
554            + u32::from(second.root_side.upper) * second_width;
555        for point in first_nodes {
556            let uv = [point.coordinates[a], point.coordinates[b]];
557            let mapped = first_to_second(uv, denominator, *orientation);
558            if !(0..2).all(|axis| {
559                second_rect.ranges[axis][0] <= mapped[axis]
560                    && mapped[axis] <= second_rect.ranges[axis][1]
561            }) {
562                continue;
563            }
564            let first_id = chart_indices[&point];
565            let mut coordinates = [0; 3];
566            coordinates[second.root_side.axis] = second_normal;
567            coordinates[sa] = mapped[0];
568            coordinates[sb] = mapped[1];
569            let second_point = ChartPoint {
570                root: second.leaf.root.clone(),
571                coordinates,
572            };
573            let second_id = add_chart_point(second_point, &mut chart_indices, &mut union);
574            union.join(first_id, second_id);
575        }
576        let second_nodes = chart_indices
577            .keys()
578            .filter(|point| {
579                if point.root != second.leaf.root
580                    || point.coordinates[second.root_side.axis] != second_normal
581                {
582                    return false;
583                }
584                let uv = [point.coordinates[sa], point.coordinates[sb]];
585                (0..2).all(|axis| {
586                    second_rect.ranges[axis][0] <= uv[axis]
587                        && uv[axis] <= second_rect.ranges[axis][1]
588                })
589            })
590            .cloned()
591            .collect::<Vec<_>>();
592        for point in second_nodes {
593            let uv = [point.coordinates[sa], point.coordinates[sb]];
594            let mapped = second_to_first(uv, denominator, *orientation);
595            if !(0..2).all(|axis| {
596                face.first_rect.ranges[axis][0] <= mapped[axis]
597                    && mapped[axis] <= face.first_rect.ranges[axis][1]
598            }) {
599                continue;
600            }
601            let second_id = chart_indices[&point];
602            let mut coordinates = [0; 3];
603            coordinates[face.first.root_side.axis] = normal;
604            coordinates[a] = mapped[0];
605            coordinates[b] = mapped[1];
606            let first_point = ChartPoint {
607                root: face.first.leaf.root.clone(),
608                coordinates,
609            };
610            let first_id = add_chart_point(first_point, &mut chart_indices, &mut union);
611            union.join(first_id, second_id);
612        }
613    }
614
615    let mut minimum_chart = BTreeMap::<usize, ChartPoint>::new();
616    let mut representative_by_chart = BTreeMap::new();
617    for (point, &index) in &chart_indices {
618        let representative = union.find(index);
619        representative_by_chart.insert(point.clone(), representative);
620        minimum_chart
621            .entry(representative)
622            .and_modify(|old| {
623                if point < old {
624                    *old = point.clone();
625                }
626            })
627            .or_insert_with(|| point.clone());
628    }
629    let mut ordered_representatives = minimum_chart.into_iter().collect::<Vec<_>>();
630    ordered_representatives.sort_by(|a, b| a.1.cmp(&b.1));
631    let representative_ids = ordered_representatives
632        .iter()
633        .enumerate()
634        .map(|(id, (representative, _))| (*representative, id))
635        .collect::<BTreeMap<_, _>>();
636    let global_vertex_by_chart = representative_by_chart
637        .iter()
638        .map(|(point, representative)| (point.clone(), representative_ids[representative]))
639        .collect::<BTreeMap<_, _>>();
640    let mut charts_by_vertex = vec![Vec::new(); representative_ids.len()];
641    for (point, &id) in &global_vertex_by_chart {
642        charts_by_vertex[id].push(point.clone());
643    }
644    let vertices = charts_by_vertex
645        .into_iter()
646        .enumerate()
647        .map(|(id, mut charts)| {
648            charts.sort();
649            charts.dedup();
650            ComplexVertex { id, charts }
651        })
652        .collect::<Vec<_>>();
653
654    let mut raw_edges = BTreeMap::<(usize, usize), RawEdge>::new();
655    let mut glued_segments = BTreeSet::<(ChartSegment, ChartSegment)>::new();
656    let mut face_cycles = Vec::with_capacity(raw_faces.len());
657    let mut boundary_face_ids = Vec::new();
658    let (_, cell_ids) = cell_id_map(&report.leaves, adaptive)?;
659    for face in raw_faces {
660        let first_cycle = face_cycle(
661            &face.first,
662            face.first_rect,
663            &chart_indices,
664            &global_vertex_by_chart,
665            &mut raw_edges,
666            denominator,
667        )?;
668        let first_leaf = cell_ids[&face.first.leaf];
669        let incident_cells = if let Some((second, second_rect, _)) = &face.second {
670            let second_cycle = face_cycle(
671                second,
672                *second_rect,
673                &chart_indices,
674                &global_vertex_by_chart,
675                &mut raw_edges,
676                denominator,
677            )?;
678            let first_chain = first_cycle
679                .iter()
680                .map(|(edge, segment, sign)| (*edge, (segment.clone(), *sign)))
681                .collect::<BTreeMap<_, _>>();
682            let second_chain = second_cycle
683                .iter()
684                .map(|(edge, segment, sign)| (*edge, (segment.clone(), *sign)))
685                .collect::<BTreeMap<_, _>>();
686            if first_chain.len() != first_cycle.len()
687                || second_chain.len() != second_cycle.len()
688                || first_chain.len() != second_chain.len()
689                || first_chain.iter().any(|(edge, (_, sign))| {
690                    second_chain
691                        .get(edge)
692                        .is_none_or(|(_, other_sign)| *other_sign != -*sign)
693                })
694            {
695                return Err("D4 interface face cycles are not oppositely oriented");
696            }
697            for (edge, (segment, _)) in &first_chain {
698                let (other, _) = &second_chain[edge];
699                let pair = if segment < other {
700                    (segment.clone(), other.clone())
701                } else {
702                    (other.clone(), segment.clone())
703                };
704                glued_segments.insert(pair);
705            }
706            vec![
707                SignedId {
708                    id: first_leaf,
709                    sign: 1,
710                },
711                SignedId {
712                    id: cell_ids[&second.leaf],
713                    sign: -1,
714                },
715            ]
716        } else {
717            boundary_face_ids.push(face_cycles.len());
718            vec![SignedId {
719                id: first_leaf,
720                sign: 1,
721            }]
722        };
723        face_cycles.push(FaceCycle {
724            source: face.source,
725            boundary: first_cycle,
726            incident_cells,
727        });
728    }
729
730    for edge in raw_edges.values() {
731        if edge.provenance.len() < 2 {
732            continue;
733        }
734        if !segment_provenance_connected(&edge.provenance, &glued_segments) {
735            return Err("ambiguous multiedge: distinct chart segments share endpoints without D4 identification");
736        }
737    }
738
739    let edges = raw_edges
740        .into_iter()
741        .enumerate()
742        .map(|(id, ((first, second), edge))| ComplexEdge {
743            id,
744            vertices: [first, second],
745            provenance: edge.provenance.into_iter().collect(),
746        })
747        .collect::<Vec<_>>();
748    let edge_ids = edges
749        .iter()
750        .map(|edge| ((edge.vertices[0], edge.vertices[1]), edge.id))
751        .collect::<BTreeMap<_, _>>();
752    let faces = face_cycles
753        .into_iter()
754        .enumerate()
755        .map(|(id, cycle)| ComplexFace {
756            id,
757            boundary: cycle
758                .boundary
759                .into_iter()
760                .map(|(edge, _, sign)| SignedId {
761                    id: edge_ids[&edge],
762                    sign,
763                })
764                .collect(),
765            incident_cells: cycle.incident_cells,
766            source: cycle.source,
767        })
768        .collect::<Vec<_>>();
769    let mut cells = cell_id_map(&report.leaves, adaptive)?.0;
770    for face in &faces {
771        for incidence in &face.incident_cells {
772            cells[incidence.id].boundary.push(SignedId {
773                id: face.id,
774                sign: incidence.sign,
775            });
776        }
777    }
778    for cell in &mut cells {
779        cell.boundary.sort_by_key(|face| face.id);
780    }
781    let complex = OrientedComplex {
782        grid_denominator: denominator,
783        vertices,
784        edges,
785        faces,
786        cells,
787        boundary_faces: boundary_face_ids,
788    };
789    validate_chains(&complex)?;
790    Ok(complex)
791}

793fn text(value: &str) -> Json {
794    Json::String(value.to_owned())
795}
796fn number(value: impl ToString) -> Json {
797    Json::Number(value.to_string())
798}
799fn signed_ids(values: &[SignedId]) -> Json {
800    Json::Array(
801        values
802            .iter()
803            .map(|value| Json::Array(vec![number(value.id), number(value.sign)]))
804            .collect(),
805    )
806}
807fn leaf_json(leaf: &LeafKey) -> Json {
808    object([
809        ("root", text(&leaf.root)),
810        ("level", number(leaf.level)),
811        (
812            "index",
813            Json::Array(leaf.index.iter().copied().map(number).collect()),
814        ),
815    ])
816}
817fn side_json(side: &LeafSide) -> Json {
818    object([
819        ("root_block", text(&side.root_side.block)),
820        ("axis", number(side.root_side.axis)),
821        ("upper", Json::Bool(side.root_side.upper)),
822        ("leaf", leaf_json(&side.leaf)),
823    ])
824}
825fn rect_json(rect: FaceRect) -> Json {
826    Json::Array(
827        rect.ranges
828            .iter()
829            .map(|range| Json::Array(range.iter().copied().map(number).collect()))
830            .collect(),
831    )
832}

834impl OrientedComplex {
835    pub fn to_json(&self) -> Json {
836        let vertices = self
837            .vertices
838            .iter()
839            .map(|vertex| {
840                object([
841                    ("id", number(vertex.id)),
842                    (
843                        "charts",
844                        Json::Array(
845                            vertex
846                                .charts
847                                .iter()
848                                .map(|point| {
849                                    object([
850                                        ("root", text(&point.root)),
851                                        (
852                                            "coordinates",
853                                            Json::Array(
854                                                point
855                                                    .coordinates
856                                                    .iter()
857                                                    .copied()
858                                                    .map(number)
859                                                    .collect(),
860                                            ),
861                                        ),
862                                    ])
863                                })
864                                .collect(),
865                        ),
866                    ),
867                ])
868            })
869            .collect();
870        let edges = self
871            .edges
872            .iter()
873            .map(|edge| {
874                object([
875                    ("id", number(edge.id)),
876                    (
877                        "vertices",
878                        Json::Array(edge.vertices.iter().copied().map(number).collect()),
879                    ),
880                    (
881                        "boundary",
882                        Json::Array(vec![
883                            Json::Array(vec![number(edge.vertices[0]), number(-1)]),
884                            Json::Array(vec![number(edge.vertices[1]), number(1)]),
885                        ]),
886                    ),
887                    (
888                        "provenance",
889                        Json::Array(
890                            edge.provenance
891                                .iter()
892                                .map(|segment| {
893                                    object([
894                                        ("root", text(&segment.root)),
895                                        (
896                                            "start",
897                                            Json::Array(
898                                                segment.start.iter().copied().map(number).collect(),
899                                            ),
900                                        ),
901                                        (
902                                            "end",
903                                            Json::Array(
904                                                segment.end.iter().copied().map(number).collect(),
905                                            ),
906                                        ),
907                                    ])
908                                })
909                                .collect(),
910                        ),
911                    ),
912                ])
913            })
914            .collect();
915        let faces = self
916            .faces
917            .iter()
918            .map(|face| {
919                let (kind, source) = match &face.source {
920                    FaceSource::Interface {
921                        first,
922                        second,
923                        first_rect,
924                        second_rect,
925                        orientation,
926                    } => (
927                        "interface",
928                        object([
929                            ("first", side_json(first)),
930                            ("second", side_json(second)),
931                            ("first_rect", rect_json(*first_rect)),
932                            ("second_rect", rect_json(*second_rect)),
933                            (
934                                "orientation",
935                                object([
936                                    ("swap", Json::Bool(orientation.swap)),
937                                    ("flip_first", Json::Bool(orientation.flip_first)),
938                                    ("flip_second", Json::Bool(orientation.flip_second)),
939                                ]),
940                            ),
941                        ]),
942                    ),
943                    FaceSource::Exterior {
944                        side,
945                        rect,
946                        features,
947                    } => (
948                        "exterior",
949                        object([
950                            ("side", side_json(side)),
951                            ("rect", rect_json(*rect)),
952                            (
953                                "features",
954                                Json::Array(features.iter().map(|feature| text(feature)).collect()),
955                            ),
956                        ]),
957                    ),
958                };
959                object([
960                    ("id", number(face.id)),
961                    ("kind", text(kind)),
962                    ("source", source),
963                    ("boundary", signed_ids(&face.boundary)),
964                    ("incident_cells", signed_ids(&face.incident_cells)),
965                ])
966            })
967            .collect();
968        let cells = self
969            .cells
970            .iter()
971            .map(|cell| {
972                object([
973                    ("id", number(cell.id)),
974                    ("leaf", leaf_json(&cell.leaf)),
975                    ("block_id", text(&cell.block_id)),
976                    ("boundary", signed_ids(&cell.boundary)),
977                ])
978            })
979            .collect();
980        object([
981            ("schema", text("adaptive_oriented_cell_complex_v1")),
982            ("grid_denominator", number(self.grid_denominator)),
983            ("orientation_convention", text("block_chart_u_v_w_positive; each exterior face uses its outward chart orientation")),
984            ("vertices", Json::Array(vertices)),
985            ("edges", Json::Array(edges)),
986            ("faces", Json::Array(faces)),
987            ("cells", Json::Array(cells)),
988            ("boundary_faces", Json::Array(self.boundary_faces.iter().copied().map(number).collect())),
989            ("boundary_verified", Json::Bool(true)),
990        ])
991    }
992}
