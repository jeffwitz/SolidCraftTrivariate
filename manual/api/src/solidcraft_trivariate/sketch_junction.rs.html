1//! Geometric continuity at an oriented planar rational Bézier junction.
2//! The left span ends at the join and the right span starts there.
3
4use crate::constraint_system::Ratio;
5
6#[derive(Clone, Copy, Debug, Eq, PartialEq)]
7pub enum Continuity {
8    Disconnected,
9    Singular,
10    Corner,
11    Tangent,
12    Curvature,
13}
14
15type Pair = [Ratio; 2];
16type Error = &'static str;
17
18struct Jet {
19    position: Pair,
20    velocity: Pair,
21    acceleration: Pair,
22}
23
24fn quotient(a: Ratio, b: Ratio) -> Result<Ratio, Error> {
25    a.checked_mul(Ratio::new(b.denominator(), b.numerator())?)
26}
27
28fn endpoint(controls: &[[i64; 3]], end: bool) -> Result<Jet, Error> {
29    if controls.len() < 2 || controls.iter().any(|p| p[2] <= 0) {
30        return Err("a junction requires positive-weight spans of degree at least one");
31    }
32    let n = controls.len() - 1;
33    let degree = Ratio::new(n as i128, 1)?;
34    let second_factor = degree.checked_mul(Ratio::new((n - 1) as i128, 1)?)?;
35    let point = |i: usize| controls[if end { n - i } else { i }].map(i128::from);
36    let h = point(0);
37    let near = point(1);
38    let far = point(n.min(2));
39    let inv_w = Ratio::new(1, h[2])?;
40    // Differences first point inward at both ends. Reversing the parameter
41    // changes the first derivative's sign and preserves the second derivative.
42    let first = |d: usize| Ratio::new(near[d] - h[d], 1)?.checked_mul(degree);
43    let second = |d: usize| Ratio::new(far[d] - 2 * near[d] + h[d], 1)?.checked_mul(second_factor);
44    let dw = first(2)?;
45    let ddw = second(2)?;
46    let mut jet = Jet {
47        position: [Ratio::zero(); 2],
48        velocity: [Ratio::zero(); 2],
49        acceleration: [Ratio::zero(); 2],
50    };
51    for (d, x) in jet.position.iter_mut().enumerate() {
52        *x = Ratio::new(h[d], h[2])?;
53        let v = first(d)?
54            .checked_sub(x.checked_mul(dw)?)?
55            .checked_mul(inv_w)?;
56        jet.acceleration[d] = second(d)?
57            .checked_sub(x.checked_mul(ddw)?)?
58            .checked_sub(v.checked_mul(dw)?.checked_mul(Ratio::new(2, 1)?)?)?
59            .checked_mul(inv_w)?;
60        jet.velocity[d] = if end {
61            Ratio::zero().checked_sub(v)?
62        } else {
63            v
64        };
65    }
66    Ok(jet)
67}

69fn cross(a: Pair, b: Pair) -> Result<Ratio, Error> {
70    a[0].checked_mul(b[1])?.checked_sub(a[1].checked_mul(b[0])?)
71}

73/// Classify using exact endpoint jets of the stored homogeneous (wx, wy, w)
74/// controls. Curvature means at least G2. Singular means a stationary endpoint
75/// prevents this regular-jet test from deciding geometric continuity.
76pub fn classify(left: &[[i64; 3]], right: &[[i64; 3]]) -> Result<Continuity, Error> {
77    let a = endpoint(left, true)?;
78    let b = endpoint(right, false)?;
79    if a.position != b.position {
80        return Ok(Continuity::Disconnected);
81    }
82    let Some(axis) = a.velocity.iter().position(|v| v.numerator() != 0) else {
83        return Ok(Continuity::Singular);
84    };
85    if b.velocity == [Ratio::zero(); 2] {
86        return Ok(Continuity::Singular);
87    }
88    if cross(a.velocity, b.velocity)? != Ratio::zero()
89        || a.velocity[axis].numerator().signum() != b.velocity[axis].numerator().signum()
90    {
91        return Ok(Continuity::Corner);
92    }
93    // If v_b = speed * v_a, G2 permits arbitrary tangential acceleration.
94    // The normal part of a_b must equal that of speed^2 * a_a.
95    let speed = quotient(b.velocity[axis], a.velocity[axis])?;
96    let right_bend = cross(a.velocity, b.acceleration)?;
97    let left_bend = cross(a.velocity, a.acceleration)?
98        .checked_mul(speed)?
99        .checked_mul(speed)?;
100    Ok(if right_bend == left_bend {
101        Continuity::Curvature
102    } else {
103        Continuity::Tangent
104    })
105}

107/// A candidate partition event at the start of the named right-hand span.
108/// At closure the incoming span ends at the upper limit of the active domain.
109#[derive(Clone, Debug, Eq, PartialEq)]
110pub struct SketchJunction {
111    pub right_span: usize,
112    pub parameter: i64,
113    pub multiplicity: usize,
114    pub closure: bool,
115    pub continuity: Continuity,
116}

118/// Inspect every junction, including closure, before building any volume.
119/// All events are preserved; callers decide which ones require partition lines.
120/// Disconnected and singular results are diagnostics, not admissions.
121pub fn plan_loop(
122    degree: usize,
123    knots: &[i64],
124    controls: &[[i64; 3]],
125) -> Result<Vec<SketchJunction>, Error> {
126    let spans = crate::nurbs_curve::extract_active_bezier(degree, knots, controls)?;
127    let mut events = Vec::with_capacity(spans.len());
128    for (index, right) in spans.iter().enumerate() {
129        let left = &spans[if index == 0 {
130            spans.len() - 1
131        } else {
132            index - 1
133        }];
134        let parameter = right.interval[0];
135        let first = knots.partition_point(|k| *k < parameter);
136        let after = knots.partition_point(|k| *k <= parameter);
137        events.push(SketchJunction {
138            right_span: index,
139            parameter,
140            multiplicity: after - first,
141            closure: index == 0,
142            continuity: classify(&left.controls, &right.controls)?,
143        });
144    }
145    Ok(events)
146}
