1//! Construct planar section patches and oriented boundary spans directly from
2//! sketch primitives. Cell counts and joins are determined before sweeping.
3use crate::{
4    bracket::BracketBlock, constraint_system::Ratio as R, nurbs_cross::NurbsCurve2,
5    nurbs_curve::normalize, sketch_junction::Continuity,
6};
7
8type Error = &'static str;
9type H = [R; 3];
10
11#[derive(Debug, Clone, Copy)]
12pub enum Primitive {
13    Rectangle { min: [R; 2], max: [R; 2] },
14    Circle { center: [R; 2], radius: R },
15    Ellipse { center: [R; 2], radii: [R; 2] },
16}
17
18#[derive(Debug, Clone, Copy)]
19pub enum Operation {
20    /// Translate the filled XY sketch along positive Z.
21    Extrude { length: i64 },
22    /// Full turn about sketch Y, with the whole section strictly in x > 0.
23    Revolve,
24}
25
26#[derive(Debug, Clone)]
27pub struct SectionPatch {
28    pub degrees: [usize; 2],
29    pub controls: Vec<[i64; 3]>,
30}
31
32pub struct PrimitivePlan {
33    pub contour: NurbsCurve2,
34    /// Geometric join at the start of each emitted boundary span.
35    pub junctions: Vec<Continuity>,
36    pub section_patches: Vec<SectionPatch>,
37    operation: Operation,
38}
39
40fn int(n: i64) -> R {
41    R::new(n.into(), 1).expect("integer")
42}
43
44fn contour(spans: &[Vec<H>]) -> Result<NurbsCurve2, Error> {
45    let degree = spans[0].len() - 1;
46    let mut controls = vec![spans[0][0]];
47    let mut knots = vec![0; degree + 1];
48    for (k, s) in spans.iter().enumerate() {
49        controls.extend_from_slice(&s[1..]);
50        knots.extend(std::iter::repeat_n(k as i64 + 1, degree));
51    }
52    knots.push(spans.len() as i64);
53    Ok(NurbsCurve2 {
54        degree,
55        knots,
56        controls: normalize(&controls)?,
57    })
58}
59
60fn rectangle(min: [R; 2], max: [R; 2], operation: Operation) -> Result<PrimitivePlan, Error> {
61    for d in 0..2 {
62        if max[d].checked_sub(min[d])?.numerator() <= 0 {
63            return Err("rectangle dimensions must be positive");
64        }
65    }
66    let point = |x, y| [x, y, R::one()];
67    let [a, b, c, d] = [
68        point(min[0], min[1]),
69        point(max[0], min[1]),
70        point(max[0], max[1]),
71        point(min[0], max[1]),
72    ];
73    Ok(PrimitivePlan {
74        contour: contour(&[vec![a, b], vec![b, c], vec![c, d], vec![d, a]])?,
75        junctions: vec![Continuity::Corner; 4],
76        section_patches: vec![SectionPatch {
77            degrees: [1, 1],
78            controls: normalize(&[a, d, b, c])?,
79        }],
80        operation,
81    })
82}
83
84fn rotate([mut x, mut y, w]: [i64; 3], quarter: usize) -> [i64; 3] {
85    for _ in 0..quarter {
86        (x, y) = (-y, x);
87    }
88    [x, y, w]
89}

91// Both parameter directions run counterclockwise. Alternating the gauges
92// makes the endpoint weights match around a closed four-quarter ring.
93fn arc(q: usize) -> [[i64; 3]; 3] {
94    let base = if q.is_multiple_of(2) {
95        [[1, 0, 1], [1, 1, 1], [0, 2, 2]]
96    } else {
97        [[2, 0, 2], [1, 1, 1], [0, 1, 1]]
98    };
99    base.map(|p| rotate(p, q))
100}

102fn ellipse(center: [R; 2], radii: [R; 2], operation: Operation) -> Result<PrimitivePlan, Error> {
103    if radii.iter().any(|r| r.numerator() <= 0) {
104        return Err("ellipse radii must be positive");
105    }
106    let transform = |[x, y, w]: [i64; 3]| -> Result<H, Error> {
107        let w = int(w);
108        Ok([
109            center[0]
110                .checked_mul(w)?
111                .checked_add(radii[0].checked_mul(int(x))?)?,
112            center[1]
113                .checked_mul(w)?
114                .checked_add(radii[1].checked_mul(int(y))?)?,
115            w,
116        ])
117    };
118    let mut core = Vec::new();
119    for i in 0..3 {
120        for j in 0..3 {
121            core.push(transform([2 - i - j, i - j, 4])?);
122        }
123    }
124    let mut section_patches = vec![SectionPatch {
125        degrees: [2, 2],
126        controls: normalize(&core)?,
127    }];
128    let mut boundary = Vec::new();
129    for q in 0..4 {
130        let mut net = Vec::new();
131        for p in [[2, 0, 4], [1, 1, 4], [0, 2, 4]] {
132            net.push(transform(rotate(p, q))?);
133        }
134        // The inner edge has weight four. Give the unit circle the same
135        // global gauge so the radial seams agree from both neighboring cells.
136        let outer = arc(q)
137            .map(|p| transform(p.map(|c| 4 * c)))
138            .into_iter()
139            .collect::<Result<Vec<_>, _>>()?;
140        net.extend_from_slice(&outer);
141        section_patches.push(SectionPatch {
142            degrees: [1, 2],
143            controls: normalize(&net)?,
144        });
145        boundary.push(outer);
146    }
147    Ok(PrimitivePlan {
148        contour: contour(&boundary)?,
149        junctions: vec![Continuity::Curvature; 4],
150        section_patches,
151        operation,
152    })
153}

155impl Primitive {
156    pub fn plan(self, operation: Operation) -> Result<PrimitivePlan, Error> {
157        match operation {
158            Operation::Extrude { length } if length <= 0 => {
159                return Err("extrusion length must be positive")
160            }
161            Operation::Revolve => {
162                let min_r = match self {
163                    Self::Rectangle { min, .. } => min[0],
164                    Self::Circle { center, radius } => center[0].checked_sub(radius)?,
165                    Self::Ellipse { center, radii } => center[0].checked_sub(radii[0])?,
166                };
167                if min_r.numerator() <= 0 {
168                    return Err("full-turn template requires positive radial clearance");
169                }
170            }
171            _ => {}
172        }
173        match self {
174            Self::Rectangle { min, max } => rectangle(min, max, operation),
175            Self::Circle { center, radius } => ellipse(center, [radius; 2], operation),
176            Self::Ellipse { center, radii } => ellipse(center, radii, operation),
177        }
178    }
179}
180
181impl PrimitivePlan {
182    pub fn block_count(&self) -> usize {
183        self.section_patches.len()
184            * match self.operation {
185                Operation::Extrude { .. } => 1,
186                Operation::Revolve => 4,
187            }
188    }
189
190    /// Instantiate the already partitioned section. No fitting or search for
191    /// knots, seams, or cylinder intersections is needed for an isolated sweep.
192    pub fn build(&self) -> Result<Vec<BracketBlock>, Error> {
193        let mut blocks = Vec::with_capacity(self.block_count());
194        for (section, patch) in self.section_patches.iter().enumerate() {
195            let [u, v] = patch.degrees;
196            match self.operation {
197                Operation::Extrude { length } => {
198                    let mut net = Vec::new();
199                    for &[x, y, w] in &patch.controls {
200                        for z in [0, length] {
201                            net.push([int(x), int(y), int(z).checked_mul(int(w))?, int(w)]);
202                        }
203                    }
204                    blocks.push(BracketBlock {
205                        id: format!("section_{section}"),
206                        degrees: [u, v, 1],
207                        controls: normalize(&net)?,
208                    });
209                }
210                Operation::Revolve => {
211                    for q in 0..4 {
212                        let mut net = Vec::new();
213                        for &[x, y, w] in &patch.controls {
214                            for [c, s, k] in arc(q) {
215                                net.push([
216                                    int(x).checked_mul(int(c))?,
217                                    int(y).checked_mul(int(k))?,
218                                    int(x).checked_mul(int(s))?,
219                                    int(w).checked_mul(int(k))?,
220                                ]);
221                            }
222                        }
223                        blocks.push(BracketBlock {
224                            id: format!("section_{section}_turn_{q}"),
225                            degrees: [u, v, 2],
226                            controls: normalize(&net)?,
227                        });
228                    }
229                }
230            }
231        }
232        Ok(blocks)
233    }
