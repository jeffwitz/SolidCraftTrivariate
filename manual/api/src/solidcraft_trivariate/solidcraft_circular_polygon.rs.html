26struct DisjointSet(Vec<usize>);
27impl DisjointSet {
28    fn new(n: usize) -> Self {
29        Self((0..n).collect())
30    }
31    fn find(&mut self, x: usize) -> usize {
32        if self.0[x] != x {
33            self.0[x] = self.find(self.0[x]);
34        }
35        self.0[x]
36    }
37    fn union(&mut self, a: usize, b: usize) {
38        let (a, b) = (self.find(a), self.find(b));
39        if a != b {
40            self.0[b] = a;
41        }
42    }
43}

45fn text(value: &Json) -> Result<&str, &'static str> {
46    value.text()
47}
48fn exact_i32(value: &Json, allow_string: bool) -> Result<i32, &'static str> {
49    let token = match value {
50        Json::Number(s) => s.as_str(),
51        Json::String(s) if allow_string => s.as_str(),
52        _ => return Err("SolidCraft circular profile values must be literal numbers"),
53    };
54    // SolidCraft serializes integral dimensions as decimals (including 0.0).
55    // The shared box parser intentionally rejects some zero spellings, so
56    // recognize a valid all-zero JSON decimal before delegating other values.
57    let mantissa = token.split_once(['e', 'E']).map_or(token, |(m, _)| m);
58    if mantissa
59        .bytes()
60        .all(|byte| matches!(byte, b'0' | b'.' | b'-'))
61    {
62        return Ok(0);
63    }
64    let (numerator, denominator) = crate::solidcraft_boxes::decimal_rational(token)?;
65    if denominator != 1 {
66        return Err("SolidCraft circular profile values must be integral millimetres");
67    }
68    i32::try_from(numerator).map_err(|_| "SolidCraft circular profile value outside integer range")
69}
70fn index(value: &Json, limit: usize) -> Result<usize, &'static str> {
71    let index: usize = value.number()?;
72    if index >= limit {
73        return Err("SolidCraft sketch point index out of range");
74    }
75    Ok(index)
76}
77fn safe_vertex_id(source: &str) -> Result<String, &'static str> {
78    if source.is_empty()
79        || source.len() > 22
80        || !source
81            .bytes()
82            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'.')
83    {
84        return Err("unsupported SolidCraft polygon point id");
85    }
86    Ok(format!("v_{}", source.replace('.', "_")))
87}
88fn timeline_feature<'a>(command: &'a Json, kind: &str) -> Result<&'a Json, &'static str> {
89    command.keys(&["command", "params"])?;
90    if text(command.field("command")?)? != "timeline.add" {
91        return Err("unsupported SolidCraft timeline command");
92    }
93    let params = command.field("params")?;
94    params.keys(&["feature"])?;
95    let feature = params.field("feature")?;
96    if text(feature.field("type")?)? != kind {
97        return Err("unsupported SolidCraft feature order");
98    }
99    Ok(feature)
100}
101fn feature_id(feature: &Json) -> Result<usize, &'static str> {
102    let id: usize = feature.field("id")?.number()?;
103    if id == 0 {
104        return Err("invalid SolidCraft feature id");
105    }
106    Ok(id)
107}
108fn signed_area2(vertices: &[Vertex]) -> i64 {
109    (0..vertices.len())
110        .map(|i| {
111            let a = vertices[i].xy_mm;
112            let b = vertices[(i + 1) % vertices.len()].xy_mm;
113            i64::from(a[0]) * i64::from(b[1]) - i64::from(a[1]) * i64::from(b[0])
114        })
115        .sum()
116}

118fn parse_profile(sketch_feature: &Json) -> Result<ParsedProfile, &'static str> {
119    sketch_feature.keys(&["id", "name", "plane", "sketch", "suppressed", "type"])?;
120    let sketch_id = feature_id(sketch_feature)?;
121    if sketch_feature.field("name")?.text()?.is_empty()
122        || sketch_feature.field("suppressed")? != &Json::Bool(false)
123    {
124        return Err("invalid or suppressed SolidCraft sketch");
125    }
126    let plane = sketch_feature.field("plane")?;
127    plane.keys(&["name", "type"])?;
128    if text(plane.field("type")?)? != "origin" || text(plane.field("name")?)? != "XY" {
129        return Err("SolidCraft circular profile must lie on origin XY");
130    }
131    let sketch = sketch_feature.field("sketch")?;
132    sketch.keys(&["constraints", "counters", "curves", "points"])?;
133    let points = sketch.field("points")?.items()?;
134    let curves = sketch.field("curves")?.items()?;
135    let constraints = sketch.field("constraints")?.items()?;
136    if !(4..=32).contains(&points.len()) || curves.len() < 4 || curves.len() > 36 {
137        return Err("SolidCraft circular sketch size out of range");
138    }
139    let counters = sketch.field("counters")?;
140    let nk: usize = if counters.field("k").is_ok() {
141        counters.keys(&["c", "k", "l"])?;
142        counters.field("k")?.number()?
143    } else {
144        counters.keys(&["c", "l"])?;
145        if !constraints.is_empty() {
146            return Err("missing SolidCraft sketch constraint counter");
147        }
148        0
149    };
150    let (nc, nl): (usize, usize) = (
151        counters.field("c")?.number()?,
152        counters.field("l")?.number()?,
153    );
154    if nk < constraints.len() {
155        return Err("invalid SolidCraft sketch constraint counter");
156    }
157    let mut coordinates = Vec::with_capacity(points.len());
158    let mut point_ids = BTreeSet::new();
159    for point in points {
160        point.keys(&["fixed", "id", "pos"])?;
161        if !matches!(point.field("fixed")?, Json::Bool(_)) {
162            return Err("invalid SolidCraft point fixed flag");
163        }
164        let id = text(point.field("id")?)?;
165        if id.is_empty() || !point_ids.insert(id.to_owned()) {
166            return Err("duplicate SolidCraft point id");
167        }
168        let pos = point.field("pos")?.items()?;
169        if pos.len() != 2 {
170            return Err("SolidCraft sketch point must have two coordinates");
171        }
172        coordinates.push([exact_i32(&pos[0], false)?, exact_i32(&pos[1], false)?]);
173    }
174    let mut dsu = DisjointSet::new(points.len());
175    let mut constraint_ids = BTreeSet::new();
176    for constraint in constraints {
177        constraint.keys(&["id", "p", "q", "type"])?;
178        let id = text(constraint.field("id")?)?;
179        if id.is_empty() || !constraint_ids.insert(id.to_owned()) {
180            return Err("duplicate SolidCraft constraint id");
181        }
182        if text(constraint.field("type")?)? != "coincident" {
183            return Err("unsupported SolidCraft sketch constraint");
184        }
185        let p = index(constraint.field("p")?, points.len())?;
186        let q = index(constraint.field("q")?, points.len())?;
187        if coordinates[p] != coordinates[q] {
188            return Err("inconsistent SolidCraft coincident constraint");
189        }
190        dsu.union(p, q);
191    }
192    let mut point_names = BTreeMap::<usize, String>::new();
193    let mut point_coordinates = BTreeMap::<usize, [i32; 2]>::new();
194    for (i, point) in points.iter().enumerate() {
195        let root = dsu.find(i);
196        let source_name = text(point.field("id")?)?;
197        if let Some(existing) = point_coordinates.insert(root, coordinates[i]) {
198            if existing != coordinates[i] {
199                return Err("coincident point coordinates differ");
200            }
201        }
202        point_names
203            .entry(root)
204            .and_modify(|name| {
205                if source_name < name {
206                    *name = source_name.to_owned();
207                }
208            })
209            .or_insert_with(|| source_name.to_owned());
210    }
211    let mut line_curves = Vec::new();
212    let mut circles = Vec::new();
213    let mut curve_ids = BTreeSet::new();
214    let mut used_roots = BTreeSet::new();
215    for curve in curves {
216        let kind = text(curve.field("type")?)?;
217        let id = text(curve.field("id")?)?;
218        if id.is_empty() || !curve_ids.insert(id.to_owned()) {
219            return Err("duplicate SolidCraft curve id");
220        }
221        if curve.field("construction")? != &Json::Bool(false) {
222            return Err("construction geometry is outside SC6");
223        }
224        match kind {
225            "line" => {
226                curve.keys(&["a", "b", "construction", "id", "type"])?;
227                if !id
228                    .bytes()
229                    .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'.')
230                {
231                    return Err("invalid SolidCraft line id");
232                }
233                let a = dsu.find(index(curve.field("a")?, points.len())?);
234                let b = dsu.find(index(curve.field("b")?, points.len())?);
235                if a == b {
236                    return Err("degenerate SolidCraft profile edge");
237                }
238                used_roots.insert(a);
239                used_roots.insert(b);
240                line_curves.push(LineCurve {
241                    id: id.to_owned(),
242                    a,
243                    b,
244                });
245            }
246            "circle" => {
247                curve.keys(&["c", "construction", "id", "r", "type"])?;
248                let center = dsu.find(index(curve.field("c")?, points.len())?);
249                used_roots.insert(center);
250                let radius_mm = exact_i32(curve.field("r")?, false)?;
251                if radius_mm < 1 {
252                    return Err("circle radius must be a positive integral millimetre");
253                }
254                circles.push(CircleCurve {
255                    id: id.to_owned(),
256                    center,
257                    radius_mm,
258                });
259            }
260            _ => return Err("SC6 supports only straight lines and circle sketch curves"),
261        }
262    }
263    if circles.is_empty() || circles.len() > 4 || nc < circles.len() {
264        return Err("SC6 requires one to four stored circle curves");
265    }
266    if nl < line_curves.len() {
267        return Err("invalid SolidCraft sketch curve counter");
268    }
269    for (i, point) in points.iter().enumerate() {
270        let root = dsu.find(i);
271        if !used_roots.contains(&root) && text(point.field("id")?)? != "origin" {
272            return Err("unused SolidCraft sketch point");
273        }
274    }
275    let mut adjacency = BTreeMap::<usize, Vec<(usize, usize)>>::new();
276    let mut unique_edges = BTreeSet::new();
277    for (ci, line) in line_curves.iter().enumerate() {
278        if !unique_edges.insert((line.a.min(line.b), line.a.max(line.b))) {
279            return Err("duplicate SolidCraft profile edge");
280        }
281        adjacency.entry(line.a).or_default().push((line.b, ci));
282        adjacency.entry(line.b).or_default().push((line.a, ci));
283    }
284    if adjacency.values().any(|neighbors| neighbors.len() != 2) {
285        return Err("outer SolidCraft profile must be one closed line cycle");
286    }
287    let mut consumed = BTreeSet::new();
288    let mut cycle_roots = Vec::new();
289    let mut selected_lines = BTreeSet::new();
290    let start = *adjacency
291        .keys()
292        .min_by_key(|root| point_names[root].clone())
293        .ok_or("missing polygon profile")?;
294    let mut current = start;
295    let mut previous = None;
296    loop {
297        cycle_roots.push(current);
298        let next = adjacency[&current]
299            .iter()
300            .filter(|(_, edge)| Some(*edge) != previous)
301            .min_by_key(|(root, edge)| (point_names[root].clone(), line_curves[*edge].id.clone()))
302            .copied()
303            .ok_or("open polygon profile")?;
304        if !consumed.insert(next.1) {
305            if next.0 == start {
306                break;
307            }
308            return Err("invalid polygon curve graph");
309        }
310        selected_lines.insert(line_curves[next.1].id.clone());
311        previous = Some(next.1);
312        current = next.0;
313        if current == start {
314            break;
315        }
316        if cycle_roots.len() > line_curves.len() {
317            return Err("invalid polygon profile cycle");
318        }
319    }
320    if consumed.len() != line_curves.len() || cycle_roots.len() < 3 {
321        return Err("SC6 requires exactly one simple outer line contour");
322    }
323    let vertices = cycle_roots
324        .iter()
325        .map(|root| {
326            Ok(Vertex {
327                id: safe_vertex_id(&point_names[root])?,
328                xy_mm: point_coordinates[root],
329            })
330        })
331        .collect::<Result<Vec<_>, &'static str>>()?;
332    if signed_area2(&vertices) == 0 {
333        return Err("zero-area SolidCraft outer profile");
334    }
335    let bores = circles
336        .into_iter()
337        .map(|circle| {
338            let center = point_coordinates[&circle.center];
339            let guard_mm = circle
340                .radius_mm
341                .checked_mul(2)
342                .ok_or("circular bore guard overflow")?;
343            Ok((circle.id, center, circle.radius_mm, guard_mm))
344        })
345        .collect::<Result<Vec<_>, &'static str>>()?;
346    let _ = nl;
347    Ok((vertices, selected_lines, bores, sketch_id))
348}

350fn validate_extrude(
351    feature: &Json,
352    sketch_id: usize,
353    line_ids: &BTreeSet<String>,
354) -> Result<i32, &'static str> {
355    feature.keys(&[
356        "body_names",
357        "extent",
358        "id",
359        "name",
360        "operation",
361        "param_names",
362        "profiles",
363        "sketch",
364        "suppressed",
365        "type",
366    ])?;
367    feature_id(feature)?;
368    if text(feature.field("type")?)? != "extrude"
369        || feature.field("suppressed")? != &Json::Bool(false)
370        || feature.field("name")?.text()?.is_empty()
371    {
372        return Err("invalid SolidCraft extrusion feature");
373    }
374    if feature.field("sketch")?.number::<usize>()? != sketch_id {
375        return Err("SolidCraft extrusion references another sketch");
376    }
377    if text(feature.field("operation")?)? != "new_body" {
378        return Err("SolidCraft circular extrusion must create a body");
379    }
380    let bodies = feature.field("body_names")?.items()?;
381    if bodies.len() != 1 || bodies[0].text()?.is_empty() {
382        return Err("SolidCraft circular extrusion must target one new body");
383    }
384    let params = feature.field("param_names")?.items()?;
385    let mut names = BTreeSet::new();
386    if params.is_empty()
387        || params.iter().any(|p| {
388            p.text()
389                .map_or(true, |s| s.is_empty() || !names.insert(s.to_owned()))
390        })
391    {
392        return Err("invalid SolidCraft extrusion parameter names");
393    }
394    let extent = feature.field("extent")?;
395    extent.keys(&["convention", "direction", "distance"])?;
396    if text(extent.field("convention")?)? != "total_distance"
397        || text(extent.field("direction")?)? != "positive"
398    {
399        return Err("unsupported SolidCraft extrusion direction");
400    }
401    let height = exact_i32(extent.field("distance")?, true)?;
402    if !(1..=64).contains(&height) {
403        return Err("SolidCraft extrusion height outside native range");
404    }
405    let profiles = feature.field("profiles")?;
406    profiles.keys(&["loops", "type"])?;
407    if text(profiles.field("type")?)? != "curves" {
408        return Err("circular profile selection must be explicit");
409    }
410    let loops = profiles.field("loops")?.items()?;
411    if loops.len() != 1 {
412        return Err("SolidCraft extrusion must select one outer contour");
413    }
414    let selected = loops[0]
415        .items()?
416        .iter()
417        .map(|v| v.text().map(str::to_owned))
418        .collect::<Result<BTreeSet<_>, _>>()?;
419    if selected.len() != loops[0].items()?.len() || selected != *line_ids {
420        return Err("SolidCraft extrusion selection differs from outer contour");
421    }
422    Ok(height)
423}

425/// Import a canonical two-feature SolidCraft circle-sketch extrusion as G20.
426pub fn import_circular_polygon_timeline(script: &Json) -> Result<Json, &'static str> {
427    script.keys(&["about", "commands"])?;
428    let about = script.field("about")?;
429    about.keys(&["design", "format", "replay", "version"])?;
430    if text(about.field("design")?)?.is_empty()
431        || text(about.field("format")?)? != "solidcraft/66"
432        || text(about.field("version")?)? != "0.48.1"
433        || text(about.field("replay")?)?.is_empty()
434    {
435        return Err("unsupported SolidCraft history metadata");
436    }
437    let commands = script.field("commands")?.items()?;
438    if commands.len() != 2 {
439        return Err("SC6 accepts exactly one sketch and one extrusion");
440    }
441    let sketch = timeline_feature(&commands[0], "sketch")?;
442    let (vertices, line_ids, circles, sketch_id) = parse_profile(sketch)?;
443    let extrude = timeline_feature(&commands[1], "extrude")?;
444    let height = validate_extrude(extrude, sketch_id, &line_ids)?;
445    let bores = circles
446        .into_iter()
447        .map(|(id, center_mm, radius_mm, guard_mm)| {
448            object([
449                ("id", Json::String(id)),
450                (
451                    "center_mm",
452                    Json::Array(
453                        center_mm
454                            .into_iter()
455                            .map(|v| Json::Number(v.to_string()))
456                            .collect(),
457                    ),
458                ),
459                ("radius_mm", Json::Number(radius_mm.to_string())),
460                ("guard_mm", Json::Number(guard_mm.to_string())),
461            ])
462        })
463        .collect();
464    let recipe = object([
465        ("family", Json::String("polygon_circular_bores_v1".into())),
466        (
467            "outer",
468            object([
469                ("id", Json::String("outer".into())),
470                (
471                    "vertices",
472                    Json::Array(
473                        vertices
474                            .into_iter()
475                            .map(|v| {
476                                object([
477                                    ("id", Json::String(v.id)),
478                                    (
479                                        "xy_mm",
480                                        Json::Array(
481                                            v.xy_mm
482                                                .into_iter()
483                                                .map(|n| Json::Number(n.to_string()))
484                                                .collect(),
485                                        ),
486                                    ),
487                                ])
488                            })
489                            .collect(),
490                    ),
491                ),
492            ]),
493        ),
494        ("bores", Json::Array(bores)),
495        ("height_mm", Json::Number(height.to_string())),
496        (
497            "top_shift_mm",
498            Json::Array(vec![Json::Number("0".into()), Json::Number("0".into())]),
499        ),
500    ]);
501    circular_polygon_json::build(&recipe)
