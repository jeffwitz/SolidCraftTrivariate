1//! Versioned exchange for polygonal extrusions with bounded axial bore intervals.
2use crate::{
3    bracket_json::{decode_blocks, design, encode, topology},
4    circular_polygon_json::bore_values,
5    interval_bores::{build_interval_bore_prism, IntervalBore, IntervalBorePrism},
6    interval_bores_audit::audit_interval_bore_prism,
7    json::{object, Json},
8    polygon_json::{pair, profile},
9};
10
11pub(crate) const FAMILY: &str = "polygon_interval_bores_v1";
12pub(crate) const METHOD: &str = "certified_polygon_interval_bores";
13const PARAMETERS: [&str; 4] = ["outer", "bores", "height_mm", "top_shift_mm"];
14
15pub(crate) fn parse(data: &Json) -> Result<IntervalBorePrism, &'static str> {
16    let list = data.field("bores")?.items()?;
17    if !(1..=4).contains(&list.len()) {
18        return Err("interval_bore_count");
19    }
20    let mut bores = Vec::with_capacity(list.len());
21    for v in list {
22        v.keys(&["id", "center_mm", "radius_mm", "guard_mm", "z_interval_mm"])?;
23        bores.push(IntervalBore {
24            bore: bore_values(v)?,
25            z_interval_mm: pair(v.field("z_interval_mm")?)?,
26        });
27    }
28    Ok(IntervalBorePrism {
29        outer: profile(data.field("outer")?)?,
30        bores,
31        height_mm: data.field("height_mm")?.number()?,
32        top_shift_mm: pair(data.field("top_shift_mm")?)?,
33    })
34}
35
36pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
37    recipe.keys(&[&["family"][..], &PARAMETERS].concat())?;
38    let r = parse(recipe)?;
39    let blocks = build_interval_bore_prism(&r)?;
40    let certified = audit_interval_bore_prism(&r, &blocks)?;
41    encoded_model(recipe, blocks, &certified.certificate)
42}
43
44pub(crate) fn encoded_model(
45    recipe: &Json,
46    blocks: Vec<crate::bracket::BracketBlock>,
47    certificate: &crate::bracket_audit::BracketCertificate,
48) -> Result<Json, &'static str> {
49    let mut model = encode(FAMILY, blocks, certificate);
50    if let Json::Object(fields) = &mut model {
51        for k in PARAMETERS {
52            fields.insert(k.into(), recipe.field(k)?.clone());
53        }
54    }
55    Ok(model)
56}
57
58pub(crate) fn audit(data: &Json) -> Result<Json, &'static str> {
59    data.keys(
60        &[
61            &["schema", "blocks", "topology", "design_faces"][..],
62            &PARAMETERS,
63        ]
64        .concat(),
65    )?;
66    let r = parse(data)?;
67    let blocks = decode_blocks(data)?;
68    let cert = audit_interval_bore_prism(&r, &blocks)?;
69    let c = &cert.certificate;
70    if data.field("topology")? != &topology(&c.incidence)
71        || data.field("design_faces")? != &design(c)
72    {
73        return Err("interval_bore_topology_or_provenance_mismatch");
74    }
75    let n = |v: i64| Json::Number(v.to_string());
76    Ok(object([
77        ("status", Json::String("valid".into())),
78        ("method", Json::String(METHOD.into())),
79        ("blocks", n(blocks.len() as i64)),
80        ("shared_faces", n(c.incidence.interfaces.len() as i64)),
81        ("exterior_sides", n(c.incidence.exterior.len() as i64)),
82        ("through_bores", n(cert.through_bores as i64)),
83        ("cavities", n(cert.enclosed_voids as i64)),
84        (
85            "determinant_lower_bound_mm3",
86            Json::Number(c.determinant_lower_bound_mm3.to_string()),
87        ),
88        ("hausdorff_error_upper_bound_mm", n(0)),
89        (
90            "volume_exact",
91            object([
92                (
93                    "twice_rational_part_mm3",
94                    n(cert.outer_area_twice_mm2 * i64::from(r.height_mm)),
95                ),
96                (
97                    "pi_coefficient_mm3",
98                    n(-cert.removed_pi_volume_coefficient_mm3),
99                ),
100            ]),
101        ),
102    ]))
