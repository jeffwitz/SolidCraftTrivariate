40fn int(value: i64) -> R {
41    R::new(i128::from(value), 1).expect("integer ratio")

44fn div(a: R, b: R) -> Result<R, Error> {
45    if b.numerator() == 0 {
46        return Err("division by zero in S3 exact geometry");
47    }
48    a.checked_mul(R::new(b.denominator(), b.numerator())?)
49}

51fn read_pair(value: &Json) -> Result<[i64; 2], Error> {
52    let [a, b] = value.items()? else {
53        return Err("expected a two-integer pair");
54    };
55    Ok([a.number()?, b.number()?])
56}

58fn read_ratio(value: &Json) -> Result<R, Error> {
59    let [n, d] = read_pair(value)?;
60    if d <= 0 {
61        return Err("rational denominator must be positive");
62    }
63    R::new(i128::from(n), i128::from(d))
64}

70fn validate_parent(parent: &Json) -> Result<Output, Error> {
71    let source = parent.field("source_recipe")?;
72    if source.field("family")?.text()? != g2_cross_plate_json::FAMILY {
73        return Err("S3 parent must be a delivered g2_cross_plate_v1 model");
74    }
75    g2_cross_plate_json::audit(parent)?;
76    let blocks = crate::bracket_json::decode_blocks(parent)?;
77    let faces = crate::bracket_json::decode_design_faces(parent)?;
78    Ok((blocks, faces))
79}

81fn validate_operation(
82    operation: &Json,
83    support_z: R,
84    stock_bounds: [[R; 2]; 2],
85) -> Result<(R, R, R, R, R), Error> {
86    operation.keys(&["support_feature", "frame", "direction", "depth_mm", "loop"])?;
87    if operation.field("support_feature")?.text()? != "rect_x/zmin" {
88        return Err("S3 pocket support must be rect_x/zmin");
89    }
90    let frame = operation.field("frame")?;
91    frame.keys(&["origin_mm", "u", "v"])?;
92    let origin = vector(frame.field("origin_mm")?)?;
93    let support_z_i64 = i64::try_from(support_z.numerator() / support_z.denominator())
94        .map_err(|_| "S3 support height exceeds integer frame range")?;
95    if origin != [0, 0, support_z_i64]
96        || vector(frame.field("u")?)? != [1, 0, 0]
97        || vector(frame.field("v")?)? != [0, 1, 0]
98        || vector(operation.field("direction")?)? != [0, 0, -1]
99    {
100        return Err("S3 pocket must use rect_x/zmin and run in the negative Z direction");
101    }
102    let depth = int(operation.field("depth_mm")?.number::<i64>()?);
103    if !depth.cmp_value(&R::zero()).is_gt() || !depth.cmp_value(&support_z).is_lt() {
104        return Err("S3 pocket depth must remain strictly inside the stock");
105    }
106    let loop_json = operation.field("loop")?;
107    loop_json.keys(&["id", "primitive"])?;
108    let primitive = loop_json.field("primitive")?;
109    primitive.keys(&["kind", "min", "max"])?;
110    if primitive.field("kind")?.text()? != "rectangle" {
111        return Err("S3 pocket loop must be a native rectangle");
112    }
113    let [x_min_json, y_min_json] = primitive.field("min")?.items()? else {
114        return Err("rectangle min must contain x and y");
115    };
116    let [x_max_json, y_max_json] = primitive.field("max")?.items()? else {
117        return Err("rectangle max must contain x and y");
118    };
119    let x_min = read_ratio(x_min_json)?;
120    let y_min = read_ratio(y_min_json)?;
121    let x_max = read_ratio(x_max_json)?;
122    let y_max = read_ratio(y_max_json)?;
123    if !x_min.cmp_value(&stock_bounds[0][0]).is_ge()
124        || !x_max.cmp_value(&stock_bounds[0][1]).is_le()
125        || !y_min.cmp_value(&stock_bounds[1][0]).is_ge()
126        || !y_max.cmp_value(&stock_bounds[1][1]).is_le()
127        || !x_min.cmp_value(&x_max).is_lt()
128        || !y_min.cmp_value(&y_max).is_lt()
129    {
130        return Err("S3 rectangle is outside the stock bounds");
131    }
132    Ok((x_min, x_max, y_min, y_max, depth))
133}

135fn metadata(block: &BracketBlock) -> Result<(usize, usize, usize, usize), Error> {
136    // Stable S2 IDs encode sector, piece, Z layer, and source Bézier span.
137    let fields: Vec<_> = block.id.split('_').collect();
138    if fields.len() != 5 || fields[0] != "g2" || fields[1].len() < 2 || fields[2].len() < 2 {
139        return Err("S3 requires canonical S2 block identifiers");
140    }
141    let sector = fields[1][1..].parse().map_err(|_| "invalid S2 sector ID")?;
142    let piece = fields[2][1..].parse().map_err(|_| "invalid S2 piece ID")?;
143    let layer = fields[3]
144        .strip_prefix('z')
145        .ok_or("invalid S2 layer ID")?
146        .parse()
147        .map_err(|_| "invalid S2 layer ID")?;
148    let source = fields[4]
149        .strip_prefix("src")
150        .ok_or("invalid S2 span ID")?
151        .parse()
152        .map_err(|_| "invalid S2 span ID")?;
153    Ok((sector, piece, layer, source))
154}

156fn flat_index(degree: [usize; 3], i: usize, j: usize, k: usize) -> usize {
157    (i * (degree[1] + 1) + j) * (degree[2] + 1) + k
158}

160fn interpolate(a: H4, b: H4, t: R) -> Result<H4, Error> {
161    let mut q = [R::zero(); 4];
162    for d in 0..4 {
163        q[d] = a[d].checked_add(b[d].checked_sub(a[d])?.checked_mul(t)?)?;
164    }
165    Ok(q)
166}

168fn split_axis(
169    block: &BracketBlock,
170    axis: usize,
171    t: R,
172) -> Result<(BracketBlock, BracketBlock), Error> {
173    let degree = block.degrees[axis];
174    if !(1..=3).contains(&degree) || t == R::zero() || t == R::one() {
175        return Err("S3 subdivision must be interior to a linear or cubic axis");
176    }
177    let mut left = vec![[R::zero(); 4]; block.controls.len()];
178    let mut right = vec![[R::zero(); 4]; block.controls.len()];
179    let mut other = Vec::new();
180    for a in 0..3 {
181        if a != axis {
182            other.push(a);
183        }
184    }
185    let count0 = block.degrees[other[0]] + 1;
186    let count1 = block.degrees[other[1]] + 1;
187    for a in 0..count0 {
188        for b in 0..count1 {
189            let mut row: Vec<H4> = (0..=degree)
190                .map(|s| {
191                    let mut q = [0; 3];
192                    q[axis] = s;
193                    q[other[0]] = a;
194                    q[other[1]] = b;
195                    block.controls[flat_index(block.degrees, q[0], q[1], q[2])].map(int)
196                })
197                .collect();
198            let mut l = vec![row[0]];
199            let mut r = vec![*row.last().ok_or("empty S3 subdivision row")?];
200            while row.len() > 1 {
201                row = row
202                    .windows(2)
203                    .map(|pair| interpolate(pair[0], pair[1], t))
204                    .collect::<Result<_, _>>()?;
205                l.push(row[0]);
206                r.push(*row.last().ok_or("empty S3 subdivision row")?);
207            }
208            r.reverse();
209            for s in 0..=degree {
210                for (target, value) in [(&mut left, l[s]), (&mut right, r[s])] {
211                    let mut q = [0; 3];
212                    q[axis] = s;
213                    q[other[0]] = a;
214                    q[other[1]] = b;
215                    target[flat_index(block.degrees, q[0], q[1], q[2])] = value;
216                }
217            }
218        }
219    }
220    let prefix = block.id.clone();
221    Ok((
222        BracketBlock {
223            id: format!("{prefix}_cut{axis}a"),
224            degrees: block.degrees,
225            controls: integerize(left)?,
226        },
227        BracketBlock {
228            id: format!("{prefix}_cut{axis}b"),
229            degrees: block.degrees,
230            controls: integerize(right)?,
231        },
232    ))
233}

235fn gcd(mut a: u128, mut b: u128) -> u128 {
236    while b != 0 {
237        (a, b) = (b, a % b);
238    }
239    a
240}

242fn lcm(a: i128, b: i128) -> Result<i128, Error> {
243    let g = i128::try_from(gcd(a.unsigned_abs(), b.unsigned_abs()))
244        .map_err(|_| "S3 scale exceeds i128")?;
245    (a / g).checked_mul(b).ok_or("S3 scale exceeds i128")
246}

248fn integerize(net: Vec<H4>) -> Result<Vec<[i64; 4]>, Error> {
249    let mut scale = 1_i128;
250    for p in &net {
251        for c in p {
252            scale = lcm(scale, c.denominator())?;
253        }
254    }
255    let mut integers = Vec::with_capacity(net.len());
256    let mut common = 0_u128;
257    for p in net {
258        let q = p.map(|v| {
259            v.numerator()
260                .checked_mul(scale / v.denominator())
261                .ok_or("S3 control exceeds i128")
262        });
263        let q = q.into_iter().collect::<Result<Vec<_>, _>>()?;
264        for v in &q {
265            common = gcd(common, v.unsigned_abs());
266        }
267        integers.push(q);
268    }
269    let common = i128::try_from(common).map_err(|_| "S3 gauge exceeds i128")?;
270    integers
271        .into_iter()
272        .map(|p| {
273            p.into_iter()
274                .map(|v| i64::try_from(v / common).map_err(|_| "S3 control exceeds i64"))
275                .collect::<Result<Vec<_>, _>>()?
276                .try_into()
277                .map_err(|_| "invalid S3 homogeneous control")
278        })
279        .collect()
280}

282fn extract_curves(block: &BracketBlock) -> Result<(Vec<Point2H>, Vec<Point2H>, R, R), Error> {
283    if block.degrees != [3, 1, 1] {
284        return Err("S3 parent blocks must be cubic radial charts");
285    }
286    let mut outer = Vec::with_capacity(4);
287    let mut inner = Vec::with_capacity(4);
288    let mut z_bounds = None;
289    for u in 0..4 {
290        for v in 0..2 {
291            let control = block.controls[flat_index(block.degrees, u, v, 0)];
292            let h = [int(control[0]), int(control[1]), int(control[3])];
293            if v == 0 {
294                outer.push(h);
295            } else {
296                inner.push(h);
297            }
298            let lo = R::new(i128::from(control[2]), i128::from(control[3]))?;
299            let high_control = block.controls[flat_index(block.degrees, u, v, 1)];
300            let hi = R::new(i128::from(high_control[2]), i128::from(high_control[3]))?;
301            if let Some((z0, z1)) = z_bounds {
302                if z0 != lo || z1 != hi {
303                    return Err("S3 source chart is not an axial extrusion");
304                }
305            } else {
306                z_bounds = Some((lo, hi));
307            }
308        }
309    }
310    let (z0, z1) = z_bounds.ok_or("empty S3 parent chart")?;
311    Ok((outer, inner, z0, z1))
312}

314fn patch_block(
315    id: String,
316    first: &[Point2H],
317    second: &[Point2H],
318    z0: R,
319    z1: R,
320) -> Result<BracketBlock, Error> {
321    if first.len() != 4 || second.len() != 4 {
322        return Err("S3 guide must be cubic");
323    }
324    let mut net = Vec::with_capacity(16);
325    for i in 0..4 {
326        for h in [first[i], second[i]] {
327            for z in [z0, z1] {
328                net.push([h[0], h[1], z.checked_mul(h[2])?, h[2]]);
329            }
330        }
331    }
332    Ok(BracketBlock {
333        id,
334        degrees: [3, 1, 1],
335        controls: integerize(net)?,
336    })
337}

357fn homogeneous_face_y(block: &BracketBlock, side: &BlockSide) -> Result<Option<R>, Error> {
358    if side.axis != 0 {
359        return Ok(None);
360    }
361    let u = if side.upper { block.degrees[0] } else { 0 };
362    let mut y = None;
363    for v in 0..=block.degrees[1] {
364        for w in 0..=block.degrees[2] {
365            let p = block.controls[flat_index(block.degrees, u, v, w)];
366            let value = R::new(i128::from(p[1]), i128::from(p[3]))?;
367            if y.is_some_and(|old| old != value) {
368                return Ok(None);
369            }
370            y = Some(value);
371        }
372    }
373    Ok(y)
374}

376fn homogeneous_face_x_bounds(
377    block: &BracketBlock,
378    side: &BlockSide,
379) -> Result<Option<[R; 2]>, Error> {
380    if side.axis != 0 {
381        return Ok(None);
382    }
383    let u = if side.upper { block.degrees[0] } else { 0 };
384    let mut bounds: Option<[R; 2]> = None;
385    for v in 0..=block.degrees[1] {
386        for w in 0..=block.degrees[2] {
387            let p = block.controls[flat_index(block.degrees, u, v, w)];
388            let value = R::new(i128::from(p[0]), i128::from(p[3]))?;
389            bounds = Some(bounds.map_or([value, value], |[lo, hi]| {
390                [
391                    if value.cmp_value(&lo).is_lt() {
392                        value
393                    } else {
394                        lo
395                    },
396                    if value.cmp_value(&hi).is_gt() {
397                        value
398                    } else {
399                        hi
400                    },
401                ]
402            }));
403        }
404    }
405    Ok(bounds)
406}

408/// Build the S3 model only from an audited, embedded S2 parent model.
409pub fn construct(recipe: &Json) -> Result<Output, Error> {
410    recipe.keys(&["family", "parent_model", "operation"])?;
411    if recipe.field("family")?.text()? != "s3_face_pocket_v1" {
412        return Err("unsupported S3 face-pocket family");
413    }
414    let parent = recipe.field("parent_model")?;
415    // Reject non-S2 parents before invoking the comparatively expensive audit.
416    if parent.field("source_recipe")?.field("family")?.text()? != g2_cross_plate_json::FAMILY {
417        return Err("S3 parent must be a delivered g2_cross_plate_v1 model");
418    }
419    let (base_blocks, base_faces) = validate_parent(parent)?;
420    if !base_faces
421        .iter()
422        .any(|(_, feature)| feature == "rect_x/zmin")
423    {
424        return Err("S3 parent has no rect_x/zmin support face");
425    }
426
427    let mut stock_bounds = [[R::zero(), R::zero()]; 2];
428    let mut first_control = true;
429    for block in &base_blocks {
430        for p in &block.controls {
431            let x = R::new(i128::from(p[0]), i128::from(p[3]))?;
432            let y = R::new(i128::from(p[1]), i128::from(p[3]))?;
433            if first_control {
434                stock_bounds = [[x, x], [y, y]];
435                first_control = false;
436            } else {
437                if x.cmp_value(&stock_bounds[0][0]).is_lt() {
438                    stock_bounds[0][0] = x;
439                }
440                if x.cmp_value(&stock_bounds[0][1]).is_gt() {
441                    stock_bounds[0][1] = x;
442                }
443                if y.cmp_value(&stock_bounds[1][0]).is_lt() {
444                    stock_bounds[1][0] = y;
445                }
446                if y.cmp_value(&stock_bounds[1][1]).is_gt() {
447                    stock_bounds[1][1] = y;
448                }
449            }
450        }
451    }
452    let support = base_faces
453        .iter()
454        .find(|(_, feature)| feature == "rect_x/zmin")
455        .ok_or("S3 parent has no rect_x/zmin support face")?;
456    if support.0.axis != 2 || !support.0.upper {
457        return Err("S3 support feature must be the upper-Z face of the recessed rectangle");
458    }
459    let support_block = base_blocks
460        .iter()
461        .find(|b| b.id == support.0.block)
462        .ok_or("S3 support face references an unknown block")?;
463    let (_, _, _, support_z) = extract_curves(support_block)?;
464    let (x_min, x_max, y_min, y_max, depth) =
465        validate_operation(recipe.field("operation")?, support_z, stock_bounds)?;
466    let floor_z = support_z.checked_sub(depth)?;
467    let support_blocks: std::collections::BTreeSet<_> = base_faces
468        .iter()
469        .filter(|(_, feature)| feature == "rect_x/zmin")
470        .map(|(side, _)| side.block.clone())
471        .collect();
472    let mut right_outer = Vec::new();
473    let mut right_inner = Vec::new();
474    let mut target_plans = BTreeMap::<(usize, usize, usize), SectorPlan>::new();
475    let mut auxiliary_inner_min: Option<R> = None;
476    for block in &base_blocks {
477        let (_, _, layer, _) = metadata(block)?;
478        if layer != 0 || !support_blocks.contains(&block.id) {
479            continue;
480        }
481        let (outer, inner, _, _) = extract_curves(block)?;
482        for p in &inner {
483            let x = div(p[0], p[2])?;
484            auxiliary_inner_min = Some(auxiliary_inner_min.map_or(x, |old| {
485                if x.cmp_value(&old).is_lt() {
486                    x
487                } else {
488                    old
489                }
490            }));
491        }
492        let on_stock_xmax = outer
493            .iter()
494            .all(|p| div(p[0], p[2]).ok() == Some(stock_bounds[0][1]));
495        if on_stock_xmax {
496            right_outer.push(outer);
497            right_inner.push(inner);
498            continue;
499        }
500        if !outer
501            .iter()
502            .all(|p| div(p[0], p[2]).ok() == Some(stock_bounds[0][0]))
503        {
504            continue;
505        }
506        if !outer
507            .iter()
508            .zip(&inner)
509            .all(|(a, b)| div(a[1], a[2]).ok() == div(b[1], b[2]).ok())
510        {
511            continue;
512        }
513        let (sector, piece, _, source_span) = metadata(block)?;
514        let plan = plan_sector(
515            &outer,
516            &inner,
517            stock_bounds[0][0],
518            x_min,
519            x_max,
520            y_min,
521            y_max,
522        )?;
523        if plan.remove_interval.is_some() {
524            target_plans.insert((sector, piece, source_span), plan);
525        }
526    }
527    certify_xmax_void(&right_outer, &right_inner, x_max, y_min, y_max)?;
528    if target_plans.is_empty() {
529        return Err("S3 rectangle does not meet a certified left-wing bore sector");
530    }
531    let auxiliary_guide_x = auxiliary_inner_min
532        .ok_or("S3 parent has no inner profile")?
533        .checked_add(stock_bounds[0][0])?
534        .checked_mul(R::new(1, 2)?)?;
535    let old_faces = base_face_map(&base_faces);
536    let mut output = Vec::new();
537    let mut tags = BTreeMap::<String, BlockTag>::new();
538    for base in &base_blocks {
539        let (sector, piece, layer, source_span) = metadata(base)?;
540        let mut u_parts = vec![(
541            base.clone(),
542            None,
543            [R::zero(), R::one()],
544            false,
545            false,
546            false,
547            false,
548        )];
549        if let Some(plan) = target_plans.get(&(sector, piece, source_span)) {
550            let plan = plan.clone();
551            let _interval = plan
552                .remove_interval
553                .ok_or("S3 pocket does not intersect its certified sector")?;
554            let mut cuts = plan.u_cuts.clone();
555            cuts.sort_by(|a, b| a.cmp_value(b));
556            let mut remaining = base.clone();
557            let mut previous = R::zero();
558            let mut pieces = Vec::new();
559            for (index, cut) in cuts
560                .iter()
561                .copied()
562                .enumerate()
563                .skip(1)
564                .take(cuts.len().saturating_sub(2))
565            {
566                let local = div(cut.checked_sub(previous)?, R::one().checked_sub(previous)?)?;
567                let (left, right) = split_axis(&remaining, 0, local)?;
568                let inside = plan
569                    .slices
570                    .iter()
571                    .any(|s| s.interval == [previous, cut] && s.status == SliceStatus::InsideBand);
572                pieces.push((
573                    left,
574                    Some(index - 1),
575                    [previous, cut],
576                    inside,
577                    plan.omit_outer,
578                    plan.omit_inner,
579                    plan.wall_xmin,
580                ));
581                remaining = right;
582                previous = cut;
583            }
584            let last = *cuts.last().ok_or("S3 planner returned no cuts")?;
585            let inside = plan
586                .slices
587                .iter()
588                .any(|s| s.interval == [previous, last] && s.status == SliceStatus::InsideBand);
589            pieces.push((
590                remaining,
591                Some(pieces.len()),
592                [previous, last],
593                inside,
594                plan.omit_outer,
595                plan.omit_inner,
596                plan.wall_xmin,
597            ));
598            u_parts = pieces;
599        }
600        for (u_block, u_half, u_interval, pocket_slice, omit_outer, omit_inner, wall_xmin) in
601            u_parts
602        {
603            let (outer, inner, z0, z1) = extract_curves(&u_block)?;
604            let guide_x = if x_min == R::zero() {
605                auxiliary_guide_x
606            } else {
607                x_min
608            };
609            let guide = radial_guide(&outer, &inner, stock_bounds[0][0], guide_x)?;
610            if guide.len() != 4 {
611                return Err("radial guide degree does not match its source chart");
612            }
613            let mut z_parts = vec![(z0, z1, None)];
614            if layer == 0 {
615                if z0 != R::zero() || z1 != support_z {
616                    return Err("S3 expects the registered S2 lower axial layer");
617                }
618                z_parts = vec![(R::zero(), floor_z, Some(0)), (floor_z, support_z, Some(1))];
619            }
620            for (za, zb, z_half) in z_parts {
621                for radial in [RadialPart::Outer, RadialPart::Inner] {
622                    let omitted = pocket_slice
623                        && z_half == Some(1)
624                        && match radial {
625                            RadialPart::Outer => omit_outer,
626                            RadialPart::Inner => omit_inner,
627                        };
628                    if omitted {
629                        continue;
630                    }
631                    let (a, b) = match radial {
632                        RadialPart::Outer => (&outer, &guide),
633                        RadialPart::Inner => (&guide, &inner),
634                    };
635                    let radial_name = if radial == RadialPart::Outer {
636                        "o"
637                    } else {
638                        "i"
639                    };
640                    let half_name = u_half.map_or_else(|| "u".to_owned(), |i| format!("u{i}"));
641                    let z_name = match z_half {
642                        Some(0) => format!("z0_{}", floor_z.numerator() / floor_z.denominator()),
643                        Some(1) => format!("z{}_4", floor_z.numerator() / floor_z.denominator()),
644                        _ => "zbase".to_owned(),
645                    };
646                    let id = format!("s3_{radial_name}_{half_name}_{z_name}_{}", u_block.id);
647                    let block = patch_block(id.clone(), a, b, za, zb)?;
648                    let certificate =
649                        certify_spatial(block.degrees, &block.controls, Default::default());
650                    if certificate.status != CertificateStatus::Valid {
651                        return Err("S3 chart lacks a positive Bernstein determinant certificate");
652                    }
653                    tags.insert(
654                        id.clone(),
655                        BlockTag {
656                            parent_id: base.id.clone(),
657                            z_half,
658                            z0: za,
659                            z1: zb,
660                            u_interval,
661                            pocket_slice,
662                            omit_outer,
663                            wall_xmin,
664                            radial,
665                        },
666                    );
667                    output.push(block);
668                }
669            }
670        }
671    }
672
673    let incidence = enumerate_faces(&output)?;
674    let by_id: BTreeMap<_, _> = output.iter().map(|b| (b.id.as_str(), b)).collect();
675    let mut faces = Vec::new();
676    for side in incidence.exterior {
677        let tag = tags.get(&side.block).ok_or("missing S3 block provenance")?;
678        let mut feature = None;
679        // Preserve all original S2 exterior features through the exact chart split.
680        match side.axis {
681            0 => {
682                let is_parent_end = (!side.upper && tag.u_interval[0] == R::zero())
683                    || (side.upper && tag.u_interval[1] == R::one());
684                if is_parent_end {
685                    feature = parent_label(&tags, &old_faces, &side.block, 0, side.upper);
686                }
687                if feature.is_none()
688                    && tag.z_half == Some(1)
689                    && tag.z0 == floor_z
690                    && tag.z1 == support_z
691                {
692                    let y = homogeneous_face_y(by_id[side.block.as_str()], &side)?;
693                    let x_bounds = homogeneous_face_x_bounds(by_id[side.block.as_str()], &side)?;
694                    let within_rectangle = x_bounds.is_some_and(|[lo, hi]| {
695                        lo.cmp_value(&x_min).is_ge() && hi.cmp_value(&x_max).is_le()
696                    });
697                    if side.upper
698                        && y == Some(y_max)
699                        && within_rectangle
700                        && (tag.radial == RadialPart::Inner || x_min == stock_bounds[0][0])
701                    {
702                        feature = Some("pocket/wall_ymax".to_owned());
703                    } else if !side.upper
704                        && y == Some(y_min)
705                        && within_rectangle
706                        && (tag.radial == RadialPart::Inner || x_min == stock_bounds[0][0])
707                    {
708                        feature = Some("pocket/wall_ymin".to_owned());
709                    }
710                }
711            }
712            1 => {
713                let parent_side = match tag.radial {
714                    RadialPart::Outer => !side.upper,
715                    RadialPart::Inner => side.upper,
716                };
717                if parent_side {
718                    feature = parent_label(&tags, &old_faces, &side.block, 1, side.upper);
719                } else if tag.wall_xmin
720                    && tag.pocket_slice
721                    && tag.z_half == Some(1)
722                    && tag.z0 == floor_z
723                    && tag.z1 == support_z
724                    && tag.radial == RadialPart::Outer
725                    && side.upper
726                {
727                    feature = Some("pocket/wall_xmin".to_owned());
728                }
729            }
730            2 => {
731                if let Some(z_half) = tag.z_half {
732                    if z_half == 0 && !side.upper {
733                        feature = parent_label(&tags, &old_faces, &side.block, 2, false);
734                    } else if z_half == 1 && side.upper {
735                        feature = parent_label(&tags, &old_faces, &side.block, 2, true);
736                    }
737                    if feature.is_none()
738                        && tag.pocket_slice
739                        && z_half == 0
740                        && side.upper
741                        && (tag.radial == RadialPart::Inner || tag.omit_outer)
742                    {
743                        feature = Some("pocket/floor".to_owned());
744                    }
745                } else {
746                    feature = parent_label(&tags, &old_faces, &side.block, 2, side.upper);
747                }
748            }
749            _ => return Err("unexpected S3 exterior axis"),
750        }
751        faces.push((side, feature.ok_or("unclassified S3 exterior face")?));
752    }
753    Ok((output, faces))
754}
