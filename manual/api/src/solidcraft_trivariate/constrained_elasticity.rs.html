1//! Matrix-free solution of elasticity with exact affine scalar constraints.
2//!
3//! The three components use independent [`Lift`] maps. The reduced system is
4//! applied as C^T K C and its right hand side is C^T(f-Kg), without forming a
5//! dense constraint matrix or a dense reduced stiffness matrix.
6
7use crate::constraint_system::Lift;
8use crate::elasticity::Stiffness;
9
10/// A displacement satisfying the supplied affine constraints.
11#[derive(Debug, Clone)]
12pub struct ConstrainedSolution {
13    /// Reconstructed interleaved displacement, in mm.
14    pub displacement_mm: Vec<f64>,
15    /// Reduced unknowns, concatenated in x, y, z component order.
16    pub reduced_values: Vec<f64>,
17    /// Norm of the reduced residual divided by the reduced right-hand-side norm.
18    pub relative_residual: f64,
19    /// Number of conjugate-gradient iterations.
20    pub iterations: usize,
21    /// Strain energy in N mm, computed as 0.5 u^T K u.
22    pub strain_energy_n_mm: f64,
23}
24
25#[derive(Clone, Copy)]
26struct Term {
27    index: usize,
28    coefficient: f64,
29}
30
31/// Solve `K u = f` in the affine space defined independently for each
32/// displacement component by `component_lifts`.
33///
34/// The CSR stiffness uses interleaved xyz vectors. Each lift describes the
35/// exact affine map from its free roots to all scalar nodes. The reduced
36/// system is solved by diagonally preconditioned conjugate gradients.
37pub fn solve_constrained(
38    stiffness: &Stiffness,
39    component_lifts: [&Lift; 3],
40    force_n: &[f64],
41    tolerance: f64,
42    max_iterations: usize,
43) -> Result<ConstrainedSolution, &'static str> {
44    solve_with_guess(
45        stiffness,
46        component_lifts,
47        force_n,
48        tolerance,
49        max_iterations,
50        None,
51        None,
52    )
53}

77/// A fixed symmetric positive definite operator on the mounted reduced space.
78pub trait ReducedPreconditioner {

82/// Solve with a supplied reduced preconditioner in the same component-major
83/// ordering as the returned reduced values. Convergence uses the full measured
84/// force residual projected back through the exact affine constraints.
85pub fn solve_preconditioned(
86    stiffness: &Stiffness,
87    component_lifts: [&Lift; 3],
88    force_n: &[f64],
89    tolerance: f64,
90    max_iterations: usize,
91    preconditioner: &dyn ReducedPreconditioner,
92) -> Result<ConstrainedSolution, &'static str> {
93    solve_with_guess(
94        stiffness,
95        component_lifts,
96        force_n,
97        tolerance,
98        max_iterations,
99        None,
100        Some(preconditioner),
101    )

104fn solve_with_guess(
105    stiffness: &Stiffness,
106    component_lifts: [&Lift; 3],
107    force_n: &[f64],
108    tolerance: f64,
109    max_iterations: usize,
110    guess: Option<&[f64]>,
111    preconditioner: Option<&dyn ReducedPreconditioner>,
112) -> Result<ConstrainedSolution, &'static str> {
113    if !tolerance.is_finite() || tolerance <= 0. || tolerance >= 1. {
114        return Err("tolerance must be finite and in (0,1)");
115    }
116    if max_iterations == 0 {
117        return Err("iteration limit must be positive");
118    }
119    let n = stiffness
120        .offsets
121        .len()
122        .checked_sub(1)
123        .ok_or("invalid CSR offsets")?;
124    if n == 0 || stiffness.size() != force_n.len() || force_n.iter().any(|x| !x.is_finite()) {
125        return Err("invalid stiffness or force dimensions");
126    }
127    stiffness.validate()?;
128
129    let mut component_maps: Vec<Vec<Vec<Term>>> = Vec::with_capacity(3);
130    let mut component_offsets: Vec<Vec<f64>> = Vec::with_capacity(3);
131    let mut reduced_size = 0usize;
132    for lift in component_lifts {
133        if lift.lines.len() != n {
134            return Err("constraint lift has wrong number of rows");
135        }
136        let base = reduced_size;
137        reduced_size = reduced_size
138            .checked_add(lift.free_roots.len())
139            .ok_or("reduced size overflow")?;
140        let mut maps = Vec::with_capacity(n);
141        let mut offsets = Vec::with_capacity(n);
142        for (dof, line) in lift.lines.iter().enumerate() {
143            if line.dof != dof {
144                return Err("constraint lift rows are not in DOF order");
145            }
146            let mut terms = Vec::with_capacity(line.terms.len());
147            let mut previous = None;
148            for &(index, ratio) in &line.terms {
149                if index >= lift.free_roots.len() || previous.is_some_and(|p| index <= p) {
150                    return Err("invalid or unsorted constraint lift terms");
151                }
152                previous = Some(index);
153                let coefficient = ratio.numerator() as f64 / ratio.denominator() as f64;
154                if !coefficient.is_finite() {
155                    return Err("nonfinite constraint coefficient");
156                }
157                if coefficient != 0. {
158                    terms.push(Term {
159                        index: base + index,
160                        coefficient,
161                    });
162                }
163            }
164            let offset = line.offset.numerator() as f64 / line.offset.denominator() as f64;
165            if !offset.is_finite() {
166                return Err("nonfinite constraint offset");
167            }
168            maps.push(terms);
169            offsets.push(offset);
170        }
171        component_maps.push(maps);
172        component_offsets.push(offsets);
173    }
174
175    let mut full_offsets = vec![0.; 3 * n];
176    for c in 0..3 {
177        for i in 0..n {
178            full_offsets[3 * i + c] = component_offsets[c][i];
179        }
180    }
181    let shifted = stiffness.equilibrium_residual(&full_offsets, force_n)?;
182    let rhs = project_equilibrium(&component_maps, &shifted, reduced_size)?;
183
184    let diagonal = if preconditioner.is_none() {
185        let mut diagonal = vec![0.; reduced_size];
186        for i in 0..n {
187            for entry in stiffness.offsets[i]..stiffness.offsets[i + 1] {
188                let j = stiffness.columns[entry];
189                let block = stiffness.values[entry];
190                for (c, maps) in component_maps.iter().enumerate() {
191                    for left in &maps[i] {
192                        for right in &maps[j] {
193                            if left.index == right.index {
194                                diagonal[left.index] +=
195                                    left.coefficient * block[c][c] * right.coefficient;
196                            }
197                        }
198                    }
199                }
200            }
201        }
202        if diagonal.iter().any(|d| !d.is_finite() || *d <= 0.) {
203            return Err("reduced stiffness has nonpositive or nonfinite diagonal");
204        }
205        diagonal
206    } else {
207        Vec::new()
208    };
209    let precondition = |r: &[f64]| -> Result<Vec<f64>, &'static str> {
210        let z = match preconditioner {
211            Some(prec) => prec.apply(r)?,
212            None => r.iter().zip(&diagonal).map(|(x, d)| x / d).collect(),
213        };
214        if z.len() != r.len() || z.iter().any(|x| !x.is_finite()) {
215            return Err("invalid reduced preconditioner output");
216        }
217        Ok(z)
218    };
219
220    if guess.is_some_and(|v| v.len() != reduced_size || v.iter().any(|x| !x.is_finite())) {
221        return Err("invalid initial reduced vector");
222    }
223    let mut reduced = guess.map_or_else(|| vec![0.; reduced_size], |v| v.to_vec());
224    let mut iterations = 0usize;
225    let mut relative_residual = 0.;
226    if reduced_size > 0 {
227        let rhs_norm = norm(&rhs);
228        if !rhs_norm.is_finite() {
229            return Err("nonfinite reduced load norm");
230        }
231        let target = tolerance * rhs_norm;
232        let equilibrium = |reduced: &[f64]| {
233            let full = reconstruct(&component_maps, &component_offsets, reduced, n);
234            let r = stiffness.equilibrium_residual(&full, force_n)?;
235            project_equilibrium(&component_maps, &r, reduced_size)
236        };
237        let mut residual = if rhs_norm == 0. {
238            reduced.fill(0.);
239            vec![0.; reduced_size]
240        } else if guess.is_some() {
241            equilibrium(&reduced)?
242        } else {
243            rhs.clone()
244        };
245        if rhs_norm > 0. {
246            relative_residual = norm(&residual) / rhs_norm;
247        }
248        let mut z: Vec<f64> = precondition(&residual)?;
249        let mut direction = z.clone();
250        let mut rz = dot(&residual, &z);
251        if !rz.is_finite() || rz < 0. {
252            return Err("invalid preconditioned residual");
253        }
254        while norm(&residual) > target {
255            if iterations >= max_iterations {
256                return Err("conjugate-gradient iteration limit reached");
257            }
258            let applied = apply_reduced(stiffness, &component_maps, n, &direction)?;
259            let curvature = dot(&direction, &applied);
260            if !curvature.is_finite() || curvature <= 0. {
261                return Err("reduced stiffness is singular or indefinite");
262            }
263            let alpha = rz / curvature;
264            if !alpha.is_finite() {
265                return Err("nonfinite conjugate-gradient step");
266            }
267            axpy(&mut reduced, alpha, &direction);
268            axpy(&mut residual, -alpha, &applied);
269            iterations += 1;
270            if norm(&residual) <= target {
271                residual = equilibrium(&reduced)?;
272                relative_residual = norm(&residual) / rhs_norm;
273                if !relative_residual.is_finite() {
274                    return Err("nonfinite measured reduced residual");
275                }
276                if relative_residual <= tolerance {
277                    break;
278                }
279                z = precondition(&residual)?;
280                direction.clone_from(&z);
281                rz = dot(&residual, &z);
282                if !rz.is_finite() || rz <= 0. {
283                    return Err("nonpositive preconditioned equilibrium residual");
284                }
285                continue;
286            }
287            z = precondition(&residual)?;
288            let next_rz = dot(&residual, &z);
289            if !next_rz.is_finite() || next_rz <= 0. {
290                return Err("invalid conjugate-gradient residual");
291            }
292            let beta = next_rz / rz;
293            if !beta.is_finite() {
294                return Err("nonfinite conjugate-gradient recurrence");
295            }
296            for (p, &zi) in direction.iter_mut().zip(&z) {
297                *p = zi + beta * *p;
298            }
299            rz = next_rz;
300        }
301    }
302    if reduced.iter().any(|x| !x.is_finite()) {
303        return Err("nonfinite reduced solution");
304    }
305
306    let displacement = reconstruct(&component_maps, &component_offsets, &reduced, n);
307    if displacement.iter().any(|x| !x.is_finite()) {
308        return Err("nonfinite reconstructed displacement");
309    }
310    let kd = stiffness.apply(&displacement)?;
311    let quadratic = dot(&displacement, &kd);
312    if !quadratic.is_finite() || quadratic < -64. * f64::EPSILON * norm(&displacement) * norm(&kd) {
313        return Err("invalid nonpositive strain energy");
314    }
315    Ok(ConstrainedSolution {
316        displacement_mm: displacement,
317        reduced_values: reduced,
318        relative_residual,
319        iterations,
320        strain_energy_n_mm: 0.5 * quadratic.max(0.),
321    })
322}

324/// Project a measured full force residual, retaining low-order terms where
325/// neighboring hanging-constraint forces cancel on a common master.
326fn project_equilibrium(
327    maps: &[Vec<Vec<Term>>],
328    full: &[f64],
329    size: usize,
330) -> Result<Vec<f64>, &'static str> {
331    let mut sum = vec![0_f64; size];
332    let mut correction = vec![0.; size];
333    for (i, vector) in full.as_chunks::<3>().0.iter().enumerate() {
334        for (component, &force) in vector.iter().enumerate() {
335            for term in &maps[component][i] {
336                let j = term.index;
337                let product = term.coefficient * force;
338                let next = sum[j] + product;
339                let roundoff = if sum[j].abs() >= product.abs() {
340                    (sum[j] - next) + product
341                } else {
342                    (product - next) + sum[j]
343                };
344                correction[j] += roundoff + term.coefficient.mul_add(force, -product);
345                sum[j] = next;
346            }
347        }
348    }
349    for (s, c) in sum.iter_mut().zip(correction) {
350        *s += c;
351        if !s.is_finite() {
352            return Err("nonfinite projected force residual");
353        }
354    }
355    Ok(sum)
356}

358fn reconstruct(
359    maps: &[Vec<Vec<Term>>],
360    offsets: &[Vec<f64>],
361    reduced: &[f64],
362    n: usize,
363) -> Vec<f64> {
364    let mut full = vec![0.; 3 * n];
365    for i in 0..n {
366        for c in 0..3 {
367            full[3 * i + c] = offsets[c][i]
368                + maps[c][i]
369                    .iter()
370                    .map(|t| t.coefficient * reduced[t.index])
371                    .sum::<f64>();
372        }
373    }
374    full
375}

377fn apply_reduced(
378    stiffness: &Stiffness,
379    maps: &[Vec<Vec<Term>>],
380    n: usize,
381    x: &[f64],
382) -> Result<Vec<f64>, &'static str> {
383    let mut full = vec![0.; 3 * n];
384    for i in 0..n {
385        for c in 0..3 {
386            full[3 * i + c] = maps[c][i].iter().map(|t| t.coefficient * x[t.index]).sum();
387        }
388    }
389    let product = stiffness.apply(&full)?;
390    let mut result = vec![0.; x.len()];
391    for i in 0..n {
392        for c in 0..3 {
393            for term in &maps[c][i] {
394                result[term.index] += term.coefficient * product[3 * i + c];
395            }
396        }
397    }
398    if result.iter().any(|v| !v.is_finite()) {
399        return Err("nonfinite reduced matrix product");
400    }
401    Ok(result)
402}

404fn dot(a: &[f64], b: &[f64]) -> f64 {
405    a.iter().zip(b).map(|(x, y)| x * y).sum()
406}
407fn norm(a: &[f64]) -> f64 {
408    dot(a, a).sqrt()
409}
410fn axpy(y: &mut [f64], alpha: f64, x: &[f64]) {
411    for (yi, xi) in y.iter_mut().zip(x) {
412        *yi += alpha * xi;
413    }
414}

422    fn ratio(n: i128, d: i128) -> Ratio {
423        Ratio::new(n, d).unwrap()
424    }
425    fn lift(rows: Vec<(Vec<(usize, Ratio)>, Ratio)>, roots: Vec<usize>) -> Lift {
426        Lift {
427            free_roots: roots,
428            lines: rows
429                .into_iter()
430                .enumerate()
431                .map(|(dof, (terms, offset))| LiftLine { dof, terms, offset })
432                .collect(),
433        }
434    }
435    fn identity_stiffness(n: usize) -> Stiffness {
436        let eye: Matrix = [[1., 0., 0.], [0., 1., 0.], [0., 0., 1.]];
437        Stiffness {
438            offsets: (0..=n).collect(),
439            columns: (0..n).collect(),
440            values: vec![eye; n],
441            integrated_volume: 1.,
442        }
443    }

445    #[test]
446    fn solves_componentwise_fractional_affine_lifts_and_offsets() {
447        let half = ratio(1, 2);
448        let three_quarters = ratio(3, 4);
449        let x = lift(
450            vec![
451                (vec![(0, ratio(1, 1))], ratio(0, 1)),
452                (vec![(0, half)], half),
453                (vec![], ratio(1, 1)),
454            ],
455            vec![0],
456        );
457        let y = lift(
458            vec![
459                (vec![(0, three_quarters)], ratio(1, 2)),
460                (vec![], ratio(2, 1)),
461                (vec![(0, ratio(1, 1))], ratio(0, 1)),
462            ],
463            vec![2],
464        );
465        // The z component is entirely prescribed at nonzero offsets.
466        let z = lift(
467            vec![
468                (vec![], ratio(1, 1)),
469                (vec![], ratio(2, 1)),
470                (vec![], ratio(3, 1)),
471            ],
472            vec![],
473        );
474        let f = vec![1., 1., 0., 2., 2., 0., 3., 4., 0.];
475        let result =
476            solve_constrained(&identity_stiffness(3), [&x, &y, &z], &f, 1e-12, 20).unwrap();
477        let u = result.displacement_mm;
478        assert!((u[0] - 1.4).abs() < 1e-12);
479        assert!((u[3] - 1.2).abs() < 1e-12);
480        assert!((u[6] - 1.).abs() < 1e-12);
481        assert!((u[1] - 2.6).abs() < 1e-12);
482        assert!((u[4] - 2.).abs() < 1e-12);
483        assert!((u[7] - 2.8).abs() < 1e-12);
484        assert_eq!([u[2], u[5], u[8]], [1., 2., 3.]);
485        assert!(result.relative_residual <= 1e-12);
486    }

488    #[test]
489    fn initialized_reduced_solution_is_checked_against_the_full_affine_equilibrium() {
490        let k = identity_stiffness(2);
491        let x = lift(
492            vec![
493                (vec![(0, ratio(1, 1))], ratio(0, 1)),
494                (vec![(0, ratio(1, 2))], ratio(3, 2)),

497        );
498        let fixed = lift(vec![(vec![], ratio(0, 1)), (vec![], ratio(0, 1))], vec![]);
499        let force = [2., 0., 0., 1., 0., 0.];
500        let cold = solve_constrained(&k, [&x, &fixed, &fixed], &force, 1e-12, 10).unwrap();
501        let warm = solve_constrained_from(
502            &k,
503            [&x, &fixed, &fixed],

515            solve_constrained_from(&k, [&x, &fixed, &fixed], &force, 1e-12, 10, &[f64::NAN])

517        );
