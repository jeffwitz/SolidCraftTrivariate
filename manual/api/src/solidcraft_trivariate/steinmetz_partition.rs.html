1//! Certify the stored charts of the equal-radius orthogonal through-bore junction.
2//!
3//! The exact nonnegative cylinder polynomials confine all charts to target material.
4//! The oriented volume boundaries cancel pairwise on interior interfaces. Physical
5//! exterior traces lie on the target stock planes or the actual cylinder surfaces.
6//! Thus the sum of local degrees is constant on the connected interior of the target.
7//! A unique affine brick occupies a neighborhood of each stock corner, fixing that
8//! degree at one. Strict positive Jacobians then exclude missing or multiply covered
9//! open volumes. No quadrature or point cloud is used for this conclusion.
10use crate::{
11    bracket::BracketBlock,
12    bracket_audit::BracketCertificate,
13    incidence::{enumerate_faces, BlockSide},
14    orthogonal_sketch::{OrthogonalSketch, SweepPartition},
15    spatial_certificate::{certify_spatial, CertificateBudget, CertificateStatus},
16};
17use std::collections::BTreeMap;

19fn binomial(d: usize, k: usize) -> i128 {
20    if d == 2 && k == 1 {
21        2
22    } else {
23        1
24    }
25}
26fn axis_index(i: usize, d: [usize; 3]) -> [usize; 3] {
27    [
28        i / ((d[1] + 1) * (d[2] + 1)),
29        i / (d[2] + 1) % (d[1] + 1),
30        i % (d[2] + 1),
31    ]
32}
33fn cylinder_exclusion(
34    b: &BracketBlock,
35    axis: usize,
36    center: [i32; 3],
37    r: i32,
38) -> Result<(), &'static str> {
39    let d = b.degrees;
40    let dims = d.map(|v| 2 * v + 1);
41    let n = dims.iter().product();
42    let mut coeff = vec![0i128; n];
43    let a = (axis + 1) % 3;
44    let c = (axis + 2) % 3;
45    let mut terms = Vec::new();
46    for (i, h) in b.controls.iter().enumerate() {
47        let q = axis_index(i, d);
48        let weight = (0..3).map(|k| binomial(d[k], q[k])).product::<i128>();
49        let w = i128::from(h[3]);
50        // The validated design window and normalized constructor nets keep these products in i128.
51        let p = [
52            i128::from(h[a]) - i128::from(center[a]) * w,
53            i128::from(h[c]) - i128::from(center[c]) * w,
54            i128::from(r) * w,
55        ];
56        terms.push((q, p, weight));
57    }
58    for &(i, p, v) in &terms {
59        for &(j, q, w) in &terms {
60            let k = ((i[0] + j[0]) * dims[1] + i[1] + j[1]) * dims[2] + i[2] + j[2];
61            let prod = |k: usize| p[k].checked_mul(q[k]).ok_or("steinmetz_exclusion_overflow");
62            let t = prod(0)?
63                .checked_add(prod(1)?)
64                .and_then(|s| s.checked_sub(prod(2).ok()?))
65                .and_then(|s| s.checked_mul(v * w))
66                .ok_or("steinmetz_exclusion_overflow")?;
67            coeff[k] = coeff[k]
68                .checked_add(t)
69                .ok_or("steinmetz_exclusion_overflow")?;
70        }
71    }
72    if coeff.iter().any(|v| *v < 0) {
73        return Err("steinmetz_cylinder_exclusion_indeterminate");
74    }
75    Ok(())
76}
77fn corner_degree(blocks: &[BracketBlock], bounds: [[i32; 2]; 3]) -> Result<(), &'static str> {
78    for octant in 0..8 {
79        let p: [i32; 3] = std::array::from_fn(|k| bounds[k][(octant >> k) & 1]);
80        let mut candidates = blocks.iter().filter(|b| {
81            (0..3).all(|k| {
82                b.controls
83                    .iter()
84                    .any(|h| i128::from(h[k]) == i128::from(p[k]) * i128::from(h[3]))
85            })
86        });
87        let Some(b) = candidates.next() else {
88            return Err("steinmetz_missing_corner");
89        };
90        if candidates.next().is_some()
91            || b.degrees != [1; 3]
92            || b.controls.iter().any(|h| h[3] != 1)
93        {
94            return Err("steinmetz_ambiguous_corner_degree");
95        }
96        let mut used = [false; 3];
97        for k in 0..3 {
98            let p0 = b.controls[0][k];
99            let mut direction = None;
100            for (axis, step) in [4, 2, 1].into_iter().enumerate() {
101                if b.controls[step][k] != p0 && direction.replace((axis, step)).is_some() {
102                    return Err("steinmetz_corner_not_affine");
103                }
104            }
105            let (axis, step) = direction.ok_or("steinmetz_degenerate_corner")?;
106            if used[axis]
107                || b.controls.iter().enumerate().any(|(i, h)| {
108                    h[k] != if i & step != 0 {
109                        b.controls[step][k]
110                    } else {
111                        p0
112                    }
113                })
114            {
115                return Err("steinmetz_corner_not_affine");
116            }
117            used[axis] = true;
118        }
119    }
120    Ok(())
121}
122fn face_on_plane(b: &BracketBlock, s: &BlockSide, k: usize, plane: i32) -> bool {
123    b.controls
124        .iter()
125        .enumerate()
126        .filter(|(i, _)| {
127            axis_index(*i, b.degrees)[s.axis] == usize::from(s.upper) * b.degrees[s.axis]
128        })
129        .all(|(_, h)| i128::from(h[k]) == i128::from(plane) * i128::from(h[3]))
130}

132pub(crate) fn try_build(j: &OrthogonalSketch) -> Result<Option<SweepPartition>, &'static str> {
133    let [a, b] = j.bores.as_slice() else {
134        return Ok(None);
135    };
136    if a.axis == b.axis
137        || a.axis >= 3
138        || b.axis >= 3
139        || !j.fillets.is_empty()
140        || a.radius != b.radius
141    {
142        return Ok(None);
143    }
144    let normal = 3 - a.axis - b.axis;
145    if a.center[normal] != b.center[normal] {
146        return Ok(None);
147    }
148    if a.radius <= 0
149        || a.radius > 1024
150        || a.center
151            .into_iter()
152            .chain(b.center)
153            .any(|x| !(-4096..=4096).contains(&x))
154    {
155        return Err("invalid_steinmetz_bore");
156    }
157    let bounds = std::array::from_fn(|k| {
158        if k == 2 {
159            [0, j.height]
160        } else {
161            let pts = j.outer.vertices.iter().map(|v| v[k]);
162            [pts.clone().min().unwrap(), pts.max().unwrap()]
163        }
164    });
165    let size = bounds.map(|v| i64::from(v[1] - v[0]));
166    if crate::sketch_layers::area2(&j.outer) != 2 * size[0] * size[1] {
167        return Ok(None);
168    }
169    let mut center = a.center;
170    center[a.axis] = b.center[a.axis];
171    for p in [a.center, b.center] {
172        if (0..3).any(|k| p[k] <= bounds[k][0] || p[k] >= bounds[k][1]) {
173            return Err("steinmetz_axis_seed_outside_stock");
174        }
175    }
176    let guard = a.radius + 1;
177    let axes = [a.axis, b.axis];
178    let mut sorted_axes = axes;
179    sorted_axes.sort();
180    let mut blocks = crate::steinmetz_frame::intersecting_cylinders(
181        bounds,
182        center,
183        a.radius,
184        guard,
185        sorted_axes,
186        "junction",
187    )?;
188    blocks.sort_by(|u, v| u.id.cmp(&v.id));
189    if blocks.len() != 40 {
190        return Err("steinmetz_reported_partition_size");
191    }
192    let mut det = f64::INFINITY;
193    for block in &blocks {
194        if block.controls.iter().any(|h| {
195            h[3] <= 0
196                || (0..3).any(|k| {
197                    let x = i128::from(h[k]);
198                    let w = i128::from(h[3]);
199                    x < i128::from(bounds[k][0]) * w || x > i128::from(bounds[k][1]) * w
200                })
201        }) {
202            return Err("steinmetz_stock_convex_hull_violation");
203        }
204        for ax in axes {
205            cylinder_exclusion(block, ax, center, a.radius)?;
206        }
207        let cert = certify_spatial(block.degrees, &block.controls, CertificateBudget::default());
208        if cert.status != CertificateStatus::Valid {
209            return Err("steinmetz_spatial_certificate_incomplete");
210        }
211        det = det.min(
212            cert.determinant_lower_bound_mm3
213                .ok_or("steinmetz_positive_bound_missing")?,
214        );
215    }
216    corner_degree(&blocks, bounds)?;
217    let incidence = enumerate_faces(&blocks)?;
218    for link in &incidence.interfaces {
219        let side = |s: &BlockSide| s.upper ^ (s.axis == 1);
220        let o = link.orientation;
221        if !(side(&link.first) ^ side(&link.second) ^ o.swap ^ o.flip_first ^ o.flip_second) {
222            return Err("steinmetz_oriented_interfaces_do_not_cancel");
223        }
224    }
225    let index: BTreeMap<_, _> = blocks.iter().map(|c| (&c.id, c)).collect();
226    let mut design_faces = Vec::new();
227    for s in &incidence.exterior {
228        let block = index[&s.block];
229        let mut label = None;
230        for (k, ends) in bounds.iter().enumerate() {
231            for (i, &h) in ends.iter().enumerate() {
232                if face_on_plane(block, s, k, h) {
233                    if label.is_some() {
234                        return Err("steinmetz_ambiguous_plane");
235                    }
236                    label = Some(if k == 2 {
237                        ["start", "end"][i].to_owned()
238                    } else {
239                        format!("outer/{}", j.outer.id)
240                    });
241                }
242            }
243        }
244        if label.is_none() {
245            for feature in [a, b] {
246                if crate::steinmetz_audit::certify_boundary(
247                    block,
248                    s,
249                    feature.axis,
250                    center,
251                    feature.radius,
252                )? {
253                    if label.is_some() {
254                        return Err("steinmetz_ambiguous_cylinder");
255                    }
256                    label = Some(format!("bore/{}/wall", feature.id));
257                }
258            }
259        }
260        design_faces.push((
261            s.clone(),
262            label.ok_or("steinmetz_unclassified_exterior_face")?,
263        ));
264    }
265    let box_volume = size.iter().product::<i64>();
266    let r = i64::from(a.radius);
267    let planes = std::array::from_fn(|k| vec![bounds[k][0], center[k], bounds[k][1]]);
268    Ok(Some(SweepPartition {
269        cut_comparison: None,
270        cut_rounds: Vec::new(),
271        cut_sources: Vec::new(),
272        compact: false,
273        fallback_reason: None,
274        blocks,
275        certificate: BracketCertificate {
276            incidence,
277            design_faces,
278            determinant_lower_bound_mm3: det,
279            radial_error_upper_mm: 0.,
280        },
281        twice_constant: 2 * box_volume,
282        four_pi: -4 * r * r * (size[a.axis] + size[b.axis]),
283        planes,
284        guards: Vec::new(),
285        junction: Some((center, sorted_axes, a.radius, guard)),
286    }))
287}
