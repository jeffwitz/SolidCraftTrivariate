1//! Surface loads on whole or partial parameter faces of a certified mesh.
2use crate::{
3    analysis_space::AnalysisSpace,
4    elasticity::{basis_geometry, gauss, Vector},
5    incidence::BlockSide,
6};
7use std::collections::{BTreeMap, BTreeSet};
8
9/// Coordinates use increasing local parameter axes after omitting the normal.
10#[derive(Clone, Debug, PartialEq, Eq)]
11pub struct LoadPatch {
12    pub side: BlockSide,
13    pub depth: u8,
14    pub intervals: [[u32; 2]; 2],
15}
16
17impl LoadPatch {
18    pub fn whole(side: BlockSide) -> Self {
19        Self {
20            side,
21            depth: 0,
22            intervals: [[0, 1]; 2],
23        }
24    }
25
26    fn bounds(&self) -> Result<[[f64; 2]; 2], &'static str> {
27        if self.depth > 30 {
28            return Err("surface patch depth exceeds the dyadic coordinate range");
29        }
30        let scale = 1_u32 << self.depth;
31        if self
32            .intervals
33            .iter()
34            .any(|&[lo, hi]| lo >= hi || hi > scale)
35        {
36            return Err("empty or invalid surface patch rectangle");
37        }
38        Ok(self
39            .intervals
40            .map(|r| r.map(|n| f64::from(n) / f64::from(scale))))
41    }
42}
43
44pub struct SurfaceLoad {
45    pub nodal_force_n: Vec<f64>,
46    pub area_mm2: f64,
47    pub resultant_n: Vector,
48    /// Moment about the physical coordinate origin.
49    pub moment_n_mm: Vector,
50}
51
52fn cross(a: Vector, b: Vector) -> Vector {
53    std::array::from_fn(|k| a[(k + 1) % 3] * b[(k + 2) % 3] - a[(k + 2) % 3] * b[(k + 1) % 3])
54}
55
56/// Integrate a traction (N/mm²), evaluated at the physical position and
57/// outward unit normal, into the interleaved full nodal force. The caller
58/// supplies the certified exterior sides of these analysis blocks.
59pub fn integrate_traction(
60    space: &AnalysisSpace,
61    exterior: &BTreeSet<BlockSide>,
62    patches: &[LoadPatch],
63    order: usize,
64    traction: impl Fn(Vector, Vector) -> Vector,
65) -> Result<SurfaceLoad, &'static str> {
66    let quadrature = gauss(order)?;
67    if patches.is_empty() {
68        return Err("empty traction selection");
69    }
70    let mut known = BTreeMap::new();
71    for b in &space.blocks {
72        if known.insert(&b.id, b).is_some() {
73            return Err("duplicate analysis block");
74        }
75    }
76    let mut used = BTreeMap::<&BlockSide, Vec<[[f64; 2]; 2]>>::new();
77    let len = space
78        .ndof
79        .checked_mul(3)
80        .ok_or("force vector dimension overflow")?;
81    let mut result = SurfaceLoad {
82        nodal_force_n: vec![0.; len],
83        area_mm2: 0.,
84        resultant_n: [0.; 3],
85        moment_n_mm: [0.; 3],
86    };
87    for patch in patches {
88        let face = &patch.side;
89        if face.axis >= 3 || !exterior.contains(face) {
90            return Err("traction requires a certified exterior side");
91        }
92        let b = known.get(&face.block).ok_or("unknown traction block")?;
93        if b.dofs.len() != 27 || b.dofs.iter().any(|&d| d >= space.ndof) {
94            return Err("invalid traction block DOFs");
95        }
96        let rect = patch.bounds()?;
97        let previous = used.entry(face).or_default();
98        if previous
99            .iter()
100            .any(|r| (0..2).all(|i| rect[i][0] < r[i][1] && r[i][0] < rect[i][1]))
101        {
102            return Err("overlapping traction patches");
103        }
104        previous.push(rect);
105        let axes = match face.axis {
106            0 => [1, 2],
107            1 => [0, 2],
108            _ => [0, 1],
109        };
110        let sign = if face.upper ^ (face.axis == 1) {
111            1.
112        } else {
113            -1.
114        };
115        let width = rect.map(|[lo, hi]| hi - lo);
116        for &(u, wu) in &quadrature {
117            for &(v, wv) in &quadrature {
118                let mut t = [0.; 3];
119                t[face.axis] = f64::from(u8::from(face.upper));
120                t[axes[0]] = rect[0][0] + width[0] * u;
121                t[axes[1]] = rect[1][0] + width[1] * v;
122                let sample = basis_geometry(b, t)?;
123                let area = cross(
124                    sample.jacobian_columns[axes[0]],
125                    sample.jacobian_columns[axes[1]],
126                );
127                let jac = area[0].hypot(area[1]).hypot(area[2]);
128                if jac <= 0. || !jac.is_finite() {
129                    return Err("invalid surface Jacobian");
130                }
131                let measure = jac * wu * wv * width[0] * width[1];
132                let normal = area.map(|s| sign * s / jac);
133                let df = traction(sample.position, normal).map(|s| s * measure);
134                if df.iter().any(|x| !x.is_finite()) {
135                    return Err("nonfinite integrated traction");
136                }
137                let moment = cross(sample.position, df);
138                result.area_mm2 += measure;
139                for d in 0..3 {
140                    result.resultant_n[d] += df[d];
141                    result.moment_n_mm[d] += moment[d];
142                }
143                for (&id, value) in b.dofs.iter().zip(sample.values) {
144                    for (d, f) in df.iter().enumerate() {
145                        result.nodal_force_n[3 * id + d] += value * f;
146                    }
147                }
148            }
149        }
150    }
151    if !result.area_mm2.is_finite()
152        || result
153            .nodal_force_n
154            .iter()
155            .chain(&result.resultant_n)
156            .chain(&result.moment_n_mm)
157            .any(|v| !v.is_finite())
158    {
159        return Err("nonfinite assembled surface load");
160    }
161    Ok(result)
162}
