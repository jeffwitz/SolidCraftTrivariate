1//! Recipe interface to native sketch primitives and their construction-time plans.
2use crate::{
3    constraint_system::Ratio,
4    json::{object, Json},
5    primitive_template::{Operation, Primitive, PrimitivePlan},
6    sketch_junction::Continuity,
7};
8type Error = &'static str;
9fn number(v: impl ToString) -> Json {
10    Json::Number(v.to_string())
11}
12fn coordinate(v: &Json) -> Result<Ratio, Error> {
13    let [n, d] = v.items()? else {
14        return Err("expected exact coordinate [numerator,denominator]");
15    };
16    Ratio::new(n.number()?, d.number()?)
17}
18fn point(v: &Json) -> Result<[Ratio; 2], Error> {
19    let [x, y] = v.items()? else {
20        return Err("expected two sketch coordinates");
21    };
22    Ok([coordinate(x)?, coordinate(y)?])
23}
24pub(crate) fn primitive(v: &Json) -> Result<Primitive, Error> {
25    match v.field("kind")?.text()? {
26        "rectangle" => {
27            v.keys(&["kind", "min", "max"])?;
28            Ok(Primitive::Rectangle {
29                min: point(v.field("min")?)?,
30                max: point(v.field("max")?)?,
31            })
32        }
33        "circle" => {
34            v.keys(&["kind", "center", "radius"])?;
35            Ok(Primitive::Circle {
36                center: point(v.field("center")?)?,
37                radius: coordinate(v.field("radius")?)?,
38            })
39        }
40        "ellipse" => {
41            v.keys(&["kind", "center", "radii"])?;
42            Ok(Primitive::Ellipse {
43                center: point(v.field("center")?)?,
44                radii: point(v.field("radii")?)?,
45            })
46        }
47        _ => Err("unknown sketch primitive"),
48    }
49}
50pub(crate) fn contour(plan: &PrimitivePlan) -> Json {
51    let c = &plan.contour;
52    object([
53        ("degree", number(c.degree)),
54        ("knots", Json::Array(c.knots.iter().map(number).collect())),
55        (
56            "controls",
57            Json::Array(
58                c.controls
59                    .iter()
60                    .map(|p| Json::Array(p.map(number).into()))
61                    .collect(),
62            ),
63        ),
64    ])
65}
66pub(crate) fn continuity(c: Continuity) -> &'static str {
67    match c {
68        Continuity::Corner => "G0",
69        Continuity::Tangent => "G1",
70        Continuity::Curvature => "G2",
71        Continuity::Disconnected => "disconnected",
72        Continuity::Singular => "singular",
73    }
74}
75fn template(recipe: &Json) -> Result<PrimitivePlan, Error> {
76    recipe.keys(&["family", "primitive", "operation"])?;
77    if recipe.field("family")?.text()? != "primitive_sweep_v1" {
78        return Err("unknown primitive planning recipe");
79    }
80    let op = recipe.field("operation")?;
81    let operation = match op.field("kind")?.text()? {
82        "extrude" => {
83            op.keys(&["kind", "length_mm"])?;
84            Operation::Extrude {
85                length: op.field("length_mm")?.number()?,
86            }
87        }
88        "revolve" => {
89            op.keys(&["kind"])?;
90            Operation::Revolve
91        }
92        _ => return Err("unknown primitive sweep operation"),
93    };
94    primitive(recipe.field("primitive")?)?.plan(operation)
95}

97pub fn plan(recipe: &Json) -> Result<Json, Error> {
98    let p = template(recipe)?;
99    let op = recipe.field("operation")?;
100    let patches = p
101        .section_patches
102        .iter()
103        .map(|s| {
104            object([
105                ("degrees", Json::Array(s.degrees.map(number).into())),
106                (
107                    "controls",
108                    Json::Array(
109                        s.controls
110                            .iter()
111                            .map(|c| Json::Array(c.map(number).into()))
112                            .collect(),
113                    ),
114                ),
115            ])
116        })
117        .collect();
118    Ok(object([
119        ("source_primitive", recipe.field("primitive")?.clone()),
120        ("operation", op.clone()),
121        ("contour", contour(&p)),
122        (
123            "junctions",
124            Json::Array(
125                p.junctions
126                    .iter()
127                    .map(|c| Json::String(continuity(*c).into()))
128                    .collect(),
129            ),
130        ),
131        ("section_patches", Json::Array(patches)),
132        ("block_count", number(p.block_count())),
133    ]))
134}

136type Geometry = (
137    Vec<crate::bracket::BracketBlock>,
138    Vec<(crate::incidence::BlockSide, String)>,
139);
140pub(crate) fn construct(recipe: &Json) -> Result<Geometry, Error> {
141    let template = template(recipe)?;
142    let blocks = template.build()?;
143    let exterior = crate::incidence::enumerate_faces(&blocks)?.exterior;
144    let mut faces = Vec::with_capacity(exterior.len());
145    for side in exterior {
146        let feature = if side.axis == 2 {
147            if recipe.field("operation")?.field("kind")?.text()? != "extrude" {
148                return Err("unpaired revolution section face");
149            }
150            if side.upper {
151                "end".to_string()
152            } else {
153                "start".to_string()
154            }
155        } else {
156            let section = side.block.split("_turn_").next().unwrap_or(&side.block);
157            format!(
158                "{section}:{}{}",
159                ["u", "v"][side.axis],
160                usize::from(side.upper)
161            )
162        };
163        faces.push((side, feature));
164    }
165    Ok((blocks, faces))
166}
