21type V = [f64; 3];
22fn index(d: [usize; 3], at: [usize; 3]) -> usize {
23    (at[0] * (d[1] + 1) + at[1]) * (d[2] + 1) + at[2]
24}
25pub(crate) fn val(r: Q) -> f64 {
26    r.numerator() as f64 / r.denominator() as f64
27}
28pub(crate) fn ratio_div(a: Q, b: Q) -> Result<Q> {
29    a.checked_mul(Q::new(b.denominator(), b.numerator())?)
30}
31fn n(i: impl ToString) -> Json {
32    Json::Number(i.to_string())
33}
34fn cross(a: V, b: V) -> V {
35    [
36        a[1] * b[2] - a[2] * b[1],
37        a[2] * b[0] - a[0] * b[2],
38        a[0] * b[1] - a[1] * b[0],
39    ]
40}
41fn dot(a: V, b: V) -> f64 {
42    a.iter().zip(b).map(|(x, y)| x * y).sum()
43}

58impl InheritedCell {
59    pub fn sample(&self, uvw: V) -> Result<(ElementGeometrySample, ElementBasisSample)> {
60        let g = self
61            .geometry
62            .evaluate(uvw)
63            .map_err(|_| "geometry evaluation unavailable")?;
64        let j = g.jacobian_columns;
65        let cof = [cross(j[1], j[2]), cross(j[2], j[0]), cross(j[0], j[1])];
66        let det = dot(j[0], cof[0]);
67        if !det.is_finite() || det <= 0. {
68            return Err("nonpositive sampled geometry Jacobian");
69        }
70        let b: [_; 3] = std::array::from_fn(|a| basis(self.degrees[a], uvw[a]));
71        let mut values = vec![0.; self.active.len()];
72        let mut grads = vec![[0.; 3]; self.active.len()];
73        let mut w = 0.;
74        let mut dw = [0.; 3];
75        for i in 0..=self.degrees[0] {
76            for j in 0..=self.degrees[1] {
77                for k in 0..=self.degrees[2] {
78                    let l = index(self.degrees, [i, j, k]);
79                    let weight = self.weights[l];
80                    let p = weight * b[0].0[i] * b[1].0[j] * b[2].0[k];
81                    let dp = [
82                        weight * b[0].1[i] * b[1].0[j] * b[2].0[k],
83                        weight * b[0].0[i] * b[1].1[j] * b[2].0[k],
84                        weight * b[0].0[i] * b[1].0[j] * b[2].1[k],
85                    ];
86                    w += p;
87                    for a in 0..3 {
88                        dw[a] += dp[a];
89                    }
90                    for &(d, c) in &self.stencils[l] {
91                        values[d] += p * c;
92                        for a in 0..3 {
93                            grads[d][a] += dp[a] * c;
94                        }
95                    }
96                }
97            }
98        }
99        if w <= 0. || !w.is_finite() {
100            return Err("invalid rational denominator");
101        }
102        for (r, d) in values.iter_mut().zip(&mut grads) {
103            *r /= w;
104            let ref_grad: [f64; 3] = std::array::from_fn(|a| (d[a] - *r * dw[a]) / w);
105            *d = std::array::from_fn(|c| {
106                (ref_grad[0] * cof[0][c] + ref_grad[1] * cof[1][c] + ref_grad[2] * cof[2][c]) / det
107            });
108        }
109        Ok((
110            ElementGeometrySample {
111                position: g.position,
112                jacobian_columns: g.jacobian_columns,
113                determinant: det,
114            },
115            ElementBasisSample {
116                dofs: self.active.clone(),
117                values,
118                physical_gradients: grads,
119            },
120        ))
121    }
122}

188/// A section in x,y held fixed over an affine physical z interval, including weights.
189pub(crate) struct Section<'a> {
190    pub block: &'a BracketBlock,
191    pub d: [usize; 3],
192    pub z: [Q; 2],
193    pub shear: [Q; 2],
194    pub offset: usize,
195}
196impl Section<'_> {
197    fn len(&self) -> usize {
198        self.d.iter().map(|v| v + 1).product()
199    }
200    fn point(&self, at: [usize; 3]) -> Result<[Q; 3]> {
201        let h = self.block.controls[index(self.block.degrees, [at[0], at[1], 0])];
202        let z = self.z[0].checked_add(
203            self.z[1]
204                .checked_sub(self.z[0])?
205                .checked_mul(Q::new(at[2] as i128, self.d[2] as i128)?)?,
206        )?;
207        let dz = z.checked_sub(self.z[0])?;
208        Ok([
209            Q::new(h[0].into(), h[3].into())?.checked_add(self.shear[0].checked_mul(dz)?)?,
210            Q::new(h[1].into(), h[3].into())?.checked_add(self.shear[1].checked_mul(dz)?)?,
211            z,
212        ])
213    }
214    pub fn flat(&self, at: [usize; 3]) -> usize {
215        self.offset + index(self.d, at)
216    }
217    fn height(&self) -> Result<Q> {
218        self.z[1].checked_sub(self.z[0])
219    }
220}

222fn section(block: &BracketBlock, p: usize, offset: usize) -> Result<Section<'_>> {
223    if block.degrees[2] != 1 {
224        return Err("unsupported: source w is not a degree-one extrusion");
225    }
226    let mut bounds = None;
227    let mut direction = None;
228    for pair in block.controls.as_chunks::<2>().0.iter() {
229        let (a, b) = (pair[0], pair[1]);
230        if a[3] <= 0 || a[3] != b[3] {
231            return Err("unsupported: varying homogeneous section along w");
232        }
233        let z = [
234            Q::new(a[2].into(), a[3].into())?,
235            Q::new(b[2].into(), b[3].into())?,
236        ];
237        if z[0].cmp_value(&z[1]).is_ge() || bounds.is_some_and(|v| v != z) {
238            return Err("unsupported: extrusion requires common ascending physical z planes");
239        }
240        let dz = z[1].checked_sub(z[0])?;
241        let mut shear = [Q::zero(); 2];
242        for c in 0..2 {
243            shear[c] = ratio_div(
244                Q::new(i128::from(b[c]) - i128::from(a[c]), i128::from(a[3]))?,
245                dz,
246            )?;
247        }
248        if direction.is_some_and(|old| old != shear) {
249            return Err("unsupported: non-translational ruled section");
250        }
251        direction = Some(shear);
252        bounds = Some(z);
253    }
254    Ok(Section {
255        block,
256        d: [block.degrees[0], block.degrees[1], p],
257        z: bounds.ok_or("empty extrusion")?,
258        shear: direction.ok_or("empty extrusion")?,
259        offset,
260    })
261}

263pub(crate) fn face_coord(side: &BlockSide, i: usize, j: usize, d: [usize; 3]) -> [usize; 3] {
264    let axes = face_axes(side.axis);
265    let mut at = [0; 3];
266    at[axes[0]] = i;
267    at[axes[1]] = j;
268    at[side.axis] = if side.upper { d[side.axis] } else { 0 };
269    at
270}

272fn paired_coords(
273    link: &Interface,
274    a: &Section,
275    b: &Section,
276) -> Result<Vec<([usize; 3], [usize; 3])>> {
277    let [s, t] = face_axes(link.first.axis).map(|axis| a.d[axis]);
278    let dims = face_axes(link.second.axis).map(|axis| b.d[axis]);
279    let o = link.orientation;
280    if dims != if o.swap { [t, s] } else { [s, t] } {
281        return Err("unsupported: different field trace degrees");
282    }
283    let mut pairs = Vec::new();
284    for i in 0..=s {
285        for j in 0..=t {
286            let k = if o.flip_first { s - i } else { i };
287            let l = if o.flip_second { t - j } else { j };
288            let [x, y] = if o.swap { [l, k] } else { [k, l] };
289            pairs.push((
290                face_coord(&link.first, i, j, a.d),
291                face_coord(&link.second, x, y, b.d),
292            ));
293        }
294    }
295    Ok(pairs)
296}

298fn choose(n: usize, k: usize) -> i128 {
299    match (n, k) {
300        (2, 1) => 2,
301        (3, 1 | 2) => 3,
302        _ => 1,
303    }
304}

306/// Enforce homogeneous numerator derivatives divided by each trace's gauge.
307/// A common positive denominator with the same jets implies the rational field identity.
308fn jet_row(
309    row: &mut Row,
310    s: &Section,
311    mut at: [usize; 3],
312    side: &BlockSide,
313    order: usize,
314    interval: (Q, i128),
315    ids: &[usize],
316) -> Result<()> {
317    let axis = side.axis;
318    let p = s.d[axis];
319    let face_weight = s.block.controls[index(s.block.degrees, [at[0], at[1], 0])][3];
320    let (length, sign) = interval;
321    let mut scale = Q::new(sign, 1)?;
322    for _ in 0..order {
323        scale = ratio_div(scale, length)?;
324    }
325    for k in 0..=order {
326        at[axis] = if side.upper { p - order + k } else { k };
327        let h = s.block.controls[index(s.block.degrees, [at[0], at[1], 0])];
328        let coefficient = choose(order, k) * if (order - k).is_multiple_of(2) { 1 } else { -1 };
329        let c =
330            Q::new(coefficient * i128::from(h[3]), i128::from(face_weight))?.checked_mul(scale)?;
331        add(row, ids[s.flat(at)], c)?;
332    }
333    Ok(())
334}

336/// Identify an affine change of the normal coordinate from the complete first
337/// homogeneous jet. Face coefficients already share a constant gauge by
338/// incidence. The ratio compares outward normals with opposite signs.
339fn normal_scale(link: &Interface, a: &Section, b: &Section) -> Result<Option<Q>> {
340    let outward = |section: &Section, side: &BlockSide, mut at: [usize; 3]| -> Result<[Q; 4]> {
341        at[2] = usize::from(at[2] == section.d[2]);
342        let block = section.block;
343        let face = block.controls[index(block.degrees, at)];
344        if side.upper {
345            at[side.axis] -= 1;
346        } else {
347            at[side.axis] += 1;
348        }
349        let inner = block.controls[index(block.degrees, at)];
350        let mut result = [Q::zero(); 4];
351        for (k, value) in result.iter_mut().enumerate() {
352            *value = Q::new(
353                (i128::from(face[k]) - i128::from(inner[k])) * block.degrees[side.axis] as i128,
354                i128::from(face[3]),
355            )?;
356        }
357        Ok(result)
358    };
359    let mut scale = None;
360    for (i, j) in paired_coords(link, a, b)? {
361        if i[2] != 0 && i[2] != a.d[2] {
362            continue;
363        }
364        let first = outward(a, &link.first, i)?;
365        let second = outward(b, &link.second, j)?;
366        for (x, y) in first.into_iter().zip(second) {
367            if y == Q::zero() {
368                if x != Q::zero() {
369                    return Ok(None);
370                }
371            } else {
372                let r = ratio_div(Q::zero().checked_sub(x)?, y)?;
373                if r.numerator() <= 0 || scale.is_some_and(|v| v != r) {
374                    return Ok(None);
375                }
376                scale = Some(r);
377            }
378        }
379    }
380    Ok(scale)
381}

419fn compile_extrusion(
420    model: &Json,
421    degree: usize,
422    continuity: usize,
423    restore_strip: bool,
424    transverse: &[(String, String)],
425    plane_degree: usize,
426    match_chart_jets: bool,
427) -> Result<InheritedSpace> {
428    if plane_degree > 3 {
429        return Err("unsupported transverse degree");
430    }
431    if !(1..=3).contains(&degree) || continuity > degree {
432        return Err("invalid field degree or continuity");
433    }
434    if audit(model)?.field("status")?.text()? != "valid" {
435        return Err("source geometry is not certified");
436    }
437    let mut blocks = decode_blocks(model)?;
438    for block in &mut blocks {
439        let target = [
440            block.degrees[0].max(plane_degree),
441            block.degrees[1].max(plane_degree),
442            block.degrees[2],
443        ];
444        if target != block.degrees {
445            let net = crate::nurbs_space::elevate_to(block, target);
446            let gcd = |mut a: i128, mut b: i128| {
447                while b != 0 {
448                    (a, b) = (b, a % b);
449                }
450                a.abs()
451            };
452            let divisor = net.iter().flatten().copied().fold(0, gcd);
453            if divisor == 0 {
454                return Err("zero elevated net");
455            }
456            let mut controls = Vec::with_capacity(net.len());
457            for h in net {
458                let mut point = [0i64; 4];
459                for k in 0..4 {
460                    point[k] = i64::try_from(h[k] / divisor)
461                        .map_err(|_| "indeterminate: elevated net exceeds i64")?;
462                }
463                controls.push(point);
464            }
465            block.degrees = target;
466            block.controls = controls;
467        }
468    }
469    if blocks.is_empty() || blocks.len() > 2048 {
470        return Err("indeterminate: extrusion block budget");
471    }
472    let links = enumerate_faces(&blocks)?;
473    let mut offset = 0;
474    let mut sections = Vec::new();
475    let mut by_id = BTreeMap::new();
476    for (i, b) in blocks.iter().enumerate() {
477        let s = section(b, degree, offset)?;
478        offset += s.len();
479        sections.push(s);
480        by_id.insert(b.id.as_str(), i);
481    }
482    if offset > 120000 {
483        return Err("indeterminate: raw scalar field budget");
484    }
485    let mut uf = UnionFind::new(offset);
486    let mut shared_plane = BTreeSet::new();
487    for link in &links.interfaces {
488        let a = &sections[by_id[link.first.block.as_str()]];
489        let b = &sections[by_id[link.second.block.as_str()]];
490        for (i, j) in paired_coords(link, a, b)? {
491            uf.join(a.flat(i), b.flat(j));
492            if link.first.axis == 2 {
493                shared_plane.insert(a.flat(i));
494            }
495        }
496    }
497    let mut c0 = BTreeMap::new();
498    let mut ids = Vec::new();
499    for i in 0..offset {
500        let next = c0.len();
501        ids.push(*c0.entry(uf.find(i)).or_insert(next));
502    }
503    let n0 = c0.len();
504    let mut xyz = vec![None; n0];
505    let mut priority = vec![1; n0];
506    for s in &sections {
507        for i in 0..=s.d[0] {
508            for j in 0..=s.d[1] {
509                for k in 0..=degree {
510                    let at = [i, j, k];
511                    let id = ids[s.flat(at)];
512                    let pt = s.point(at)?;
513                    if xyz[id].is_some_and(|v| v != pt) {
514                        return Err("field coefficient trace positions differ");
515                    }
516                    xyz[id] = Some(pt);
517                    if k == 0 || k == degree {
518                        priority[id] = 2;
519                    }
520                }
521            }
522        }
523    }
524    for i in shared_plane {
525        priority[ids[i]] = 0;
526    }
527    // Pivot on internal planes first, retaining bottom/top trace roots whenever possible.
528    let mut order: Vec<_> = (0..n0).collect();
529    order.sort_by_key(|&i| (priority[i], i));
530    let mut priority_id = vec![0; n0];
531    for (j, &i) in order.iter().enumerate() {
532        priority_id[i] = j;
533    }
534    let pids: Vec<_> = ids.iter().map(|&i| priority_id[i]).collect();
535    let mut eq = Vec::new();
536    let mut lateral = Vec::new();
537    let mut evidence = Vec::new();
538    let mut inherited = 0;
539    let mut coupled = 0;
540    let wanted: BTreeSet<_> = transverse
541        .iter()
542        .map(|(a, b)| if a < b { (a, b) } else { (b, a) })
543        .collect();
544    if transverse.len() > 256 {
545        return Err("indeterminate: transverse interface budget");
546    }
547    if wanted.len() != transverse.len() {
548        return Err("duplicate transverse join request");
549    }
550    let strip_parent = if restore_strip {
551        let parent = crate::spline_support::recover_nurbs_strip_parent(model)?;
552        if plane_degree > parent.degrees[0] {
553            return Err("unsupported elevated source-curve continuity");
554        }
555        Some(parent)
556    } else {
557        None
558    };
559    for link in &links.interfaces {
560        let ai = by_id[link.first.block.as_str()];
561        let bi = by_id[link.second.block.as_str()];
562        let (a, b) = (&sections[ai], &sections[bi]);
563        let key = if link.first.block < link.second.block {
564            (&link.first.block, &link.second.block)
565        } else {
566            (&link.second.block, &link.first.block)
567        };
568        let mut record = object([
569            ("first", link_side(&link.first)),
570            ("second", link_side(&link.second)),
571            ("role", Json::String("whole_face_join".into())),
572            ("field_order", n(0)),
573            (
574                "basis_evidence",
575                Json::String("oriented_proportional_homogeneous_face".into()),
576            ),
577        ]);
578        let mut set_evidence = |role: &str, method: &str, r: usize| {
579            if let Json::Object(fields) = &mut record {
580                fields.insert("role".into(), Json::String(role.into()));
581                fields.insert("basis_evidence".into(), Json::String(method.into()));
582                fields.insert("field_order".into(), n(r));
583            }
584        };
585        if match_chart_jets && link.first.axis < 2 && link.second.axis < 2 {
586            if let Some(scale) = normal_scale(link, a, b)? {
587                let sign_b = if link.first.upper == link.second.upper {
588                    1
589                } else {
590                    -1
591                };
592                for (i, j) in paired_coords(link, a, b)? {
593                    let mut row = Row::new();
594                    jet_row(
595                        &mut row,
596                        a,
597                        i,
598                        &link.first,
599                        1,
600                        (scale, a.d[link.first.axis] as i128),
601                        &pids,
602                    )?;
603                    jet_row(
604                        &mut row,
605                        b,
606                        j,
607                        &link.second,
608                        1,
609                        (Q::one(), sign_b * b.d[link.second.axis] as i128),
610                        &pids,
611                    )?;
612                    lateral.push(row);
613                }
614                set_evidence(
615                    "distinct_support_join",
616                    "exact_proportional_complete_homogeneous_normal_jet",
617                    1,
618                );
619                if let Json::Object(fields) = &mut record {
620                    fields.insert(
621                        "normal_interval_ratio".into(),
622                        Json::Array(vec![n(scale.numerator()), n(scale.denominator())]),
623                    );
624                }
625                coupled += 1;
626                evidence.push(record);
627                continue;
628            }
629        }
630        if wanted.contains(&key) {
631            if strip_parent.is_some() && link.first.axis == 0 && link.second.axis == 0 {
632                return Err("duplicate source-parent and transverse join request");
633            }
634            if link.first.axis == 2 || link.second.axis == 2 {
635                return Err("expected lateral transverse join");
636            }
637            crate::transverse_jets::append(link, a, b, &pids, &mut lateral)?;
638            coupled += 1;
639            set_evidence(
640                "distinct_support_join",
641                "exact_physical_gradient_unisolvence",
642                1,
643            );
644            evidence.push(record);
645            continue;
646        }
647        let (r, ha, hb) = if link.first.axis == 2
648            && link.second.axis == 2
649            && a.shear == b.shear
650            && continuity > 0
651        {
652            // Incidence proves the entire rational section matches up to a constant
653            // homogeneous gauge and the section check proves extrusion in physical z.
654            (continuity, a.height()?, b.height()?)
655        } else if restore_strip && link.first.axis == 0 && link.second.axis == 0 {
656            let parent = strip_parent.as_ref().ok_or("missing source parent")?;
657            let restriction = |id: &str| {
658                parent
659                    .source_spans
660                    .iter()
661                    .find(|span| span.source_block_id == id)
662                    .map(|span| &span.restriction)
663                    .ok_or("missing source restriction")
664            };
665            let ar = restriction(&link.first.block)?;
666            let br = restriction(&link.second.block)?;
667            let [x, y] = ar.intervals[0];
668            let [s, t] = br.intervals[0];
669            let seam = ar.intervals[0][usize::from(link.first.upper ^ ar.reversals[0])];
670            if seam != br.intervals[0][usize::from(link.second.upper ^ br.reversals[0])] {
671                return Err("parent restrictions disagree at interface");
672            }
673            let knots = model.field("lower")?.field("knots")?.items()?;
674            let m = knots
675                .iter()
676                .map(Json::number::<i64>)
677                .collect::<Result<Vec<_>>>()?
678                .into_iter()
679                .filter(|&v| v == seam)
680                .count();
681            let smooth = a.d[0].checked_sub(m).ok_or("invalid knot multiplicity")?;
682            if smooth == 0 {
683                evidence.push(record);
684                continue;
685            }
686            (
687                smooth,
688                Q::new(i128::from(y) - i128::from(x), 1)?,
689                Q::new(i128::from(t) - i128::from(s), 1)?,
690            )
691        } else {
692            evidence.push(record);
693            continue;
694        };
695        if link.first.upper == link.second.upper {
696            return Err("unsupported reversed normal parent interval");
697        }
698        for (i, j) in paired_coords(link, a, b)? {
699            for k in 1..=r {
700                let mut row = Row::new();
701                jet_row(&mut row, a, i, &link.first, k, (ha, 1), &pids)?;
702                jet_row(&mut row, b, j, &link.second, k, (hb, -1), &pids)?;
703                eq.push(row);
704            }
705        }
706        set_evidence(
707            "common_parent_subdivision",
708            if link.first.axis == 2 {
709                "same_homogeneous_section_affine_physical_z"
710            } else {
711                "recovered_source_knots_weighted_numerator_jets"
712            },
713            r,
714        );
715        evidence.push(record);
716        inherited += 1;
717    }
718    if !match_chart_jets && coupled != wanted.len() {
719        return Err("requested transverse interface not found");
720    }
721    let (prior_lift, free) = crate::spline_constraints::joint_kernel(n0, &eq, &lateral)?;
722    let mut free_original: Vec<_> = free.iter().map(|&i| order[i]).collect();
723    free_original.sort_unstable();
724    let free_indices: BTreeMap<_, _> = free_original
725        .iter()
726        .enumerate()
727        .map(|(i, &r)| (r, i))
728        .collect();
729    let coords: Vec<_> = free_original.iter().map(|&i| xyz[i].unwrap()).collect();
730    let lift: Vec<Row> = priority_id
731        .iter()
732        .map(|&i| {
733            prior_lift[i]
734                .iter()
735                .map(|(&k, &v)| (free_indices[&order[free[k]]], v))
736                .collect()
737        })
738        .collect();
739    for (i, row) in lift.iter().enumerate() {
740        let mut h = [Q::zero(); 4];
741        for (&j, &k) in row {
742            for (v, x) in h.iter_mut().zip(coords[j].into_iter().chain([Q::one()])) {
743                *v = v.checked_add(k.checked_mul(x)?)?;
744            }
745        }
746        let pos = xyz[i].unwrap();
747        if h != [pos[0], pos[1], pos[2], Q::one()] {
748            return Err("jet space lost an affine coordinate");
749        }
750    }
751    let mut cells = Vec::new();
752    for s in sections {
753        let extraction: Vec<_> = ids[s.offset..s.offset + s.len()]
754            .iter()
755            .map(|&i| lift[i].clone())
756            .collect();
757        let active: Vec<_> = extraction
758            .iter()
759            .flat_map(|r| r.keys().copied())
760            .collect::<BTreeSet<_>>()
761            .into_iter()
762            .collect();
763        let local: BTreeMap<_, _> = active.iter().enumerate().map(|(i, &j)| (j, i)).collect();
764        let stencils = extraction
765            .iter()
766            .map(|r| r.iter().map(|(i, &c)| (local[i], val(c))).collect())
767            .collect();
768        let weights = s
769            .block
770            .controls
771            .as_chunks::<2>()
772            .0
773            .iter()
774            .flat_map(|pair| std::iter::repeat_n(pair[0][3] as f64, degree + 1))
775            .collect();
776        let geometry = RationalBezierMap::new(
777            s.block.degrees,
778            s.block
779                .controls
780                .iter()
781                .map(|h| h.map(|v| v as f64))
782                .collect(),
783        )
784        .map_err(|_| "invalid source map")?;
785        cells.push(InheritedCell {
786            id: s.block.id.clone(),
787            degrees: s.d,
788            z_bounds: s.z.map(val),
789            extraction,
790            active,
791            weights,
792            stencils,
793            geometry,
794        });
795    }
796    Ok(InheritedSpace {
797        cells,
798        ndof: free.len(),
799        control_xyz: coords.into_iter().map(|p| p.map(val)).collect(),
800        c0_ndof: n0,
801        common_parent_faces: inherited,
802        retained_faces: links.interfaces.len() - inherited - coupled,
803        transverse_faces: coupled,
804        interface_evidence: evidence,
805        lift,
806    })
807}

820/// Compile an independently replayable field descriptor, retaining geometry, boundaries and job plan.
821pub fn space_from_plan(model: &Json, plan: &Json) -> Result<InheritedSpace> {
822    if plan.field("schema")?.text()? == "matched_chart_jets_v1" {
823        plan.keys(&["schema", "axial_degree", "axial_continuity"])?;
824        return compile_extrusion(
825            model,
826            plan.field("axial_degree")?.number()?,
827            plan.field("axial_continuity")?.number()?,
828            false,
829            &[],
830            0,
831            true,
832        );
833    }
834    let plane = if let Ok(p) = plan.field("plane_degree") {
835        plan.keys(&[
836            "schema",
837            "axial_degree",
838            "axial_continuity",
839            "restore_strip",
840            "transverse_interfaces",
841            "plane_degree",
842        ])?;
843        p.number()?
844    } else {
845        plan.keys(&[
846            "schema",
847            "axial_degree",
848            "axial_continuity",
849            "restore_strip",
850            "transverse_interfaces",
851        ])?;
852        0
853    };
854    if plan.field("schema")?.text()? != "inherited_field_plan_v1" {
855        return Err("unsupported inherited field plan");
856    }
857    let restore = match plan.field("restore_strip")? {
858        Json::Bool(v) => *v,
859        _ => return Err("restore_strip must be boolean"),
860    };
861    let pairs = plan
862        .field("transverse_interfaces")?
863        .items()?
864        .iter()
865        .map(|v| {
866            let a = v.items()?;
867            if a.len() != 2 {
868                return Err("expected two named blocks per transverse join");
869            }
870            Ok((a[0].text()?.to_owned(), a[1].text()?.to_owned()))
871        })
872        .collect::<Result<Vec<_>>>()?;
873    build_plane_space(
874        model,
875        plan.field("axial_degree")?.number()?,
876        plan.field("axial_continuity")?.number()?,
877        restore,
878        &pairs,
879        plane,
880    )
881}
882pub fn compile(model: &Json, plan: &Json) -> Result<Json> {
883    let space = space_from_plan(model, plan)?;
884    Ok(object([
885        (
886            "schema",
887            Json::String("solidcraft_inherited_analysis_v1".into()),
888        ),
889        ("geometry", model.clone()),
890        ("plan", plan.clone()),
891        ("field", space.document()),
892    ]))
893}
894pub fn replay(package: &Json) -> Result<InheritedSpace> {
895    package.keys(&["schema", "geometry", "plan", "field"])?;
896    let space = space_from_plan(package.field("geometry")?, package.field("plan")?)?;
897    if package.field("schema")?.text()? != "solidcraft_inherited_analysis_v1"
898        || package.field("field")? != &space.document()
899    {
900        return Err("inherited field differs from canonical replay");
901    }
902    Ok(space)
903}
904pub fn audit_package(package: &Json) -> Result<Json> {
905    let field = replay(package)?;
906    Ok(object([
907        ("status", Json::String("valid".into())),
908        ("scalar_dofs", n(field.ndof)),
909    ]))
910}

912impl InheritedSpace {
913    /// Homogeneous essential data on selected full source faces. Every
914    /// Bernstein trace row is constrained even when an eliminated coefficient
915    /// depends on several roots whose own coordinates lie off the face.
916    pub fn zero_face_lift(&self, faces: &[BlockSide]) -> Result<crate::constraint_system::Lift> {
917        use crate::constraint_system::{Lift, LiftLine};
918        let cells: BTreeMap<_, _> = self.cells.iter().map(|c| (c.id.as_str(), c)).collect();
919        let mut eq = Vec::new();
920        for side in faces {
921            let c = cells
922                .get(side.block.as_str())
923                .ok_or("unknown essential face")?;
924            if side.axis > 2 {
925                return Err("invalid essential face axis");
926            }
927            let [s, t] = face_axes(side.axis).map(|d| c.degrees[d]);
928            for i in 0..=s {
929                for j in 0..=t {
930                    eq.push(
931                        c.extraction[index(c.degrees, face_coord(side, i, j, c.degrees))].clone(),
932                    );
933                }
934            }
935        }
936        if eq.is_empty() {
937            return Err("empty essential selection");
938        }
939        let (rows, free_roots) = nullspace(self.ndof, &eq)?;
940        Ok(Lift {
941            free_roots,
942            lines: rows
943                .into_iter()
944                .enumerate()
945                .map(|(dof, r)| LiftLine {
946                    dof,
947                    terms: r.into_iter().collect(),
948                    offset: Q::zero(),
949                })
950                .collect(),
951        })
952    }
