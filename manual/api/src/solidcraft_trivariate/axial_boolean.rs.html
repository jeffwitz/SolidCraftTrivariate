1//! Regularized booleans for products of one fixed G2 section and axial intervals.
2//!
3//! The operation is performed on open axial cells between sorted interval
4//! bounds. Selected cells are retained as separate layers, including at a
5//! coplanar union contact; their common caps are represented as internal faces
6//! in the returned incidence. Cross-section intersections are outside scope.
7

15const AXIAL_LIMIT_MM: i32 = 1_000;
16
17/// Boolean operation on two products of the same section and z interval.
18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub enum AxialBooleanOperation {
20    Union,
21    Intersection,
22    Difference,
23}
24
25/// A G2 section family. Only the rectangle and pierced square are currently supported.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum AxialSectionFamily {
28    ReferenceRectangleV1,
29    CoreDiskV1,
30    FullAnnulusV1,
31    PiercedSquareV1,
32    RoundedCornerV1,
33    LayeredPocketV1,
34}
35
36impl AxialSectionFamily {
37    fn name(self) -> &'static str {
38        match self {
39            Self::ReferenceRectangleV1 => "reference_rectangle_v1",
40            Self::CoreDiskV1 => "core_disk_v1",
41            Self::FullAnnulusV1 => "full_annulus_v1",
42            Self::PiercedSquareV1 => "pierced_square_v1",
43            Self::RoundedCornerV1 => "rounded_corner_v1",
44            Self::LayeredPocketV1 => "layered_pocket_v1",
45        }
46    }
47
48    fn supported(self) -> bool {
49        matches!(self, Self::ReferenceRectangleV1 | Self::PiercedSquareV1)
50    }
51}

100/// Complete layered model for one regularized axial boolean.
101#[derive(Debug, Clone, PartialEq)]

104    pub section: AxialSectionFamily,
105    pub axis: AxialAxis,

117pub enum AxialBooleanError {

119    DifferentSections,
120    IncompatibleAxes,
121    UnsupportedAxis,
122    InvertedInterval,

126    NumericalRange,

146fn validate_interval(interval: [i32; 2]) -> Result<(), AxialBooleanError> {
147    if interval[0] < -AXIAL_LIMIT_MM || interval[1] > AXIAL_LIMIT_MM {
148        return Err(AxialBooleanError::IntervalOutOfRange);
149    }
150    if interval[0] > interval[1] {
151        return Err(AxialBooleanError::InvertedInterval);
152    }
153    if interval[0] == interval[1] {
154        return Err(AxialBooleanError::NullInterval);
155    }
156    Ok(())
157}

159fn selected(operation: AxialBooleanOperation, in_a: bool, in_b: bool) -> bool {
160    match operation {
161        AxialBooleanOperation::Union => in_a || in_b,
162        AxialBooleanOperation::Intersection => in_a && in_b,
163        AxialBooleanOperation::Difference => in_a && !in_b,
164    }
165}

167fn make_cells(
168    operation: AxialBooleanOperation,
169    interval_a: [i32; 2],
170    interval_b: [i32; 2],
171) -> Vec<AxialCell> {
172    let mut bounds = vec![interval_a[0], interval_a[1], interval_b[0], interval_b[1]];
173    bounds.sort_unstable();
174    bounds.dedup();
175    bounds
176        .windows(2)
177        .map(|w| {
178            let [lo, hi] = [w[0], w[1]];
179            let in_a = lo >= interval_a[0] && hi <= interval_a[1];
180            let in_b = lo >= interval_b[0] && hi <= interval_b[1];
181            AxialCell {
182                bounds: [lo, hi],
183                in_a: in_a && selected(operation, in_a, in_b),
184                in_b: in_b && selected(operation, in_a, in_b),
185            }
186        })
187        .collect()
188}

190fn reslice_block(
191    source: &BracketBlock,
192    id: String,
193    interval: [i32; 2],
194) -> Result<BracketBlock, AxialBooleanError> {
195    let mut controls = Vec::with_capacity(source.controls.len());
196    for [x, y, z, weight] in &source.controls {
197        let w = i128::from(*weight);
198        let z = i128::from(*z);
199        let output_z = if z == 0 {
200            i128::from(interval[0]) * w
201        } else if z == 8 * w {
202            i128::from(interval[1]) * w
203        } else {
204            return Err(AxialBooleanError::UnsupportedSectionModel);
205        };
206        controls.push([
207            *x,
208            *y,
209            i64::try_from(output_z).map_err(|_| AxialBooleanError::NumericalRange)?,
210            *weight,
211        ]);
212    }
213    Ok(BracketBlock {
214        id,
215        degrees: source.degrees,
216        controls,
217    })
218}

220fn source_faces(
221    source_blocks: &[BracketBlock],
222    section: AxialSectionFamily,
223) -> Result<BTreeMap<(String, usize, bool), String>, AxialBooleanError> {
224    let certificate = audit_section(section.name(), source_blocks)
225        .map_err(|_| AxialBooleanError::UnsupportedSectionModel)?;
226    Ok(certificate
227        .design_faces
228        .into_iter()
229        .map(|(side, label)| ((side.block, side.axis, side.upper), label))
230        .collect())
231}

233fn cell_operands(cell: AxialCell) -> Vec<OperandLabel> {
234    let mut result = Vec::with_capacity(2);
235    if cell.in_a {
236        result.push(OperandLabel::A);
237    }
238    if cell.in_b {
239        result.push(OperandLabel::B);
240    }
241    result
242}

244fn cap_sources(
245    operation: AxialBooleanOperation,
246    output_upper: bool,
247    z: i32,
248    interval_a: [i32; 2],
249    interval_b: [i32; 2],
250) -> Vec<FaceProvenance> {
251    let output_normal = if output_upper { 1 } else { -1 };
252    let mut result = Vec::new();
253    for (operand, interval) in [(OperandLabel::A, interval_a), (OperandLabel::B, interval_b)] {
254        let reversed = operation == AxialBooleanOperation::Difference && operand == OperandLabel::B;
255        let endpoints = [(interval[0], -1, "z0"), (interval[1], 1, "z8")];
256        for (coordinate, mut source_normal, source_face) in endpoints {
257            if coordinate != z {
258                continue;
259            }
260            if reversed {
261                source_normal = -source_normal;
262            }
263            if source_normal == output_normal {
264                result.push(FaceProvenance {
265                    operand,
266                    source_face: source_face.to_string(),
267                    orientation_reversed: reversed,
268                });
269            }
270        }
271    }
272    result
273}

275/// Build a regularized boolean of two solids in the same G2 cross-section.
276///
277/// Both operands must use the z axis and the same supported section family.
278/// Axial bounds are ordered integer millimeters in `[-1000,1000]`. Selected
279/// subintervals are retained as distinct layers rather than fused, so every
280/// internal cap match remains visible in the exact face incidence.
281pub fn build_axial_boolean(
282    operation: AxialBooleanOperation,
283    a: AxialOperand,
284    b: AxialOperand,
285) -> Result<AxialBooleanModel, AxialBooleanError> {
286    if a.axis != b.axis {
287        return Err(AxialBooleanError::IncompatibleAxes);
288    }
289    if a.axis != AxialAxis::Z {
290        return Err(AxialBooleanError::UnsupportedAxis);
291    }
292    if a.section != b.section {
293        return Err(AxialBooleanError::DifferentSections);
294    }
295    if !a.section.supported() {
296        return Err(AxialBooleanError::UnsupportedSection);
297    }
298    validate_interval(a.interval_mm)?;
299    validate_interval(b.interval_mm)?;
300
301    // Audit the source recipe before transforming any cells.
302    let source_blocks =
303        section_blocks(a.section.name()).map_err(|_| AxialBooleanError::UnsupportedSectionModel)?;
304    let source_labels = source_faces(&source_blocks, a.section)?;
305    let cells = make_cells(operation, a.interval_mm, b.interval_mm);
306    let mut selected_intervals = Vec::new();
307    let mut blocks = Vec::new();
308    let mut metadata = BTreeMap::<String, OutputBlockMeta>::new();
309    let mut component_count = 0usize;
310    let mut was_selected = false;
311    let mut selected_length = 0_i128;
312
313    for (index, cell) in cells.into_iter().enumerate() {
314        let is_selected = cell.in_a || cell.in_b;
315        if !is_selected {
316            was_selected = false;
317            continue;
318        }
319        if !was_selected {
320            component_count += 1;
321        }
322        was_selected = true;
323        selected_length = selected_length
324            .checked_add(i128::from(cell.bounds[1] - cell.bounds[0]))
325            .ok_or(AxialBooleanError::NumericalRange)?;
326        selected_intervals.push(SelectedAxialInterval {
327            bounds_mm: cell.bounds,
328            operands: cell_operands(cell),
329        });
330
331        for source in &source_blocks {
332            let id = format!("axial_boolean_cell_{index:02}_{}", source.id);
333            let output = reslice_block(source, id.clone(), cell.bounds)?;
334            metadata.insert(
335                id,
336                OutputBlockMeta {
337                    source_id: source.id.clone(),
338                    interval: cell.bounds,
339                    in_a: cell.in_a,
340                    in_b: cell.in_b,
341                },
342            );
343            blocks.push(output);
344        }
345    }
346
347    let incidence = enumerate_faces(&blocks).map_err(AxialBooleanError::Incidence)?;
348    let mut faces = Vec::with_capacity(incidence.exterior.len());
349    for side in &incidence.exterior {
350        let meta = metadata
351            .get(&side.block)
352            .ok_or(AxialBooleanError::ProvenanceMismatch)?;
353        let mut provenance = if side.axis == 2 {
354            let z = if side.upper {
355                meta.interval[1]
356            } else {
357                meta.interval[0]
358            };
359            cap_sources(operation, side.upper, z, a.interval_mm, b.interval_mm)
360        } else {
361            let label = source_labels
362                .get(&(meta.source_id.clone(), side.axis, side.upper))
363                .ok_or(AxialBooleanError::ProvenanceMismatch)?;
364            let mut labels = Vec::with_capacity(2);
365            if meta.in_a {
366                labels.push(FaceProvenance {
367                    operand: OperandLabel::A,
368                    source_face: label.clone(),
369                    orientation_reversed: false,
370                });
371            }
372            if meta.in_b {
373                labels.push(FaceProvenance {
374                    operand: OperandLabel::B,
375                    source_face: label.clone(),
376                    orientation_reversed: false,
377                });
378            }
379            labels
380        };
381        provenance.sort_by_key(|entry| (entry.operand, entry.source_face.clone()));
382        if provenance.is_empty() {
383            return Err(AxialBooleanError::ProvenanceMismatch);
384        }
385        faces.push(AxialBooleanFace {
386            side: side.clone(),
387            provenance,
388        });
389    }
390    faces.sort_by(|left, right| left.side.cmp(&right.side));
391
392    let area = match a.section {
393        AxialSectionFamily::ReferenceRectangleV1 => 400.0,
394        AxialSectionFamily::PiercedSquareV1 => 400.0 - 16.0 * std::f64::consts::PI,
395        _ => return Err(AxialBooleanError::UnsupportedSection),
396    };
397    let volume_mm3 = area * selected_length as f64;
398    if !volume_mm3.is_finite() || volume_mm3 < 0.0 {
399        return Err(AxialBooleanError::NumericalRange);
400    }
401
402    Ok(AxialBooleanModel {
403        operation,
404        section: a.section,
405        axis: a.axis,
406        selected_intervals,
407        component_count,
408        blocks,
409        incidence,
410        faces,
411        volume_mm3,
412    })
