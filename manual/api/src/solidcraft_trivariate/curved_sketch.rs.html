1//! Exact polynomial curves for rectangular sketch cuts on guided partitions.
2//!
3//! The scaffold constructor determines the topology. This adapter accepts only
4//! whole rectangular feature sides, so every curved trace is represented once
5//! and copied identically to all incident blocks. Through features use a 2/3
6//! corner contraction and half-span midpoint handles (C1 joins). Blind cuts
7//! use 4/5 contraction and quarter-span handles (G0 joins).
8
9use crate::bracket::BracketBlock;
10use crate::bracket_json::{decode_blocks, encode_blocks};
11use crate::json::{object, Json};

13const FAMILY: &str = "curved_sketch_layers_v1";
14// Represents thirds, fifths, eighths, and the exact Coons-center division by 4.
15const SCALE: i64 = 480;
16
17fn n(value: impl ToString) -> Json {
18    Json::Number(value.to_string())
19}
20fn text(value: &str) -> Json {
21    Json::String(value.to_owned())
22}

31#[derive(Clone)]
32struct Feature {
33    id: String,
34    vertices: Vec<[i64; 2]>,
35    bounds: [i64; 4], // xmin, xmax, ymin, ymax
36    shrink: (i64, i64),
37    handle: (i64, i64),
38    through: bool,
39}

41#[derive(Clone, Copy)]
42struct EdgeCurve {
43    a: [i64; 2],
44    b: [i64; 2],
45    outward: i64,
46    transverse_span: i64,
47    sag: (i64, i64),
48}

50fn xy(value: &Json) -> Result<[i64; 2], &'static str> {
51    let [x, y] = value.items()? else {
52        return Err("curved_sketch_expected_xy_point");
53    };
54    Ok([x.number()?, y.number()?])
55}

57fn bounds(vertices: &[[i64; 2]]) -> [i64; 4] {
58    [
59        vertices.iter().map(|p| p[0]).min().unwrap(),
60        vertices.iter().map(|p| p[0]).max().unwrap(),
61        vertices.iter().map(|p| p[1]).min().unwrap(),
62        vertices.iter().map(|p| p[1]).max().unwrap(),
63    ]
64}
65
66fn on_segment(p: [i64; 2], a: [i64; 2], b: [i64; 2]) -> bool {
67    if a[0] == b[0] {
68        p[0] == a[0] && a[1].min(b[1]) <= p[1] && p[1] <= a[1].max(b[1])
69    } else if a[1] == b[1] {
70        p[1] == a[1] && a[0].min(b[0]) <= p[0] && p[0] <= a[0].max(b[0])
71    } else {
72        false
73    }
74}

76fn features(scaffold: &Json) -> Result<(Vec<Feature>, Vec<EdgeCurve>), &'static str> {
77    let height = scaffold.field("height_mm")?.number::<i64>()?;
78    let mut features = Vec::new();
79    let mut edges = Vec::new();
80    let outer = scaffold.field("outer")?;
81    let outer_vertices = outer
82        .field("vertices")?
83        .items()?
84        .iter()
85        .map(xy)
86        .collect::<Result<Vec<_>, _>>()?;
87    if outer_vertices.len() < 4 {
88        return Err("curved_sketch_bad_outer_loop");
89    }
90    let ob = bounds(&outer_vertices);
91    for i in 0..outer_vertices.len() {
92        let a = outer_vertices[i];
93        let b = outer_vertices[(i + 1) % outer_vertices.len()];
94        if a[1] == ob[2] && b[1] == ob[2] {
95            edges.push(EdgeCurve {
96                a,
97                b,
98                outward: -1,
99                transverse_span: ob[3] - ob[2],
100                sag: (1, 8),
101            });
102        }
103    }
104    for cut in scaffold.field("cuts")?.items()? {
105        let vertices = cut
106            .field("vertices")?
107            .items()?
108            .iter()
109            .map(xy)
110            .collect::<Result<Vec<_>, _>>()?;
111        if vertices.len() != 4 {
112            return Err("curved_sketch_requires_rectangular_features");
113        }
114        let b = bounds(&vertices);
115        if b[0] == b[1] || b[2] == b[3] {
116            return Err("curved_sketch_degenerate_feature");
117        }
118        for i in 0..4 {
119            let a = vertices[i];
120            let c = vertices[(i + 1) % 4];
121            if a[0] != c[0] && a[1] != c[1] {
122                return Err("curved_sketch_requires_axis_aligned_features");
123            }
124        }
125        let interval = cut.field("z_interval_mm")?.items()?;
126        if interval.len() != 2 {
127            return Err("curved_sketch_bad_feature_interval");
128        }
129        let through = interval[0].number::<i64>()? == 0 && interval[1].number::<i64>()? == height;
130        let shrink = if through { (2, 3) } else { (4, 5) };
131        let handle = if through { (1, 2) } else { (1, 4) };
132        let id = cut.field("id")?.text()?.to_owned();
133        features.push(Feature {
134            id,
135            vertices: vertices.clone(),
136            bounds: b,
137            shrink,
138            handle,
139            through,
140        });
141        for i in 0..4 {
142            let a = vertices[i];
143            let c = vertices[(i + 1) % 4];
144            let (outward, transverse_span) = if a[1] == c[1] {
145                (if 2 * a[1] > b[2] + b[3] { 1 } else { -1 }, b[3] - b[2])
146            } else {
147                (if 2 * a[0] > b[0] + b[1] { 1 } else { -1 }, b[1] - b[0])
148            };
149            let sag = if through { (1, 3) } else { (1, 5) };
150            edges.push(EdgeCurve {
151                a,
152                b: c,
153                outward,
154                transverse_span,
155                sag,
156            });
157        }
158    }
159    for i in 0..features.len() {
160        for j in 0..i {
161            let a = features[i].bounds;
162            let b = features[j].bounds;
163            if a[0] < b[1] && b[0] < a[1] && a[2] < b[3] && b[2] < a[3] {
164                return Err("curved_sketch_intersecting_features_out_of_scope");
165            }
166        }
167    }
168    Ok((features, edges))
169}

171fn scale_ratio(value: i64, num: i64, den: i64) -> Result<i64, &'static str> {
172    let n = value
173        .checked_mul(num)
174        .and_then(|x| x.checked_mul(SCALE))
175        .ok_or("curved_sketch_integer_overflow")?;
176    if n % den != 0 {
177        return Err("curved_sketch_common_scale_not_exact");
178    }
179    Ok(n / den)
180}

182fn remap(p: [i64; 2], z: i64, features: &[Feature]) -> Result<Point, &'static str> {
183    for feature in features {
184        let on_boundary = (0..feature.vertices.len()).any(|i| {
185            on_segment(
186                p,
187                feature.vertices[i],
188                feature.vertices[(i + 1) % feature.vertices.len()],
189            )
190        });
191        if !on_boundary {
192            continue;
193        }
194        let [xmin, xmax, ymin, ymax] = feature.bounds;
195        // p' = center + shrink * (p - center), kept as an exact integer numerator.
196        let tx = (xmin + xmax) * feature.shrink.1 + (2 * p[0] - xmin - xmax) * feature.shrink.0;
197        let ty = (ymin + ymax) * feature.shrink.1 + (2 * p[1] - ymin - ymax) * feature.shrink.0;
198        let den = 2 * feature.shrink.1;
199        let x = scale_ratio(tx, 1, den)?;
200        let y = scale_ratio(ty, 1, den)?;
201        return Ok(Point { x, y, z: z * SCALE });
202    }
203    Ok(Point {
204        x: p[0] * SCALE,
205        y: p[1] * SCALE,
206        z: z * SCALE,
207    })
208}

210fn matching_curve(
211    a: [i64; 2],
212    b: [i64; 2],
213    edges: &[EdgeCurve],
214) -> Result<Option<usize>, &'static str> {
215    for (i, edge) in edges.iter().enumerate() {
216        if on_segment(a, edge.a, edge.b) && on_segment(b, edge.a, edge.b) {
217            // Curved feature sides must be whole scaffold edges. This keeps
218            // their intended parameterization and corner continuity explicit.
219            if !((a == edge.a && b == edge.b) || (a == edge.b && b == edge.a)) {
220                return Err("curved_sketch_feature_side_subdivided");
221            }
222            return Ok(Some(i));
223        }
224    }
225    Ok(None)
226}

228fn edge_control(
229    a: Point,
230    b: Point,
231    source_a: [i64; 2],
232    source_b: [i64; 2],
233    edges: &[EdgeCurve],
234    seen: &mut [bool],
235) -> Result<[i64; 4], &'static str> {
236    let mut x = (a.x + b.x) / 2;
237    let mut y = (a.y + b.y) / 2;
238    if (a.x + b.x) % 2 != 0 || (a.y + b.y) % 2 != 0 {
239        return Err("curved_sketch_edge_midpoint_not_exact");
240    }
241    if let Some(i) = matching_curve(source_a, source_b, edges)? {
242        seen[i] = true;
243        let edge = &edges[i];
244        let numerator = edge.transverse_span * edge.sag.0 * SCALE;
245        let denominator = edge.sag.1;
246        if numerator % denominator != 0 {
247            return Err("curved_sketch_handle_not_exact_on_common_scale");
248        }
249        let offset = edge.outward * numerator / denominator;
250        if edge.a[0] == edge.b[0] {
251            x += offset;
252        } else {
253            y += offset;
254        }
255    }
256    Ok([x, y, (a.z + b.z) / 2, SCALE])
257}

259fn map_controls(
260    block: BracketBlock,
261    features: &[Feature],
262    edges: &[EdgeCurve],
263    seen: &mut [bool],
264    top_shift: [i64; 2],
265    height: i64,
266) -> Result<BracketBlock, &'static str> {
267    if block.degrees != [1, 1, 1]
268        || block.controls.len() != 8
269        || block.controls.iter().any(|p| p[3] != 1)
270    {
271        return Err("curved_sketch_requires_integral_trilinear_scaffold");
272    }
273    let raw = block.controls;
274    let corner = |i: usize, j: usize, k: usize| raw[(i * 2 + j) * 2 + k];
275    let z0 = corner(0, 0, 0)[2];
276    let z1 = corner(0, 0, 1)[2];
277    let scaled_shift = top_shift.map(|delta| delta * z0);
278    if scaled_shift.iter().any(|x| x % height != 0) {
279        return Err("curved_sketch_nonintegral_layer_offset");
280    }
281    let offset = scaled_shift.map(|x| x / height);
282    let raw_xy = [
283        corner(0, 0, 0),
284        corner(1, 0, 0),
285        corner(0, 1, 0),
286        corner(1, 1, 0),
287    ]
288    .map(|p| [p[0] - offset[0], p[1] - offset[1]]);
289    if [corner(1, 0, 0)[2], corner(0, 1, 0)[2], corner(1, 1, 0)[2]]
290        .iter()
291        .any(|&z| z != z0)
292        || [corner(1, 0, 1)[2], corner(0, 1, 1)[2], corner(1, 1, 1)[2]]
293            .iter()
294            .any(|&z| z != z1)
295    {
296        return Err("curved_sketch_requires_planar_axial_layers");
297    }
298    let shift = [
299        corner(0, 0, 1)[0] - corner(0, 0, 0)[0],
300        corner(0, 0, 1)[1] - corner(0, 0, 0)[1],
301    ];
302    for (i, j) in [(1, 0), (0, 1), (1, 1)] {
303        let a = corner(i, j, 0);
304        let b = corner(i, j, 1);
305        if [b[0] - a[0], b[1] - a[1]] != shift {
306            return Err("curved_sketch_requires_uniform_axial_shift");
307        }
308    }
309    let p = raw_xy
310        .map(|q| remap(q, z0, features))
311        .into_iter()
312        .collect::<Result<Vec<_>, _>>()?;
313    let [p00, p10, p01, p11] =
314        <[Point; 4]>::try_from(p).map_err(|_| "curved_sketch_bad_corner_count")?;
315    let mids = [
316        edge_control(p00, p10, raw_xy[0], raw_xy[1], edges, seen)?,
317        edge_control(p01, p11, raw_xy[2], raw_xy[3], edges, seen)?,
318        edge_control(p00, p01, raw_xy[0], raw_xy[2], edges, seen)?,
319        edge_control(p10, p11, raw_xy[1], raw_xy[3], edges, seen)?,
320    ];
321    let corners = [p00, p10, p01, p11].map(|p| [p.x, p.y, p.z, SCALE]);
322    let sum = |axis: usize| mids.iter().map(|p| p[axis]).sum::<i64>();
323    let mut center = [0_i64; 4];
324    for axis in 0..3 {
325        let numerator = 2 * sum(axis) - corners.iter().map(|p| p[axis]).sum::<i64>();
326        if numerator % 4 != 0 {
327            return Err("curved_sketch_coons_center_not_exact");
328        }
329        center[axis] = numerator / 4;
330    }
331    center[3] = SCALE;
332    let mut controls = Vec::with_capacity(18);
333    for i in 0..3 {
334        for j in 0..3 {
335            let q = match (i, j) {
336                (0, 0) => corners[0],
337                (2, 0) => corners[1],
338                (0, 2) => corners[2],
339                (2, 2) => corners[3],
340                (1, 0) => mids[0],
341                (1, 2) => mids[1],
342                (0, 1) => mids[2],
343                (2, 1) => mids[3],
344                (1, 1) => center,
345                _ => unreachable!(),
346            };
347            let q = [
348                q[0] + offset[0] * SCALE,
349                q[1] + offset[1] * SCALE,
350                q[2],
351                SCALE,
352            ];
353            controls.push(q);
354            controls.push([
355                q[0] + shift[0] * SCALE,
356                q[1] + shift[1] * SCALE,
357                z1 * SCALE,
358                SCALE,
359            ]);
360        }
361    }
362    Ok(BracketBlock {
363        id: block.id,
364        degrees: [2, 2, 1],
365        controls,
366    })
367}

369/// Construct a rationally exact curved model from a sketch-layer scaffold.
370/// Input: `{ "family":"curved_sketch_layers_v1", "scaffold": <recipe> }`.
371pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
372    recipe.keys(&["family", "scaffold"])?;
373    if recipe.field("family")?.text()? != FAMILY {
374        return Err("curved_sketch_unknown_family");
375    }
376    let scaffold = recipe.field("scaffold")?;
377    if scaffold.field("family")?.text()? != "sketch_layer_partition_v1" {
378        return Err("curved_sketch_requires_guided_layer_scaffold");
379    }
380    let base = crate::interchange::build(scaffold)?;
381    let (features, edges) = features(scaffold)?;
382    let height = scaffold.field("height_mm")?.number::<i64>()?;
383    let shift = scaffold
384        .field("top_shift_mm")
385        .ok()
386        .map(xy)
387        .transpose()?
388        .unwrap_or([0, 0]);
389    let mut seen = vec![false; edges.len()];
390    let blocks = decode_blocks(&base)?
391        .into_iter()
392        .map(|block| map_controls(block, &features, &edges, &mut seen, shift, height))
393        .collect::<Result<Vec<_>, _>>()?;
394    if seen.contains(&false) {
395        return Err("curved_sketch_feature_side_missing_from_scaffold");
396    }
397    let topology = base.field("topology")?.clone();
398    let design_faces = base.field("design_faces")?.clone();
399    let scaffold_plan = base.field("partition_plan").ok().cloned();
400    let mut model = object([
401        ("schema", text(FAMILY)),
402        ("scaffold", scaffold.clone()),
403        ("blocks", encode_blocks(blocks)),
404        ("topology", topology),
405        ("design_faces", design_faces),
406        (
407            "degree_convention",
408            text("tensor_bernstein_xy_quadratic_z_linear"),
409        ),
410        (
411            "curve_construction",
412            object([
413                (
414                    "method",
415                    text("contracted_rectangle_quadratic_midpoint_controls"),
416                ),
417                ("common_integer_scale", n(SCALE)),
418                ("through_corner_scale", Json::Array(vec![n(2), n(3)])),
419                (
420                    "through_handle_fraction_of_reduced_span",
421                    Json::Array(vec![n(1), n(2)]),
422                ),
423                ("blind_corner_scale", Json::Array(vec![n(4), n(5)])),
424                (
425                    "blind_handle_fraction_of_reduced_span",
426                    Json::Array(vec![n(1), n(4)]),
427                ),
428                ("through_join", text("C1")),
429                ("blind_join", text("G0")),
430            ]),
431        ),
432        (
433            "feature_curve_provenance",
434            Json::Array(
435                features
436                    .iter()
437                    .map(|f| {
438                        object([
439                            ("id", text(&f.id)),
440                            ("kind", text(if f.through { "through" } else { "blind" })),
441                            (
442                                "source_loop",
443                                Json::Array(
444                                    f.vertices
445                                        .iter()
446                                        .map(|p| Json::Array(p.iter().copied().map(n).collect()))
447                                        .collect(),
448                                ),
449                            ),
450                            ("corner_scale_numerator", n(f.shrink.0)),
451                            ("corner_scale_denominator", n(f.shrink.1)),
452                            ("handle_numerator", n(f.handle.0)),
453                            ("handle_denominator", n(f.handle.1)),
454                        ])
455                    })
456                    .collect(),
457            ),
458        ),
459    ]);
460    if let (Some(plan), Json::Object(fields)) = (scaffold_plan, &mut model) {
461        fields.insert("scaffold_partition_plan".into(), plan);
462    }
463    Ok(model)
