34#[derive(Clone, PartialEq, Eq)]
35struct Trace {
36    degree: [usize; 2],
37    points: Vec<[i128; 4]>,
38}
39
40fn gcd(mut a: i128, mut b: i128) -> i128 {
41    while b != 0 {
42        (a, b) = (b, a % b);
43    }
44    a.abs()
45}
46
47impl Trace {
48    fn point(&self, i: usize, j: usize) -> [i128; 4] {
49        self.points[i * (self.degree[1] + 1) + j]
50    }
51
52    fn corner_key(&self) -> Result<[[i128; 4]; 4], &'static str> {
53        let [p, q] = self.degree;
54        let mut corners = [
55            self.point(0, 0),
56            self.point(0, q),
57            self.point(p, 0),
58            self.point(p, q),
59        ];
60        for point in &mut corners {
61            let scale = point.iter().copied().fold(0, gcd);
62            *point = point.map(|v| v / scale);
63        }
64        corners.sort();
65        if corners.windows(2).any(|pair| pair[0] == pair[1]) {
66            return Err("collapsed face");
67        }
68        Ok(corners)
69    }
70
71    fn oriented(&self, o: Orientation) -> Self {
72        let [p, q] = self.degree;
73        let degree = if o.swap { [q, p] } else { [p, q] };
74        let mut points = Vec::new();
75        for i in 0..=degree[0] {
76            for j in 0..=degree[1] {
77                let a = if o.flip_first { degree[0] - i } else { i };
78                let b = if o.flip_second { degree[1] - j } else { j };
79                points.push(if o.swap {
80                    self.point(b, a)
81                } else {
82                    self.point(a, b)
83                });
84            }
85        }
86        Self { degree, points }
87    }
88
89    fn elevated(mut self, target: usize) -> Self {
90        // Elevation in each face axis uses small integer factors. Even a
91        // linear-by-linear net elevated to cubic-by-cubic stays far inside
92        // the exact i128 range for i64 input coefficients.
93        for axis in 0..2 {
94            while self.degree[axis] < target {
95                let mut degree = self.degree;
96                degree[axis] += 1;
97                let n = degree[axis];
98                let mut points = Vec::new();
99                for i in 0..=degree[0] {
100                    for j in 0..=degree[1] {
101                        let mut prev = [i, j];
102                        let mut next = prev;
103                        let k = prev[axis];
104                        prev[axis] = k.saturating_sub(1);
105                        next[axis] = k.min(n - 1);
106                        let a = self.point(prev[0], prev[1]);
107                        let b = self.point(next[0], next[1]);
108                        points.push(std::array::from_fn(|c| {
109                            k as i128 * a[c] + (n - k) as i128 * b[c]
110                        }));
111                    }
112                }
113                self = Self { degree, points };
114            }
115        }
116        let scale = self.points.iter().flatten().copied().fold(0, gcd);
117        for p in &mut self.points {
118            *p = p.map(|v| v / scale);
119        }
120        self
121    }
122}

124fn extract(block: &BracketBlock, axis: usize, upper: bool) -> Trace {
125    let [p, q, r] = block.degrees;
126    let mut points = Vec::new();
127    let fixed = if upper { block.degrees[axis] } else { 0 };
128    for i in 0..=p {
129        for j in 0..=q {
130            for k in 0..=r {
131                if [i, j, k][axis] == fixed {
132                    points.push(block.controls[(i * (q + 1) + j) * (r + 1) + k].map(i128::from));
133                }
134            }
135        }
136    }
137    let degree = match axis {
138        0 => [q, r],
139        1 => [p, r],
140        _ => [p, q],
141    };
142    Trace { degree, points }
143}

145/// Reconstruct incidences within a whole-face template. Proving disjoint
146/// interiors and ruling out partial contacts require a geometric family audit.
147pub fn enumerate_faces(blocks: &[BracketBlock]) -> Result<Incidence, &'static str> {
148    let mut ids = BTreeSet::new();
149    let mut buckets = BTreeMap::<_, Vec<(BlockSide, Trace)>>::new();
150    for block in blocks {
151        if block.id.is_empty() || !ids.insert(&block.id) {
152            return Err("invalid block identity");
153        }
154        if block.degrees.iter().any(|d| !(1..=3).contains(d))
155            || block.controls.len() != block.degrees.iter().map(|d| d + 1).product::<usize>()
156            || block.controls.iter().any(|p| p[3] <= 0)
157        {
158            return Err("invalid block net");
159        }
160        for axis in 0..3 {
161            for upper in [false, true] {
162                let trace = extract(block, axis, upper);
163                let key = trace.corner_key()?;
164                let side = BlockSide {
165                    block: block.id.clone(),
166                    axis,
167                    upper,
168                };
169                buckets.entry(key).or_default().push((side, trace));
170            }
171        }
172    }
173    let mut result = Incidence {
174        interfaces: Vec::new(),
175        exterior: Vec::new(),
176    };
177    for mut faces in buckets.into_values() {
178        faces.sort_by(|a, b| a.0.cmp(&b.0));
179        match faces.as_slice() {
180            [(side, _)] => result.exterior.push(side.clone()),
181            [(a, ta), (b, tb)] => {
182                let target = *ta.degree.iter().chain(&tb.degree).max().unwrap();
183                let first = ta.clone().elevated(target);
184                let second = tb.clone().elevated(target);
185                let mut matches = Vec::new();
186                for swap in [false, true] {
187                    for flip_first in [false, true] {
188                        for flip_second in [false, true] {
189                            let orientation = Orientation {
190                                swap,
191                                flip_first,
192                                flip_second,
193                            };
194                            if first == second.oriented(orientation) {
195                                matches.push(orientation);
196                            }
197                        }
198                    }
199                }
200                if matches.len() != 1 {
201                    return Err("nonunique or incompatible trace");
202                }
203                result.interfaces.push(Interface {
204                    first: a.clone(),
205                    second: b.clone(),
206                    orientation: matches[0],
207                });
208            }
209            _ => return Err("nonmanifold face"),
210        }
211    }
212    result.interfaces.sort_by(|a, b| a.first.cmp(&b.first));
213    result.exterior.sort();
214    Ok(result)
215}
