56fn point(h: H) -> Result<[R; 2], Error> {
57    if h[2].numerator() <= 0 {
58        return Err("periodic profile has a nonpositive denominator");
59    }
60    Ok([div(h[0], h[2])?, div(h[1], h[2])?])
61}

67fn gcd(mut a: u128, mut b: u128) -> u128 {
68    while b != 0 {
69        (a, b) = (b, a % b);
70    }
71    a
72}

74fn lcm(a: i128, b: i128) -> Result<i128, Error> {
75    let g = i128::try_from(gcd(a.unsigned_abs(), b.unsigned_abs()))
76        .map_err(|_| "homogeneous scale exceeds i128")?;
77    (a / g)
78        .checked_mul(b)
79        .ok_or("homogeneous scale exceeds i128")
80}

106fn split_net(net: &HNet, t: R) -> Result<(HNet, HNet), Error> {
107    if t == R::zero() || t == R::one() {
108        return Err("profile subdivision must be interior");
109    }
110    let mut row = net.clone();
111    let mut left = vec![row[0]];
112    let mut right = vec![*row.last().ok_or("empty profile span")?];
113    while row.len() > 1 {
114        let mut next_row = Vec::with_capacity(row.len() - 1);
115        for pair in row.windows(2) {
116            let mut q = [R::zero(); 3];
117            for d in 0..3 {
118                q[d] =
119                    pair[0][d].checked_add(t.checked_mul(pair[1][d].checked_sub(pair[0][d])?)?)?;
120            }
121            next_row.push(q);
122        }
123        row = next_row;
124        left.push(row[0]);
125        right.push(*row.last().unwrap());
126    }
127    right.reverse();
128    Ok((left, right))
129}

131fn direction(p: [R; 2], center: [R; 2]) -> Result<usize, Error> {
132    let dx = p[0].checked_sub(center[0])?;
133    let dy = p[1].checked_sub(center[1])?;
134    if dx == R::zero() && dy == R::zero() {
135        return Err("periodic profile anchor equals its center");
136    }
137    let abs = |v: R| -> Result<R, Error> {
138        if v.numerator() < 0 {
139            R::zero().checked_sub(v)
140        } else {
141            Ok(v)
142        }
143    };
144    if abs(dx)? == abs(dy)? {
145        Ok(match (dx.numerator().signum(), dy.numerator().signum()) {
146            (1, 1) => 0,
147            (-1, 1) => 2,
148            (-1, -1) => 4,
149            (1, -1) => 6,
150            _ => return Err("invalid diagonal profile direction"),
151        })
152    } else if dx == R::zero() {
153        Ok(if dy.numerator() > 0 { 1 } else { 5 })
154    } else if dy == R::zero() {
155        Ok(if dx.numerator() < 0 { 3 } else { 7 })
156    } else {
157        Err("periodic profile anchor is not on a cage ray")
158    }
159}

161fn cross(a: [R; 2], b: [R; 2]) -> Result<R, Error> {
162    a[0].checked_mul(b[1])?.checked_sub(a[1].checked_mul(b[0])?)
163}

165fn verify_polar_monotonicity(net: &[H], center: [R; 2]) -> Result<(), Error> {
166    let points = net
167        .iter()
168        .map(|&h| point(h))
169        .collect::<Result<Vec<_>, _>>()?;
170    let vectors = points
171        .iter()
172        .map(|p| Ok([p[0].checked_sub(center[0])?, p[1].checked_sub(center[1])?]))
173        .collect::<Result<Vec<[R; 2]>, Error>>()?;
174    for i in 0..vectors.len() {
175        for j in i + 1..vectors.len() {
176            if cross(vectors[i], vectors[j])?.numerator() <= 0 {
177                return Err("periodic profile span lacks a strict single-sector polar certificate");
178            }
179        }
180    }
181    Ok(())
182}

193fn parse_curve(recipe: &Json) -> Result<(Curve, Vec<BezierSpan>), Error> {
194    recipe.keys(&[
195        "family",
196        "bounds_xyz_mm",
197        "profile",
198        "rect_y_mm",
199        "rect_z_mm",
200    ])?;
201    if recipe.field("family")?.text()? != "g2_cross_plate_v1" {
202        return Err("unsupported cross-plate family");
203    }
204    let bounds = recipe.field("bounds_xyz_mm")?.items()?;
205    let [xb, yb, zb] = bounds else {
206        return Err("expected XYZ stock bounds");
207    };
208    let pair = |value: &Json| -> Result<[i64; 2], Error> {
209        let [a, b] = value.items()? else {
210            return Err("expected two stock bounds");
211        };
212        Ok([a.number()?, b.number()?])
213    };
214    let [xmin, xmax] = pair(xb)?;
215    let [ymin, ymax] = pair(yb)?;
216    let [zmin, zmax] = pair(zb)?;
217    if [xmin, ymin, zmin] != [0; 3] || xmin >= xmax || ymin >= ymax || zmin >= zmax {
218        return Err("cross-plate stock bounds must be positive and start at the origin");
219    }
220    let rect_y = recipe.field("rect_y_mm")?.items()?;
221    let [yl, yh] = rect_y else {
222        return Err("rectangle y bounds must be rational pairs");
223    };
224    let rational = |value: &Json| -> Result<R, Error> {
225        let [n, d] = value.items()? else {
226            return Err("rectangle ordinate must be a numerator/denominator pair");
227        };
228        let (n, d) = (n.number::<i64>()?, d.number::<i64>()?);
229        if d <= 0 {
230            return Err("rectangle denominator must be positive");
231        }
232        R::new(i128::from(n), i128::from(d))
233    };
234    let y_low = rational(yl)?;
235    let y_high = rational(yh)?;
236    if compare(y_low, int(ymin)) != std::cmp::Ordering::Greater
237        || compare(y_high, int(ymax)) != std::cmp::Ordering::Less
238        || compare(y_low, y_high) != std::cmp::Ordering::Less
239    {
240        return Err("rectangle y section must be strictly inside stock");
241    }
242    let z_cut = pair(recipe.field("rect_z_mm")?)?;
243    if z_cut[0] <= zmin || z_cut[0] >= z_cut[1] || z_cut[1] >= zmax {
244        return Err("rectangle z interval must be internal to the plate");
245    }
246    let source = SourceCurve::parse(recipe.field("profile")?)?;
247    if source.degree != 3 {
248        return Err("periodic profile must be cubic");
249    }
250    let spans = profile_spans(recipe.field("profile")?)?;
251    if spans.len() < 8 {
252        return Err("periodic profile needs eight active spans");
253    }
254    for p in source.controls {
255        if p[2] <= 0
256            || compare(R::new(i128::from(p[0]), i128::from(p[2]))?, int(xmin))
257                != std::cmp::Ordering::Greater
258            || compare(R::new(i128::from(p[0]), i128::from(p[2]))?, int(xmax))
259                != std::cmp::Ordering::Less
260            || compare(R::new(i128::from(p[1]), i128::from(p[2]))?, int(ymin))
261                != std::cmp::Ordering::Greater
262            || compare(R::new(i128::from(p[1]), i128::from(p[2]))?, int(ymax))
263                != std::cmp::Ordering::Less
264        {
265            return Err("periodic profile control hull must be strictly inside stock");
266        }
267    }
268    if spans[0].controls[0] != *spans.last().unwrap().controls.last().unwrap() {
269        // Extraction spans may carry independent integer gauges, so compare
270        // physical closure below instead of rejecting unequal homogeneous nets.
271        let a = point(spans[0].controls[0].map(int))?;
272        let b = point(spans.last().unwrap().controls.last().unwrap().map(int))?;
273        if a != b {
274            return Err("periodic profile does not close geometrically");
275        }
276    }
277    let endpoints: Vec<_> = spans
278        .iter()
279        .flat_map(|s| [s.controls[0], *s.controls.last().unwrap()])
280        .map(|p| point(p.map(int)))
281        .collect::<Result<_, _>>()?;
282    let extrema = |axis: usize| -> Result<(R, R), Error> {
283        let mut values = endpoints.iter().map(|p| p[axis]);
284        let first = values.next().ok_or("profile has no endpoints")?;
285        Ok(values.fold((first, first), |(lo, hi), v| {
286            (
287                if compare(v, lo).is_lt() { v } else { lo },
288                if compare(v, hi).is_gt() { v } else { hi },
289            )
290        }))
291    };
292    let (xmin_p, xmax_p) = extrema(0)?;
293    let (ymin_p, ymax_p) = extrema(1)?;
294    let center = [
295        xmin_p.checked_add(xmax_p)?.checked_mul(R::new(1, 2)?)?,
296        ymin_p.checked_add(ymax_p)?.checked_mul(R::new(1, 2)?)?,
297    ];
298    Ok((
299        Curve {
300            bounds: [xmin, xmax, ymin, ymax],
301            center,
302            y_low,
303            y_high,
304            z_cut,
305            height: zmax,
306        },
307        spans,
308    ))
309}

318fn canonical_spans(
319    spans: &[BezierSpan],
320    center: [R; 2],
321) -> Result<Vec<(HNet, usize, bool)>, Error> {
322    let mut nets: Vec<_> = spans.iter().map(ratio_net).collect();
323    for i in 1..nets.len() {
324        let p = *nets[i - 1].last().unwrap();
325        let q = nets[i][0];
326        scale_net(&mut nets[i], div(p[2], q[2])?)?;
327        if nets[i][0] != p {
328            return Err("profile knot traces disagree after gauge alignment");
329        }
330    }
331    let first = nets[0][0];
332    let last = *nets.last().unwrap().last().unwrap();
333    if point(first)? != point(last)? || first != last {
334        return Err("periodic profile endpoints are not coincident in one homogeneous gauge");
335    }
336    let unique: Vec<_> = nets
337        .iter()
338        .flat_map(|n| [n[0], *n.last().unwrap()])
339        .map(point)
340        .collect::<Result<_, _>>()?;
341    let anchors: BTreeMap<usize, [R; 2]> = unique
342        .into_iter()
343        .filter_map(|p| direction(p, center).ok().map(|d| (d, p)))
344        .collect();
345    if anchors.len() != 8 {
346        return Err("profile must have eight distinct cardinal/diagonal cage anchors");
347    }
348
349    let start = anchors[&0]; // northeast diagonal, independent of source index.
350    let incident: Vec<_> = nets
351        .iter()
352        .enumerate()
353        .filter_map(|(i, net)| {
354            let a = point(net[0]).ok()?;
355            let b = point(*net.last()?).ok()?;
356            if a == start {
357                Some((i, false, b))
358            } else if b == start {
359                Some((i, true, a))
360            } else {
361                None
362            }
363        })
364        .collect();
365    if incident.len() != 2 {
366        return Err("periodic profile does not have a unique start anchor");
367    }
368    let mut chosen = None;
369    for option in &incident {
370        let end = option.2;
371        let a = [
372            start[0].checked_sub(center[0])?,
373            start[1].checked_sub(center[1])?,
374        ];
375        let b = [
376            end[0].checked_sub(center[0])?,
377            end[1].checked_sub(center[1])?,
378        ];
379        if cross(a, b)?.numerator() > 0 {
380            chosen = Some(option);
381            break;
382        }
383    }
384    let chosen = chosen.ok_or("profile orientation is not counterclockwise from northeast")?;
385    let mut used = BTreeSet::new();
386    let mut ordered = Vec::with_capacity(nets.len());
387    let mut current = start;
388    let mut first_piece = true;
389    let mut candidate = (chosen.0, chosen.1);
390    while ordered.len() < nets.len() {
391        let (index, reverse) = candidate;
392        if !used.insert(index) {
393            return Err("profile span graph contains a cycle before closure");
394        }
395        let mut net = nets[index].clone();
396        if reverse {
397            net = reverse_net(&net);
398        }
399        if point(net[0])? != current {
400            return Err("profile span graph has a disconnected endpoint");
401        }
402        gcd_point(&net)?;
403        verify_polar_monotonicity(&net, center)?;
404        current = point(*net.last().unwrap())?;
405        ordered.push((net, index, reverse));
406        first_piece = false;
407        if current == start {
408            break;
409        }
410        let next: Vec<_> = nets
411            .iter()
412            .enumerate()
413            .filter(|(i, _)| !used.contains(i))
414            .filter_map(|(i, net)| {
415                let a = point(net[0]).ok()?;
416                let b = point(*net.last()?).ok()?;
417                if a == current {
418                    Some((i, false))
419                } else if b == current {
420                    Some((i, true))
421                } else {
422                    None
423                }
424            })
425            .collect();
426        if next.len() != 1 {
427            return Err("profile spans do not form one simple periodic chain");
428        }
429        candidate = next[0];
430    }
431    let _ = first_piece;
432    if ordered.len() != nets.len() || current != start {
433        return Err("profile chain did not close after one revolution");
434    }
435    Ok(ordered)
436}

438fn endpoint_directions(nets: &[HNet], center: [R; 2]) -> Result<BTreeMap<usize, [R; 2]>, Error> {
439    let mut anchors = BTreeMap::new();
440    for net in nets {
441        for h in [net[0], *net.last().unwrap()] {
442            let p = point(h)?;
443            if let Ok(d) = direction(p, center) {
444                if anchors.insert(d, p).is_some_and(|old| old != p) {
445                    return Err("duplicate profile point on one cage ray");
446                }
447            }
448        }
449    }
450    if anchors.len() != 8 {
451        return Err("profile must expose one endpoint on each cage ray");
452    }
453    Ok(anchors)
454}

456fn anchor_outer(direction: usize, bounds: [i64; 4], center: [R; 2]) -> Result<[R; 2], Error> {
457    let [xmin, xmax, ymin, ymax] = bounds;
458    let [center_x, center_y] = center;
459    Ok(match direction {
460        0 => [int(xmax), int(ymax)],
461        1 => [center_x, int(ymax)],
462        2 => [int(xmin), int(ymax)],
463        3 => [int(xmin), center_y],
464        4 => [int(xmin), int(ymin)],
465        5 => [center_x, int(ymin)],
466        6 => [int(xmax), int(ymin)],
467        7 => [int(xmax), center_y],
468        _ => unreachable!(),
469    })
470}

494fn split_piece_at(net: HNet, cuts: &[R]) -> Result<Vec<(HNet, R, R)>, Error> {
495    let mut parts = vec![(net, R::zero(), R::one())];
496    let mut previous = R::zero();
497    let mut result = Vec::new();
498    for cut in cuts {
499        let local = div(cut.checked_sub(previous)?, R::one().checked_sub(previous)?)?;
500        let (rest, lo, hi) = parts.pop().ok_or("missing profile subdivision")?;
501        let (left, right) = split_net(&rest, local)?;
502        let mid = lo.checked_add(hi.checked_sub(lo)?.checked_mul(local)?)?;
503        result.push((left, lo, mid));
504        parts.push((right, mid, hi));
505        previous = *cut;
506    }
507    result.extend(parts);
508    Ok(result)
509}

511fn line_controls(a: H, b: H, degree: usize) -> Result<HNet, Error> {
512    if degree != 3 {
513        return Err("outer cage curve requires cubic elevation");
514    }
515    let mut net = vec![a, b];
516    while net.len() < degree + 1 {
517        let n = net.len() - 1;
518        let m = n + 1;
519        let mut elevated = Vec::with_capacity(m + 1);
520        elevated.push(net[0]);
521        for i in 1..=n {
522            let left = R::new(i as i128, m as i128)?;
523            let right = R::one().checked_sub(left)?;
524            let mut control = [R::zero(); 3];
525            for (d, value) in control.iter_mut().enumerate() {
526                *value = net[i - 1][d]
527                    .checked_mul(left)?
528                    .checked_add(net[i][d].checked_mul(right)?)?;
529            }
530            elevated.push(control);
531        }
532        elevated.push(*net.last().unwrap());
533        net = elevated;
534    }
535    Ok(net)
536}

538fn outer_net(inner: &HNet, endpoints: [[R; 2]; 2], lateral_x: Option<i64>) -> Result<HNet, Error> {
539    if let Some(x) = lateral_x {
540        return inner
541            .iter()
542            .map(|p| {
543                let scale = R::new(3, 1)?;
544                Ok([
545                    int(x).checked_mul(p[2])?.checked_mul(scale)?,
546                    p[1].checked_mul(scale)?,
547                    p[2].checked_mul(scale)?,
548                ])
549            })
550            .collect();
551    }
552    let scale = R::new(3, 1)?;
553    let a = [
554        endpoints[0][0]
555            .checked_mul(inner[0][2])?
556            .checked_mul(scale)?,
557        endpoints[0][1]
558            .checked_mul(inner[0][2])?
559            .checked_mul(scale)?,
560        inner[0][2].checked_mul(scale)?,
561    ];
562    let b = [
563        endpoints[1][0]
564            .checked_mul(inner.last().unwrap()[2])?
565            .checked_mul(scale)?,
566        endpoints[1][1]
567            .checked_mul(inner.last().unwrap()[2])?
568            .checked_mul(scale)?,
569        inner.last().unwrap()[2].checked_mul(scale)?,
570    ];
571    line_controls(a, b, inner.len() - 1)
572}

574fn block_controls(outer: &HNet, inner: &HNet, z0: i64, z1: i64) -> Result<Vec<[i64; 4]>, Error> {
575    if outer.len() != 4 || inner.len() != 4 {
576        return Err("cross-plate sector must have cubic profile degree");
577    }
578    let mut net = Vec::with_capacity(16);
579    for i in 0..4 {
580        for (v, p) in [outer[i], inner[i]].into_iter().enumerate() {
581            let _ = v;
582            for z in [z0, z1] {
583                net.push([p[0], p[1], int(z).checked_mul(p[2])?, p[2]]);
584            }
585        }
586    }
587    let mut scale = 1_i128;
588    for p in &net {
589        for c in p {
590            scale = lcm(scale, c.denominator())?;
591        }
592    }
593    let mut integers = Vec::with_capacity(net.len());
594    let mut common = 0_u128;
595    for p in net {
596        let mut q = [0_i128; 4];
597        for d in 0..4 {
598            q[d] = p[d]
599                .numerator()
600                .checked_mul(scale / p[d].denominator())
601                .ok_or("cross-plate homogeneous coefficient exceeds i128")?;
602            common = gcd(common, q[d].unsigned_abs());
603        }
604        integers.push(q);
605    }
606    let common =
607        i128::try_from(common).map_err(|_| "cross-plate homogeneous scale exceeds i128")?;
608    integers
609        .into_iter()
610        .map(|p| {
611            p.map(|v| i64::try_from(v / common).map_err(|_| "cross-plate control exceeds i64"))
612                .into_iter()
613                .collect::<Result<Vec<_>, _>>()?
614                .try_into()
615                .map_err(|_| "invalid cross-plate control")
616        })
617        .collect()
618}

620fn boundary_side(points: [[R; 2]; 2], bounds: [i64; 4]) -> Result<usize, Error> {
621    let [xmin, xmax, ymin, ymax] = bounds;
622    if points[0][0] == int(xmin) && points[1][0] == int(xmin) {
623        Ok(0)
624    } else if points[0][0] == int(xmax) && points[1][0] == int(xmax) {
625        Ok(1)
626    } else if points[0][1] == int(ymin) && points[1][1] == int(ymin) {
627        Ok(2)
628    } else if points[0][1] == int(ymax) && points[1][1] == int(ymax) {
629        Ok(3)
630    } else {
631        Err("outer anchor pair is not on a stock rectangle side")
632    }
633}
634
635fn transverse_face_y(block: &BracketBlock, side: &BlockSide) -> Result<R, Error> {
636    let u = if side.upper { 3 } else { 0 };
637    let mut common = None;
638    for v in 0..=1 {
639        for z in 0..=1 {
640            let control = block.controls[(u * 2 + v) * 2 + z];
641            let y = R::new(i128::from(control[1]), i128::from(control[3]))?;
642            if common.is_some_and(|value| value != y) {
643                return Err("transverse exposed face is not a constant-y sketch edge");
644            }
645            common = Some(y);
646        }
647    }
648    common.ok_or("transverse exposed face has no controls")
649}

651/// Construct the full stock minus the two intersecting through bores.
652pub fn construct(recipe: &Json) -> Result<Construction, Error> {
653    let (parameters, _) = crate::sketch_partition_plan::plan(recipe)?;
654    construct_partition(&parameters)
655}
656
657/// Construct from a normalized plan emitted by the shared sketch planner.
658pub(crate) fn construct_partition(parameters: &Json) -> Result<Construction, Error> {
659    let (curve, spans) = parse_curve(parameters)?;
660    let nets = canonical_spans(&spans, curve.center)?;
661    let net_values: Vec<_> = nets.iter().map(|x| x.0.clone()).collect();
662    let ray_anchors = endpoint_directions(&net_values, curve.center)?;
663    let start = ray_anchors[&0];
664    let end = ray_anchors[&0];
665    let _ = (start, end);
666    let mut anchor_map = BTreeMap::new();
667    for (d, p) in &ray_anchors {
668        add_ratio_point(
669            &mut anchor_map,
670            *p,
671            anchor_outer(*d, curve.bounds, curve.center)?,
672            Some(*d),
673        )?;
674    }
675    let low_roots = crossings(&spans, curve.y_low)?;
676    let high_roots = crossings(&spans, curve.y_high)?;
677    let mut crossing_points = BTreeMap::new();
678    for (roots, y) in [(&low_roots, curve.y_low), (&high_roots, curve.y_high)] {
679        for root in roots {
680            let side = if compare(root.position[0], curve.center[0]) == std::cmp::Ordering::Less {
681                curve.bounds[0]
682            } else {
683                curve.bounds[1]
684            };
685            let outer = [int(side), y];
686            add_ratio_point(&mut anchor_map, root.position, outer, None)?;
687            crossing_points.insert((root.span, root.local), root.position);
688        }
689    }
690    let mut anchor_list: Vec<_> = anchor_map.values().copied().collect();
691    // The anchor order is the geometrically prescribed CCW outer cage. This
692    // also makes source cyclic shifts and reversals produce the same plan.
693    let mut ordered_anchors = Vec::with_capacity(ANCHOR_COUNT);
694    let expected_anchors = [
695        (Some(0), None, None),
696        (Some(1), None, None),
697        (Some(2), None, None),
698        (None, Some(curve.bounds[0]), Some(curve.y_high)),
699        (Some(3), None, None),
700        (None, Some(curve.bounds[0]), Some(curve.y_low)),
701        (Some(4), None, None),
702        (Some(5), None, None),
703        (Some(6), None, None),
704        (None, Some(curve.bounds[1]), Some(curve.y_low)),
705        (Some(7), None, None),
706        (None, Some(curve.bounds[1]), Some(curve.y_high)),
707    ];
708    for (direction, side, y) in expected_anchors {
709        if let Some(d) = direction {
710            ordered_anchors.push(
711                anchor_map
712                    .values()
713                    .find(|a| a.direction == Some(d))
714                    .copied()
715                    .ok_or("missing periodic cage ray")?,
716            );
717        } else {
718            let target = [int(side.unwrap()), y.unwrap()];
719            ordered_anchors.push(
720                anchor_map
721                    .values()
722                    .find(|a| a.direction.is_none() && a.outer == target)
723                    .copied()
724                    .ok_or("missing transverse intersection anchor")?,
725            );
726        }
727    }
728    anchor_list.clear();
729    let anchor_index: BTreeMap<_, _> = ordered_anchors
730        .iter()
731        .enumerate()
732        .map(|(i, a)| (a.point, i))
733        .collect();
734
735    // Split source spans exactly at the four plan events and orient each piece
736    // in canonical CCW order. The active parameter translation is immaterial:
737    // geometric endpoint anchors, rather than knot indices, select the start.
738    let mut source_cuts = BTreeMap::<usize, Vec<R>>::new();
739    for root in low_roots.iter().chain(&high_roots) {
740        source_cuts.entry(root.span).or_default().push(root.local);
741    }
742    for v in source_cuts.values_mut() {
743        v.sort_by(|a, b| compare(*a, *b));
744        v.dedup();
745    }
746    let mut split_by_span = Vec::<Vec<(HNet, R, R)>>::new();
747    for (net, source_index, reversed) in &nets {
748        let mut cuts = source_cuts.get(source_index).cloned().unwrap_or_default();
749        if *reversed {
750            cuts = cuts
751                .into_iter()
752                .map(|t| R::one().checked_sub(t))
753                .collect::<Result<_, _>>()?;
754        }
755        cuts.sort_by(|a, b| compare(*a, *b));
756        split_by_span.push(split_piece_at(net.clone(), &cuts)?);
757    }
758    // Recover the geometric chain from endpoint connectivity. This keeps
759    // sector numbering invariant under cyclic reindexing and parameter reversal.
760    let mut pieces = Vec::<ProfilePiece>::new();
761    for (span_index, subnets) in split_by_span.iter().enumerate() {
762        let source_index = nets[span_index].1;
763        let knots = &spans[source_index].interval;
764        for (net, lo, hi) in subnets {
765            pieces.push(ProfilePiece {
766                controls: net.clone(),
767                start: point(net[0])?,
768                end: point(*net.last().unwrap())?,
769                duration: int(knots[1]
770                    .checked_sub(knots[0])
771                    .ok_or("source knot interval exceeds i64")?)
772                .checked_mul(hi.checked_sub(*lo)?)?,
773                source_span: source_index,
774            });
775        }
776    }
777    // `canonical_spans` already orders complete source spans counterclockwise
778    // from the northeast anchor; de Casteljau children retain that order.
779    let ordered_pieces = pieces;
780    if ordered_pieces.first().map(|p| p.start) != Some(ordered_anchors[0].point) {
781        return Err("profile does not start at the canonical northeast anchor");
782    }
783
784    // Assign every oriented piece to one of the twelve ordered anchor arcs.
785    let mut sectors: [Vec<ProfilePiece>; ANCHOR_COUNT] = std::array::from_fn(|_| Vec::new());
786    let mut current_sector = 0;
787    for piece in ordered_pieces {
788        if let Some(&index) = anchor_index.get(&piece.start) {
789            current_sector = index;
790        }
791        if current_sector >= ANCHOR_COUNT {
792            return Err("invalid cross-plate sector assignment");
793        }
794        sectors[current_sector].push(piece.clone());
795        if anchor_index.contains_key(&piece.end) {
796            current_sector = anchor_index[&piece.end];
797        }
798    }
799    if sectors.iter().any(Vec::is_empty) {
800        return Err("periodic profile omitted a required cage sector");
801    }
802
803    let mut blocks = Vec::new();
804    let mut tags = BTreeMap::<String, Tag>::new();
805    let z_events = [0, curve.z_cut[0], curve.z_cut[1], curve.height];
806    for layer in 0..3 {
807        let z0 = z_events[layer];
808        let z1 = z_events[layer + 1];
809        for sector in 0..ANCHOR_COUNT {
810            if layer == 1 && LATERAL.contains(&sector) {
811                continue;
812            }
813            let left_anchor = ordered_anchors[sector];
814            let right_anchor = ordered_anchors[(sector + 1) % ANCHOR_COUNT];
815            let is_lateral = LATERAL.contains(&sector);
816            let line_side = if is_lateral {
817                None
818            } else {
819                Some(boundary_side(
820                    [left_anchor.outer, right_anchor.outer],
821                    curve.bounds,
822                )?)
823            };
824            let duration: R = sectors[sector]
825                .iter()
826                .try_fold(R::zero(), |sum, p| sum.checked_add(p.duration))?;
827            let mut offset = R::zero();
828            for (piece_index, piece) in sectors[sector].iter().enumerate() {
829                let next_offset = offset.checked_add(piece.duration)?;
830                let q0 = div(offset, duration)?;
831                let q1 = div(next_offset, duration)?;
832                let outer_points = if let Some(side) = line_side {
833                    let interpolate = |q: R| -> Result<[R; 2], Error> {
834                        Ok([
835                            left_anchor.outer[0].checked_add(q.checked_mul(
836                                right_anchor.outer[0].checked_sub(left_anchor.outer[0])?,
837                            )?)?,
838                            left_anchor.outer[1].checked_add(q.checked_mul(
839                                right_anchor.outer[1].checked_sub(left_anchor.outer[1])?,
840                            )?)?,
841                        ])
842                    };
843                    let pts = [interpolate(q0)?, interpolate(q1)?];
844                    let _ = side;
845                    pts
846                } else {
847                    [
848                        [
849                            int(if sector < 6 {
850                                curve.bounds[0]
851                            } else {
852                                curve.bounds[1]
853                            }),
854                            piece.start[1],
855                        ],
856                        [
857                            int(if sector < 6 {
858                                curve.bounds[0]
859                            } else {
860                                curve.bounds[1]
861                            }),
862                            piece.end[1],
863                        ],
864                    ]
865                };
866                let outer = outer_net(
867                    &piece.controls,
868                    outer_points,
869                    if is_lateral {
870                        Some(if sector < 6 {
871                            curve.bounds[0]
872                        } else {
873                            curve.bounds[1]
874                        })
875                    } else {
876                        None
877                    },
878                )?;
879                let controls = block_controls(&outer, &piece.controls, z0, z1)?;
880                let id = format!(
881                    "g2_s{sector}_p{piece_index}_z{layer}_src{}",
882                    piece.source_span
883                );
884                let block = BracketBlock {
885                    id: id.clone(),
886                    degrees: [3, 1, 1],
887                    controls,
888                };
889                let certificate =
890                    certify_spatial(block.degrees, &block.controls, Default::default());
891                if certificate.status != CertificateStatus::Valid {
892                    return Err(
893                        "cross-plate sector lacks a positive Bernstein determinant certificate",
894                    );
895                }
896                tags.insert(
897                    id,
898                    Tag {
899                        sector,
900                        layer,
901                        side_min: outer_points[0],
902                        side_max: outer_points[1],
903                    },
904                );
905                blocks.push(block);
906                offset = next_offset;
907            }
908        }
909    }
910
911    let incidence = enumerate_faces(&blocks)?;
912    let blocks_by_id: BTreeMap<_, _> = blocks
913        .iter()
914        .map(|block| (block.id.as_str(), block))
915        .collect();
916    let mut faces = Vec::new();
917    for side in incidence.exterior {
918        let tag = tags
919            .get(&side.block)
920            .ok_or("missing cross-plate block provenance")?;
921        let z0 = z_events[tag.layer];
922        let z1 = z_events[tag.layer + 1];
923        let feature = match side.axis {
924            0 => {
925                let neighbor = if side.upper {
926                    (tag.sector + 1) % ANCHOR_COUNT
927                } else {
928                    (tag.sector + ANCHOR_COUNT - 1) % ANCHOR_COUNT
929                };
930                if tag.layer == 1 && LATERAL.contains(&neighbor) {
931                    let block = blocks_by_id
932                        .get(side.block.as_str())
933                        .ok_or("missing exposed cross-plate block")?;
934                    let face_y = transverse_face_y(block, &side)?;
935                    if face_y == curve.y_low {
936                        "rect_x/ymin".to_owned()
937                    } else if face_y == curve.y_high {
938                        "rect_x/ymax".to_owned()
939                    } else {
940                        return Err(
941                            "exposed transverse side is not on a registered rectangle edge",
942                        );
943                    }
944                } else {
945                    return Err("unexpected exposed periodic sector seam");
946                }
947            }
948            1 if side.upper => "nurbs_z_hole/wall".to_owned(),
949            1 => {
950                let p = tag.side_min;
951                let q = tag.side_max;
952                if p[0] == int(curve.bounds[0]) && q[0] == int(curve.bounds[0]) {
953                    "stock/xmin".to_owned()
954                } else if p[0] == int(curve.bounds[1]) && q[0] == int(curve.bounds[1]) {
955                    "stock/xmax".to_owned()
956                } else if p[1] == int(curve.bounds[2]) && q[1] == int(curve.bounds[2]) {
957                    "stock/ymin".to_owned()
958                } else if p[1] == int(curve.bounds[3]) && q[1] == int(curve.bounds[3]) {
959                    "stock/ymax".to_owned()
960                } else {
961                    return Err("outer guard face is not on the stock boundary");
962                }
963            }
964            2 if !side.upper && z0 == 0 => "stock/zmin".to_owned(),
965            2 if side.upper && z1 == curve.height => "stock/zmax".to_owned(),
966            2 if side.upper && z1 == curve.z_cut[0] && LATERAL.contains(&tag.sector) => {
967                "rect_x/zmin".to_owned()
968            }
969            2 if !side.upper && z0 == curve.z_cut[1] && LATERAL.contains(&tag.sector) => {
970                "rect_x/zmax".to_owned()
971            }
972            2 => return Err("unexpected exposed cross-plate layer face"),
973            _ => return Err("unexpected cross-plate boundary side"),
974        };
975        faces.push((side, feature));
976        let _ = z0;
977        let _ = z1;
978    }
979    Ok((blocks, faces))
980}
