13pub(crate) const FAMILY: &str = "nurbs_cross_bores_v1";
14pub(crate) const BOUNDED_FAMILY: &str = "nurbs_cross_bores_bounded_v1";
15pub(crate) const OVAL_FAMILY: &str = "nurbs_cross_oval_v1";
16pub(crate) const METHOD: &str = "canonical_nurbs_cross_reconstruction_and_exact_incidence";
17pub(crate) const BOUNDED_METHOD: &str = "certified_rational_nurbs_graph_approximation";
18pub(crate) const OVAL_METHOD: &str = "exact_symmetric_nurbs_oval_reconstruction";
19const PARAMETERS: [&str; 2] = ["bounds_mm", "profiles"];

21fn number(n: impl ToString) -> Json {
22    Json::Number(n.to_string())
23}
24fn text(s: &str) -> Json {
25    Json::String(s.to_owned())
26}
27fn array<T>(xs: impl IntoIterator<Item = T>, f: impl FnMut(T) -> Json) -> Json {
28    Json::Array(xs.into_iter().map(f).collect())
29}
30fn pair(v: &Json) -> Result<[i64; 2], &'static str> {
31    v.items()?
32        .iter()
33        .map(Json::number)
34        .collect::<Result<Vec<_>, _>>()?
35        .try_into()
36        .map_err(|_| "expected_two_bounds")
37}
38fn parse_bounds(v: &Json) -> Result<[[i64; 2]; 3], &'static str> {
39    v.items()?
40        .iter()
41        .map(pair)
42        .collect::<Result<Vec<_>, _>>()?
43        .try_into()
44        .map_err(|_| "expected_three_axis_bounds")
45}
46fn curve_json(c: &NurbsCurve2) -> Json {
47    object([
48        ("degree", number(c.degree)),
49        ("knots", array(c.knots.iter().copied(), number)),
50        (
51            "controls",
52            array(c.controls.iter(), |p| array(p.iter().copied(), number)),
53        ),
54    ])
55}
56fn parse_curve(v: &Json) -> Result<NurbsCurve2, &'static str> {
57    v.keys(&["degree", "knots", "controls"])?;
58    let curve = NurbsCurve2 {
59        degree: v.field("degree")?.number()?,
60        knots: v
61            .field("knots")?
62            .items()?
63            .iter()
64            .map(Json::number)
65            .collect::<Result<_, _>>()?,
66        controls: v
67            .field("controls")?
68            .items()?
69            .iter()
70            .map(|p| {
71                p.items()?
72                    .iter()
73                    .map(Json::number)
74                    .collect::<Result<Vec<i64>, _>>()?
75                    .try_into()
76                    .map_err(|_| "expected_homogeneous_2d_control")
77            })
78            .collect::<Result<_, _>>()?,
79    };
80    if !(1..=3).contains(&curve.degree) {
81        return Err("unsupported_nurbs_profile_degree");
82    }
83    Ok(curve)
84}
85fn parse_profiles(v: &Json) -> Result<[NurbsCurve2; 4], &'static str> {
86    v.keys(&["x_min", "x_max", "z_min", "z_max"])?;
87    Ok([
88        parse_curve(v.field("x_min")?)?,
89        parse_curve(v.field("x_max")?)?,
90        parse_curve(v.field("z_min")?)?,
91        parse_curve(v.field("z_max")?)?,
92    ])
93}
94fn parse_center(v: &Json) -> Result<[i64; 3], &'static str> {
95    v.items()?
96        .iter()
97        .map(Json::number)
98        .collect::<Result<Vec<_>, _>>()?
99        .try_into()
100        .map_err(|_| "expected_three_center_coordinates")
101}
102fn parse_meridians(v: &Json) -> Result<[NurbsCurve2; 2], &'static str> {
103    v.keys(&["x", "z"])?;
104    Ok([parse_curve(v.field("x")?)?, parse_curve(v.field("z")?)?])
105}
106fn parse_tolerance(v: &Json) -> Result<Ratio, &'static str> {
107    let [numerator, denominator] = v.items()? else {
108        return Err("expected_rational_lateral_tolerance");
109    };
110    let ratio = Ratio::new(numerator.number()?, denominator.number()?)?;
111    if ratio.numerator() <= 0 {
112        return Err("lateral_tolerance_must_be_positive");
113    }
114    Ok(ratio)
115}
116fn params_json(bounds: [[i64; 2]; 3], profiles: &[NurbsCurve2; 4]) -> (Json, Json) {
117    let b = array(bounds, |axis| array(axis, number));
118    let p = object([
119        ("x_min", curve_json(&profiles[0])),
120        ("x_max", curve_json(&profiles[1])),
121        ("z_min", curve_json(&profiles[2])),
122        ("z_max", curve_json(&profiles[3])),
123    ]);
124    (b, p)
125}
126fn build_parts(
127    bounds: [[i64; 2]; 3],
128    profiles: &[NurbsCurve2; 4],
129) -> Result<Vec<BracketBlock>, &'static str> {
130    for interval in bounds {
131        if interval[0] >= interval[1] {
132            return Err("cross_bounds_must_be_strictly_increasing");
133        }
134    }
135    build_cross(bounds, std::array::from_fn(|i| &profiles[i]))
136}
137fn build_bounded_parts(
138    bounds: [[i64; 2]; 3],
139    profiles: &[NurbsCurve2; 4],
140    tolerance: Ratio,
141    max_depth: usize,
142) -> Result<Vec<BracketBlock>, &'static str> {
143    for interval in bounds {
144        if interval[0] >= interval[1] {
145            return Err("cross_bounds_must_be_strictly_increasing");
146        }
147    }
148    if max_depth > 12 {
149        return Err("max_depth_exceeds_12");
150    }
151    crate::nurbs_cross::build_cross_bounded(
152        bounds,
153        std::array::from_fn(|i| &profiles[i]),
154        tolerance,
155        max_depth,
156    )
157}
158fn certificate(blocks: &[BracketBlock]) -> Result<BracketCertificate, &'static str> {
159    let incidence = enumerate_faces(blocks)?;
160    let design_faces = incidence
161        .exterior
162        .iter()
163        .cloned()
164        .map(|side| (side, "boundary".to_owned()))
165        .collect();
166    Ok(BracketCertificate {
167        incidence,
168        determinant_lower_bound_mm3: 0.0,
169        radial_error_upper_mm: 0.0,
170        design_faces,
171    })
172}

174/// Construct the candidate exclusively through the native NURBS-cross builder.
175pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
176    match recipe.field("family")?.text()? {
177        FAMILY => build_exact(recipe),
178        BOUNDED_FAMILY => build_bounded(recipe),
179        OVAL_FAMILY => build_oval(recipe),
180        _ => Err("unsupported_family"),
181    }
182}
183fn build_exact(recipe: &Json) -> Result<Json, &'static str> {
184    recipe.keys(&[&["family"][..], &PARAMETERS].concat())?;
185    let bounds = parse_bounds(recipe.field("bounds_mm")?)?;
186    let profiles = parse_profiles(recipe.field("profiles")?)?;
187    let blocks = build_parts(bounds, &profiles)?;
188    let cert = certificate(&blocks)?;
189    let mut model = encode(FAMILY, blocks, &cert);
190    if let Json::Object(fields) = &mut model {
191        let (b, p) = params_json(bounds, &profiles);
192        fields.insert("bounds_mm".into(), b);
193        fields.insert("profiles".into(), p);
194    }
195    Ok(model)
196}
197fn build_bounded(recipe: &Json) -> Result<Json, &'static str> {
198    recipe.keys(
199        &[
200            &["family"][..],
201            &PARAMETERS,
202            &["lateral_tolerance_mm", "max_depth"][..],
203        ]
204        .concat(),
205    )?;
206    let bounds = parse_bounds(recipe.field("bounds_mm")?)?;
207    let profiles = parse_profiles(recipe.field("profiles")?)?;
208    let tolerance = parse_tolerance(recipe.field("lateral_tolerance_mm")?)?;
209    let max_depth = recipe.field("max_depth")?.number()?;
210    let blocks = build_bounded_parts(bounds, &profiles, tolerance, max_depth)?;
211    let cert = certificate(&blocks)?;
212    let mut model = encode(BOUNDED_FAMILY, blocks, &cert);
213    if let Json::Object(fields) = &mut model {
214        let (b, p) = params_json(bounds, &profiles);
215        fields.insert("bounds_mm".into(), b);
216        fields.insert("profiles".into(), p);
217        fields.insert(
218            "lateral_tolerance_mm".into(),
219            array([tolerance.numerator(), tolerance.denominator()], number),
220        );
221        fields.insert("max_depth".into(), number(max_depth));
222    }
223    Ok(model)
224}
225fn build_oval(recipe: &Json) -> Result<Json, &'static str> {
226    recipe.keys(&["family", "bounds_mm", "center_mm", "meridians"])?;
227    let bounds = parse_bounds(recipe.field("bounds_mm")?)?;
228    let center = parse_center(recipe.field("center_mm")?)?;
229    let meridians = parse_meridians(recipe.field("meridians")?)?;
230    for interval in bounds {
231        if interval[0] >= interval[1] {
232            return Err("cross_bounds_must_be_strictly_increasing");
233        }
234    }
235    let blocks = crate::nurbs_cross::build_oval(bounds, center, &meridians[0], &meridians[1])?;
236    let cert = certificate(&blocks)?;
237    let mut model = encode(OVAL_FAMILY, blocks, &cert);
238    if let Json::Object(fields) = &mut model {
239        fields.insert("bounds_mm".into(), array(bounds, |v| array(v, number)));
240        fields.insert("center_mm".into(), array(center, number));
241        fields.insert(
242            "meridians".into(),
243            object([
244                ("x", curve_json(&meridians[0])),
245                ("z", curve_json(&meridians[1])),
246            ]),
247        );
248    }
249    Ok(model)
250}

252/// Rebuild from the stored recipe, compare the complete coefficient ledger, and
253/// independently re-enumerate exact whole-face incidences. This is a structural
254/// certificate; continuous geometric claims remain with the construction audit.
255pub(crate) fn audit(model: &Json) -> Result<Json, &'static str> {
256    match model.field("schema")?.text()? {
257        FAMILY => audit_exact(model),
258        BOUNDED_FAMILY => audit_bounded(model),
259        OVAL_FAMILY => audit_oval(model),
260        _ => Err("unsupported_schema"),
261    }
262}
263fn audit_exact(model: &Json) -> Result<Json, &'static str> {
264    model.keys(
265        &[
266            &["schema", "blocks", "topology", "design_faces"][..],
267            &PARAMETERS,
268        ]
269        .concat(),
270    )?;
271    let bounds = parse_bounds(model.field("bounds_mm")?)?;
272    let profiles = parse_profiles(model.field("profiles")?)?;
273    let expected = build_parts(bounds, &profiles)?;
274    let actual = decode_blocks(model)?;
275    if actual != expected {
276        return Err("nurbs_cross_block_reconstruction_mismatch");
277    }
278    let cert = certificate(&actual)?;
279    if model.field("topology")? != &topology(&cert.incidence)
280        || model.field("design_faces")? != &design(&cert)
281    {
282        return Err("nurbs_cross_incidence_or_provenance_mismatch");
283    }
284    Ok(object([
285        ("status", text("valid")),
286        ("method", text(METHOD)),
287        ("fidelity", text("EXACT_REPRESENTATION")),
288        ("blocks", number(actual.len())),
289        ("shared_faces", number(cert.incidence.interfaces.len())),
290        ("exterior_sides", number(cert.incidence.exterior.len())),
291        ("weights_status", text("positive_on_closed_domain")),
292        (
293            "ordinates_status",
294            text("strictly_increasing_on_closed_domain"),
295        ),
296        (
297            "lateral_order_status",
298            text("strict_order_certified_on_closed_domain"),
299        ),
300        (
301            "structural_reconstruction",
302            text("exact_integer_coefficients_and_whole_face_incidence"),
303        ),
304        (
305            "continuous_geometry",
306            text("exact_rational_bernstein_controls_checked_for_order_and_positive_weights"),
307        ),
308        (
309            "certificate_scope",
310            text("recipe_reconstruction_and_exact_whole_face_incidence_only"),
311        ),
312    ]))
313}
314fn audit_bounded(model: &Json) -> Result<Json, &'static str> {
315    model.keys(
316        &[
317            &["schema", "blocks", "topology", "design_faces"][..],
318            &PARAMETERS,
319            &["lateral_tolerance_mm", "max_depth"][..],
320        ]
321        .concat(),
322    )?;
323    let bounds = parse_bounds(model.field("bounds_mm")?)?;
324    let profiles = parse_profiles(model.field("profiles")?)?;
325    let tolerance = parse_tolerance(model.field("lateral_tolerance_mm")?)?;
326    let max_depth = model.field("max_depth")?.number()?;
327    let expected = build_bounded_parts(bounds, &profiles, tolerance, max_depth)?;
328    let actual = decode_blocks(model)?;
329    if actual != expected {
330        return Err("nurbs_cross_bounded_reconstruction_mismatch");
331    }
332    let cert = certificate(&actual)?;
333    if model.field("topology")? != &topology(&cert.incidence)
334        || model.field("design_faces")? != &design(&cert)
335    {
336        return Err("nurbs_cross_incidence_or_provenance_mismatch");
337    }
338    let upper = tolerance.checked_mul(Ratio::new(2, 1)?)?;
339    Ok(object([
340        ("status", text("valid")),
341        ("method", text(BOUNDED_METHOD)),
342        ("blocks", number(actual.len())),
343        ("shared_faces", number(cert.incidence.interfaces.len())),
344        ("exterior_sides", number(cert.incidence.exterior.len())),
345        ("fidelity", text("APPROX_CERTIFIED")),
346        ("weights_status", text("positive_on_closed_domain")),
347        (
348            "ordinates_status",
349            text("strictly_increasing_on_closed_domain"),
350        ),
351        (
352            "lateral_order_status",
353            text("strict_order_certified_for_constructed_rational_envelope"),
354        ),
355        (
356            "hausdorff_error_upper_bound_mm",
357            array([upper.numerator(), upper.denominator()], number),
358        ),
359        (
360            "structural_reconstruction",
361            text("exact_integer_coefficients_and_whole_face_incidence"),
362        ),
363        (
364            "certificate_scope",
365            text("bounded_lateral_profile_fidelity_and_exact_whole_face_incidence"),
366        ),
367        ("max_depth", number(max_depth)),
368    ]))
369}
