1//! Exact recovery of one tensor Bézier parent from two adjacent restrictions.
2//!
3//! This is a pairwise algebraic test. It proves only that the two supplied
4//! full-face restrictions are exact restrictions of one parent net; it does
5//! not assign a historical feature identity or prove global material
6//! coverage, regularity, or disjoint interiors.

36fn parse_orientation(value: &Json) -> Result<Orientation> {
37    value.keys(&["swap", "flip_first", "flip_second"])?;
38    let get = |key| match value.field(key)? {
39        Json::Bool(v) => Ok(*v),
40        _ => Err("pair parent orientation fields must be booleans"),
41    };
42    Ok(Orientation {
43        swap: get("swap")?,
44        flip_first: get("flip_first")?,
45        flip_second: get("flip_second")?,
46    })
47}

68/// Audit all declared whole-face interfaces of an already audited model.
69/// Recoverable pairs carry exact parent nets; every other pair remains in the
70/// report as a refusal or an indeterminate arithmetic/ratio result.
71pub fn recover_model_interfaces(model: &Json) -> Result<Json> {
72    let mut blocks = BTreeMap::new();
73    for block in crate::bracket_json::decode_blocks(model)? {
74        validate(&block)?;
75        if blocks.insert(block.id.clone(), block).is_some() {
76            return Err("pair parent model has duplicate block identifiers");
77        }
78    }
79    let interfaces = model.field("topology")?.field("interfaces")?.items()?;
80    let mut rows = Vec::with_capacity(interfaces.len());
81    let mut recovered = 0usize;
82    let mut refused = 0usize;
83    let mut indeterminate = 0usize;
84    let mut axis_counts = [[0usize; 4]; 3];
85    for (interface_index, entry) in interfaces.iter().enumerate() {
86        let first_side = parse_side(entry.field("first")?)?;
87        let second_side = parse_side(entry.field("second")?)?;
88        let orientation = parse_orientation(entry.field("orientation")?)?;
89        let first = blocks
90            .get(&first_side.block)
91            .ok_or("pair parent interface references an unknown first block")?;
92        let second = blocks
93            .get(&second_side.block)
94            .ok_or("pair parent interface references an unknown second block")?;
95        let attempt = recover_pair(
96            first,
97            first_side.clone(),
98            second,
99            second_side.clone(),
100            orientation,
101        );
102        let (status, reason, recovery) = match attempt {
103            Ok(parent) => {
104                recovered += 1;
105                axis_counts[first_side.axis][0] += 1;
106                ("recovered", Json::Null, parent.document())
107            }
108            Err(reason)
109                if reason.contains("indeterminate")
110                    || reason.contains("overflow")
111                    || reason.contains("exceeds i128")
112                    || reason.contains("budget") =>
113            {
114                indeterminate += 1;
115                axis_counts[first_side.axis][2] += 1;
116                ("indeterminate", string(reason), Json::Null)
117            }
118            Err(reason) => {
119                refused += 1;
120                axis_counts[first_side.axis][1] += 1;
121                ("refused", string(reason), Json::Null)
122            }
123        };
124        rows.push(object([
125            ("interface_index", number(interface_index)),
126            ("first", side_json(&first_side)),
127            ("second", side_json(&second_side)),
128            ("orientation", orientation_json(orientation)),
129            ("status", string(status)),
130            ("reason", reason),
131            ("recovery", recovery),
132        ]));
133    }
134    let model_schema = model
135        .field("schema")
136        .and_then(Json::text)
137        .ok()
138        .map(string)
139        .unwrap_or(Json::Null);
140    let axis_summary = ["u", "v", "w"]
141        .iter()
142        .enumerate()
143        .map(|(axis, name)| {
144            object([
145                ("axis", string(*name)),
146                (
147                    "interface_count",
148                    number(axis_counts[axis].iter().sum::<usize>()),
149                ),
150                ("recovered_count", number(axis_counts[axis][0])),
151                ("refused_count", number(axis_counts[axis][1])),
152                ("indeterminate_count", number(axis_counts[axis][2])),
153            ])
154        })
155        .collect();
156    Ok(object([
157        ("schema", string("pair_bezier_parent_report_v1")),
158        ("model_schema", model_schema),
159        ("claim_scope", string("pairwise exact mathematical support only; no historical feature provenance or global geometric certification")),
160        ("interface_count", number(interfaces.len())),
161        ("recovered_count", number(recovered)),
162        ("refused_count", number(refused)),
163        ("indeterminate_count", number(indeterminate)),
164        ("counts_by_first_interface_axis", Json::Array(axis_summary)),
165        ("interfaces", Json::Array(rows)),
166    ]))
167}

169fn index(degrees: [usize; 3], p: [usize; 3]) -> usize {
170    (p[0] * (degrees[1] + 1) + p[1]) * (degrees[2] + 1) + p[2]
171}

173fn checked_scale(point: HPoint, scale: Ratio) -> Result<HPoint> {
174    let mut out = [Ratio::zero(); 4];
175    for c in 0..4 {
176        out[c] = point[c].checked_mul(scale)?;
177    }
178    Ok(out)
179}

181fn checked_add_scaled(a: HPoint, sa: Ratio, b: HPoint, sb: Ratio) -> Result<HPoint> {
182    let mut out = [Ratio::zero(); 4];
183    for c in 0..4 {
184        out[c] = a[c].checked_mul(sa)?.checked_add(b[c].checked_mul(sb)?)?;
185    }
186    Ok(out)
187}

189fn validate(block: &BracketBlock) -> Result<()> {
190    let expected = block
191        .degrees
192        .iter()
193        .try_fold(1usize, |n, d| n.checked_mul(d.checked_add(1)?))
194        .ok_or("pair parent control count overflow")?;
195    if block.id.is_empty()
196        || block.degrees.iter().any(|d| !(1..=3).contains(d))
197        || block.controls.len() != expected
198    {
199        return Err("invalid pair parent input block");
200    }
201    if block.controls.iter().any(|p| p[3] <= 0) {
202        return Err("pair parent requires positive homogeneous weights");
203    }
204    Ok(())
205}

207fn elevate_axis(mut net: Net, degrees: &mut [usize; 3], axis: usize, target: usize) -> Result<Net> {
208    if axis >= 3 || target > 3 || target < degrees[axis] {
209        return Err("unsupported pair parent degree elevation");
210    }
211    while degrees[axis] < target {
212        let old_degrees = *degrees;
213        let mut new_degrees = old_degrees;
214        new_degrees[axis] += 1;
215        let mut new_net = vec![[Ratio::zero(); 4]; (0..3).map(|a| new_degrees[a] + 1).product()];
216        let other: Vec<_> = (0..3).filter(|&a| a != axis).collect();
217        for a in 0..=old_degrees[other[0]] {
218            for b in 0..=old_degrees[other[1]] {
219                for k in 0..=new_degrees[axis] {
220                    let mut at = [0; 3];
221                    at[other[0]] = a;
222                    at[other[1]] = b;
223                    at[axis] = k;
224                    let destination = index(new_degrees, at);
225                    if k == 0 {
226                        at[axis] = 0;
227                        new_net[destination] = net[index(old_degrees, at)];
228                    } else if k == new_degrees[axis] {
229                        at[axis] = old_degrees[axis];
230                        new_net[destination] = net[index(old_degrees, at)];
231                    } else {
232                        let alpha = Ratio::new(k as i128, new_degrees[axis] as i128)?;
233                        let complement = Ratio::one().checked_sub(alpha)?;
234                        at[axis] = k - 1;
235                        let previous = net[index(old_degrees, at)];
236                        at[axis] = k;
237                        let current = net[index(old_degrees, at)];
238                        new_net[destination] =
239                            checked_add_scaled(previous, alpha, current, complement)?;
240                    }
241                }
242            }
243        }
244        *degrees = new_degrees;
245        net = new_net;
246    }
247    Ok(net)
248}

250fn reverse_axis(net: Net, degrees: [usize; 3], axis: usize) -> Net {
251    let mut out = vec![[Ratio::zero(); 4]; net.len()];
252    for i in 0..=degrees[0] {
253        for j in 0..=degrees[1] {
254            for k in 0..=degrees[2] {
255                let output = [i, j, k];
256                let mut source = output;
257                source[axis] = degrees[axis] - output[axis];
258                out[index(degrees, output)] = net[index(degrees, source)];
259            }
260        }
261    }
262    out
263}

273fn orient_second(
274    block: &BracketBlock,
275    first_axis: usize,
276    second_side: &BlockSide,
277    orientation: Orientation,
278) -> Result<(Net, [usize; 3])> {
279    if first_axis >= 3 || second_side.axis >= 3 {
280        return Err("pair parent interface axis outside 0..3");
281    }
282    let mut source_axis = [usize::MAX; 3];
283    source_axis[first_axis] = second_side.axis;
284    let first_free: Vec<_> = (0..3).filter(|&a| a != first_axis).collect();
285    let second_free: Vec<_> = (0..3).filter(|&a| a != second_side.axis).collect();
286    let mut flip = [false; 3];
287    if second_side.upper {
288        flip[first_axis] = true;
289    }
290    for output_tangent in 0..2 {
291        let source_tangent = if orientation.swap {
292            1 - output_tangent
293        } else {
294            output_tangent
295        };
296        source_axis[first_free[output_tangent]] = second_free[source_tangent];
297        flip[first_free[output_tangent]] = if orientation.swap {
298            if output_tangent == 0 {
299                orientation.flip_second
300            } else {
301                orientation.flip_first
302            }
303        } else if output_tangent == 0 {
304            orientation.flip_first
305        } else {
306            orientation.flip_second
307        };
308    }
309    if source_axis.iter().any(|&a| a >= 3) {
310        return Err("invalid pair parent axis permutation");
311    }
312    let degrees = std::array::from_fn(|a| block.degrees[source_axis[a]]);
313    let source = input_net(block);
314    let mut out = vec![[Ratio::zero(); 4]; source.len()];
315    for i in 0..=degrees[0] {
316        for j in 0..=degrees[1] {
317            for k in 0..=degrees[2] {
318                let output = [i, j, k];
319                let mut input = [0; 3];
320                for axis in 0..3 {
321                    let source_id = source_axis[axis];
322                    input[source_id] = if flip[axis] {
323                        block.degrees[source_id] - output[axis]
324                    } else {
325                        output[axis]
326                    };
327                }
328                out[index(degrees, output)] = source[index(block.degrees, input)];
329            }
330        }
331    }
332    Ok((out, degrees))
333}

335fn face_points(net: &Net, degrees: [usize; 3], axis: usize, upper: bool) -> Vec<HPoint> {
336    let free: Vec<_> = (0..3).filter(|&a| a != axis).collect();
337    let fixed = if upper { degrees[axis] } else { 0 };
338    let mut result = Vec::new();
339    for i in 0..=degrees[free[0]] {
340        for j in 0..=degrees[free[1]] {
341            let mut p = [0; 3];
342            p[axis] = fixed;
343            p[free[0]] = i;
344            p[free[1]] = j;
345            result.push(net[index(degrees, p)]);
346        }
347    }
348    result
349}

351fn common_gauge(a: &[HPoint], b: &[HPoint]) -> Result<Ratio> {
352    if a.len() != b.len() || a.is_empty() {
353        return Err("pair parent interface trace dimensions differ");
354    }
355    let gauge = a[0][3].checked_div(b[0][3])?;
356    if gauge <= Ratio::zero() {
357        return Err("pair parent interface gauge must be positive");
358    }
359    for (left, right) in a.iter().zip(b) {
360        if *left != checked_scale(*right, gauge)? {
361            return Err("pair parent traces differ beyond a constant homogeneous gauge");
362        }
363    }
364    Ok(gauge)
365}

367fn infer_split(a: &Net, b: &Net, degrees: [usize; 3], axis: usize) -> Result<Ratio> {
368    let p = degrees[axis];
369    let free: Vec<_> = (0..3).filter(|&x| x != axis).collect();
370    let mut t: Option<Ratio> = None;
371    for i in 0..=degrees[free[0]] {
372        for j in 0..=degrees[free[1]] {
373            let mut seam = [0; 3];
374            seam[free[0]] = i;
375            seam[free[1]] = j;
376            let mut left_inner = seam;
377            left_inner[axis] = p - 1;
378            seam[axis] = p;
379            let left_delta = std::array::from_fn::<_, 4, _>(|c| {
380                a[index(degrees, seam)][c].checked_sub(a[index(degrees, left_inner)][c])
381            });
382            let left_delta = left_delta.into_iter().collect::<Result<Vec<_>>>()?;
383            let mut right_inner = seam;
384            right_inner[axis] = 1;
385            seam[axis] = 0;
386            let right_delta = std::array::from_fn::<_, 4, _>(|c| {
387                b[index(degrees, right_inner)][c].checked_sub(b[index(degrees, seam)][c])
388            });
389            let right_delta = right_delta.into_iter().collect::<Result<Vec<_>>>()?;
390            for c in 0..4 {
391                let sum = left_delta[c].checked_add(right_delta[c])?;
392                if sum != Ratio::zero() {
393                    let candidate = left_delta[c].checked_div(sum)?;
394                    if t.is_some_and(|known| known != candidate) {
395                        return Err("pair parent first jets imply inconsistent interval ratios");
396                    }
397                    t = Some(candidate);
398                } else if left_delta[c] != Ratio::zero() || right_delta[c] != Ratio::zero() {
399                    return Err("pair parent first jets imply no positive interval ratio");
400                }
401            }
402        }
403    }
404    let t = t.ok_or("pair parent split ratio is indeterminate from zero first jets")?;
405    if t.cmp_value(&Ratio::zero()).is_le() || t.cmp_value(&Ratio::one()).is_ge() {
406        return Err("pair parent split ratio is outside the open unit interval");
407    }
408    for i in 0..=degrees[free[0]] {
409        for j in 0..=degrees[free[1]] {
410            let mut seam = [0; 3];
411            seam[free[0]] = i;
412            seam[free[1]] = j;
413            let mut left_inner = seam;
414            left_inner[axis] = p - 1;
415            seam[axis] = p;
416            let left = a[index(degrees, seam)];
417            let left_previous = a[index(degrees, left_inner)];
418            let mut right_inner = seam;
419            right_inner[axis] = 1;
420            seam[axis] = 0;
421            let right = b[index(degrees, right_inner)];
422            let right_start = b[index(degrees, seam)];
423            for c in 0..4 {
424                let da = left[c].checked_sub(left_previous[c])?;
425                let db = right[c].checked_sub(right_start[c])?;
426                if da != t.checked_mul(da.checked_add(db)?)? {
427                    return Err("pair parent first jets are not one positive scaled trace");
428                }
429            }
430        }
431    }
432    Ok(t)
433}

435fn binomial(n: usize, k: usize) -> i128 {
436    if k > n {
437        return 0;
438    }
439    let k = k.min(n - k);
440    (1..=k).fold(1_i128, |v, i| v * (n - k + i) as i128 / i as i128)
441}

443fn power(base: Ratio, exponent: usize) -> Result<Ratio> {
444    let mut result = Ratio::one();
445    for _ in 0..exponent {
446        result = result.checked_mul(base)?;
447    }
448    Ok(result)

451fn recover_parent(first: &Net, degrees: [usize; 3], axis: usize, t: Ratio) -> Result<Net> {
452    let mut parent = vec![[Ratio::zero(); 4]; first.len()];
453    let p = degrees[axis];
454    let complement = Ratio::one().checked_sub(t)?;
455    let free: Vec<_> = (0..3).filter(|&x| x != axis).collect();
456    for i in 0..=degrees[free[0]] {
457        for j in 0..=degrees[free[1]] {
458            let mut at = [0; 3];
459            at[free[0]] = i;
460            at[free[1]] = j;
461            let mut line = Vec::with_capacity(p + 1);
462            for q in 0..=p {
463                at[axis] = q;
464                line.push(first[index(degrees, at)]);
465            }
466            let mut recovered = vec![[Ratio::zero(); 4]; p + 1];
467            recovered[0] = line[0];
468            for k in 1..=p {
469                let mut known = [Ratio::zero(); 4];
470                for (j, previous) in recovered.iter().take(k).enumerate() {
471                    let coefficient = Ratio::new(binomial(k, j), 1)?
472                        .checked_mul(power(t, j)?)?
473                        .checked_mul(power(complement, k - j)?)?;
474                    for c in 0..4 {
475                        known[c] = known[c].checked_add(coefficient.checked_mul(previous[c])?)?;
476                    }
477                }
478                let denominator = power(t, k)?;
479                for c in 0..4 {
480                    recovered[k][c] = line[k][c].checked_sub(known[c])?.checked_div(denominator)?;
481                }
482            }
483            for (k, control) in recovered.iter().copied().enumerate() {
484                at[axis] = k;
485                parent[index(degrees, at)] = control;
486            }
487        }
488    }
489    Ok(parent)
490}

492fn split_line(line: &[HPoint], t: Ratio) -> Result<(Vec<HPoint>, Vec<HPoint>)> {
493    let p = line.len() - 1;
494    let complement = Ratio::one().checked_sub(t)?;
495    let mut level = line.to_vec();
496    let mut left = vec![line[0]];
497    let mut right = vec![line[p]];
498    while level.len() > 1 {
499        let mut next = Vec::with_capacity(level.len() - 1);
500        for pair in level.windows(2) {
501            next.push(checked_add_scaled(pair[0], complement, pair[1], t)?);
502        }
503        left.push(next[0]);
504        right.push(*next.last().expect("nonempty split level"));
505        level = next;
506    }
507    right.reverse();
508    Ok((left, right))
509}

511fn split_net(parent: &Net, degrees: [usize; 3], axis: usize, t: Ratio) -> Result<(Net, Net)> {
512    let mut left = vec![[Ratio::zero(); 4]; parent.len()];
513    let mut right = left.clone();
514    let free: Vec<_> = (0..3).filter(|&a| a != axis).collect();
515    for i in 0..=degrees[free[0]] {
516        for j in 0..=degrees[free[1]] {
517            let mut at = [0; 3];
518            at[free[0]] = i;
519            at[free[1]] = j;
520            let mut line = Vec::with_capacity(degrees[axis] + 1);
521            for k in 0..=degrees[axis] {
522                at[axis] = k;
523                line.push(parent[index(degrees, at)]);
524            }
525            let (l, r) = split_line(&line, t)?;
526            for k in 0..=degrees[axis] {
527                at[axis] = k;
528                left[index(degrees, at)] = l[k];
529                right[index(degrees, at)] = r[k];
530            }
531        }
532    }
533    Ok((left, right))
534}

572/// Recover one tensor Bézier map from two audited, adjacent full-face blocks.
573///
574/// The first block is oriented so its declared interface is the upper face;
575/// the second is oriented so its interface is the lower face. `orientation`
576/// maps the second tangential trace into the first trace convention. Tangential
577/// and normal degrees are harmonized by exact Bernstein elevation (up to cubic).
578/// The first-jet ratio is inferred from the complete homogeneous face net, then
579/// every parent control line is recovered and both block nets are reconstructed
580/// by exact de Casteljau restriction. Unsupported or inconsistent pairs refuse.
581pub fn recover_pair(
582    first: &BracketBlock,
583    first_side: BlockSide,
584    second: &BracketBlock,
585    second_side: BlockSide,
586    orientation: Orientation,
587) -> Result<PairParentRecovery> {
588    validate(first)?;
589    validate(second)?;
590    if first.id == second.id || first_side.block != first.id || second_side.block != second.id {
591        return Err("pair parent side identities do not match input blocks");
592    }
593    if first_side.axis >= 3 || second_side.axis >= 3 {
594        return Err("pair parent interface axis outside 0..3");
595    }
596    let axis = first_side.axis;
597    let mut a = input_net(first);
598    let mut degree_a = first.degrees;
599    if !first_side.upper {
600        a = reverse_axis(a, degree_a, axis);
601    }
602    let (mut b, mut degree_b) = orient_second(second, axis, &second_side, orientation)?;
603    let mut target = degree_a;
604    for ax in 0..3 {
605        target[ax] = degree_a[ax].max(degree_b[ax]);
606    }
607    for (ax, &target_degree) in target.iter().enumerate() {
608        a = elevate_axis(a, &mut degree_a, ax, target_degree)?;
609        b = elevate_axis(b, &mut degree_b, ax, target_degree)?;
610    }
611    let face_a = face_points(&a, target, axis, true);
612    let face_b = face_points(&b, target, axis, false);
613    let gauge = common_gauge(&face_a, &face_b)?;
614    for point in &mut b {
615        *point = checked_scale(*point, gauge)?;
616    }
617    let t = infer_split(&a, &b, target, axis)?;
618    let parent = recover_parent(&a, target, axis, t)?;
619    if parent.iter().any(|p| p[3] <= Ratio::zero()) {
620        return Err("recovered pair parent has a nonpositive homogeneous weight");
621    }
622    let (reconstructed_a, reconstructed_b) = split_net(&parent, target, axis, t)?;
623    if reconstructed_a != a || reconstructed_b != b {
624        return Err("full restrictions do not reproduce both supplied homogeneous nets");
625    }
626    let ratio = t.checked_div(Ratio::one().checked_sub(t)?)?;
627    Ok(PairParentRecovery {
628        source_ids: [first.id.clone(), second.id.clone()],
629        degrees: target,
630        controls: parent,
631        split_parameter: t,
632        first_to_second_interval_ratio: ratio,
633        second_to_first_gauge: gauge,
634    })
635}
