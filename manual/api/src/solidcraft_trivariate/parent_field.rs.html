17/// Rows index Bernstein coefficients on the requested subinterval. Columns
18/// index functions in the original knot vector. A subinterval may be shorter
19/// than a knot span. Crossing a nonempty knot interval is refused.
20pub(crate) fn axis_rows(knots: &[Q], degree: usize, interval: [Q; 2]) -> Result<Vec<Row>> {
21    if !(1..=3).contains(&degree)
22        || knots.len() < 2 * (degree + 1)
23        || knots
24            .windows(2)
25            .any(|pair| pair[0].cmp_value(&pair[1]).is_gt())
26        || !interval[0].cmp_value(&interval[1]).is_lt()
27    {
28        return Err("invalid knot interval for polynomial extraction");
29    }
30    let n = knots.len() - degree - 1;
31    let span = (degree..n)
32        .find(|&i| {
33            !knots[i].cmp_value(&interval[0]).is_gt()
34                && !interval[1].cmp_value(&knots[i + 1]).is_gt()
35        })
36        .ok_or("extraction interval crosses a knot or leaves the active domain")?;
37    let mut previous = BTreeMap::from([(span, vec![Q::one()])]);
38    for d in 1..=degree {
39        let mut next = BTreeMap::new();
40        for i in span - d..=span {
41            let mut coeffs = vec![Q::zero(); d + 1];
42            for (j, left) in [(i, true), (i + 1, false)] {
43                let Some(source) = previous.get(&j) else {
44                    continue;
45                };
46                let width = knots[j + d].checked_sub(knots[j])?;
47                if width == Q::zero() {
48                    continue;
49                }
50                let affine = interval.map(|t| {
51                    if left {
52                        t.checked_sub(knots[j])
53                    } else {
54                        knots[j + d].checked_sub(t)
55                    }
56                });
57                let a = affine[0]?.checked_div(width)?;
58                let b = affine[1]?.checked_div(width)?;
59                for (k, &value) in source.iter().enumerate() {
60                    let low = value
61                        .checked_mul(a)?
62                        .checked_mul(Q::new((d - k) as i128, d as i128)?)?;
63                    let high = value
64                        .checked_mul(b)?
65                        .checked_mul(Q::new((k + 1) as i128, d as i128)?)?;
66                    coeffs[k] = coeffs[k].checked_add(low)?;
67                    coeffs[k + 1] = coeffs[k + 1].checked_add(high)?;
68                }
69            }
70            next.insert(i, coeffs);
71        }
72        previous = next;
73    }
74    let mut rows = vec![Row::new(); degree + 1];
75    for (i, values) in previous {
76        for (k, v) in values.into_iter().enumerate() {
77            add(&mut rows[k], i, v)?;
78        }
79    }
80    for row in &rows {
81        let sum = row.values().try_fold(Q::zero(), |s, &v| s.checked_add(v))?;
82        if sum != Q::one() || row.values().any(|v| v.numerator() < 0) {
83            return Err("invalid polynomial partition of unity");
84        }
85    }
86    Ok(rows)
87}
88fn numeric_axis(rows: &[Row]) -> (Vec<usize>, AxisExtraction) {
89    let active: Vec<usize> = rows
90        .iter()
91        .flat_map(|r| r.keys().copied())
92        .collect::<BTreeSet<_>>()
93        .into_iter()
94        .collect();
95    let mut columns = Vec::new();
96    let mut coefficients = Vec::new();
97    let mut row_offsets = vec![0];
98    for r in rows {
99        for (&i, &v) in r {
100            columns.push(active.binary_search(&i).unwrap());
101            coefficients.push(v);
102        }
103        row_offsets.push(columns.len());
104    }
105    let axis = AxisExtraction {
106        degree: rows.len() - 1,
107        basis_count: active.len(),
108        columns,
109        coefficients,
110        row_offsets,
111    };
112    (active, axis)
113}
114fn row_json(row: &Row) -> Json {
115    Json::Array(
116        row.iter()
117            .map(|(i, c)| {
118                Json::Array(vec![
119                    Json::Number(i.to_string()),
120                    Json::Array(vec![
121                        Json::String(c.numerator().to_string()),
122                        Json::String(c.denominator().to_string()),
123                    ]),
124                ])
125            })
126            .collect(),
127    )
128}

130pub struct ParentField {
131    pub elements: Vec<ExtractedElement>,
132    pub ndof: usize,
133    pub extraction: Json,
134    cell_rows: BTreeMap<String, (usize, Vec<Row>)>,
135    boundary: Vec<(usize, usize, bool)>,
136    clamp: Lift,
137    z_planes: Option<[f64; 2]>,
138}
139impl ParentField {
140    pub fn solve(&self, job: &Json) -> Result<Json> {
141        let domain = IntegrationDomain {
142            scalar_dofs: self.ndof,
143            cell_count: self.elements.len(),
144            z_bounds: self
145                .z_planes
146                .ok_or("unsupported clamp: no physical z end planes")?,
147            boundary: &self.boundary,
148            essential: &self.clamp,
149        };
150        solve_sampled(&domain, job, |i, p| self.elements[i].sample(p))
151    }
152}
153pub fn build(model: &Json) -> Result<ParentField> {
154    let support = recover_complete_parent(model)?;
155    let ndof = support.homogeneous_controls.len();
156    let [_, nv, nw] = support.control_shape;
157    let blocks = decode_blocks(model)?;
158    let faces = enumerate_faces(&blocks)?;
159    let blocks: BTreeMap<_, _> = blocks.iter().map(|b| (b.id.as_str(), b)).collect();
160    let mut elements = Vec::new();
161    let mut records = Vec::new();
162    let mut cell_rows = BTreeMap::new();
163    let mut indices = BTreeMap::new();
164    for span in &support.source_spans {
165        let r = &span.restriction;
166        if r.axes != [0, 1, 2] {
167            return Err("unsupported parent field span orientation");
168        }
169        let mut factors = Vec::new();
170        for a in 0..3 {
171            let knot = support.knots[a]
172                .iter()
173                .map(|&x| Q::new(i128::from(x), 1))
174                .collect::<Result<Vec<_>>>()?;
175            let [lo, hi] = r.intervals[a];
176            let mut restriction = axis_rows(
177                &knot,
178                support.degrees[a],
179                [Q::new(i128::from(lo), 1)?, Q::new(i128::from(hi), 1)?],
180            )?;
181            if r.reversals[a] {
182                restriction.reverse();
183            }
184            factors.push(restriction);
185        }
186        let axes = [&factors[0], &factors[1], &factors[2]];
187        let [(u, au), (v, av), (w, aw)] = axes.map(|rs| numeric_axis(rs));
188        let mut dofs = Vec::new();
189        let mut controls = Vec::new();
190        for &i in &u {
191            for &j in &v {
192                for &k in &w {
193                    let id = (i * nv + j) * nw + k;
194                    dofs.push(id);
195                    controls.push(support.homogeneous_controls[id].map(|x| x as f64));
196                }
197            }
198        }
199        let e = ExtractedElement {
200            axes: [au, av, aw],
201            dofs,
202            field_weights: controls.iter().map(|c| c[3]).collect(),
203            geometry_controls: controls,
204        };
205        e.validate()?;
206        let block = blocks[span.source_block_id.as_str()];
207        let gauge = Q::new(span.gauge[0], span.gauge[1])?;
208        let mut polynomial = Vec::new();
209        let mut rational = Vec::new();
210        let mut physical_rows = Vec::new();
211        for a in axes[0] {
212            for b in axes[1] {
213                for c in axes[2] {
214                    let mut row = Row::new();
215                    let mut control = [Q::zero(); 4];
216                    for (&i, &lu) in a {
217                        for (&j, &lv) in b {
218                            for (&k, &lw) in c {
219                                let id = (i * nv + j) * nw + k;
220                                let basis = lu.checked_mul(lv)?.checked_mul(lw)?;
221                                row.insert(id, basis);
222                                for (dst, h) in
223                                    control.iter_mut().zip(support.homogeneous_controls[id])
224                                {
225                                    *dst = dst.checked_add(basis.checked_mul(Q::new(h, 1)?)?)?;
226                                }
227                            }
228                        }
229                    }
230                    let expected = block.controls[polynomial.len()];
231                    for (&h, x) in control.iter().zip(expected) {
232                        if h.checked_mul(gauge)? != Q::new(i128::from(x), 1)? {
233                            return Err("canonical field extraction disagrees with the source net");
234                        }
235                    }
236                    let physical = row
237                        .iter()
238                        .map(|(&id, &v)| {
239                            Ok((
240                                id,
241                                v.checked_mul(Q::new(support.homogeneous_controls[id][3], 1)?)?
242                                    .checked_div(control[3])?,
243                            ))
244                        })
245                        .collect::<Result<Row>>()?;
246                    polynomial.push(row_json(&row));
247                    rational.push(row_json(&physical));
248                    physical_rows.push(physical);
249                }
250            }
251        }
252        indices.insert(span.source_block_id.as_str(), elements.len());
253        records.push(object([
254            ("source_block", Json::String(span.source_block_id.clone())),
255            ("E", Json::Array(polynomial)),
256            ("C", Json::Array(rational)),
257        ]));
258        cell_rows.insert(
259            span.source_block_id.clone(),
260            (elements.len(), physical_rows),
261        );
262        elements.push(e);
263    }
264    let free_roots: Vec<_> = (0..ndof).filter(|i| i % nw != 0).collect();
265    let lines = (0..ndof)
266        .map(|i| LiftLine {
267            dof: i,
268            terms: free_roots
269                .binary_search(&i)
270                .map(|j| vec![(j, Q::one())])
271                .unwrap_or_default(),
272            offset: Q::zero(),
273        })
274        .collect();
275    let boundary = faces
276        .exterior
277        .iter()
278        .map(|s| (indices[s.block.as_str()], s.axis, s.upper))
279        .collect();
280    let h = &support.homogeneous_controls;
281    let planes = [0, nw - 1]
282        .map(|k| Q::new(h[k][2], h[k][3]))
283        .into_iter()
284        .collect::<Result<Vec<_>>>()?;
285    let regular = h.iter().enumerate().all(|(i, p)| {
286        i % nw != 0 && i % nw != nw - 1
287            || Q::new(p[2], p[3]) == Ok(planes[usize::from(i % nw != 0)])
288    });
289    let z_planes = (regular && planes[0].cmp_value(&planes[1]).is_lt()).then(|| {
290        [
291            crate::inherited_space::val(planes[0]),
292            crate::inherited_space::val(planes[1]),
293        ]
294    });
295    let extraction = object([
296        ("schema", Json::String("canonical_parent_field_v1".into())),
297        ("support", Json::String(support.id)),
298        ("scalar_dofs", Json::Number(ndof.to_string())),
299        ("elements", Json::Array(records)),
300    ]);
301    Ok(ParentField {
302        elements,
303        ndof,
304        extraction,
305        cell_rows,
306        boundary,
307        clamp: Lift { free_roots, lines },
308        z_planes,
309    })
310}

312/// Compare the canonical NURBS space to its exact embedding in the full-face
313/// C0 coefficients. Both integrals use identical source geometry evaluations.
314pub fn check_galerkin(model: &Json, order: usize) -> Result<Json> {
315    use crate::extracted_space::{assemble_sampled, compare_assembled};
316    let field = build(model)?;
317    let c0 = crate::inherited_space::build_axial_space(model, 1, 0)?;
318    let mut lift = vec![Row::new(); c0.ndof];
319    let mut element_order = Vec::new();
320    for cell in &c0.cells {
321        let (element, rows) = field
322            .cell_rows
323            .get(&cell.id)
324            .ok_or("source field lineage differs")?;
325        element_order.push(*element);
326        if rows.len() != cell.extraction.len() {
327            return Err("incompatible C0 degree");
328        }
329        for (c, r) in cell.extraction.iter().zip(rows) {
330            if c.len() != 1 {
331                return Err("source C0 row is not nodal");
332            }
333            let (&i, &v) = c.first_key_value().unwrap();
334            if v != Q::one() || (!lift[i].is_empty() && lift[i] != *r) {
335                return Err("parent transfer changed on shared coefficient");
336            }
337            lift[i] = r.clone();
338        }
339    }
340    let (_, kernel) = crate::spline_constraints::nullspace(field.ndof, &lift)?;
341    if !kernel.is_empty() {
342        return Err("parent coefficient transfer lacks full column rank");
343    }
344    let body = |x: [f64; 3]| [1. + x[0], x[2] - x[1], 2. * x[1] + x[0]];
345    let large = assemble_sampled(
346        c0.ndof,
347        c0.cells.len(),
348        1.5,
349        1.,
350        order,
351        |i, p| c0.cells[i].sample(p),
352        body,
353    )?;
354    let small = assemble_sampled(
355        field.ndof,
356        c0.cells.len(),
357        1.5,
358        1.,
359        order,
360        |i, p| {
361            let (common, _) = c0.cells[i].sample(p)?;
362            let (canonical, basis) = field.elements[element_order[i]].sample(p)?;
363            if common
364                .position
365                .iter()
366                .zip(canonical.position)
367                .any(|(a, b)| (a - b).abs() > 1e-10 * (1. + a.abs()))
368                || common
369                    .jacobian_columns
370                    .iter()
371                    .flatten()
372                    .zip(canonical.jacobian_columns.iter().flatten())
373                    .any(|(a, b)| (a - b).abs() > 1e-10 * (1. + a.abs()))
374            {
375                return Err("parent and source quadrature geometry differ");
376            }
377            Ok((common, basis))
378        },
379        body,
380    )?;
381    let result = compare_assembled(&large, &small, c0.ndof, field.ndof, &lift)?;
382    if result.stiffness_relative_frobenius_defect > 1e-10 || result.force_relative_l2_defect > 1e-10
383    {
384        return Err("parent Galerkin identity exceeds tolerance");
385    }
386    Ok(object([
387        ("status", Json::String("valid".into())),
388        ("c0_scalar_dofs", Json::Number(c0.ndof.to_string())),
389        (
390            "canonical_scalar_dofs",
391            Json::Number(field.ndof.to_string()),
392        ),
393        ("exact_transfer_rank", Json::Number(field.ndof.to_string())),
394        (
395            "relative_stiffness_defect",
396            Json::Number(result.stiffness_relative_frobenius_defect.to_string()),
397        ),
398        (
399            "relative_load_defect",
400            Json::Number(result.force_relative_l2_defect.to_string()),
401        ),
402    ]))
403}

405/// Small-system oracle interchange. The native assembly remains block CSR.
406pub fn stiffness_package(model: &Json, order: usize) -> Result<Json> {
407    use crate::extracted_space::assemble_elements;
408    let parent = recover_complete_parent(model)?;
409    let field = build(model)?;
410    if field.ndof > 256 {
411        return Err("indeterminate: small-system spectrum audit budget");
412    }
413    if field.z_planes.is_none() {
414        return Err("unsupported spectrum clamp orientation");
415    }
416    let material = assemble_elements(&field.elements, field.ndof, 2.6, 0.3, order)?;
417    let m = material.stiffness;
418    let n = |v: f64| Json::Number(v.to_string());
419    let array =
420        |v: Vec<usize>| Json::Array(v.into_iter().map(|i| Json::Number(i.to_string())).collect());
421    let values = m
422        .values
423        .into_iter()
424        .map(|block| {
425            Json::Array(
426                block
427                    .into_iter()
428                    .map(|r| Json::Array(r.map(n).to_vec()))
429                    .collect(),
430            )
431        })
432        .collect();
433    let points = parent
434        .homogeneous_controls
435        .into_iter()
436        .map(|h| Json::Array((0..3).map(|j| n(h[j] as f64 / h[3] as f64)).collect()))
437        .collect();
438    Ok(object([
439        ("schema", Json::String("canonical_elasticity_csr_v1".into())),
440        ("offsets", array(m.offsets)),
441        ("columns", array(m.columns)),
442        ("values", Json::Array(values)),
443        ("control_positions_mm", Json::Array(points)),
444        ("free_scalar_roots", array(field.clamp.free_roots)),
445        ("volume_mm3", n(m.integrated_volume)),
446        ("young_mpa", n(2.6)),
447        ("poisson", n(0.3)),
448    ]))
449}

451#[cfg(test)]
452mod tests {
453    use super::*;
454    fn q(n: i128, d: i128) -> Q {
455        Q::new(n, d).unwrap()

457    #[test]
458    fn subinterval_restriction_uses_actual_source_knots() {
459        let knot = [q(-2, 1), q(-2, 1), q(2, 1), q(5, 1), q(5, 1)];
460        let map = axis_rows(&knot, 1, [q(-1, 1), q(0, 1)]).unwrap();
461        assert_eq!(
462            map,
463            vec![
464                Row::from([(0, q(3, 4)), (1, q(1, 4))]),
465                Row::from([(0, q(1, 2)), (1, q(1, 2))])
466            ]
467        );
468        assert!(axis_rows(&knot, 1, [q(-1, 1), q(3, 1)]).is_err());
469        assert!(axis_rows(&knot, 1, [q(0, 1), q(0, 1)]).is_err());
470    }
471    #[test]
472    fn canonical_parent_stiffness_and_load_pull_back_from_c0() {
473        let recipe =
