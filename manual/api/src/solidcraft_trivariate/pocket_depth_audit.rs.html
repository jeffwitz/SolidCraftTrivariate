1//! Independent G2-transfer audit for variable-depth square pockets.
2//!
3//! Every accepted layer is mapped exactly back to the frozen G2 section
4//! templates. Rational axial maps are evaluated in i128 and a single positive
5//! homogeneous scale is removed per block before conversion back to i64.

19    Blind,
20    Through,

23/// Certify one pocket model at an integer half-millimetre depth.
24///
25/// `depth_ticks` is in `0..=18`. The returned certificate covers local
26/// regularity, exact interfaces, exterior faces, and the G2-qualified domain
27/// partition for the selected solid, blind-pocket, or through-bore regime.
28pub fn audit_depth(
29    depth_ticks: i32,
30    blocks: &[BracketBlock],
31) -> Result<BracketCertificate, &'static str> {
32    if !(0..=18).contains(&depth_ticks) {
33        return Err("pocket_depth_out_of_range");
34    }
35    let regime = if depth_ticks == 0 {
36        Regime::Solid
37    } else if depth_ticks < 16 {
38        Regime::Blind
39    } else {
40        Regime::Through
41    };
42    validate_ids(blocks, regime)?;
43    let (canonical, axial_scale) = canonical_g2(blocks, depth_ticks, regime)?;
44    let g2 = if regime == Regime::Through {
45        audit_section("pierced_square_v1", &canonical)?
46    } else {
47        audit_section("layered_pocket_v1", &canonical)?
48    };
49
50    let incidence = enumerate_faces(blocks)?;
51    let (expected_interfaces, expected_exterior) = match regime {
52        Regime::Solid => (16, 22),
53        Regime::Blind => (24, 30),
54        Regime::Through => (4, 16),
55    };
56    if incidence.interfaces.len() != expected_interfaces
57        || incidence.exterior.len() != expected_exterior
58    {
59        return Err("pocket_incidence_mismatch");
60    }
61    let design_faces = classify_faces(blocks, &incidence.exterior, regime)?;
62    let determinant_lower_bound_mm3 = g2.determinant_lower_bound_mm3 * axial_scale;
63    if !determinant_lower_bound_mm3.is_finite() || determinant_lower_bound_mm3 <= 0. {
64        return Err("pocket_determinant_certificate");
65    }
66    Ok(BracketCertificate {
67        incidence,
68        determinant_lower_bound_mm3,
69        radial_error_upper_mm: g2.radial_error_upper_mm,
70        design_faces,
71    })
72}

74fn validate_ids(blocks: &[BracketBlock], regime: Regime) -> Result<(), &'static str> {
75    let expected: BTreeSet<String> = match regime {
76        Regime::Solid => {
77            let mut ids = BTreeSet::from(["pocket_core_lower".to_string()]);
78            for sector in SECTORS {
79                ids.insert(format!("pocket_disk_sector_lower_{sector}"));
80                ids.insert(format!("pocket_outer_sector_lower_{sector}"));
81            }
82            ids
83        }
84        Regime::Blind => {
85            let mut ids = BTreeSet::from(["pocket_core_lower".to_string()]);
86            for sector in SECTORS {
87                ids.insert(format!("pocket_disk_sector_lower_{sector}"));
88                ids.insert(format!("pocket_outer_sector_lower_{sector}"));
89                ids.insert(format!("pocket_outer_sector_upper_{sector}"));
90            }
91            ids
92        }
93        Regime::Through => SECTORS
94            .iter()
95            .map(|sector| format!("pocket_outer_sector_upper_{sector}"))
96            .collect(),
97    };
98    if blocks.len() != expected.len() {
99        return Err("pocket_block_inventory");
100    }
101    let mut actual = BTreeSet::new();
102    for block in blocks {
103        if !actual.insert(block.id.clone()) {
104            return Err("duplicate_pocket_block_id");
105        }
106        if !expected.contains(&block.id) {
107            return Err("unexpected_pocket_block_id");
108        }
109        let required_degree = if block.id == "pocket_core_lower" {
110            [1, 1, 1]
111        } else {
112            [2, 1, 1]
113        };
114        if block.degrees != required_degree
115            || block.controls.len() != required_degree.iter().map(|d| d + 1).product::<usize>()
116        {
117            return Err("pocket_block_dimensions");
118        }
119        if block.controls.iter().any(|p| p[3] <= 0) {
120            return Err("nonpositive_pocket_weight");
121        }
122    }
123    if actual != expected {
124        return Err("pocket_block_inventory");
125    }
126    Ok(())
127}

129fn canonical_g2(
130    blocks: &[BracketBlock],
131    depth_ticks: i32,
132    regime: Regime,
133) -> Result<(Vec<BracketBlock>, f64), &'static str> {
134    match regime {
135        Regime::Solid => {
136            let mut lower = Vec::with_capacity(9);
137            for block in blocks {
138                let id = lower_id(&block.id)?;
139                lower.push(transform_lower(block, id, 16)?);
140            }
141            // Build the synthetic upper half from the same exact four square
142            // sectors. Its addition lets the independent G2 layered-pocket
143            // oracle qualify the full square cross-section at every height.
144            let mut canonical = lower.clone();
145            for sector in SECTORS {
146                let source_id = format!("pocket_outer_sector_{sector}_z0_4");
147                let source = lower
148                    .iter()
149                    .find(|block| block.id == source_id)
150                    .ok_or("missing_solid_outer_sector")?;
151                canonical.push(shift_z(
152                    source,
153                    format!("pocket_outer_sector_{sector}_z4_8"),
154                    4,
155                )?);
156            }
157            Ok((canonical, 2.0))
158        }
159        Regime::Blind => {
160            let floor_ticks = 16 - depth_ticks;
161            let mut canonical = Vec::with_capacity(13);
162            for block in blocks {
163                if block.id == "pocket_core_lower" || block.id.contains("_lower_") {
164                    canonical.push(transform_lower(block, lower_id(&block.id)?, floor_ticks)?);
165                } else if block.id.contains("_upper_") {
166                    canonical.push(transform_upper(
167                        block,
168                        upper_id(&block.id)?,
169                        floor_ticks,
170                        depth_ticks,
171                    )?);
172                } else {
173                    return Err("unknown_blind_pocket_layer");
174                }
175            }
176            let scale = f64::from(floor_ticks.min(depth_ticks)) / 8.0;
177            Ok((canonical, scale))
178        }
179        Regime::Through => {
180            let mut canonical = Vec::with_capacity(4);
181            for block in blocks {
182                let sector = sector_from_id(&block.id)?;
183                canonical.push(BracketBlock {
184                    id: format!("pierced_sector_{sector}_z0_8"),
185                    degrees: block.degrees,
186                    controls: block.controls.clone(),
187                });
188            }
189            Ok((canonical, 1.0))
190        }
191    }
192}

194fn lower_id(id: &str) -> Result<String, &'static str> {
195    if id == "pocket_core_lower" {
196        return Ok("pocket_core_z0_4".into());
197    }
198    if let Some(sector) = id.strip_prefix("pocket_disk_sector_lower_") {
199        return Ok(format!("pocket_disk_sector_{sector}_z0_4"));
200    }
201    if let Some(sector) = id.strip_prefix("pocket_outer_sector_lower_") {
202        return Ok(format!("pocket_outer_sector_{sector}_z0_4"));
203    }
204    Err("invalid_lower_id")
205}

207fn upper_id(id: &str) -> Result<String, &'static str> {
208    let sector = id
209        .strip_prefix("pocket_outer_sector_upper_")
210        .ok_or("invalid_upper_id")?;
211    Ok(format!("pocket_outer_sector_{sector}_z4_8"))
212}

214fn sector_from_id(id: &str) -> Result<&'static str, &'static str> {
215    for sector in SECTORS {
216        if id.contains(sector) {
217            return Ok(sector);
218        }
219    }
220    Err("unknown_pocket_sector")
221}

223fn checked_mul(a: i128, b: i128) -> Result<i128, &'static str> {
224    a.checked_mul(b).ok_or("pocket_transform_overflow")
225}
226fn checked_add(a: i128, b: i128) -> Result<i128, &'static str> {
227    a.checked_add(b).ok_or("pocket_transform_overflow")
228}
229fn checked_sub(a: i128, b: i128) -> Result<i128, &'static str> {
230    a.checked_sub(b).ok_or("pocket_transform_overflow")
231}

233fn normalize_controls(points: Vec<[i128; 4]>) -> Result<Vec<[i64; 4]>, &'static str> {
234    let mut divisor = 0_i128;
235    for point in &points {
236        for &value in point {
237            divisor = gcd_i128(divisor, value)?;
238        }
239    }
240    if divisor <= 0 {
241        return Err("zero_pocket_homogeneous_scale");
242    }
243    let mut result = Vec::with_capacity(points.len());
244    for point in points {
245        let mut converted = [0_i64; 4];
246        for axis in 0..4 {
247            converted[axis] =
248                i64::try_from(point[axis] / divisor).map_err(|_| "pocket_transform_exceeds_i64")?;
249        }
250        result.push(converted);
251    }
252    Ok(result)
253}
254
255fn gcd_i128(mut a: i128, mut b: i128) -> Result<i128, &'static str> {
256    a = a.checked_abs().ok_or("pocket_transform_overflow")?;
257    b = b.checked_abs().ok_or("pocket_transform_overflow")?;
258    while b != 0 {
259        (a, b) = (b, a % b);
260    }
261    Ok(a)
262}

264fn transformed(
265    block: &BracketBlock,
266    id: String,
267    transform: impl Fn([i128; 4]) -> Result<[i128; 4], &'static str>,
268) -> Result<BracketBlock, &'static str> {
269    let points = block
270        .controls
271        .iter()
272        .map(|p| transform(p.map(i128::from)))
273        .collect::<Result<Vec<_>, _>>()?;
274    Ok(BracketBlock {
275        id,
276        degrees: block.degrees,
277        controls: normalize_controls(points)?,
278    })
279}

281/// Map z ticks `[0,span]` to the G2 lower layer `[0,4]` mm.
282fn transform_lower(
283    block: &BracketBlock,
284    id: String,
285    span_ticks: i32,
286) -> Result<BracketBlock, &'static str> {
287    if span_ticks <= 0 {
288        return Err("zero_height_pocket_layer");
289    }
290    transformed(block, id, |[x, y, z, w]| {
291        Ok([
292            checked_mul(i128::from(span_ticks), x)?,
293            checked_mul(i128::from(span_ticks), y)?,
294            checked_mul(8, z)?,
295            checked_mul(i128::from(span_ticks), w)?,
296        ])
297    })
298}

300/// Map the blind upper layer `[floorTicks,16]` to G2's `[4,8]` mm.
301fn transform_upper(
302    block: &BracketBlock,
303    id: String,
304    floor_ticks: i32,
305    span_ticks: i32,
306) -> Result<BracketBlock, &'static str> {
307    if span_ticks <= 0 {
308        return Err("zero_height_pocket_layer");
309    }
310    transformed(block, id, |[x, y, z, w]| {
311        let z = checked_add(
312            checked_sub(
313                checked_mul(8, z)?,
314                checked_mul(4 * i128::from(floor_ticks), w)?,
315            )?,
316            checked_mul(4 * i128::from(span_ticks), w)?,
317        )?;
318        Ok([
319            checked_mul(i128::from(span_ticks), x)?,
320            checked_mul(i128::from(span_ticks), y)?,
321            z,
322            checked_mul(i128::from(span_ticks), w)?,
323        ])
324    })
325}

327fn shift_z(block: &BracketBlock, id: String, dz_mm: i128) -> Result<BracketBlock, &'static str> {
328    transformed(block, id, |[x, y, z, w]| {
329        Ok([x, y, checked_add(z, checked_mul(dz_mm, w)?)?, w])
330    })

333fn classify_faces(
334    blocks: &[BracketBlock],
335    exterior: &[BlockSide],
336    regime: Regime,
337) -> Result<Vec<(BlockSide, String)>, &'static str> {
338    let by_id: BTreeMap<&str, &BracketBlock> = blocks
339        .iter()
340        .map(|block| (block.id.as_str(), block))
341        .collect();
342    let mut result = Vec::with_capacity(exterior.len());
343    for side in exterior {
344        let block = by_id
345            .get(side.block.as_str())
346            .ok_or("unknown_pocket_boundary_face")?;
347        let id = block.id.as_str();
348        let tag = if id.starts_with("pocket_outer_sector_") && side.axis == 1 {
349            if side.upper {
350                if regime == Regime::Solid {
351                    return Err("unexpected_solid_bore_face");
352                }
353                "bore_wall".to_string()
354            } else {
355                let sector = sector_from_id(id)?;
356                match sector {
357                    "east" => "x10",
358                    "north" => "y10",
359                    "west" => "x-10",
360                    "south" => "y-10",
361                    _ => return Err("unknown_pocket_sector"),
362                }
363                .to_string()
364            }
365        } else if side.axis == 2 && !side.upper {
366            "bottom".to_string()
367        } else if side.axis == 2 && side.upper {
368            if regime == Regime::Blind
369                && (id == "pocket_core_lower" || id.starts_with("pocket_disk_sector_"))
370            {
371                "pocket_floor".to_string()
372            } else {
373                "top".to_string()
374            }
375        } else {
376            return Err("unexpected_pocket_boundary_face");
377        };
378        result.push((side.clone(), tag));
379    }
380    result.sort_by(|a, b| a.0.cmp(&b.0));
381    Ok(result)
382}

389    fn audited(depth: i32) -> (Vec<BracketBlock>, BracketCertificate) {
390        let model = build_pocket_depth(depth).unwrap();
391        let blocks: Vec<_> = model.blocks.iter().map(|item| item.block.clone()).collect();
392        let certificate = audit_depth(depth, &blocks)
393            .unwrap_or_else(|error| panic!("audit rejected depth {depth}: {error}"));
394        (blocks, certificate)
395    }

397    #[test]
398    fn certifies_all_nineteen_depth_ticks_and_expected_volumes() {
399        for depth in 0..=18 {
400            let (blocks, cert) = audited(depth);
401            assert!(cert.determinant_lower_bound_mm3.is_finite());
402            assert!(cert.determinant_lower_bound_mm3 > 0.);
403            assert!(cert.radial_error_upper_mm <= 1e-12);
404            let effective = f64::from(depth.min(16)) * 0.5;
405            let reference_volume = 3200. - 16. * std::f64::consts::PI * effective;
406            let built = build_pocket_depth(depth).unwrap();
407            assert_eq!(blocks.len(), built.blocks.len());
408            assert!((built.volume_mm3 - reference_volume).abs() <= 4. * f64::EPSILON * 3200.);
409            let certified_faces: BTreeSet<_> = cert
410                .design_faces
411                .iter()
412                .map(|(side, label)| (side.clone(), label.as_str()))
413                .collect();
414            let constructed_faces: BTreeSet<_> = built
415                .design_faces
416                .iter()
417                .map(|face| (face.side.clone(), face.label))
418                .collect();
419            assert_eq!(certified_faces, constructed_faces);
420            for selected in &built.design_faces {
421                assert!(cert
422                    .incidence
423                    .exterior
424                    .iter()
425                    .any(|side| side == &selected.side));
426                assert!(!cert.incidence.interfaces.iter().any(|interface| {
427                    interface.first == selected.side || interface.second == selected.side
428                }));
429            }
430            match depth {
431                0 => {
432                    assert_eq!(built.regime, PocketRegime::Solid);
433                    assert!(!cert
434                        .design_faces
435                        .iter()
436                        .any(|(_, label)| label == "pocket_floor"));
437                    assert!(!cert
438                        .design_faces
439                        .iter()

441                }
442                1..=15 => {

459                    assert!(cert

489    #[test]
490    fn verifies_control_jets_within_the_blind_regime() {
491        for depth in 2..=14 {
492            let center = build_pocket_depth(depth).unwrap();
493            let lower = build_pocket_depth(depth - 1).unwrap();
494            let upper = build_pocket_depth(depth + 1).unwrap();
495            for ((current, lo), hi) in center.blocks.iter().zip(&lower.blocks).zip(&upper.blocks) {
496                assert_eq!(current.block.id, lo.block.id);
497                assert_eq!(current.block.id, hi.block.id);
498                let jet = current.dcontrols_dd.as_ref().unwrap();
499                assert_eq!(jet.len(), current.block.controls.len());
500                for ((derivative, before), after) in
501                    jet.iter().zip(&lo.block.controls).zip(&hi.block.controls)
502                {
503                    // Two half-millimetre ticks span exactly one physical mm.
504                    for axis in 0..4 {
505                        assert_eq!(
506                            i128::from(after[axis]) - i128::from(before[axis]),
507                            i128::from(derivative[axis])
508                        );
509                    }
510                }
511            }
512        }
513        assert!(build_pocket_depth(0)
514            .unwrap()
515            .blocks
516            .iter()
517            .all(|b| b.dcontrols_dd.is_none()));
518        assert!(build_pocket_depth(16)
519            .unwrap()
520            .blocks
521            .iter()
522            .all(|b| b.dcontrols_dd.is_none()));
523    }

526    fn rejects_wrong_depth_inventory_geometry_and_topology() {
527        assert!(audit_depth(-1, &[]).is_err());
528        assert!(audit_depth(19, &[]).is_err());
529        let (mut blocks, _) = audited(7);

534        assert!(audit_depth(8, &blocks).is_err());

537        assert!(audit_depth(16, &blocks).is_err());
538        let (mut blocks, _) = audited(7);
539        blocks.reverse();
540        assert!(audit_depth(7, &blocks).is_ok());
