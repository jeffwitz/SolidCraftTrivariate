1//! Compact ten-block candidate for the orthogonal filleted bracket.
2//!
3//! The candidate is deliberately restricted to a six-vertex L section with one
4//! equal-clearance re-entrant corner, a longitudinal round on that corner, and
5//! one through-bore in each straight wing. Each wing and the elbow is a complete
6//! untrimmed partition of its portion of the design domain.
7
8use crate::{
9    bracket::BracketBlock,
10    orthogonal_sketch::{self, OrthogonalSketch, SweepPartition},
11    sketch_layers::{area2, edges},
12};
13use std::collections::{BTreeMap, BTreeSet};

15type Bounds = [[i32; 2]; 3];
16type CutClosureResult = (
17    Vec<BracketBlock>,
18    BTreeMap<(String, usize, bool), String>,
19    Vec<[usize; 3]>,
20    Vec<orthogonal_sketch::CutSource>,
21);

23fn ordered(a: i32, b: i32) -> [i32; 2] {
24    [a.min(b), a.max(b)]
25}

27fn standard_l(j: &OrthogonalSketch, c: &orthogonal_sketch::Corner<'_>) -> Option<[i32; 2]> {
28    if j.outer.vertices.len() != 6 {
29        return None;
30    }
31    let f = c.f.vertex;
32    let x_ext = if c.sign[0] > 0 {
33        j.outer.vertices.iter().map(|p| p[0]).max()
34    } else {
35        j.outer.vertices.iter().map(|p| p[0]).min()
36    };
37    let y_ext = if c.sign[1] > 0 {
38        j.outer.vertices.iter().map(|p| p[1]).max()
39    } else {
40        j.outer.vertices.iter().map(|p| p[1]).min()
41    };
42    let (Some(x_ext), Some(y_ext)) = (x_ext, y_ext) else {
43        return None;
44    };
45    let back = c.origin;
46    let expected = [
47        back,
48        [x_ext, back[1]],
49        [x_ext, f[1]],
50        f,
51        [f[0], y_ext],
52        [back[0], y_ext],
53    ];
54    let actual: BTreeSet<_> = j.outer.vertices.iter().copied().collect();
55    (actual.len() == 6 && expected.into_iter().collect::<BTreeSet<_>>() == actual)
56        .then_some([x_ext, y_ext])
57}

59fn make_arm_bounds(
60    j: &OrthogonalSketch,
61    c: &orthogonal_sketch::Corner<'_>,
62    ends: [i32; 2],
63) -> Result<[Bounds; 2], &'static str> {
64    let f = c.f.vertex;
65    let [x_far, y_far] = std::array::from_fn(|k| c.origin[k] + c.sign[k] * c.size_xy[k]);
66    let [x_end, y_end] = ends;
67    if (x_end - x_far) * c.sign[0] <= 0 || (y_end - y_far) * c.sign[1] <= 0 {
68        return Err("compact_elbow_nonpositive_wing_length");
69    }
70    let [x_band, y_band] = c.origin;
71    let east = [ordered(x_far, x_end), ordered(y_band, f[1]), [0, j.height]];
72    let north = [ordered(x_band, f[0]), ordered(y_far, y_end), [0, j.height]];
73    if east[0][0] == east[0][1]
74        || east[1][0] == east[1][1]
75        || north[0][0] == north[0][1]
76        || north[1][0] == north[1][1]
77    {
78        return Err("compact_elbow_empty_wing");
79    }
80    Ok([east, north])
81}

83fn supported_bore(b: &orthogonal_sketch::Bore, arm: &Bounds, expected_axis: usize) -> bool {
84    if b.axis != expected_axis || !(1..=1024).contains(&b.radius) {
85        return false;
86    }
87    if !(0..3).all(|k| arm[k][0] < b.center[k] && b.center[k] < arm[k][1]) {
88        return false;
89    }
90    // This template represents a complete transverse through-hole. Its axis
91    // spans the entire wing interval, and its circular section stays strictly
92    // inside the other two sides of the rectangular arm.
93    (0..3)
94        .filter(|&k| k != b.axis)
95        .all(|k| b.center[k] - b.radius > arm[k][0] && b.center[k] + b.radius < arm[k][1])
96}

98fn elbow_faces(
99    j: &OrthogonalSketch,
100    c: &orthogonal_sketch::Corner<'_>,
101    physical: &mut BTreeMap<(String, usize, bool), String>,
102    blocks: &mut Vec<BracketBlock>,
103) -> Result<(), &'static str> {
104    let z = [0, j.height];
105    let id = "elbow";
106    for mut block in
107        crate::sketch_fillet_chart::fillet_charts_rect([0, 0], c.size_xy, c.f.radius, z, id)?
108    {
109        orthogonal_sketch::transform_fillet(&mut block, c)?;
110        let center = [
111            c.f.vertex[0] + c.sign[0] * c.f.radius,
112            c.f.vertex[1] + c.sign[1] * c.f.radius,
113            0,
114        ];
115        orthogonal_sketch::certify_circular_side(&block, 2, center, c.f.radius)?;
116        physical.insert(
117            (block.id.clone(), 1, true),
118            format!("fillet/{}/wall", c.f.id),
119        );
120        blocks.push(block);
121    }
122    Ok(())
123}

125/// Try the compact elbow candidate. `Ok(None)` means the geometry is outside
126/// this template's declared class and the caller may use its established
127/// partitioner. A construction or certification error discards this candidate.
128pub(crate) fn try_build(j: &OrthogonalSketch) -> Result<Option<SweepPartition>, &'static str> {
129    if j.fillets.len() != 1 || !(2..=8).contains(&j.bores.len()) || j.height <= 0 {
130        return Ok(None);
131    }
132    if edges(&j.outer).any(|(a, b)| a[0] != b[0] && a[1] != b[1]) {
133        return Ok(None);
134    }
135    let c = match orthogonal_sketch::corner(j, &j.fillets[0]) {
136        Ok(c) => c,
137        Err(_) => return Ok(None),
138    };
139    let Some(ends) = standard_l(j, &c) else {
140        return Ok(None);
141    };
142    let arms = make_arm_bounds(j, &c, ends)?;
143    if j.bores.len() > 2 && j.bores.iter().any(|b| b.axis != j.bores[0].axis) {
144        return try_multi_arm_hosts(j, &c, arms);
145    }
146    if j.bores.iter().all(|b| b.axis == j.bores[0].axis) {
147        return try_multi_arm_hosts(j, &c, arms);
148    }
149    let mut east_bore = None;
150    let mut north_bore = None;
151    for bore in &j.bores {
152        match bore.axis {
153            1 if east_bore.replace(bore).is_none() => {}
154            0 if north_bore.replace(bore).is_none() => {}
155            _ => return Ok(None),
156        }
157    }
158    let (Some(east_bore), Some(north_bore)) = (east_bore, north_bore) else {
159        return Ok(None);
160    };
161    if [east_bore, north_bore].iter().any(|b| {
162        !(1..=1024).contains(&b.radius) || b.center.iter().any(|v| !(-4096..=4096).contains(v))
163    }) {
164        return Ok(None);
165    }
166    if !supported_bore(east_bore, &arms[0], 1) || !supported_bore(north_bore, &arms[1], 0) {
167        return Ok(None);
168    }
169    let mut blocks = Vec::with_capacity(10);
170    let mut physical = BTreeMap::new();
171    elbow_faces(j, &c, &mut physical, &mut blocks)?;
172    for (b, bounds, wing) in [(east_bore, arms[0], "east"), (north_bore, arms[1], "north")] {
173        let charts = crate::compact_wing::build(
174            b.axis,
175            bounds,
176            b.center,
177            b.radius,
178            &format!("wing_{wing}"),
179        )?;
180        if charts.len() != 4 {
181            return Err("compact_wing_chart_count");
182        }
183        for block in charts {
184            orthogonal_sketch::certify_circular_side(&block, b.axis, b.center, b.radius)?;
185            // Compact wing charts expose their bore wall on the radial v=1
186            // side. The circle identity is checked by the wing builder.
187            physical.insert((block.id.clone(), 1, true), format!("bore/{}/wall", b.id));
188            blocks.push(block);
189        }
190    }
191    if blocks.len() != 10 {
192        return Err("compact_orthogonal_block_count");
193    }
194    let mut planes: [BTreeSet<i32>; 3] = std::array::from_fn(|_| BTreeSet::new());
195    for bounds in [c.bounds, arms[0], arms[1]] {
196        for axis in 0..3 {
197            planes[axis].extend(bounds[axis]);
198        }
199    }
200    let planes = planes.map(|axis| axis.into_iter().collect());
201    let guards = vec![
202        (c.f.id.clone(), c.bounds),
203        (east_bore.id.clone(), arms[0]),
204        (north_bore.id.clone(), arms[1]),
205    ];
206    let h = i128::from(j.height);
207    let r_f = i128::from(c.f.radius);
208    let mut twice_constant = i128::from(area2(&j.outer)) * h;
209    let mut four_pi = -r_f * r_f * h;
210    twice_constant = twice_constant
211        .checked_add(2 * r_f * r_f * h)
212        .ok_or("compact_orthogonal_volume_overflow")?;
213    for b in [east_bore, north_bore] {
214        let r = i128::from(b.radius);
215        let wing = if b.axis == east_bore.axis {
216            arms[0]
217        } else {
218            arms[1]
219        };
220        let length = i128::from(wing[b.axis][1] - wing[b.axis][0]);
221        four_pi = four_pi
222            .checked_sub(4 * r * r * length)
223            .ok_or("compact_orthogonal_volume_overflow")?;
224    }
225    // Retain the exact area identity as a precondition check: the corner
226    // decomposition adds r²(1-π/4), and each through-bore removes πr²L.
227    if twice_constant < 0 || four_pi >= 0 {
228        return Err("compact_orthogonal_volume_sign");
229    }
230    let partition = orthogonal_sketch::finish_partition(
231        j,
232        blocks,
233        physical,
234        twice_constant,
235        four_pi,
236        planes,
237        guards,
238    )?;
239    // The per-block Bernstein bounds returned by finish_partition certify
240    // regularity; exact shared traces and the prescribed physical labels
241    // certify conformity and provenance for this assembled candidate.
242    Ok(Some(partition))
243}

245/// Candidate using the mapped-cut closure policy. The caller compares it with
246/// the frozen compact plan before selection; this entry point keeps failures
247/// separate from the historical builder.
248pub(crate) fn try_propagated(j: &OrthogonalSketch) -> Result<Option<SweepPartition>, &'static str> {
249    if j.fillets.len() != 1
250        || j.bores.len() < 2
251        || j.height <= 0
252        || j.bores
253            .iter()
254            .any(|b| b.axis > 2 || !(1..=1024).contains(&b.radius))
255        || edges(&j.outer).any(|(a, b)| a[0] != b[0] && a[1] != b[1])
256    {
257        return Ok(None);
258    }
259    let c = match orthogonal_sketch::corner(j, &j.fillets[0]) {
260        Ok(c) => c,
261        Err(_) => return Ok(None),
262    };
263    let Some(ends) = standard_l(j, &c) else {
264        return Ok(None);
265    };
266    let arms = make_arm_bounds(j, &c, ends)?;
267    let mut by_arm: [Vec<&orthogonal_sketch::Bore>; 2] = std::array::from_fn(|_| Vec::new());
268    for bore in &j.bores {
269        let arm_index = match bore.axis {
270            1 => Some(0),
271            0 => Some(1),
272            2 => {
273                let fits = (0..2)
274                    .filter(|&i| {
275                        bore.center[2] > arms[i][2][0]
276                            && bore.center[2] < arms[i][2][1]
277                            && bore.center[0] - bore.radius > arms[i][0][0]
278                            && bore.center[0] + bore.radius < arms[i][0][1]
279                            && bore.center[1] - bore.radius > arms[i][1][0]
280                            && bore.center[1] + bore.radius < arms[i][1][1]
281                    })
282                    .collect::<Vec<_>>();
283                if fits.len() != 1 {
284                    return Ok(None);
285                }
286                fits.first().copied()
287            }
288            _ => None,
289        };
290        let Some(index) = arm_index else {
291            return Ok(None);
292        };
293        by_arm[index].push(bore);
294    }
295    // A host is one tensor extrusion direction. Mixed bore axes in the same
296    // arm require a non-isoparametric T-junction and are deliberately refused.
297    let mut blocks = Vec::new();
298    let mut physical = BTreeMap::new();
299    elbow_faces(j, &c, &mut physical, &mut blocks)?;
300    let mut guards = vec![(c.f.id.clone(), c.bounds)];
301    for arm_index in 0..2 {
302        let arm_axis = if arm_index == 0 { 1 } else { 0 };
303        let group = &by_arm[arm_index];
304        if group.is_empty() {
305            blocks.push(crate::compact_hosts::plain_box(
306                format!("propagated_wing_free_{arm_index}"),
307                arms[arm_index],
308            ));
309            continue;
310        }
311        let bore_axes = group.iter().map(|b| b.axis).collect::<BTreeSet<_>>();
312        let hosted = if bore_axes.len() > 1 {
313            crate::mixed_axis_hosts::build(
314                arms[arm_index],
315                group,
316                &format!("propagated_wing_{arm_index}"),
317            )?
318        } else {
319            let bore_axis = group[0].axis;
320            crate::compact_hosts::holed_box_propagated(
321                bore_axis,
322                arms[arm_index],
323                group,
324                &format!("propagated_wing_{arm_index}"),
325                bore_axis == arm_axis,
326            )?
327        };
328        let Some(crate::compact_hosts::HoledBox {
329            blocks: mut hosted,
330            physical: mut labels,
331            guards: mut host_guards,
332        }) = hosted
333        else {
334            return Ok(None);
335        };
336        blocks.append(&mut hosted);
337        physical.append(&mut labels);
338        guards.append(&mut host_guards);
339    }
340
341    let mut planes: [BTreeSet<i32>; 3] = std::array::from_fn(|_| BTreeSet::new());
342    for bounds in [c.bounds, arms[0], arms[1]] {
343        for axis in 0..3 {
344            planes[axis].extend(bounds[axis]);
345        }
346    }
347    for (_, bounds) in guards.iter().skip(1) {
348        for axis in 0..3 {
349            planes[axis].extend(bounds[axis]);
350        }
351    }
352    let h = i128::from(j.height);
353    let rf = i128::from(c.f.radius);
354    let mut twice_constant = i128::from(area2(&j.outer))
355        .checked_mul(h)
356        .ok_or("mapped_cut_volume_overflow")?;
357    twice_constant = twice_constant
358        .checked_add(2 * rf * rf * h)
359        .ok_or("mapped_cut_volume_overflow")?;
360    let mut four_pi = -rf * rf * h;
361    for bore in &j.bores {
362        let length = if bore.axis == 2 {
363            j.height
364        } else {
365            arms[if bore.axis == 1 { 0 } else { 1 }][bore.axis][1]
366                - arms[if bore.axis == 1 { 0 } else { 1 }][bore.axis][0]
367        };
368        let r = i128::from(bore.radius);
369        four_pi = four_pi
370            .checked_sub(4 * r * r * i128::from(length))
371            .ok_or("mapped_cut_volume_overflow")?;
372    }
373    let (blocks, physical, cut_rounds, cut_sources) = close_mapped_cuts(blocks, physical, 1024)?;
374    let mut result = orthogonal_sketch::finish_partition(
375        j,
376        blocks,
377        physical,
378        twice_constant,
379        four_pi,
380        planes.map(|p| p.into_iter().collect()),
381        guards,
382    )?;
383    result.cut_rounds = cut_rounds;
384    result.cut_sources = cut_sources;
385    Ok(Some(result))
386}

388fn close_mapped_cuts(
389    mut blocks: Vec<BracketBlock>,
390    mut physical: BTreeMap<(String, usize, bool), String>,
391    budget: usize,
392) -> Result<CutClosureResult, &'static str> {
393    let mut rounds = Vec::new();
394    let mut sources = Vec::new();
395    for round in 0..8 {
396        let (faces, requests) = crate::mapped_seams::discover(&blocks)?;
397        if requests.is_empty() {
398            return Ok((blocks, physical, rounds, sources));
399        }
400        let previous_count = blocks.len();
401        let plan = crate::mapped_cut_propagation::propagate_exact_cuts(
402            &blocks, &faces, &requests, budget,
403        )?;
404        if plan.child_count <= previous_count || plan.child_count > budget {
405            return Err("mapped_cut_no_progress_or_budget");
406        }
407        rounds.push([
408            plan.requested_cut_count,
409            plan.propagated_cut_count,
410            plan.child_count,
411        ]);
412        for cut in &plan.cuts {
413            sources.push(orthogonal_sketch::CutSource {
414                round,
415                block: blocks[cut.block].id.clone(),
416                axis: cut.axis,
417                numerator: cut.parameter.numerator,
418                denominator: cut.parameter.denominator,
419                sources: cut.provenance.clone(),
420            });
421        }
422        let mut transferred = BTreeMap::new();
423        for ((id, axis, upper), label) in &physical {
424            let source_index = blocks
425                .iter()
426                .position(|b| b.id == *id)
427                .ok_or("mapped_cut_lost_physical_source")?;
428            let mut count = 0;
429            for child in plan
430                .boundary_children
431                .iter()
432                .filter(|c| c.source_block == source_index && c.axis == *axis && c.upper == *upper)
433            {
434                let child_id = plan.blocks[child.child_block].id.clone();
435                transferred.insert((child_id, *axis, *upper), label.clone());
436                count += 1;
437            }
438            if count == 0 {
439                return Err("mapped_cut_lost_physical_boundary");
440            }
441        }
442        blocks = plan.blocks;
443        physical = transferred;
444    }
445    Err("mapped_cut_round_budget")
446}

448fn try_multi_arm_hosts(
449    j: &OrthogonalSketch,
450    c: &orthogonal_sketch::Corner<'_>,
451    arms: [Bounds; 2],
452) -> Result<Option<SweepPartition>, &'static str> {
453    if j.bores.iter().any(|b| {
454        b.axis > 1
455            || !(1..=1024).contains(&b.radius)
456            || b.center.iter().any(|v| !(-4096..=4096).contains(v))
457    }) {
458        return Ok(None);
459    }
460    let mut blocks = Vec::new();
461    let mut physical = BTreeMap::new();
462    elbow_faces(j, c, &mut physical, &mut blocks)?;
463    let mut guards = vec![(c.f.id.clone(), c.bounds)];
464    for (arm_index, arm_bounds) in arms.iter().copied().enumerate() {
465        let axis = if arm_index == 0 { 1 } else { 0 };
466        let bores = j
467            .bores
468            .iter()
469            .filter(|b| b.axis == axis)
470            .collect::<Vec<_>>();
471        if bores.is_empty() {
472            blocks.push(crate::compact_hosts::plain_box(
473                format!("wing_free_{arm_index}"),
474                arm_bounds,
475            ));
476        } else {
477            let Some(crate::compact_hosts::HoledBox {
478                blocks: mut hosted,
479                physical: mut labels,
480                guards: mut host_guards,
481            }) = crate::compact_hosts::holed_box(
482                axis,
483                arm_bounds,
484                &bores,
485                &format!("wing_{arm_index}"),
486                true,
487            )?
488            else {
489                return Ok(None);
490            };
491            blocks.append(&mut hosted);
492            physical.append(&mut labels);
493            guards.append(&mut host_guards);
494        }
495    }
496    let mut planes: [BTreeSet<i32>; 3] = std::array::from_fn(|_| BTreeSet::new());
497    for b in [c.bounds, arms[0], arms[1]] {
498        for k in 0..3 {
499            planes[k].extend(b[k]);
500        }
501    }
502    for (_, host) in guards.iter().skip(1) {
503        for k in 0..3 {
504            planes[k].extend(host[k]);
505        }
506    }
507    let mut twice_constant = i128::from(area2(&j.outer)) * i128::from(j.height);
508    let r_f = i128::from(c.f.radius);
509    let h = i128::from(j.height);
510    twice_constant = twice_constant
511        .checked_add(2 * r_f * r_f * h)
512        .ok_or("compact_orthogonal_volume_overflow")?;
513    let mut four_pi = -r_f * r_f * h;
514    for bore in &j.bores {
515        let arm_index = if bore.axis == 1 { 0 } else { 1 };
516        let r = i128::from(bore.radius);
517        four_pi = four_pi
518            .checked_sub(
519                4 * r
520                    * r
521                    * i128::from(arms[arm_index][bore.axis][1] - arms[arm_index][bore.axis][0]),
522            )
523            .ok_or("compact_orthogonal_volume_overflow")?;
524    }
525    if twice_constant < 0 || four_pi >= 0 {
526        return Err("compact_orthogonal_volume_sign");
527    }
528    let partition = orthogonal_sketch::finish_partition(
529        j,
530        blocks,
531        physical,
532        twice_constant,
533        four_pi,
534        planes.map(|p| p.into_iter().collect()),
535        guards,
536    )?;
537    Ok(Some(partition))
538}
