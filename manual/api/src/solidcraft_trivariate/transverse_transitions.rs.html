41fn qnew(numerator: i128, denominator: i128) -> Option<Q> {
42    Q::new(numerator, denominator).ok()
43}

45fn qtext(value: Q) -> String {
46    if value.denominator() == 1 {
47        value.numerator().to_string()
48    } else {
49        format!("{}/{}", value.numerator(), value.denominator())
50    }

53type P = Vec<Q>;
54fn trim(mut p: P) -> P {
55    while p.last().is_some_and(|value| value.numerator() == 0) {
56        p.pop();
57    }
58    p
59}
60fn padd(a: &P, b: &P, scale: i128) -> Option<P> {
61    let mut c = vec![qnew(0, 1)?; a.len().max(b.len())];
62    for (i, x) in a.iter().enumerate() {
63        c[i] = c[i].add(*x)?
64    }
65    for (i, x) in b.iter().enumerate() {
66        c[i] = c[i].add(x.mul(qnew(scale, 1)?)?)?
67    }
68    Some(trim(c))
69}
70fn pmul(a: &P, b: &P) -> Option<P> {
71    if a.is_empty() || b.is_empty() {
72        return Some(vec![]);
73    }
74    let mut c = vec![qnew(0, 1)?; a.len() + b.len() - 1];
75    for (i, x) in a.iter().enumerate() {
76        for (j, y) in b.iter().enumerate() {
77            c[i + j] = c[i + j].add(x.mul(*y)?)?
78        }
79    }
80    Some(trim(c))
81}
82fn deriv(a: &P) -> Option<P> {
83    let mut out = Vec::new();
84    for (i, x) in a.iter().enumerate().skip(1) {
85        out.push(x.mul(qnew(i as i128, 1)?)?)
86    }
87    Some(trim(out))
88}
89fn cross(a: &[P; 2], b: &[P; 2]) -> Option<P> {
90    padd(&pmul(&a[0], &b[1])?, &pmul(&a[1], &b[0])?, -1)
91}
92fn bernstein(row: &[Q]) -> Option<P> {
93    if row.is_empty() {
94        return Some(vec![]);
95    }
96    let d = row.len() - 1;
97    let mut out = vec![qnew(0, 1)?; d + 1];
98    for (i, v) in row.iter().enumerate() {
99        let choose = binom(d, i)?;
100        for j in 0..=d - i {
101            let term = v.mul(qnew(
102                choose
103                    .checked_mul(binom(d - i, j)?)?
104                    .checked_mul(if j % 2 == 0 { 1 } else { -1 })?,
105                1,
106            )?)?;
107            out[i + j] = out[i + j].add(term)?;
108        }
109    }
110    Some(trim(out))
111}
112fn binom(n: usize, k: usize) -> Option<i128> {
113    let k = k.min(n - k);
114    let mut r = 1i128;
115    for i in 0..k {
116        r = r
117            .checked_mul((n - i) as i128)?
118            .checked_div((i + 1) as i128)?;
119    }
120    Some(r)
121}
122fn quotient_affine(a: &P, b: &P) -> Option<Option<(Q, Q)>> {
123    if b.is_empty() || b[0].numerator() == 0 {
124        return None;
125    }
126    let c0 = if a.is_empty() {
127        qnew(0, 1)?
128    } else {
129        a[0].div(b[0])?
130    };
131    let a1 = if a.len() > 1 { a[1] } else { qnew(0, 1)? };
132    let b1 = if b.len() > 1 { b[1] } else { qnew(0, 1)? };
133    let c1 = a1.sub(c0.mul(b1)?)?.div(b[0])?;
134    let product = pmul(&vec![c0, c1], b)?;
135    let residual = padd(a, &product, -1)?;
136    Some(if residual.is_empty() {
137        Some((c0, c1))
138    } else {
139        None
140    })
141}

143fn pdiv_rem(a: &P, b: &P) -> Option<(P, P)> {
144    if b.is_empty() {
145        return None;
146    }
147    let mut r = a.clone();
148    let mut q = vec![qnew(0, 1)?; a.len().saturating_sub(b.len()) + 1];
149    while !r.is_empty() && r.len() >= b.len() {
150        let shift = r.len() - b.len();
151        let factor = r.last()?.div(*b.last()?)?;
152        q[shift] = factor;
153        let mut term = vec![qnew(0, 1)?; shift];
154        term.extend(
155            b.iter()
156                .map(|x| x.mul(factor))
157                .collect::<Option<Vec<_>>>()?,
158        );
159        r = padd(&r, &term, -1)?;
160    }
161    Some((trim(q), r))
162}

164fn pgcd(a: &P, b: &P) -> Option<P> {
165    let (mut x, mut y) = (trim(a.clone()), trim(b.clone()));
166    while !y.is_empty() {
167        let (_, r) = pdiv_rem(&x, &y)?;
168        x = y;
169        y = r;
170    }
171    if x.is_empty() {
172        return Some(x);
173    }
174    let lead = *x.last()?;
175    x.iter().map(|v| v.div(lead)).collect()
176}

178fn split_beta(a: &P, b: &P, beta: &P) -> Option<(P, P)> {
179    // Solve beta = a * beta_second + b * beta_first for linear factors.
180    let z = qnew(0, 1)?;
181    let at = |p: &P, i: usize| p.get(i).copied().unwrap_or(z);
182    let mut m = vec![
183        vec![at(b, 0), z, at(a, 0), z, at(beta, 0)],
184        vec![at(b, 1), at(b, 0), at(a, 1), at(a, 0), at(beta, 1)],
185        vec![z, at(b, 1), z, at(a, 1), at(beta, 2)],
186    ];
187    let mut row = 0;
188    for col in 0..4 {
189        let Some(pivot) = (row..3).find(|r| m[*r][col].numerator() != 0) else {
190            continue;
191        };
192        m.swap(row, pivot);
193        let d = m[row][col];
194        for k in col..5 {
195            m[row][k] = m[row][k].div(d)?;
196        }
197        for r in 0..3 {
198            if r == row {
199                continue;
200            }
201            let f = m[r][col];
202            for k in col..5 {
203                m[r][k] = m[r][k].sub(f.mul(m[row][k])?)?;
204            }
205        }
206        row += 1;
207        if row == 3 {
208            break;
209        }
210    }
211    for r in row..3 {
212        if (0..4).all(|c| m[r][c].numerator() == 0) && m[r][4].numerator() != 0 {
213            return None;
214        }
215    }
216    let mut x = [z; 4];
217    for r in (0..row).rev() {
218        let col = (0..4).find(|c| m[r][*c].numerator() != 0)?;
219        let rhs = (col + 1..4).try_fold(m[r][4], |v, c| v.sub(m[r][c].mul(x[c])?))?;
220        x[col] = rhs.div(m[r][col])?;
221    }
222    Some((vec![x[0], x[1]], vec![x[2], x[3]]))
223}

225fn as_g1_result(
226    a: &Json,
227    sa: &str,
228    b: &Json,
229    sb: &str,
230    reverse: bool,
231) -> Result<Json, &'static str> {
232    let unsupported = || object([("status", Json::String("unsupported_family".into()))]);
233    let da = a.field("degrees")?.items()?;
234    let db = b.field("degrees")?.items()?;
235    if da.len() != 3
236        || db.len() != 3
237        || da
238            .iter()
239            .chain(db)
240            .any(|d| d.number::<usize>().ok() != Some(1))
241    {
242        return Ok(unsupported());
243    }
244    for block in [a, b] {
245        let controls = block.field("controls")?.items()?;
246        let first_weight = controls
247            .first()
248            .and_then(|p| p.items().ok())
249            .and_then(|p| p.get(3))
250            .and_then(|w| w.number::<i128>().ok());
251        if first_weight.is_none_or(|w| w <= 0)
252            || controls.iter().any(|p| {
253                p.items()
254                    .ok()
255                    .and_then(|p| p.get(3))
256                    .and_then(|w| w.number::<i128>().ok())
257                    != first_weight
258            })
259        {
260            return Ok(object([(
261                "status",
262                Json::String("unsupported_projective_weights".into()),
263            )]));
264        }
265    }
266    let (ta, na, _) = block_jet(a, sa, false)?;
267    let (tb, nb, _) = block_jet(b, sb, reverse)?;
268    if tb != ta {
269        return Ok(object([(
270            "status",
271            Json::String("unsupported_trace".into()),
272        )]));
273    }
274    // The determinant syzygy constructs gluing data:
275    // det(n_b,t)n_a + det(t,n_a)n_b + det(n_a,n_b)t = 0.
276    let mut alpha_a = cross(&nb, &ta).ok_or("rational overflow")?;
277    let mut alpha_b = cross(&ta, &na).ok_or("rational overflow")?;
278    let mut beta = cross(&na, &nb).ok_or("rational overflow")?;
279    let common = pgcd(&pgcd(&alpha_a, &alpha_b).ok_or("rational overflow")?, &beta)
280        .ok_or("rational overflow")?;
281    if common.is_empty() {
282        return Ok(object([("status", Json::String("indeterminate".into()))]));
283    }
284    for p in [&mut alpha_a, &mut alpha_b, &mut beta] {
285        let (q, r) = pdiv_rem(p, &common).ok_or("rational overflow")?;
286        if !r.is_empty() {
287            return Err("inexact gluing factor division");
288        }
289        *p = q;
290    }
291    let coeff = |p: &P| -> Json {
292        Json::Array(if p.is_empty() {
293            vec![Json::String("0".into())]
294        } else {
295            p.iter().map(|x| Json::String(qtext(*x))).collect()
296        })
297    };
298    // AS-G1 permits quadratic beta; the two alpha gluing functions are linear.
299    let linear = alpha_a.len() <= 2 && alpha_b.len() <= 2 && beta.len() <= 3;
300    let relatively_prime = pgcd(&alpha_a, &alpha_b).is_some_and(|p| p.len() == 1);
301    let Some((beta_first, beta_second)) = split_beta(&alpha_a, &alpha_b, &beta) else {
302        return Ok(object([
303            ("status", Json::String("not_as_g1".into())),
304            ("reason", Json::String("no_linear_beta_split".into())),
305        ]));
306    };
307    let split_check = padd(
308        &padd(
309            &pmul(&alpha_a, &beta_second).ok_or("rational overflow")?,
310            &pmul(&alpha_b, &beta_first).ok_or("rational overflow")?,
311            1,
312        )
313        .ok_or("rational overflow")?,
314        &beta,
315        -1,
316    )
317    .ok_or("rational overflow")?;
318    let identity_check = (0..2).all(|c| {
319        let left = pmul(&alpha_a, &na[c])
320            .and_then(|x| pmul(&alpha_b, &nb[c]).and_then(|y| padd(&x, &y, 1)))
321            .and_then(|x| pmul(&beta, &ta[c]).and_then(|y| padd(&x, &y, 1)));
322        left.is_some_and(|p| p.is_empty())
323    });
324    let endpoint_signs = |p: &P| -> Option<(i8, i8)> {
325        let x = p.first()?.numerator();
326        let y = p
327            .iter()
328            .try_fold(qnew(0, 1)?, |sum, c| sum.add(*c))?
329            .numerator();
330        if x == 0 || y == 0 || x.signum() != y.signum() {
331            None
332        } else {
333            Some((x.signum() as i8, y.signum() as i8))
334        }
335    };
336    // Normals are directed inward on each chart; compatible inward gluing has
337    // equal-sign alpha factors because the two physical inward vectors oppose.
338    let orientable = endpoint_signs(&alpha_a)
339        .zip(endpoint_signs(&alpha_b))
340        .is_some_and(|(x, y)| x.0 == y.0 && x.1 == y.1);
341    let no_zero = |p: &P| -> Option<bool> {
342        let (s0, s1) = endpoint_signs(p)?;
343        if p.len() <= 2 {
344            return Some(s0 == s1);
345        }
346        if p.len() != 3 {
347            return None;
348        }
349        let (a, b) = (p[2], p[1]);
350        if a.numerator() == 0 {
351            return Some(s0 == s1);
352        }
353        let vertex = b.mul(qnew(-1, 2)?)?.div(a)?;
354        if vertex.numerator() <= 0 || vertex.sub(qnew(1, 1)?)?.numerator() >= 0 {
355            return Some(s0 == s1);
356        }
357        let value = p[0].sub(b.mul(b)?.div(a.mul(qnew(4, 1)?)?)?)?;
358        Some(if s0 > 0 {
359            value.numerator() > 0
360        } else {
361            value.numerator() < 0
362        })
363    };
364    let regular = no_zero(&cross(&ta, &na).ok_or("rational overflow")?).unwrap_or(false)
365        && no_zero(&cross(&ta, &nb).ok_or("rational overflow")?).unwrap_or(false);
366    let admitted = linear
367        && relatively_prime
368        && orientable
369        && regular
370        && split_check.is_empty()
371        && identity_check;
372    Ok(object([
373        (
374            "status",
375            Json::String(if admitted { "as_g1" } else { "not_as_g1" }.into()),
376        ),
377        (
378            "gluing_factors",
379            object([
380                ("alpha_first", coeff(&alpha_a)),
381                ("alpha_second", coeff(&alpha_b)),
382                ("beta_first", coeff(&beta_first)),
383                ("beta_second", coeff(&beta_second)),
384                ("beta", coeff(&beta)),
385            ]),
386        ),
387        ("common_factor", coeff(&common)),
388        ("relatively_prime", Json::Bool(relatively_prime)),
389        ("orientable", Json::Bool(orientable)),
390        ("regular_on_closed_edge", Json::Bool(regular)),
391        ("trace_tangent_match", Json::Bool(true)),
392        ("beta_split_verified", Json::Bool(split_check.is_empty())),
393        ("syzygy_verified", Json::Bool(identity_check)),
394        (
395            "proof",
396            Json::String(
397                "exact determinant syzygy; polynomial gcd division has zero remainder".into(),
398            ),
399        ),
400    ]))
401}

403fn parse_face(v: &Json) -> Result<(&str, &str), &'static str> {
404    let a = v.items()?;
405    if a.len() != 2 {
406        return Err("expected block id and side");
407    };
408    Ok((a[0].text()?, a[1].text()?))
409}
410fn block_jet(block: &Json, side: &str, reverse: bool) -> Result<([P; 2], [P; 2], P), &'static str> {
411    let deg = block.field("degrees")?.items()?;
412    if deg.len() != 3 {
413        return Err("expected three degrees");
414    };
415    let du = deg[0].number::<usize>()?;
416    let dv = deg[1].number::<usize>()?;
417    let dw = deg[2].number::<usize>()?;
418    if du == 0 || dv == 0 || du > 64 || dv > 64 || dw != 1 {
419        return Err("unsupported transition degree");
420    };
421    let controls = block.field("controls")?.items()?;
422    if controls.len() != 2 * (du + 1) * (dv + 1)
423        || controls
424            .iter()
425            .any(|point| point.items().map_or(true, |values| values.len() != 4))
426    {
427        return Err("invalid transverse control net");
428    }
429    let common_weight = controls
430        .first()
431        .and_then(|c| c.items().ok())
432        .and_then(|c| c.get(3))
433        .and_then(|w| w.number::<i128>().ok())
434        .filter(|w| *w > 0)
435        .filter(|w| {
436            controls.iter().all(|c| {
437                c.items()
438                    .ok()
439                    .and_then(|v| v.get(3))
440                    .and_then(|w| w.number::<i128>().ok())
441                    == Some(*w)
442            })
443        });
444    let mut grid = vec![vec![Vec::<Q>::new(); dv + 1]; du + 1];
445    for i in 0..=du {
446        for j in 0..=dv {
447            let v = &controls[2 * (i * (dv + 1) + j)];
448            let mut c = Vec::new();
449            for x in v.items()? {
450                let value = x.number::<i128>()?;
451                c.push(if let Some(w) = common_weight {
452                    qnew(value, w).ok_or("rational overflow")?
453                } else {
454                    qnew(value, 1).ok_or("rational overflow")?
455                })
456            }
457            grid[i][j] = c;
458        }
459    }
460    let axis = match side.as_bytes() {
461        [b'u', b'0'] => 0,
462        [b'u', b'1'] => 0,
463        [b'v', b'0'] => 1,
464        [b'v', b'1'] => 1,
465        _ => return Err("unsupported side"),
466    };
467    let end = side.as_bytes()[1] == b'1';
468    let d = if axis == 0 { du } else { dv };
469    let e = if end { d } else { 0 };
470    let n = if end {
471        d.saturating_sub(1)
472    } else {
473        (d + 1).min(d)
474    };
475    let len = if axis == 0 { dv + 1 } else { du + 1 };
476    let mut h: [Vec<Q>; 4] = std::array::from_fn(|_| Vec::new());
477    let mut near: [Vec<Q>; 4] = std::array::from_fn(|_| Vec::new());
478    for t in 0..len {
479        let (i, j, ni, nj) = if axis == 0 {
480            (e, t, n, t)
481        } else {
482            (t, e, t, n)
483        };
484        for c in 0..4 {
485            h[c].push(grid[i][j][c]);
486            near[c].push(grid[ni][nj][c]);
487        }
488    }
489    // Parameter reversal changes every polynomial's argument. Bernstein reversal
490    // is exact by reversing the control row before conversion.
491    if reverse {
492        for c in 0..4 {
493            h[c].reverse();
494            near[c].reverse();
495        }
496    }
497    let hp: [P; 4] = std::array::from_fn(|c| bernstein(&h[c]).unwrap_or_default());
498    let np: [P; 4] = std::array::from_fn(|c| bernstein(&near[c]).unwrap_or_default());
499    let weight = hp[3].clone();
500    let wt = deriv(&weight).ok_or("rational overflow")?;
501    let weight_normal = padd(&np[3], &hp[3], -1)
502        .ok_or("rational overflow")?
503        .into_iter()
504        .map(|x| x.mul(qnew(d as i128, 1).unwrap()).unwrap())
505        .collect::<P>();
506    let mut derivs: [[P; 2]; 2] = std::array::from_fn(|_| std::array::from_fn(|_| Vec::new()));
507    for c in 0..2 {
508        let tangent = deriv(&hp[c]).ok_or("rational overflow")?;
509        let normal = padd(&np[c], &hp[c], -1).ok_or("rational overflow")?;
510        let normal = normal
511            .into_iter()
512            .map(|x| x.mul(qnew(d as i128, 1).unwrap()).unwrap())
513            .collect::<P>();
514        let dt = padd(
515            &pmul(&tangent, &weight).ok_or("rational overflow")?,
516            &pmul(&hp[c], &wt).ok_or("rational overflow")?,
517            -1,
518        )
519        .ok_or("rational overflow")?;
520        let dn = padd(
521            &pmul(&normal, &weight).ok_or("rational overflow")?,
522            &pmul(&hp[c], &weight_normal).ok_or("rational overflow")?,
523            -1,
524        )
525        .ok_or("rational overflow")?;
526        derivs[0][c] = dt;
527        derivs[1][c] = dn;
528    }
529    let weight2 = pmul(&weight, &weight).ok_or("rational overflow")?;
530    Ok((derivs[0].clone(), derivs[1].clone(), weight2))
531}

537fn directional(a: &Json, sa: &str, b: &Json, sb: &str, reverse: bool) -> DirectionalResult {
538    let (ta, na, wa) = block_jet(a, sa, false)?;
539    let (_, nb, wb) = block_jet(b, sb, reverse)?;
540    let ca = cross(&na, &ta).ok_or("rational overflow")?;
541    let cb = cross(&nb, &ta).ok_or("rational overflow")?;
542    let den = pmul(&ca, &wb).ok_or("rational overflow")?;
543    let nalpha = pmul(&cb, &wa).ok_or("rational overflow")?;
544    let beta_num =
545        pmul(&cross(&na, &nb).ok_or("rational overflow")?, &wa).ok_or("rational overflow")?;
546    let Some(alpha) = quotient_affine(&nalpha, &den).ok_or("rational overflow")? else {
547        return Ok(("outside_affine_transition_test", None, None));
548    };
549    let Some(beta) = quotient_affine(&beta_num, &den).ok_or("rational overflow")? else {
550        return Ok(("outside_affine_transition_test", None, None));
551    };
552    if alpha
553        .0
554        .mul(alpha.0.add(alpha.1).ok_or("rational overflow")?)
555        .ok_or("rational overflow")?
556        .numerator()
557        <= 0
558    {
559        return Ok(("transition_singular_on_face", None, None));
560    }
561    let kind = if alpha.1.numerator() != 0 || beta.1.numerator() != 0 {
562        "affine_transition"
563    } else {
564        "constant_transition"
565    };
566    Ok((kind, Some(alpha), Some(beta)))
567}
568fn coeffs(alpha: Option<(Q, Q)>, beta: Option<(Q, Q)>) -> Json {
569    let pair = |p: Option<(Q, Q)>| {
570        p.map_or(Json::Array(vec![]), |(a, b)| {
571            Json::Array(vec![Json::String(qtext(a)), Json::String(qtext(b))])
572        })
573    };
574    if alpha.is_none() || beta.is_none() {
575        object([])
576    } else {
577        object([("alpha", pair(alpha)), ("beta", pair(beta))])
578    }
579}

581/// Classify every interface in one model, returning the same per-link schema as
582/// `scripts/census_transverse_transitions.py`.
583pub fn classify(model: &Json) -> Result<Json, &'static str> {
584    let blocks = model.field("blocks")?.items()?;
585    let mut by_id = std::collections::BTreeMap::new();
586    for b in blocks {
587        by_id.insert(b.field("id")?.text()?, b);
588    }
589    let mut records = Vec::new();
590    for link in model.field("topology")?.field("interfaces")?.items()? {
591        let first = link.field("first")?;
592        let second = link.field("second")?;
593        let (a, sa) = parse_face(first)?;
594        let (b, sb) = parse_face(second)?;
595        let orient = link.field("orientation")?;
596        let swap = orient.field("swap")?.bool()?;
597        let ff = orient.field("flip_first")?.bool()?;
598        let fs = orient.field("flip_second")?.bool()?;
599        let side_axial = sa.starts_with('w') || sb.starts_with('w');
600        let aa = by_id.get(a).ok_or("unknown block")?;
601        let bb = by_id.get(b).ok_or("unknown block")?;
602        let mut coeff = object([]);
603        let mut reverse_coeff = object([]);
604        let mut reversed = None;
605        let mut as_g1 = object([("status", Json::String("unsupported_frame".into()))]);
606        let status;
607        if side_axial {
608            status = "axial_or_nonlateral";
609            as_g1 = object([("status", Json::String("nonlateral_interface".into()))]);
610        } else {
611            let shift = |x: &Json| -> Result<Option<[Q; 3]>, &'static str> {
612                let dg = x.field("degrees")?.items()?;
613                if dg[2].number::<usize>()? != 1 {
614                    return Ok(None);
615                }
616                let c = x.field("controls")?.items()?;
617                let mut got = None;
618                let (pairs, _) = c.as_chunks::<2>();
619                for pair in pairs {
620                    let u = pair[0].items()?;
621                    let v = pair[1].items()?;
622                    let mut s = [Q::zero(); 3];
623                    for i in 0..3 {
624                        let wu = u[3].number::<i128>()?;
625                        let wv = v[3].number::<i128>()?;
626                        if wu != wv || wu <= 0 {
627                            return Ok(None);
628                        }
629                        s[i] = qnew(
630                            v[i].number::<i128>()?
631                                .checked_sub(u[i].number::<i128>()?)
632                                .ok_or("rational overflow")?,
633                            wu,
634                        )
635                        .ok_or("rational overflow")?;
636                    }
637                    if got.is_some_and(|old| old != s) {
638                        return Ok(None);
639                    }
640                    got = Some(s);
641                }
642                Ok(got)
643            };
644            let x = shift(aa)?;
645            let y = shift(bb)?;
646            if swap || fs || x.is_none() || x != y {
647                status = "unsupported_transition_frame"
648            } else {
649                as_g1 = as_g1_result(aa, sa, bb, sb, ff)?;
650                let (s, c1, c2) = directional(aa, sa, bb, sb, ff)?;
651                status = s;
652                coeff = coeffs(c1, c2);
653                let (s, c1, c2) = directional(bb, sb, aa, sa, ff)?;
654                reversed = Some(s);
655                reverse_coeff = coeffs(c1, c2);
656            }
657        }
658        let face = |f: &Json| Json::Array(f.items().unwrap().to_vec());
659        let mut fields = std::collections::BTreeMap::new();
660        fields.insert("faces".into(), Json::Array(vec![face(first), face(second)]));
661        fields.insert("status".into(), Json::String(status.into()));
662        fields.insert("as_g1".into(), as_g1);
663        if let Json::Object(values) = coeff {
664            fields.extend(values);
665        }
666        if let Some(reversed) = reversed {
667            fields.insert("reversed_status".into(), Json::String(reversed.into()));
668            fields.insert("reversed_coefficients".into(), reverse_coeff);
669        }
670        records.push(Json::Object(fields));
671    }
672    Ok(Json::Array(records))
673}
