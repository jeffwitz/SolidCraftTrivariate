24fn text(value: &Json) -> Result<&str, &'static str> {
25    value.text()
26}

28fn integer(value: &Json, allow_string: bool) -> Result<i32, &'static str> {
29    let token = match value {
30        Json::Number(value) => value.as_str(),
31        Json::String(value) if allow_string => value.as_str(),
32        _ => return Err("SC9 values must be literal numbers"),
33    };
34    let (numerator, denominator) = decimal_rational(token)?;
35    if numerator % denominator != 0 {
36        return Err("SC9 geometry must use integer millimetres");
37    }
38    i32::try_from(numerator / denominator).map_err(|_| "SC9 integer range")
39}

41fn vector3(value: &Json, allow_string: bool) -> Result<[i32; 3], &'static str> {
42    let [x, y, z] = value.items()? else {
43        return Err("SC9 point must have three coordinates");
44    };
45    Ok([
46        integer(x, allow_string)?,
47        integer(y, allow_string)?,
48        integer(z, allow_string)?,
49    ])
50}

52fn box_dimensions(length: i32, width: i32, height: i32) -> Result<[i32; 3], &'static str> {
53    if !(1..=64).contains(&length) || !(1..=64).contains(&width) || !(1..=64).contains(&height) {
54        return Err("SC9 box dimensions outside native range");
55    }
56    Ok([length, width, height])
57}

59fn direct_box(command: &Json) -> Result<BoxGeometry, &'static str> {
60    command.keys(&["command", "params"])?;
61    if text(command.field("command")?)? != "solid.box" {
62        return Err("SC9 requires solid.box first");
63    }
64    let params = command.field("params")?;
65    params.keys(&["corner", "height", "length", "name", "operation", "width"])?;
66    if text(params.field("operation")?)? != "new" {
67        return Err("SC9 requires a new base box");
68    }
69    let _ = text(params.field("name")?)?;
70    let lo = vector3(params.field("corner")?, false)?;
71    let size = box_dimensions(
72        integer(params.field("length")?, false)?,
73        integer(params.field("width")?, false)?,
74        integer(params.field("height")?, false)?,
75    )?;
76    validate_box(BoxGeometry { lo, size })
77}
78
79fn validate_box(box_: BoxGeometry) -> Result<BoxGeometry, &'static str> {
80    let [x, y, z] = box_.lo;
81    let [length, width, _] = box_.size;
82    if z != 0
83        || !(-32..=32).contains(&x)
84        || !(-32..=32).contains(&y)
85        || x.checked_add(length).is_none_or(|v| v > 64)
86        || y.checked_add(width).is_none_or(|v| v > 64)
87    {
88        return Err("SC9 base box must be a bounded origin-based XY prism");
89    }
90    Ok(box_)
91}

93fn direct_counterbore(command: &Json) -> Result<CounterboreGeometry, &'static str> {
94    command.keys(&["command", "params"])?;
95    if text(command.field("command")?)? != "solid.hole" {
96        return Err("SC9 requires one solid.hole after the box");
97    }
98    let params = command.field("params")?;
99    params.keys(&[
100        "cb_depth",
101        "cb_diameter",
102        "diameter",
103        "name",
104        "position",
105        "type",
106    ])?;
107    if text(params.field("type")?)? != "counterbore" {
108        return Err("SC9 supports native counterbore holes only");
109    }
110    let _ = text(params.field("name")?)?;
111    let [x, y, z] = vector3(params.field("position")?, false)?;
112    Ok(CounterboreGeometry {
113        center: [x, y],
114        pilot_diameter: integer(params.field("diameter")?, false)?,
115        cb_diameter: integer(params.field("cb_diameter")?, false)?,
116        cb_depth: integer(params.field("cb_depth")?, false)?,
117        opening_z: z,
118        from_top: z != 0,
119    })
120}

122fn timeline_feature<'a>(command: &'a Json, expected: &str) -> Result<&'a Json, &'static str> {
123    command.keys(&["command", "params"])?;
124    if text(command.field("command")?)? != "timeline.add" {
125        return Err("SC9 accepts timeline.add commands only");
126    }
127    let params = command.field("params")?;
128    params.keys(&["feature"])?;
129    let feature = params.field("feature")?;
130    if text(feature.field("type")?)? != expected {
131        return Err("SC9 feature sequence type mismatch");
132    }
133    Ok(feature)
134}

136fn timeline_box(feature: &Json) -> Result<BoxGeometry, &'static str> {
137    feature.keys(&[
138        "body_names",
139        "corner",
140        "height",
141        "id",
142        "length",
143        "name",
144        "operation",
145        "param_names",
146        "suppressed",
147        "type",
148        "width",
149    ])?;
150    if integer(feature.field("id")?, false)? != 1
151        || feature.field("suppressed")? != &Json::Bool(false)
152        || text(feature.field("operation")?)? != "new_body"
153        || feature
154            .field("param_names")?
155            .items()?
156            .iter()
157            .map(text)
158            .collect::<Result<Vec<_>, _>>()?
159            != ["d1", "d2", "d3"]
160    {
161        return Err("SC9 requires one canonical unsuppressed initial box");
162    }
163    let bodies = feature.field("body_names")?.items()?;
164    if bodies.len() != 1 || text(&bodies[0])?.is_empty() {
165        return Err("SC9 base feature must create exactly one body");
166    }
167    let _ = text(feature.field("name")?)?;
168    validate_box(BoxGeometry {
169        lo: vector3(feature.field("corner")?, false)?,
170        size: box_dimensions(
171            integer(feature.field("length")?, true)?,
172            integer(feature.field("width")?, true)?,
173            integer(feature.field("height")?, true)?,
174        )?,
175    })
176}

178fn timeline_counterbore(feature: &Json) -> Result<CounterboreGeometry, &'static str> {
179    feature.keys(&[
180        "diameter",
181        "direction",
182        "face_names",
183        "hole",
184        "id",
185        "name",
186        "param_names",
187        "position",
188        "suppressed",
189        "type",
190    ])?;
191    if integer(feature.field("id")?, false)? != 2
192        || feature.field("suppressed")? != &Json::Bool(false)
193        || feature
194            .field("param_names")?
195            .items()?
196            .iter()
197            .map(text)
198            .collect::<Result<Vec<_>, _>>()?
199            != ["d4", "d5", "d6"]
200    {
201        return Err("SC9 requires one unsuppressed canonical counterbore feature");
202    }
203    let _ = text(feature.field("name")?)?;
204    let hole = feature.field("hole")?;
205    hole.keys(&["cb_depth", "cb_diameter", "type"])?;
206    if text(hole.field("type")?)? != "counterbore" {
207        return Err("SC9 supports native counterbore holes only");
208    }
209    let faces = feature.field("face_names")?.items()?;
210    if faces.len() != 1 {
211        return Err("SC9 counterbore must select one box end face");
212    }
213    let [x, y, z] = vector3(feature.field("position")?, false)?;
214    let direction = vector3(feature.field("direction")?, false)?;
215    let from_top = match (text(&faces[0])?, direction) {
216        ("F1:+z", [0, 0, -1]) => true,
217        ("F1:-z", [0, 0, 1]) => false,
218        _ => return Err("SC9 counterbore face and direction are inconsistent"),
219    };
220    Ok(CounterboreGeometry {
221        center: [x, y],
222        pilot_diameter: integer(feature.field("diameter")?, true)?,
223        cb_diameter: integer(hole.field("cb_diameter")?, true)?,
224        cb_depth: integer(hole.field("cb_depth")?, true)?,
225        opening_z: z,
226        from_top,
227    })
228}

230fn make_recipe(box_: BoxGeometry, bore: CounterboreGeometry) -> Result<Json, &'static str> {
231    let [x0, y0, z0] = box_.lo;
232    let [length, width, height] = box_.size;
233    let x1 = x0
234        .checked_add(length)
235        .ok_or("SC9 box coordinate overflow")?;
236    let y1 = y0.checked_add(width).ok_or("SC9 box coordinate overflow")?;
237    if z0 != 0
238        || bore.pilot_diameter <= 0
239        || bore.pilot_diameter % 2 != 0
240        || bore.cb_diameter <= 0
241        || bore.cb_diameter % 2 != 0
242    {
243        return Err("SC9 diameters must be positive even integer millimetres");
244    }
245    let r = bore.pilot_diameter / 2;
246    let big_r = bore.cb_diameter / 2;
247    if big_r <= r || bore.cb_depth <= 0 || bore.cb_depth >= height {
248        return Err("SC9 counterbore radius and depth are outside the native domain");
249    }
250    let guard = big_r.checked_mul(2).ok_or("SC9 guard overflow")?;
251    let [cx, cy] = bore.center;
252    if i64::from(cx) - i64::from(guard) <= i64::from(x0)
253        || i64::from(cx) + i64::from(guard) >= i64::from(x1)
254        || i64::from(cy) - i64::from(guard) <= i64::from(y0)
255        || i64::from(cy) + i64::from(guard) >= i64::from(y1)
256    {
257        return Err("SC9 counterbore guard must lie strictly inside the box profile");
258    }
259    let opening = if bore.from_top {
260        if bore.opening_z != height {
261            return Err("SC9 top counterbore is not on the top face");
262        }
263        "end"
264    } else {
265        if bore.opening_z != 0 {
266            return Err("SC9 bottom counterbore is not on the bottom face");
267        }
268        "start"
269    };
270    let points = [[x0, y0], [x1, y0], [x1, y1], [x0, y1]];
271    let vertices = points
272        .into_iter()
273        .enumerate()
274        .map(|(i, xy)| {
275            object([
276                ("id", Json::String(format!("sc9_v{i}"))),
277                (
278                    "xy_mm",
279                    Json::Array(xy.map(|n| Json::Number(n.to_string())).into()),
280                ),
281            ])
282        })
283        .collect();
284    let recipe = object([
285        ("family", Json::String("polygon_counterbores_v1".into())),
286        (
287            "outer",
288            object([
289                ("id", Json::String("sc9_outer".into())),
290                ("vertices", Json::Array(vertices)),
291            ]),
292        ),
293        (
294            "bores",
295            Json::Array(vec![object([
296                ("id", Json::String("hole_2".into())),
297                (
298                    "center_mm",
299                    Json::Array([cx, cy].map(|n| Json::Number(n.to_string())).into()),
300                ),
301                ("radius_mm", Json::Number(r.to_string())),
302                ("guard_mm", Json::Number(guard.to_string())),
303                ("counterbore_radius_mm", Json::Number(big_r.to_string())),
304                ("depth_mm", Json::Number(bore.cb_depth.to_string())),
305                ("opening", Json::String(opening.into())),
306            ])]),
307        ),
308        ("height_mm", Json::Number(height.to_string())),
309        (
310            "top_shift_mm",
311            Json::Array(vec![Json::Number("0".into()), Json::Number("0".into())]),
312        ),
313    ]);
314    interchange::build(&recipe)
315}

317/// Import a literal script containing exactly one base box and one counterbore.
318pub fn import_counterbore_direct(script: &Json) -> Result<Json, &'static str> {
319    script.keys(&["commands"])?;
320    let [box_command, hole_command] = script.field("commands")?.items()? else {
321        return Err("SC9 direct script requires exactly one box and one counterbore");
322    };
323    make_recipe(direct_box(box_command)?, direct_counterbore(hole_command)?)
324}

326/// Import a saved SolidCraft 0.48.1 box plus native counterbore feature history.
327pub fn import_counterbore_timeline(script: &Json) -> Result<Json, &'static str> {
328    script.keys(&["about", "commands"])?;
329    let about = script.field("about")?;
330    about.keys(&["design", "format", "replay", "version"])?;
331    if text(about.field("design")?)?.is_empty()
332        || text(about.field("format")?)? != "solidcraft/66"
333        || text(about.field("version")?)? != "0.48.1"
334        || text(about.field("replay")?)?.is_empty()
335    {
336        return Err("unsupported SC9 SolidCraft history metadata");
337    }
338    let commands = script.field("commands")?.items()?;
339    if commands.len() != 2 {
340        return Err("SC9 timeline requires one box and one counterbore");
341    }
342    make_recipe(
343        timeline_box(timeline_feature(&commands[0], "box")?)?,
344        timeline_counterbore(timeline_feature(&commands[1], "hole")?)?,
345    )
346}

353    fn real_host_top_and_bottom_histories_map_to_g23() {
354        for (name, expected_center, opening, shoulder, feature) in [
355            ("centered_top", [0, 0], "end", 8, "bore/hole_2/shoulder"),
356            ("decentered_top", [4, -3], "end", 8, "bore/hole_2/shoulder"),
357            (
358                "opposite_bottom",
359                [-5, 6],
360                "start",
361                4,
362                "bore/hole_2/shoulder",
363            ),
364        ] {
365            let source = match name {
366                "centered_top" => crate::json::parse(include_str!(
367                    "../tests/references/solidcraft-counterbore/centered_top.history.json"
368                )),
369                "decentered_top" => crate::json::parse(include_str!(
370                    "../tests/references/solidcraft-counterbore/decentered_top.history.json"
371                )),
372                _ => crate::json::parse(include_str!(
373                    "../tests/references/solidcraft-counterbore/opposite_bottom.history.json"
374                )),
375            }
376            .unwrap();
377            let model = import_counterbore_timeline(&source).unwrap();
378            let bore = &model.field("bores").unwrap().items().unwrap()[0];
379            assert_eq!(
380                bore.field("center_mm").unwrap(),
381                &Json::Array(expected_center.map(|v| Json::Number(v.to_string())).into())
382            );
383            assert_eq!(
384                bore.field("opening").unwrap(),
385                &Json::String(opening.into())
386            );
387            assert_eq!(bore.field("depth_mm").unwrap(), &Json::Number("4".into()));
388            assert!(model
389                .field("design_faces")
390                .unwrap()
391                .items()
392                .unwrap()
393                .iter()
394                .any(|face| face.field("feature").unwrap() == &Json::String(feature.into())));
395            assert_eq!(
396                model.field("schema").unwrap(),
397                &Json::String("polygon_counterbores_v1".into())
398            );
399            assert_eq!(shoulder, if opening == "end" { 8 } else { 4 });
400        }
