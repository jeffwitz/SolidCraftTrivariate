2//! Detailed exact extraction is an optional export. A solve always rebuilds

12type Result<T> = std::result::Result<T, &'static str>;
13fn n(v: impl ToString) -> Json {
14    Json::Number(v.to_string())
15}
16fn string(s: impl Into<String>) -> Json {
17    Json::String(s.into())

20fn mount(model: &Json, plan: &Json) -> Result<ThbStrip> {
21    plan.keys(&["schema", "start", "end"])?;
22    if plan.field("schema")?.text()? != "interior_thb_refinement_v1" {
23        return Err("unsupported hierarchical refinement plan");
24    }
25    let index = |key: &str| -> Result<[usize; 3]> {
26        let [x, y, z] = plan.field(key)?.items()? else {
27            return Err("refinement bounds need three cell indices");
28        };
29        Ok([x.number()?, y.number()?, z.number()?])
30    };
31    build_thb_strip_for_cells(model, index("start")?, index("end")?)
32}
33fn summary(model: &Json, field: &ThbStrip) -> Result<Json> {
34    let support = recover_nurbs_strip_parent(model)?;
35    let mut sources = Vec::new();
36    for leaf in &field.leaves {
37        let mut found = None;
38        for span in &support.source_spans {
39            let mut inside = true;
40            for (axis, [a, b]) in span.restriction.intervals.into_iter().enumerate() {
41                inside &= !leaf.intervals[axis][0]
42                    .cmp_value(&Ratio::new(i128::from(a), 1)?)
43                    .is_lt()
44                    && !leaf.intervals[axis][1]
45                        .cmp_value(&Ratio::new(i128::from(b), 1)?)
46                        .is_gt();
47            }
48            if inside && found.replace(span).is_some() {
49                return Err("ambiguous integration leaf lineage");
50            }
51        }
52        sources.push(found.ok_or("leaf crosses a construction span")?);
53    }
54    let mut boundary = Vec::new();
55    let mut covered_faces = std::collections::BTreeSet::new();
56    for &(i, axis, upper) in &field.boundary {
57        let span = sources[i];
58        let local = span
59            .restriction
60            .axes
61            .iter()
62            .position(|&a| a == axis)
63            .ok_or("missing leaf boundary axis")?;
64        let side = format!(
65            "{}{}",
66            ["u", "v", "w"][local],
67            usize::from(upper ^ span.restriction.reversals[local])
68        );
69        let design = model
70            .field("design_faces")?
71            .items()?
72            .iter()
73            .find(|f| {
74                f.field("block").and_then(Json::text) == Ok(span.source_block_id.as_str())
75                    && f.field("side").and_then(Json::text) == Ok(side.as_str())
76            })
77            .ok_or("missing source design face")?;
78        covered_faces.insert((span.source_block_id.clone(), side));
79        boundary.push(object([
80            ("leaf", string(field.leaves[i].id.clone())),
81            ("leaf_axis", n(axis)),
82            ("upper", Json::Bool(upper)),
83            ("design_face", design.clone()),
84        ]));
85    }
86    if covered_faces.len() != model.field("design_faces")?.items()?.len() {
87        return Err("local refinement lost a source design face");
88    }
89    let fine = field.leaves.iter().filter(|c| c.level == 1).count();
90    Ok(object([
91        ("profile", string("interior_dyadic_thb_cubic_v1")),
92        ("support_id", string(support.id)),
93        (
94            "parent_geometry_controls",
95            n(support.homogeneous_controls.len()),
96        ),
97        ("initial_scalar_dofs", n(field.initial_ndof)),
98        ("prepared_scalar_dofs", n(field.prepared_ndof)),
99        ("local_scalar_dofs", n(field.ndof)),
100        ("uniform_fine_scalar_dofs", n(field.uniform_fine_ndof)),
101        ("prepared_transfer_rank", n(field.transfer_rank)),
102        (
103            "local_basis_rank_in_uniform_fine_space",
104            n(field.uniform_fine_rank),
105        ),
106        ("fine_active", n(field.fine_active_count)),
107        ("coarse_active", n(field.coarse_active_count)),
108        ("coarse_leaves", n(field.leaves.len() - fine)),
109        ("fine_leaves", n(fine)),
110        (
111            "common_partition_verified",
112            Json::Bool(field.common_partition_verified),
113        ),
114        (
115            "integration_sources",
116            Json::Array(
117                field
118                    .leaves
119                    .iter()
120                    .zip(sources)
121                    .map(|(l, s)| {
122                        Json::Array(vec![
123                            string(l.id.clone()),
124                            string(s.source_block_id.clone()),
125                        ])
126                    })
127                    .collect(),
128            ),
129        ),
130        ("boundary_sources", Json::Array(boundary)),
131    ]))
132}
133/// A field refusal never changes the validity status of the source geometry.
134pub fn refusal(model: &Json, reason: &str) -> Json {
135    let validity = crate::interchange::audit(model)
136        .ok()
137        .and_then(|r| r.field("status").ok().cloned())
138        .unwrap_or_else(|| string("unverified"));
139    let outside = model.field("schema").and_then(Json::text) != Ok("nurbs_sketch_strip_v1");
140    let status = if reason.contains("overflow")
141        || reason.contains("budget")
142        || reason.contains("uncertified")
143    {
144        "indeterminate"
145    } else if outside
146        || reason.starts_with("THB marked")
147        || reason.contains("requires a cubic")
148        || reason.contains("nonuniform")
149        || reason.contains("repeated source")
150    {
151        "unsupported_profile"
152    } else {
153        "error"
154    };
155    object([
156        ("status", string(status)),
157        ("source_validity", validity),
158        ("field_eligibility", string(status)),
159        ("reason", string(reason)),
160    ])
161}

163pub fn compile(model: &Json, plan: &Json) -> Result<Json> {
164    let field = mount(model, plan)?;
165    Ok(object([
166        ("schema", string("local_thb_analysis_v1")),
167        ("geometry", model.clone()),
168        ("plan", plan.clone()),
169        ("field", summary(model, &field)?),
170    ]))
171}
172pub fn replay(package: &Json) -> Result<ThbStrip> {
173    package.keys(&["schema", "geometry", "plan", "field"])?;
174    if package.field("schema")?.text()? != "local_thb_analysis_v1" {
175        return Err("invalid local THB package schema");
176    }
177    let geometry = package.field("geometry")?;
178    let field = mount(geometry, package.field("plan")?)?;
179    if package.field("field")? != &summary(geometry, &field)? {
180        return Err("hierarchical field or source-face lineage differs from replay");
181    }
182    Ok(field)
183}
184pub fn solve(package: &Json, mode: &str, job: &Json) -> Result<Json> {
185    let field = replay(package)?;
186    let (count, lift) = match mode {
187        "prepared" => (field.prepared_ndof, field.prepared_zero_bottom_lift()?),
188        "local" => {
189            let bottom: Vec<_> = field
190                .boundary
191                .iter()
192                .filter(|(_, axis, upper)| *axis == 2 && !upper)
193                .copied()
194                .collect();
195            (field.ndof, field.zero_face_lift(&bottom)?)
196        }
197        _ => return Err("expected prepared or local THB space"),
198    };
199    let domain = IntegrationDomain {
200        scalar_dofs: count,
201        cell_count: field.leaves.len(),
202        z_bounds: field.z_bounds,
203        boundary: &field.boundary,
204        essential: &lift,
205    };
206    let mut result = solve_sampled(&domain, job, |i, p| {
207        if mode == "local" {
208            field.sample(i, p)
209        } else {
210            field.sample_prepared(i, p)
211        }
212    })?;
213    if let Json::Object(fields) = &mut result {
214        fields.insert("field_space".into(), string(mode));
215    }
216    Ok(result)
217}
218/// Restrict an assembled block matrix and force to unconstrained scalar roots
219/// of a homogeneous, whole-feature displacement clamp.
220fn principal_system(
221    system: &crate::extracted_space::NativeAssembly,
222    free: &[usize],
223) -> crate::extracted_space::NativeAssembly {
224    use crate::{elasticity::Stiffness, extracted_space::NativeAssembly};
225    let k = &system.stiffness;
226    let mut offsets = vec![0];
227    let mut columns = Vec::new();
228    let mut values = Vec::new();
229    let mut force = Vec::new();
230    for &i in free {
231        for j in k.offsets[i]..k.offsets[i + 1] {
232            if let Ok(col) = free.binary_search(&k.columns[j]) {
233                columns.push(col);
234                values.push(k.values[j]);
235            }
236        }
237        offsets.push(columns.len());
238        force.extend_from_slice(&system.force[3 * i..3 * i + 3]);
239    }
240    NativeAssembly {
241        stiffness: Stiffness {
242            offsets,
243            columns,
244            values,
245            integrated_volume: k.integrated_volume,
246        },
247        force,
248        quadrature_points: system.quadrature_points,
249        quadrature_order: system.quadrature_order,
250        cell_count: system.cell_count,
251        quadrature_signature: system.quadrature_signature,
252    }
253}

255pub fn galerkin(package: &Json, order: usize) -> Result<Json> {
256    let field = replay(package)?;
257    let load = |x: [f64; 3]| [1. + x[0], x[1] - x[2], x[0] * x[1] + 2.];
258    let fine = assemble_sampled(
259        field.ndof,
260        field.leaves.len(),
261        1.5,
262        1.,
263        order,
264        |i, p| field.sample(i, p),
265        load,
266    )?;
267    let coarse = assemble_sampled(
268        field.prepared_ndof,
269        field.leaves.len(),
270        1.5,
271        1.,
272        order,
273        |i, p| field.sample_prepared(i, p),
274        load,
275    )?;
276    let cmp = compare_assembled(
277        &fine,
278        &coarse,
279        field.ndof,
280        field.prepared_ndof,
281        &field.rational_prepared_to_thb,
282    )?;
283    let bottom: Vec<_> = field
284        .boundary
285        .iter()
286        .copied()
287        .filter(|(_, a, upper)| *a == 2 && !upper)
288        .collect();
289    let coarse_roots = field.prepared_zero_bottom_lift()?.free_roots;
290    let fine_roots = field.zero_face_lift(&bottom)?.free_roots;
291    let mut reduced_transfer = Vec::new();
292    for (i, row) in field.rational_prepared_to_thb.iter().enumerate() {
293        let reduced: std::collections::BTreeMap<_, _> = row
294            .iter()
295            .filter_map(|(&id, &x)| coarse_roots.binary_search(&id).ok().map(|j| (j, x)))
296            .collect();
297        if fine_roots.binary_search(&i).is_ok() {
298            reduced_transfer.push(reduced);
299        } else if !reduced.is_empty() {
300            return Err("transfer violates homogeneous source-face clamp");
301        }
302    }
303    let reduced = compare_assembled(
304        &principal_system(&fine, &fine_roots),
305        &principal_system(&coarse, &coarse_roots),
306        fine_roots.len(),
307        coarse_roots.len(),
308        &reduced_transfer,
309    )?;
310    for check in [&cmp, &reduced] {
311        if check.stiffness_relative_frobenius_defect > 1e-10
312            || check.force_relative_l2_defect > 1e-10
313        {
314            return Err("THB Galerkin pullback exceeds tolerance");
315        }
316    }
317    Ok(object([
318        ("status", string("valid")),
319        ("clamped_transfer_commutes_exactly", Json::Bool(true)),
320        (
321            "clamped_relative_stiffness_defect",
322            n(reduced.stiffness_relative_frobenius_defect),
323        ),
324        (
325            "clamped_relative_load_defect",
326            n(reduced.force_relative_l2_defect),
327        ),
328        ("quadrature", n(order)),
329        ("quadrature_points", n(fine.quadrature_points)),
330        (
331            "relative_stiffness_defect",
332            n(cmp.stiffness_relative_frobenius_defect),
333        ),
334        ("relative_load_defect", n(cmp.force_relative_l2_defect)),
335    ]))
