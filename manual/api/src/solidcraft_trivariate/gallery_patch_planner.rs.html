1//! Deterministic exact NURBS-cover planning for audited gallery volume maps.
2//!
3//! This planner selects compatible source-cell chains and pairs. The output
4//! layout is always rebuilt by `nurbs_tensor_assembly::assemble_cover`; the
5//! planner never accepts or copies proposed control nets.
6

38fn triple(value: &Json) -> Result<[usize; 3]> {
39    value
40        .items()?
41        .iter()
42        .map(Json::number)
43        .collect::<Result<Vec<_>>>()?
44        .try_into()
45        .map_err(|_| "expected three degrees")
46}

48fn side(value: &Json) -> Result<(String, usize, usize)> {
49    let parts = value.items()?;
50    if parts.len() != 2 {
51        return Err("invalid source side");
52    }
53    let name = parts[0].text()?.to_owned();
54    let (axis, endpoint) = side_name(parts[1].text()?)?;
55    Ok((name, axis, endpoint))
56}
57
58fn side_name(name: &str) -> Result<(usize, usize)> {
59    let bytes = name.as_bytes();
60    if bytes.len() != 2 || !b"uvw".contains(&bytes[0]) || !b"01".contains(&bytes[1]) {
61        return Err("invalid source side");
62    }
63    Ok((
64        b"uvw"
65            .iter()
66            .position(|x| *x == bytes[0])
67            .ok_or("invalid side axis")?,
68        (bytes[1] - b'0') as usize,
69    ))
70}

84fn oriented_span(
85    span: &SpanPlan,
86    target: [usize; 3],
87    source: &BTreeMap<String, [usize; 3]>,
88) -> Result<Json> {
89    let source_degree = source.get(&span.source).ok_or("unknown source block")?;
90    let elevation = std::array::from_fn(|axis| {
91        target[axis].saturating_sub(source_degree[span.axis_map[axis]])
92    });
93    Ok(object([
94        ("source_block", text(span.source.clone())),
95        ("axis_map", indices(span.axis_map)),
96        ("flips", bools(span.flips)),
97        ("degree_elevation", indices(elevation)),
98        ("parameter_interval", interval(span.interval)),
99    ]))
100}

102fn permutation_parity(axes: [usize; 3], flips: [bool; 3]) -> bool {
103    let inversions = (0..3)
104        .flat_map(|i| (i + 1..3).map(move |j| (i, j)))
105        .filter(|&(i, j)| axes[i] > axes[j])
106        .count();
107    (inversions + flips.into_iter().filter(|x| *x).count()) % 2 == 0
108}

110fn source_degrees(model: &Json) -> Result<BTreeMap<String, [usize; 3]>> {
111    let mut output = BTreeMap::new();
112    for block in model.field("blocks")?.items()? {
113        let id = block.field("id")?.text()?.to_owned();
114        let degree = triple(block.field("degrees")?)?;
115        if output.insert(id, degree).is_some() {
116            return Err("duplicate source block id");
117        }
118    }
119    if output.is_empty() {
120        return Err("model has no source blocks");
121    }
122    Ok(output)
123}

137fn chain_groups(model: &Json, degrees: &BTreeMap<String, [usize; 3]>) -> Result<Vec<Group>> {
138    let topology = model.field("topology")?;
139    let interfaces = topology.field("interfaces")?.items()?;
140    let mut pair_count = BTreeMap::<(String, String), usize>::new();
141    for iface in interfaces {
142        let (a, _, _) = side(iface.field("first")?)?;
143        let (b, _, _) = side(iface.field("second")?)?;
144        let key = if a <= b { (a, b) } else { (b, a) };
145        *pair_count.entry(key).or_default() += 1;
146    }
147    let mut forward = BTreeMap::<String, String>::new();
148    let mut predecessor = BTreeMap::<String, String>::new();
149    for iface in interfaces {
150        let orientation = iface.field("orientation")?;
151        if ["swap", "flip_first", "flip_second"].iter().any(|key| {
152            orientation
153                .field(key)
154                .ok()
155                .and_then(|x| x.bool().ok())
156                .unwrap_or(false)
157        }) {
158            continue;
159        }
160        let (a, axis_a, end_a) = side(iface.field("first")?)?;
161        let (b, axis_b, end_b) = side(iface.field("second")?)?;
162        let (lo, hi) = if axis_a == 2 && end_a == 1 && axis_b == 2 && end_b == 0 {
163            (a, b)
164        } else if axis_a == 2 && end_a == 0 && axis_b == 2 && end_b == 1 {
165            (b, a)
166        } else {
167            continue;
168        };
169        let pair_key = if lo <= hi {
170            (lo.clone(), hi.clone())
171        } else {
172            (hi.clone(), lo.clone())
173        };
174        if pair_count.get(&pair_key).copied().unwrap_or(0) != 1 {
175            continue;
176        }
177        let Some(dl) = degrees.get(&lo) else { continue };
178        let Some(dh) = degrees.get(&hi) else { continue };
179        if dl != dh {
180            continue;
181        }
182        if forward.contains_key(&lo) || predecessor.contains_key(&hi) {
183            continue;
184        }
185        forward.insert(lo.clone(), hi.clone());
186        predecessor.insert(hi, lo);
187    }
188    let mut result = Vec::new();
189    let starts: Vec<_> = forward
190        .keys()
191        .filter(|id| !predecessor.contains_key(*id))
192        .cloned()
193        .collect();
194    for start in starts {
195        let mut ids = vec![start.clone()];
196        let mut current = start;
197        let mut seen = BTreeSet::from([current.clone()]);
198        while let Some(next) = forward.get(&current) {
199            if !seen.insert(next.clone()) {
200                break;
201            }
202            ids.push(next.clone());
203            current = next.clone();
204        }
205        if ids.len() < 2 {
206            continue;
207        }
208        let spans = ids
209            .iter()
210            .enumerate()
211            .map(|(i, source)| SpanPlan {
212                source: source.clone(),
213                axis_map: [2, 0, 1],
214                flips: [false; 3],
215                interval: [i, i + 1],
216            })
217            .collect();
218        result.push(Group {
219            id: format!("axial:{}", ids.join("+")),
220            spans,
221        });
222    }
223    Ok(result)
224}

226fn pair_group(
227    a: &str,
228    b: &str,
229    iface: &Json,
230    tangent_swap_a: bool,
231    tangent_flips_a: [bool; 2],
232) -> Option<Group> {
233    let (_, normal_a, end_a) = side(iface.field("first").ok()?).ok()?;
234    let (_, normal_b, end_b) = side(iface.field("second").ok()?).ok()?;
235    let orientation = iface.field("orientation").ok();
236    let flag = |key: &str| {
237        orientation
238            .and_then(|o| o.field(key).ok())
239            .and_then(|v| v.bool().ok())
240            .unwrap_or(false)
241    };
242    let swap = flag("swap");
243    let face_flips = [flag("flip_first"), flag("flip_second")];
244    let mut tangents_a: Vec<_> = (0..3).filter(|x| *x != normal_a).collect();
245    if tangent_swap_a {
246        tangents_a.swap(0, 1);
247    }
248    let tangents_b: Vec<_> = tangents_a
249        .iter()
250        .map(|axis| {
251            let index_a = (0..3).filter(|x| *x != normal_a).position(|x| x == *axis)?;
252            let index_b = if swap { 1 - index_a } else { index_a };
253            (0..3).filter(|x| *x != normal_b).nth(index_b)
254        })
255        .collect::<Option<Vec<_>>>()?;
256    let axes_a = [normal_a, tangents_a[0], tangents_a[1]];
257    let axes_b = [normal_b, tangents_b[0], tangents_b[1]];
258    let flips_a = [end_a == 0, tangent_flips_a[0], tangent_flips_a[1]];
259    let flips_b = [
260        end_b == 1,
261        tangent_flips_a[0]
262            ^ face_flips
263                [usize::from(tangents_a[0] == (0..3).filter(|x| *x != normal_a).nth(1).unwrap())],
264        tangent_flips_a[1]
265            ^ face_flips
266                [usize::from(tangents_a[1] == (0..3).filter(|x| *x != normal_a).nth(1).unwrap())],
267    ];
268    if !permutation_parity(axes_a, flips_a) || !permutation_parity(axes_b, flips_b) {
269        return None;
270    }
271    Some(Group {
272        id: format!("band:{a}+{b}"),
273        spans: vec![
274            SpanPlan {
275                source: a.to_owned(),
276                axis_map: axes_a,
277                flips: flips_a,
278                interval: [0, 1],
279            },
280            SpanPlan {
281                source: b.to_owned(),
282                axis_map: axes_b,
283                flips: flips_b,
284                interval: [1, 2],
285            },
286        ],
287    })
288}

290fn group_degrees(group: &Group, source: &BTreeMap<String, [usize; 3]>) -> Result<[usize; 3]> {
291    let mut out = [1; 3];
292    for span in &group.spans {
293        let degree = source.get(&span.source).ok_or("unknown source block")?;
294        for axis in 0..3 {
295            out[axis] = out[axis].max(degree[span.axis_map[axis]]);
296        }
297    }
298    Ok(out)
299}

301fn subface(span: &SpanPlan, patch_id: &str, side_name: &str) -> Result<Json> {
302    let bytes = side_name.as_bytes();
303    if bytes.len() != 2 || !b"uvw".contains(&bytes[0]) || !b"01".contains(&bytes[1]) {
304        return Err("invalid interface side");
305    }
306    let source_axis = b"uvw".iter().position(|x| *x == bytes[0]).unwrap();
307    let patch_axis = span
308        .axis_map
309        .iter()
310        .position(|x| *x == source_axis)
311        .ok_or("invalid axis map")?;
312    let endpoint = (bytes[1] - b'0') as usize ^ usize::from(span.flips[patch_axis]);
313    Ok(object([
314        ("patch_id", text(patch_id.to_owned())),
315        ("source_block", text(span.source.clone())),
316        ("side", text(side_name.to_owned())),
317        (
318            "patch_side",
319            text(format!("{}{endpoint}", b"uvw"[patch_axis] as char)),
320        ),
321        ("axis_map", indices(span.axis_map)),
322        ("flips", bools(span.flips)),
323        ("parameter_interval", interval(span.interval)),
324    ]))
325}

327fn cover_layout(
328    model: &Json,
329    groups: &[Group],
330    degrees: &BTreeMap<String, [usize; 3]>,
331) -> Result<Json> {
332    let mut owner = BTreeMap::<String, (&str, &SpanPlan)>::new();
333    let mut patch_values = Vec::new();
334    for group in groups {
335        let mut source_ids = Vec::new();
336        for span in &group.spans {
337            source_ids.push(text(span.source.clone()));
338            if owner
339                .insert(span.source.clone(), (group.id.as_str(), span))
340                .is_some()
341            {
342                return Err("source block assigned more than once");
343            }
344        }
345        let degree = group_degrees(group, degrees)?;
346        let spans = group
347            .spans
348            .iter()
349            .map(|span| oriented_span(span, degree, degrees))
350            .collect::<Result<Vec<_>>>()?;
351        let patch_kind = if group.spans.len() == 1 {
352            "single_bezier"
353        } else if group.id.starts_with("axial:") {
354            "multi_span_nurbs"
355        } else {
356            "two_span_nurbs"
357        };
358        let mut patch = object([
359            ("id", text(group.id.clone())),
360            ("source_blocks", array(source_ids.clone())),
361            ("cells", array(source_ids)),
362            ("patch_kind", text(patch_kind)),
363            ("degrees", indices(degree)),
364            ("spans", array(spans)),
365        ]);
366        if let Json::Object(fields) = &mut patch {
367            if group.spans.len() > 1 {
368                fields.insert("exact_piece_extraction_expected".into(), Json::Bool(true));
369            }
370        }
371        patch_values.push(patch);
372    }
373    if owner.len() != degrees.len() {
374        return Err("patch layout does not cover every source block");
375    }
376    let mut external = Vec::new();
377    let mut self_seams = Vec::new();
378    for iface in model.field("topology")?.field("interfaces")?.items()? {
379        let (a, _, _) = side(iface.field("first")?)?;
380        let (b, _, _) = side(iface.field("second")?)?;
381        let (patch_a, span_a) = *owner.get(&a).ok_or("missing source span")?;
382        let (patch_b, span_b) = *owner.get(&b).ok_or("missing source span")?;
383        let link = object([
384            (
385                "first",
386                subface(span_a, patch_a, iface.field("first")?.items()?[1].text()?)?,
387            ),
388            (
389                "second",
390                subface(span_b, patch_b, iface.field("second")?.items()?[1].text()?)?,
391            ),
392            ("orientation", iface.field("orientation")?.clone()),
393            ("status", text("exact_match")),
394        ]);
395        if patch_a == patch_b {
396            let (id1, _, _) = side(iface.field("first")?)?;
397            let (id2, _, _) = side(iface.field("second")?)?;
398            let side1 = iface.field("first")?.items()?[1].text()?;
399            let side2 = iface.field("second")?.items()?[1].text()?;
400            if !same_grid_face(owner[&id1].1, side1, owner[&id2].1, side2)? {
401                self_seams.push(link);
402            }
403        } else {
404            external.push(link);
405        }
406    }
407    let mut boundaries = Vec::new();
408    for face in model.field("topology")?.field("exterior")?.items()? {
409        let (id, _, _) = side(face)?;
410        let span = owner.get(&id).ok_or("missing exterior source span")?;
411        boundaries.push(subface(span.1, span.0, face.items()?[1].text()?)?);
412    }
413    let model_schema = model.field("schema")?.clone();
414    let mut fields = vec![
415        ("schema", text("gallery_exact_patch_map_v1")),
416        ("method", text("deterministic_exact_pair_and_axial_chain_planning")),
417        ("scope_note", text("Exact extracted cells assembled into disjoint C0 tensor-product NURBS covers. The planner is deterministic but does not claim a globally minimum patch count, global injectivity, or material coverage beyond the audited source model.")),
418        ("source_schema", model_schema),
419        ("source_block_count", num(degrees.len())),
420        ("patches", array(patch_values)),
421        ("interfaces", array(external)),
422        ("boundary_faces", array(boundaries)),
423        ("self_seams", array(self_seams)),
424    ];
425    for key in [
426        "source_model",
427        "source_recipe",
428        "case_id",
429        "source_model_sha256",
430        "source_recipe_sha256",
431    ] {
432        if let Ok(value) = model.field(key) {
433            fields.push((key, value.clone()));
434        }
435    }
436    Ok(Json::Object(
437        fields
438            .into_iter()
439            .map(|(key, value)| (key.to_owned(), value))
440            .collect(),
441    ))
442}

444fn same_grid_face(a: &SpanPlan, side_a: &str, b: &SpanPlan, side_b: &str) -> Result<bool> {
445    let grid = |span: &SpanPlan, name: &str| -> Result<[[usize; 2]; 3]> {
446        let (axis, source_endpoint) = side_name(name)?;
447        let patch_axis = span
448            .axis_map
449            .iter()
450            .position(|x| *x == axis)
451            .ok_or("invalid map")?;
452        let endpoint = source_endpoint ^ usize::from(span.flips[patch_axis]);
453        let mut face = [[0, 1]; 3];
454        face[0] = span.interval;
455        face[patch_axis] = [span.interval[endpoint]; 2];
456        Ok(face)
457    };
458    let x = grid(a, side_a)?;
459    let y = grid(b, side_b)?;
460    Ok(x == y)
461}

463/// Plan a deterministic cover and use the native exact assembler as its feasibility check.
464pub fn plan(model: &Json) -> Result<Json> {
465    crate::interchange::audit(model)?;
466    if model.field("schema").ok().and_then(|x| x.text().ok())
467        == Some("box_cylinder_boolean_suite_v1")
468        && model.field("operation").ok().and_then(|x| x.text().ok()) == Some("cylinder")
469    {
470        // The tall cylinder has a separate periodic-cover contract. Reuse its
471        // explicitly constructed two-patch cover rather than pretending that
472        // a greedy pair cover solves the periodic closure problem.
473        return crate::cylinder_tensor_covers::build_cover(model, 2)
474            .map_err(|_| "custom tall-cylinder cover rejected");
475    }
476    let degrees = source_degrees(model)?;
477    let mut groups = chain_groups(model, &degrees)?;
478    let consumed: BTreeSet<_> = groups
479        .iter()
480        .flat_map(|g| g.spans.iter().map(|s| s.source.clone()))
481        .collect();
482    for id in degrees.keys().filter(|id| !consumed.contains(*id)) {
483        groups.push(single_group(id));
484    }
485    groups.sort_by(|a, b| a.id.cmp(&b.id));
486    // The source topology is the authority for the initial one-span cover.
487    let baseline = cover_layout(model, &groups, &degrees)?;
488    crate::nurbs_tensor_assembly::assemble_cover(model, &baseline)?;
489
490    let interfaces = model.field("topology")?.field("interfaces")?.items()?;
491    let mut counts = BTreeMap::<(String, String), usize>::new();
492    for iface in interfaces {
493        let (a, _, _) = side(iface.field("first")?)?;
494        let (b, _, _) = side(iface.field("second")?)?;
495        let key = if a <= b { (a, b) } else { (b, a) };
496        *counts.entry(key).or_default() += 1;
497    }
498    let chain_members: BTreeSet<_> = groups
499        .iter()
500        .filter(|g| g.spans.len() > 1)
501        .flat_map(|g| g.spans.iter().map(|s| s.source.clone()))
502        .collect();
503    for iface in interfaces {
504        let (a, _, _) = side(iface.field("first")?)?;
505        let (b, _, _) = side(iface.field("second")?)?;
506        if a == b || chain_members.contains(&a) || chain_members.contains(&b) {
507            continue;
508        }
509        let key = if a <= b {
510            (a.clone(), b.clone())
511        } else {
512            (b.clone(), a.clone())
513        };
514        if counts.get(&key).copied().unwrap_or(0) != 1 {
515            continue;
516        }
517        let Some(ga) = groups
518            .iter()
519            .position(|g| g.spans.len() == 1 && g.spans[0].source == a)
520        else {
521            continue;
522        };
523        let Some(gb) = groups
524            .iter()
525            .position(|g| g.spans.len() == 1 && g.spans[0].source == b)
526        else {
527            continue;
528        };
529        let mut accepted = None;
530        'orient: for swap_a in [false, true] {
531            for mask in 0..4 {
532                let group = pair_group(&a, &b, iface, swap_a, [mask & 1 != 0, mask & 2 != 0]);
533                let Some(group) = group else { continue };
534                let mut candidate = groups.clone();
535                let high = ga.max(gb);
536                let low = ga.min(gb);
537                candidate.remove(high);
538                candidate.remove(low);
539                candidate.push(group.clone());
540                candidate.sort_by(|x, y| x.id.cmp(&y.id));
541                if let Ok(layout) = cover_layout(model, &candidate, &degrees) {
542                    if crate::nurbs_tensor_assembly::assemble_cover(model, &layout).is_ok() {
543                        accepted = Some(candidate);
544                        break 'orient;
545                    }
546                }
547            }
548        }
549        if let Some(candidate) = accepted {
550            groups = candidate;
551        }
552    }
553    groups.sort_by(|a, b| a.id.cmp(&b.id));
554    let layout = cover_layout(model, &groups, &degrees)?;
555    crate::nurbs_tensor_assembly::assemble_cover(model, &layout)
556}
