1//! Four-block rational chart for a rectangular prism with an axial round bore.
2//!
3//! Each chart spans one side of the rectangle at `v = 0` and one quarter of
4//! the circular bore at `v = 1`. The angular join points are the exact
5//! rational points `(3, 4) / 5` and its quarter turns. The alternating
6//! endpoint weights make the radial traces identical at the four joins.
7
8use crate::bracket::BracketBlock;
9use std::convert::TryFrom;
10
11type HPoint = [i128; 3]; // Two homogeneous section coordinates and a weight.
12
13fn checked_i64(value: i128) -> Result<i64, &'static str> {
14    i64::try_from(value).map_err(|_| "compact wing control exceeds i64")
15}
16
17fn side_corners(bounds: [[i32; 2]; 3], plane: [usize; 2], side: usize) -> ([i128; 2], [i128; 2]) {
18    let [a, b] = plane;
19    let (lo_a, hi_a) = (i128::from(bounds[a][0]), i128::from(bounds[a][1]));
20    let (lo_b, hi_b) = (i128::from(bounds[b][0]), i128::from(bounds[b][1]));
21    // For the counter-clockwise ring the arc sectors are north, west, south,
22    // east; their endpoints follow the outer perimeter counter-clockwise.
23    let ccw = [
24        ([hi_a, hi_b], [lo_a, hi_b]),
25        ([lo_a, hi_b], [lo_a, lo_b]),
26        ([lo_a, lo_b], [hi_a, lo_b]),
27        ([hi_a, lo_b], [hi_a, hi_b]),
28    ];
29    ccw[side]
30}

32/// Build the four exact rational charts of a rectangular prism with one
33/// circular through-bore. `axis` selects the extrusion coordinate; the other
34/// two coordinates form the circular section. The bore must lie strictly
35/// inside that rectangle. The returned maps have degrees `[2, 1, 1]`, with
36/// `u` circumferential, `v=0` on the rectangle and `v=1` on the circle, and
37/// `w` along the extrusion axis.
38pub(crate) fn build(
39    axis: usize,
40    bounds: [[i32; 2]; 3],
41    center: [i32; 3],
42    radius: i32,
43    id: &str,
44) -> Result<Vec<BracketBlock>, &'static str> {
45    if axis > 2 || radius <= 0 || id.is_empty() {
46        return Err("compact_wing_invalid_parameters");
47    }
48    if bounds.iter().any(|pair| pair[0] >= pair[1]) {
49        return Err("compact_wing_invalid_bounds");
50    }
51    let plane = [(axis + 1) % 3, (axis + 2) % 3];
52    let (a, b) = (plane[0], plane[1]);
53    let (ca, cb, r) = (
54        i128::from(center[a]),
55        i128::from(center[b]),
56        i128::from(radius),
57    );
58    if ca - r <= i128::from(bounds[a][0])
59        || ca + r >= i128::from(bounds[a][1])
60        || cb - r <= i128::from(bounds[b][0])
61        || cb + r >= i128::from(bounds[b][1])
62    {
63        return Err("compact_wing_bore_must_be_strictly_inside_bounds");
64    }
65
66    // The cyclic axis order makes (plane[0], plane[1], axis) positive, so the
67    // same counter-clockwise section orientation works for every extrusion.
68    let radial = [
69        [3_i128 * r, 4_i128 * r],
70        [-4_i128 * r, 3_i128 * r],
71        [-3_i128 * r, -4_i128 * r],
72        [4_i128 * r, -3_i128 * r],
73    ];
74    let axial = [i128::from(bounds[axis][0]), i128::from(bounds[axis][1])];
75    let mut blocks = Vec::with_capacity(4);
76    for sector in 0..4 {
77        let (outer_start, outer_end) = side_corners(bounds, plane, sector);
78        let start = radial[sector];
79        let end = radial[(sector + 1) % 4];
80
81        // A quarter-circle conic has tangent-intersection control C=P+Q.
82        // With endpoint weights (10,20) or (20,10), its middle weight is 10.
83        // This avoids irrational sqrt(2) entirely.
84        let endpoint_weights = if sector % 2 == 0 {
85            [10_i128, 20_i128]
86        } else {
87            [20, 10]
88        };
89        let inner = [
90            [
91                endpoint_weights[0] * ca + endpoint_weights[0] * start[0] / 5,
92                endpoint_weights[0] * cb + endpoint_weights[0] * start[1] / 5,
93                endpoint_weights[0],
94            ],
95            [
96                10 * ca + 2 * (start[0] + end[0]),
97                10 * cb + 2 * (start[1] + end[1]),
98                10,
99            ],
100            [
101                endpoint_weights[1] * ca + endpoint_weights[1] * end[0] / 5,
102                endpoint_weights[1] * cb + endpoint_weights[1] * end[1] / 5,
103                endpoint_weights[1],
104            ],
105        ];
106
107        let outer = [
108            [10 * outer_start[0], 10 * outer_start[1], 10],
109            [
110                5 * (outer_start[0] + outer_end[0]),
111                5 * (outer_start[1] + outer_end[1]),
112                10,
113            ],
114            [10 * outer_end[0], 10 * outer_end[1], 10],
115        ];
116
117        let mut controls = Vec::with_capacity(12);
118        for u in 0..3 {
119            for v in 0..2 {
120                let h: HPoint = if v == 0 { outer[u] } else { inner[u] };
121                for z in axial {
122                    let mut xyz_w = [0_i128; 4];
123                    xyz_w[a] = h[0];
124                    xyz_w[b] = h[1];
125                    xyz_w[axis] = h[2] * z;
126                    xyz_w[3] = h[2];
127                    controls.push([
128                        checked_i64(xyz_w[0])?,
129                        checked_i64(xyz_w[1])?,
130                        checked_i64(xyz_w[2])?,
131                        checked_i64(xyz_w[3])?,
132                    ]);
133                }
134            }
135        }
136        let side = ["north", "west", "south", "east"][sector];
137        blocks.push(BracketBlock {
138            id: format!("{id}_{side}"),
139            degrees: [2, 1, 1],
140            controls,
141        });
142    }
143    Ok(blocks)
144}
