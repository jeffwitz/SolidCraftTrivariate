1//! Exact replay of a curved sketch followed by closed-cube regularity and trace checks.
2use crate::bracket_json::{decode_blocks, topology};
3use crate::incidence::enumerate_faces;
4use crate::json::{object, Json};
5use crate::spatial_certificate::{certify_spatial, CertificateBudget, CertificateStatus};
6
7pub(crate) const FAMILY: &str = "curved_sketch_layers_v1";
8pub(crate) const METHOD: &str = "curved_sketch_trace_and_bernstein";
9
10fn number(value: impl ToString) -> Json {
11    Json::Number(value.to_string())
12}
13
14fn certify(model: &Json) -> Result<Json, &'static str> {
15    let blocks = decode_blocks(model)?;
16    let faces = enumerate_faces(&blocks)?;
17    if &topology(&faces) != model.field("topology")? {
18        return Err("curved_sketch_trace_mismatch");
19    }
20    let mut minimum = f64::INFINITY;
21    for block in &blocks {
22        let certificate =
23            certify_spatial(block.degrees, &block.controls, CertificateBudget::default());
24        match certificate.status {
25            CertificateStatus::Invalid => return Err("curved_sketch_nonpositive_jacobian"),
26            CertificateStatus::Indeterminate => {
27                return Err("curved_sketch_indeterminate_bernstein")
28            }
29            CertificateStatus::Valid => {
30                minimum = minimum.min(
31                    certificate
32                        .determinant_lower_bound_mm3
33                        .ok_or("missing_jacobian_bound")?,
34                );
35            }
36        }
37    }
38    if !minimum.is_finite() || minimum <= 0.0 {
39        return Err("missing_jacobian_bound");
40    }
41    // Inputs to this family are disjoint, strictly internal feature footprints.
42    let source = model.field("scaffold")?;
43    let height = source.field("height_mm")?.number::<i32>()?;
44    let mut cavities = 0;
45    for cut in source.field("cuts")?.items()? {
46        let z = cut.field("z_interval_mm")?.items()?;
47        if let [start, end] = z {
48            cavities += usize::from(start.number::<i32>()? > 0 && end.number::<i32>()? < height);
49        } else {
50            return Err("invalid_sketch_cut_interval");
51        }
52    }
53    Ok(object([
54        ("status", Json::String("valid".into())),
55        ("method", Json::String(METHOD.into())),
56        ("blocks", number(blocks.len())),
57        ("shared_faces", number(faces.interfaces.len())),
58        ("exterior_sides", number(faces.exterior.len())),
59        ("cavities", number(cavities)),
60        ("determinant_lower_bound_mm3", number(minimum)),
61    ]))
62}
63
64pub(crate) fn build(recipe: &Json) -> Result<Json, &'static str> {
65    let model = crate::curved_sketch::build(recipe)?;
66    certify(&model)?;
67    Ok(model)
68}
69
70pub(crate) fn audit(model: &Json) -> Result<Json, &'static str> {
71    let recipe = object([
72        ("family", Json::String(FAMILY.into())),
73        ("scaffold", model.field("scaffold")?.clone()),
74    ]);
75    if crate::curved_sketch::build(&recipe)? != *model {
76        return Err("curved_sketch_replay_mismatch");
77    }
78    certify(model)
79}
