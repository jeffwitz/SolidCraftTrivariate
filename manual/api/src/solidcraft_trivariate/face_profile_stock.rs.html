1//! Complete a curved tool partition to the six planes of a rectangular stock.

5fn xy(p: [i64; 3]) -> [f64; 2] {
6    [p[0] as f64 / p[2] as f64, p[1] as f64 / p[2] as f64]
7}
8
9fn direction(p: [f64; 2], center: [f64; 2]) -> [f64; 2] {
10    let d = [p[0] - center[0], p[1] - center[1]];
11    let length = d[0].hypot(d[1]);
12    [d[0] / length, d[1] / length]
13}

15// Match the four corners to distinct curve vertices in cyclic order. Minimize
16// angular displacement about the mean boundary vertex, then divide each side
17// according to chord length. Final acceptance tests the maps themselves.
18fn perimeter(ring: &[SectionPatch], bounds: [[f64; 2]; 2]) -> Result<Vec<[f64; 2]>, &'static str> {
19    let n = ring.len();
20    if !(4..=1024).contains(&n) {
21        return Err("rectangular enclosure needs 4 to 1024 boundary spans");
22    }
23    let pts: Vec<_> = ring.iter().map(|p| xy(p.controls[0])).collect();
24    for p in ring.iter().flat_map(|p| p.controls.iter().step_by(2)) {
25        let v = xy(*p);
26        if (0..2).any(|j| v[j] <= bounds[j][0] || v[j] >= bounds[j][1]) {
27            return Err("profile control hull must be strictly inside the support rectangle");
28        }
29    }
30    let [[x0, x1], [y0, y1]] = bounds;
31    let corners = [[x0, y0], [x1, y0], [x1, y1], [x0, y1]];
32    let center = std::array::from_fn(|k| pts.iter().map(|p| p[k]).sum::<f64>() / n as f64);
33    let dirs = pts
34        .iter()
35        .map(|&p| direction(p, center))
36        .collect::<Vec<_>>();
37    let cost = |i: usize, j: usize| {
38        let a = direction(corners[j], center);
39        let b = dirs[i % n];
40        1. - (a[0] * b[0] + a[1] * b[1])
41    };
42    let mut best = f64::INFINITY;
43    let mut winner = [0; 4];
44    for start in 0..n {
45        let mut states = [(f64::INFINITY, [0; 4]); 4];
46        states[0] = (cost(start, 0), [start; 4]);
47        for k in start + 1..start + n {
48            for corner in (1..=3).rev() {
49                let (prev, mut ids) = states[corner - 1];
50                let score = prev + cost(k, corner);
51                if score < states[corner].0 {
52                    ids[corner] = k;
53                    states[corner] = (score, ids);
54                }
55            }
56        }
57        if states[3].0 < best {
58            (best, winner) = states[3];
59        }
60    }
61    if !best.is_finite() {
62        return Err("cannot find an ordered rectangular enclosure");
63    }
64    let mut out = vec![[0.; 2]; n];
65    let length = |i: usize| {
66        let a = pts[i % n];
67        let b = pts[(i + 1) % n];
68        (a[0] - b[0]).hypot(a[1] - b[1])
69    };
70    for j in 0..4 {
71        let start = winner[j];
72        let end = if j == 3 { winner[0] + n } else { winner[j + 1] };
73        let total = (start..end).map(length).sum::<f64>();
74        if total <= 0. {
75            return Err("zero-length interval on support rectangle");
76        }
77        let mut dist = 0.;
78        for i in start..end {
79            let t = dist / total;
80            out[i % n] = std::array::from_fn(|k| {
81                corners[j][k] + t * (corners[(j + 1) % 4][k] - corners[j][k])
82            });
83            dist += length(i);
84        }
85    }
86    Ok(out)
87}
88fn gcd(mut a: i128, mut b: i128) -> i128 {
89    while b != 0 {
90        (a, b) = (b, a % b);
91    }
92    a.abs()
93}

95fn enclosing_patch(
96    source: &SectionPatch,
97    a: [f64; 2],
98    b: [f64; 2],
99) -> Result<SectionPatch, &'static str> {
100    const SCALE: i128 = 1_000_000_000;
101    let d = source.degree as i128;
102    let curve = source.controls.iter().step_by(2).collect::<Vec<_>>();
103    // Give both ends of a radial edge the same homogeneous scale as its
104    // matching tool endpoint. Degree-elevate the straight outside edge exactly.
105    let point = |v: [f64; 2], w: i64| {
106        [
107            ((v[0] * SCALE as f64).round() as i128) * i128::from(w),
108            ((v[1] * SCALE as f64).round() as i128) * i128::from(w),
109            SCALE * i128::from(w),
110        ]
111    };
112    let l = point(a, curve[0][2]);
113    let r = point(b, curve[curve.len() - 1][2]);
114    let mut net = Vec::<[i128; 3]>::new();
115    for (i, h) in curve.iter().enumerate() {
116        net.push(std::array::from_fn(|k| {
117            (d - i as i128) * l[k] + i as i128 * r[k]
118        }));
119        net.push(h.map(|v| i128::from(v) * SCALE * d));
120    }
121    let g = net.iter().flatten().copied().fold(0, gcd).max(1);
122    let controls = net
123        .into_iter()
124        .map(|h| {
125            let [x, y, w] = h.map(|v| i64::try_from(v / g));
126            Ok([
127                x.map_err(|_| "stock coordinate overflow")?,
128                y.map_err(|_| "stock coordinate overflow")?,
129                w.map_err(|_| "stock weight overflow")?,
130            ])
131        })
132        .collect::<Result<Vec<_>, &'static str>>()?;
133    Ok(SectionPatch {
134        degree: source.degree,
135        controls,
136    })
137}

139pub(crate) fn layers(
140    r: &FaceProfileExtrusion,
141    core: &[SectionPatch],
142    ring: &[SectionPatch],
143) -> Result<Vec<BracketBlock>, &'static str> {
144    let tool: Vec<_> = core.iter().chain(ring.iter()).cloned().collect();
145    let points = perimeter(ring, r.bounds)?;
146    let outside = ring
147        .iter()
148        .enumerate()
149        .map(|(i, p)| enclosing_patch(p, points[i], points[(i + 1) % points.len()]))
150        .collect::<Result<Vec<_>, _>>()?;
151    let mut blocks = Vec::new();
152    let mut row = |start: f64, end: f64, inner: bool, outer: bool| -> Result<(), &'static str> {
153        if end <= start {
154            return Ok(());
155        }
156        let mut f = r.frame;
157        f[0] = std::array::from_fn(|k| f[0][k] + start * f[3][k]);
158        for patches in [
159            inner.then_some(tool.as_slice()),
160            outer.then_some(outside.as_slice()),
161        ]
162        .into_iter()
163        .flatten()
164        {
165            for mut block in lift_sections(patches, end - start, f, false)? {
166                block.id = format!("face_stock_{}", blocks.len());
167                blocks.push(block);
168            }
169        }
170        Ok(())
171    };
172    match r.operation.as_str() {
173        "add" => {
174            row(-r.thickness, 0., true, true)?;
175            row(0., r.depth, true, false)?;
176        }
177        "cut" => {
178            let bottom = (-r.depth).max(-r.thickness);
179            row(-r.thickness, bottom, true, true)?;
180            row(bottom, 0., false, true)?;
181        }
182        _ => return Err("unsupported face stock operation"),
183    }
184    Ok(blocks)
185}
