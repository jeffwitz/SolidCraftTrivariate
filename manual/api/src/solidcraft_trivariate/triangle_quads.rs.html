1//! Exact conversion of a quadratic rational triangle to three biquadratic
2//! tensor patches. Each corner owns the quadrilateral from its two incident
3//! edge midpoints to the barycenter in the reference triangle.
4fn gcd(mut a: i128, mut b: i128) -> i128 {
5    while b != 0 {
6        (a, b) = (b, a % b);
7    }
8    a.abs()
9}
10const CONTROL: [[usize; 3]; 3] = [[0, 3, 5], [3, 1, 4], [5, 4, 2]];
11
12/// Triangle coefficients in A,B,C,AB,BC,CA order. Planar homogeneous triples
13/// (wx,wy,w) must share a gauge. Returns A,B,C patches with index u*3+v.
14/// Positivity of weights is checked here; physical regularity is a separate audit.
15pub fn quadrangulate(controls: [[i64; 3]; 6]) -> Result<[[[i64; 3]; 9]; 3], &'static str> {
16    if controls.iter().any(|p| p[2] <= 0) {
17        return Err("nonpositive triangle weight");
18    }
19    let mut patches = [[[0; 3]; 9]; 3];
20    for (corner, patch) in patches.iter_mut().enumerate() {
21        let next = (corner + 1) % 3;
22        let prev = (corner + 2) % 3;
23        // Bilinear barycentric restriction. All four corners use denominator six.
24        let mut bary = [[0_i128; 3]; 4];
25        bary[0][corner] = 6;
26        bary[1][corner] = 3;
27        bary[1][prev] = 3;
28        bary[2][corner] = 3;
29        bary[2][next] = 3;
30        bary[3] = [2; 3];
31        let mut net = [[0_i128; 3]; 9];
32        for (i, a) in bary.iter().enumerate() {
33            for (j, b) in bary.iter().enumerate() {
34                let (u, v) = (i / 2 + j / 2, i % 2 + j % 2);
35                // Product of Bernstein degree-one factors, cleared by a
36                // common factor four instead of dividing the middle rows.
37                let factor = (if u == 1 { 1 } else { 2 }) * (if v == 1 { 1 } else { 2 });
38                for (s, &as_) in a.iter().enumerate() {
39                    for (t, &bt) in b.iter().enumerate() {
40                        let h = controls[CONTROL[s][t]];
41                        for (out, x) in net[u * 3 + v].iter_mut().zip(h) {
42                            *out += factor * as_ * bt * i128::from(x);
43                        }
44                    }
45                }
46            }
47        }
48        // Per coefficient the multiplier sum is 144, well inside i128 even
49        // for an extremal i64 input. One common reduction preserves the map.
50        let common = net.iter().flatten().copied().fold(0, gcd);
51        for (dst, point) in patch.iter_mut().zip(net) {
52            for (out, x) in dst.iter_mut().zip(point) {
53                *out = i64::try_from(x / common)
54                    .map_err(|_| "quadratic triangle restriction exceeds i64")?;
55            }
56        }
57    }
58    Ok(patches)
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64    #[test]
65    fn reproduces_a_weighted_triangle_and_its_parameter_restrictions() {
66        let h = [
67            [0, 0, 2],
68            [8, 0, 4],
69            [0, 6, 3],
70            [3, -1, 3],
71            [4, 4, 4],
72            [-1, 2, 2],
73        ];
74        let patches = quadrangulate(h).unwrap();
75        for (i, patch) in patches.iter().enumerate() {
76            for u in [0., 0.2, 0.5, 1.] {
77                for v in [0., 0.37, 1.] {
78                    let mut l = [0.; 3];
79                    l[i] = (1. - u) * (1. - v) + (u * (1. - v) + v * (1. - u)) / 2. + u * v / 3.;
80                    l[(i + 1) % 3] = u * (1. - v) / 2. + u * v / 3.;
81                    l[(i + 2) % 3] = v * (1. - u) / 2. + u * v / 3.;
82                    let tri: [f64; 3] = std::array::from_fn(|d| {
83                        (0..3)
84                            .flat_map(|a| {
85                                (0..3).map(move |b| l[a] * l[b] * h[CONTROL[a][b]][d] as f64)
86                            })
87                            .sum()
88                    });
89                    let basis = |x: f64| [(1. - x).powi(2), 2. * x * (1. - x), x * x];
90                    let (bu, bv) = (basis(u), basis(v));
91                    let quad: [f64; 3] = std::array::from_fn(|d| {
92                        (0..9)
93                            .map(|n| bu[n / 3] * bv[n % 3] * patch[n][d] as f64)
94                            .sum()
95                    });
96                    for d in 0..2 {
97                        assert!((tri[d] / tri[2] - quad[d] / quad[2]).abs() < 2e-14);
98                    }
99                }
100            }
101        }
102    }
