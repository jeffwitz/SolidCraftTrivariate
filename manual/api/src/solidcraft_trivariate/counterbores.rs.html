42fn add(a: i128, b: i128) -> Result<i128, &'static str> {
43    a.checked_add(b).ok_or("counterbore_integer_overflow")
44}

46fn mul(a: i128, b: i128) -> Result<i128, &'static str> {
47    a.checked_mul(b).ok_or("counterbore_integer_overflow")
48}

50fn gcd(mut a: u128, mut b: u128) -> u128 {
51    while b != 0 {
52        (a, b) = (b, a % b);
53    }
54    a
55}

57fn make_block(
58    id: String,
59    degrees: [usize; 3],
60    mut controls: Vec<[i128; 4]>,
61) -> Result<BracketBlock, &'static str> {
62    let expected = degrees.into_iter().try_fold(1_usize, |count, degree| {
63        count
64            .checked_mul(degree + 1)
65            .ok_or("counterbore_control_count_overflow")
66    })?;
67    if controls.len() != expected {
68        return Err("counterbore_control_count_mismatch");
69    }
70    if controls.iter().any(|control| control[3] <= 0) {
71        return Err("counterbore_nonpositive_weight");
72    }
73    let common = controls
74        .iter()
75        .flatten()
76        .fold(0_u128, |current, value| gcd(current, value.unsigned_abs()));
77    if common == 0 {
78        return Err("counterbore_zero_homogeneous_net");
79    }
80    let divisor = i128::try_from(common).map_err(|_| "counterbore_integer_overflow")?;
81    for control in &mut controls {
82        for coordinate in control {
83            *coordinate /= divisor;
84        }
85    }
86    Ok(BracketBlock {
87        id,
88        degrees,
89        controls: controls
90            .into_iter()
91            .map(|control| {
92                Ok([
93                    i64::try_from(control[0]).map_err(|_| "counterbore_control_exceeds_i64")?,
94                    i64::try_from(control[1]).map_err(|_| "counterbore_control_exceeds_i64")?,
95                    i64::try_from(control[2]).map_err(|_| "counterbore_control_exceeds_i64")?,
96                    i64::try_from(control[3]).map_err(|_| "counterbore_control_exceeds_i64")?,
97                ])
98            })
99            .collect::<Result<Vec<_>, _>>()?,
100    })
101}

103fn split_block(
104    block: &BracketBlock,
105    height: i32,
106    layer: Layer,
107) -> Result<BracketBlock, &'static str> {
108    let [du, dv, dw] = block.degrees;
109    if dw != 1 {
110        return Err("counterbore_source_block_must_be_linear_in_z");
111    }
112    let nv = dv + 1;
113    let mut controls = Vec::with_capacity(block.controls.len());
114    for u in 0..=du {
115        for v in 0..=dv {
116            let p0 = block.controls[(u * nv + v) * 2].map(i128::from);
117            let p1 = block.controls[(u * nv + v) * 2 + 1].map(i128::from);
118            for z in [layer.z0, layer.z1] {
119                let mut point = [0_i128; 4];
120                for k in 0..4 {
121                    point[k] = add(
122                        mul(i128::from(height - z), p0[k])?,
123                        mul(i128::from(z), p1[k])?,
124                    )?;
125                }
126                controls.push(point);
127            }
128        }
129    }
130    let suffix = format!("_z0_{height}");
131    let prefix = block
132        .id
133        .strip_suffix(&suffix)
134        .ok_or("counterbore_g21_block_has_unexpected_layer_id")?;
135    make_block(
136        format!("{prefix}_z{}_{}", layer.z0, layer.z1),
137        block.degrees,
138        controls,
139    )
140}

142fn source_index(i: usize, quadrant: usize) -> usize {
143    if quadrant % 2 == 1 {
144        2 - i
145    } else {
146        i
147    }
148}

150fn rotate_profile(point: [i128; 3], quadrant: usize) -> [i128; 3] {
151    let [x, y, weight] = point;
152    match quadrant {
153        0 => [x, y, weight],
154        1 => [-x, y, weight],
155        2 => [-x, -y, weight],
156        _ => [x, -y, weight],
157    }
158}

160fn restrict_quadratic(
161    p0: [i128; 4],
162    p1: [i128; 4],
163    p2: [i128; 4],
164    half: usize,
165    i: usize,
166) -> [i128; 4] {
167    match (half, i) {
168        (0, 0) => p0.map(|x| 4 * x),
169        (0, 1) => std::array::from_fn(|k| 2 * (p0[k] + p1[k])),
170        (0, 2) | (1, 0) => std::array::from_fn(|k| p0[k] + 2 * p1[k] + p2[k]),
171        (1, 1) => std::array::from_fn(|k| 2 * (p1[k] + p2[k])),
172        (1, 2) => p2.map(|x| 4 * x),
173        _ => unreachable!("validated quadratic restriction index"),
174    }
175}

177fn concentric_sector(
178    recipe: &CounterborePrism,
179    bore: &Counterbore,
180    layer: Layer,
181    quadrant: usize,
182    half: usize,
183) -> Result<BracketBlock, &'static str> {
184    let h = i128::from(recipe.height_mm);
185    let center = bore.center_mm.map(i128::from);
186    let shift = recipe.top_shift_mm.map(i128::from);
187    let radii = [bore.counterbore_radius_mm, bore.radius_mm].map(i128::from);
188    // The endpoint weights match the G21 quarter-circle chart exactly.
189    let arc = |radius: i128| {
190        [
191            [2 * radius, 0, 2],
192            [2 * radius, 2 * radius, 2],
193            [0, 4 * radius, 4],
194        ]
195    };
196    let outer = arc(radii[0]);
197    let inner = arc(radii[1]);
198    let mut net = [[[[0_i128; 4]; 2]; 2]; 3];
199    for (u, row) in net.iter_mut().enumerate() {
200        let source = source_index(u, quadrant);
201        for (v, pair) in row.iter_mut().enumerate() {
202            let point = if v == 0 { outer[source] } else { inner[source] };
203            let [x, y, weight] = rotate_profile(point, quadrant);
204            for (k, z) in [layer.z0, layer.z1].into_iter().enumerate() {
205                let zw = mul(i128::from(z), weight)?;
206                pair[k] = [
207                    add(
208                        mul(h, add(mul(center[0], weight)?, x)?)?,
209                        mul(zw, shift[0])?,
210                    )?,
211                    add(
212                        mul(h, add(mul(center[1], weight)?, y)?)?,
213                        mul(zw, shift[1])?,
214                    )?,
215                    mul(mul(h, i128::from(z))?, weight)?,
216                    mul(h, weight)?,
217                ];
218            }
219        }
220    }
221    let mut controls = Vec::with_capacity(12);
222    for u in 0..3 {
223        for ((v0, v1), v2) in net[0].iter().zip(&net[1]).zip(&net[2]) {
224            for ((&p0, &p1), &p2) in v0.iter().zip(v1).zip(v2) {
225                controls.push(restrict_quadratic(p0, p1, p2, half, u));
226            }
227        }
228    }
229    let quadrant_name = ["east", "north", "west", "south"][quadrant];
230    make_block(
231        format!(
232            "bore_{}_pilot_sector_{quadrant_name}_half_{half}_z{}_{}",
233            bore.id, layer.z0, layer.z1
234        ),
235        [2, 1, 1],
236        controls,
237    )
238}

240fn shoulder_z(bore: &Counterbore, height: i32) -> i32 {
241    match bore.opening {
242        CounterboreOpening::Start => bore.depth_mm,
243        CounterboreOpening::End => height - bore.depth_mm,
244    }
245}

247fn pilot_only(bore: &Counterbore, height: i32, layer: Layer) -> bool {
248    match bore.opening {
249        CounterboreOpening::Start => layer.z0 >= bore.depth_mm,
250        CounterboreOpening::End => layer.z1 <= height - bore.depth_mm,
251    }
252}

254fn validate(recipe: &CounterborePrism) -> Result<Vec<Counterbore>, &'static str> {
255    if !(1..=4).contains(&recipe.bores.len()) {
256        return Err("counterbore_count");
257    }
258    let mut bores = recipe.bores.clone();
259    bores.sort_by(|a, b| a.id.cmp(&b.id));
260    let mut ids = BTreeSet::new();
261    for bore in &bores {
262        if bore.id.is_empty()
263            || bore.id.len() > 18
264            || !bore
265                .id
266                .bytes()
267                .all(|byte| byte.is_ascii_alphanumeric() || byte == b'_')
268            || !ids.insert(bore.id.clone())
269        {
270            return Err("counterbore_invalid_or_duplicate_id");
271        }
272        if bore.radius_mm < 1
273            || bore.counterbore_radius_mm <= bore.radius_mm
274            || i64::from(bore.guard_mm) < 2 * i64::from(bore.counterbore_radius_mm)
275        {
276            return Err("counterbore_radii_or_guard");
277        }
278        if !(1..recipe.height_mm).contains(&bore.depth_mm) {
279            return Err("counterbore_depth");
280        }
281    }
282    Ok(bores)
283}

285/// Build exact start- or end-opened counterbores using G21's certified guard maps.
286pub fn build_counterbores(recipe: &CounterborePrism) -> Result<Vec<BracketBlock>, &'static str> {
287    let bores = validate(recipe)?;
288    let base_recipe = IntervalBorePrism {
289        outer: recipe.outer.clone(),
290        bores: bores
291            .iter()
292            .map(|bore| IntervalBore {
293                bore: CircularPolygonBore {
294                    id: bore.id.clone(),
295                    center_mm: bore.center_mm,
296                    radius_mm: bore.counterbore_radius_mm,
297                    guard_mm: bore.guard_mm,
298                },
299                z_interval_mm: [0, recipe.height_mm],
300            })
301            .collect(),
302        height_mm: recipe.height_mm,
303        top_shift_mm: recipe.top_shift_mm,
304    };
305    let base_blocks = build_interval_bore_prism(&base_recipe)?;
306    audit_interval_bore_prism(&base_recipe, &base_blocks)?;
307
308    let mut events = BTreeSet::from([0, recipe.height_mm]);
309    for bore in &bores {
310        events.insert(shoulder_z(bore, recipe.height_mm));
311    }
312    let events = events.into_iter().collect::<Vec<_>>();
313    let layers = events
314        .windows(2)
315        .map(|pair| Layer {
316            z0: pair[0],
317            z1: pair[1],
318        })
319        .collect::<Vec<_>>();
320
321    let mut id_order = bores.clone();
322    id_order.sort_by_key(|bore| std::cmp::Reverse(bore.id.len()));
323    let id_prefixes = id_order
324        .iter()
325        .map(|bore| {
326            (
327                format!("bore_{}_sector_", bore.id),
328                format!("bore_{}_counter_sector_", bore.id),
329            )
330        })
331        .collect::<Vec<_>>();
332
333    let mut blocks = Vec::new();
334    for layer in &layers {
335        for block in &base_blocks {
336            let mut prefix = block
337                .id
338                .strip_suffix(&format!("_z0_{}", recipe.height_mm))
339                .ok_or("counterbore_g21_block_has_unexpected_layer_id")?
340                .to_owned();
341            for (source, target) in &id_prefixes {
342                if prefix.starts_with(source) {
343                    prefix = prefix.replacen(source, target, 1);
344                    break;
345                }
346            }
347            let renamed = BracketBlock {
348                id: format!("{prefix}_z0_{}", recipe.height_mm),
349                degrees: block.degrees,
350                controls: block.controls.clone(),
351            };
352            blocks.push(split_block(&renamed, recipe.height_mm, *layer)?);
353        }
354        for bore in &bores {
355            if pilot_only(bore, recipe.height_mm, *layer) {
356                for quadrant in 0..4 {
357                    for half in 0..2 {
358                        blocks.push(concentric_sector(recipe, bore, *layer, quadrant, half)?);
359                    }
360                }
361            }
362        }
363    }
364    blocks.sort_by(|a, b| a.id.cmp(&b.id));
365    let mut ids = BTreeSet::new();
366    if blocks.iter().any(|block| !ids.insert(block.id.clone())) {
367        return Err("counterbore_duplicate_block_id");
368    }
369    Ok(blocks)
