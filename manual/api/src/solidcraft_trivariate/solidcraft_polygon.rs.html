20struct DisjointSet(Vec<usize>);
21impl DisjointSet {
22    fn new(n: usize) -> Self {
23        Self((0..n).collect())
24    }
25    fn find(&mut self, x: usize) -> usize {
26        if self.0[x] != x {
27            self.0[x] = self.find(self.0[x]);
28        }
29        self.0[x]
30    }
31    fn union(&mut self, a: usize, b: usize) {
32        let (a, b) = (self.find(a), self.find(b));
33        if a != b {
34            self.0[b] = a;
35        }
36    }
37}

39fn text(value: &Json) -> Result<&str, &'static str> {
40    value.text()
41}
42fn boolean(value: &Json, expected: bool) -> Result<(), &'static str> {
43    if value == &Json::Bool(expected) {
44        Ok(())
45    } else {
46        Err("unsupported SolidCraft boolean option")
47    }
48}
49fn exact_i32(value: &Json, allow_string: bool) -> Result<i32, &'static str> {
50    let token = match value {
51        Json::Number(value) => value.as_str(),
52        Json::String(value) if allow_string => value.as_str(),
53        _ => return Err("SolidCraft polygon values must be literal numbers"),
54    };
55    let (n, d) = crate::solidcraft_boxes::decimal_rational(token)?;
56    if d != 1 {
57        return Err("SolidCraft polygon values must be integral millimetres");
58    }
59    i32::try_from(n).map_err(|_| "SolidCraft polygon value outside native integer range")
60}
61fn index(value: &Json, limit: usize) -> Result<usize, &'static str> {
62    let n: usize = value.number()?;
63    if n >= limit {
64        return Err("SolidCraft sketch point index out of range");
65    }
66    Ok(n)
67}
68fn valid_name(s: &str) -> bool {
69    !s.is_empty()
70        && s.len() <= 22
71        && s.bytes()
72            .all(|c| c.is_ascii_alphanumeric() || c == b'_' || c == b'.')
73}
74fn native_vertex_id(source: &str) -> Result<String, &'static str> {
75    if !valid_name(source) {
76        return Err("unsupported SolidCraft sketch point id");
77    }
78    let id = format!("v_{}", source.replace('.', "_"));
79    if id.len() > 24 {
80        return Err("SolidCraft sketch point id too long");
81    }
82    Ok(id)
83}
84fn feature<'a>(command: &'a Json, kind: &str) -> Result<&'a Json, &'static str> {
85    command.keys(&["command", "params"])?;
86    if text(command.field("command")?)? != "timeline.add" {
87        return Err("unsupported SolidCraft timeline command");
88    }
89    let params = command.field("params")?;
90    params.keys(&["feature"])?;
91    let f = params.field("feature")?;
92    if text(f.field("type")?)? != kind {
93        return Err("unsupported SolidCraft timeline feature order");
94    }
95    Ok(f)
96}
97fn feature_id(feature: &Json) -> Result<usize, &'static str> {
98    let id: usize = feature.field("id")?.number()?;
99    if id == 0 {
100        return Err("invalid SolidCraft feature id");
101    }
102    Ok(id)

104fn point_inside(point: [i32; 2], vertices: &[Vertex]) -> bool {
105    let [px, py] = point.map(|v| 2 * i64::from(v));
106    let mut inside = false;
107    for i in 0..vertices.len() {
108        let a = vertices[i].xy_mm.map(i64::from);
109        let b = vertices[(i + 1) % vertices.len()].xy_mm.map(i64::from);
110        let cross = (b[0] - a[0]) * (py - 2 * a[1]) - (b[1] - a[1]) * (px - 2 * a[0]);
111        if cross == 0
112            && 2 * a[0].min(b[0]) <= px
113            && px <= 2 * a[0].max(b[0])
114            && 2 * a[1].min(b[1]) <= py
115            && py <= 2 * a[1].max(b[1])
116        {
117            return false;
118        }
119        if (2 * a[1] > py) != (2 * b[1] > py) {
120            let lhs = (px - 2 * a[0]) * (b[1] - a[1]);
121            let rhs = (py - 2 * a[1]) * (b[0] - a[0]);
122            if (b[1] > a[1]) == (lhs < rhs) {
123                inside = !inside;
124            }
125        }
126    }
127    inside
128}
129fn signed_area2(vertices: &[Vertex]) -> i64 {
130    (0..vertices.len())
131        .map(|i| {
132            let a = vertices[i].xy_mm;
133            let b = vertices[(i + 1) % vertices.len()].xy_mm;
134            i64::from(a[0]) * i64::from(b[1]) - i64::from(a[1]) * i64::from(b[0])
135        })
136        .sum()
137}

139fn parse_loops(sketch_feature: &Json) -> Result<Vec<ProfileLoop>, &'static str> {
140    sketch_feature.keys(&["id", "name", "plane", "sketch", "suppressed", "type"])?;
141    let sketch_id = feature_id(sketch_feature)?;
142    if sketch_feature.field("name")?.text()?.is_empty() {
143        return Err("empty SolidCraft sketch name");
144    }
145    boolean(sketch_feature.field("suppressed")?, false)?;
146    let plane = sketch_feature.field("plane")?;
147    plane.keys(&["name", "type"])?;
148    if text(plane.field("type")?)? != "origin" || text(plane.field("name")?)? != "XY" {
149        return Err("SolidCraft polygon sketch must lie on the origin XY plane");
150    }
151    let sketch = sketch_feature.field("sketch")?;
152    sketch.keys(&["constraints", "counters", "curves", "points"])?;
153    let points = sketch.field("points")?.items()?;
154    let curves_json = sketch.field("curves")?.items()?;
155    let constraints = sketch.field("constraints")?.items()?;
156    if points.len() < 4 || curves_json.len() < 3 || curves_json.len() > 32 {
157        return Err("SolidCraft polygon sketch size out of range");
158    }
159    let counters = sketch.field("counters")?;
160    let Json::Object(counts) = counters else {
161        return Err("invalid sketch counters");
162    };
163    for (key, count) in counts {
164        if !matches!(key.as_str(), "k" | "l" | "p") {
165            return Err("unsupported sketch counter");
166        }
167        count.number::<usize>()?;
168    }
169    let kc = counts
170        .get("k")
171        .map(Json::number::<usize>)
172        .transpose()?
173        .unwrap_or(0);
174    let lc: usize = counters.field("l")?.number()?;
175    if kc < constraints.len() || lc < curves_json.len() {
176        return Err("invalid SolidCraft sketch counters");
177    }
178
179    let mut coords = Vec::with_capacity(points.len());
180    let mut point_ids = BTreeSet::new();
181    for p in points {
182        p.keys(&["fixed", "id", "pos"])?;
183        if !matches!(p.field("fixed")?, Json::Bool(_)) {
184            return Err("invalid SolidCraft point fixed flag");
185        }
186        let id = text(p.field("id")?)?;
187        if id.is_empty() || !point_ids.insert(id.to_owned()) {
188            return Err("duplicate SolidCraft sketch point id");
189        }
190        let pos = p.field("pos")?.items()?;
191        if pos.len() != 2 {
192            return Err("SolidCraft sketch point must have two coordinates");
193        }
194        coords.push([exact_i32(&pos[0], false)?, exact_i32(&pos[1], false)?]);
195    }
196    let mut dsu = DisjointSet::new(points.len());
197    let mut constraint_ids = BTreeSet::new();
198    for c in constraints {
199        let id = text(c.field("id")?)?;
200        if id.is_empty() || !constraint_ids.insert(id.to_owned()) {
201            return Err("duplicate SolidCraft sketch constraint id");
202        }
203        let kind = text(c.field("type")?)?;
204        if matches!(kind, "horizontal" | "vertical") {
205            c.keys(&["id", "l", "type"])?;
206            let line = &curves_json[index(c.field("l")?, curves_json.len())?];
207            let a = coords[index(line.field("a")?, coords.len())?];
208            let b = coords[index(line.field("b")?, coords.len())?];
209            let axis = usize::from(kind == "horizontal");
210            if a[axis] != b[axis] {
211                return Err("inconsistent SolidCraft line direction constraint");
212            }
213            continue;
214        }
215        c.keys(&["id", "p", "q", "type"])?;
216        if kind != "coincident" {
217            return Err("unsupported SolidCraft sketch constraint");
218        }
219        let p = index(c.field("p")?, points.len())?;
220        let q = index(c.field("q")?, points.len())?;
221        if coords[p] != coords[q] {
222            return Err("inconsistent SolidCraft coincident constraint");
223        }
224        dsu.union(p, q);
225    }
226    let mut curves = Vec::with_capacity(curves_json.len());
227    let mut curve_ids = BTreeSet::new();
228    let mut used = BTreeSet::new();
229    for c in curves_json {
230        c.keys(&["a", "b", "construction", "id", "type"])?;
231        if text(c.field("type")?)? != "line" {
232            return Err("SolidCraft polygon sketch only supports straight lines");
233        }
234        boolean(c.field("construction")?, false)?;
235        let id = text(c.field("id")?)?;
236        if !valid_name(id) || !curve_ids.insert(id.to_owned()) {
237            return Err("invalid or duplicate SolidCraft curve id");
238        }
239        let a = index(c.field("a")?, points.len())?;
240        let b = index(c.field("b")?, points.len())?;
241        used.insert(a);
242        used.insert(b);
243        curves.push(Curve {
244            id: id.to_owned(),
245            a,
246            b,
247        });
248    }
249    let used_roots = used.iter().map(|&i| dsu.find(i)).collect::<BTreeSet<_>>();
250    for (i, point) in points.iter().enumerate() {
251        let root = dsu.find(i);
252        if !used_roots.contains(&root) && text(point.field("id")?)? != "origin" {
253            return Err("unused SolidCraft sketch point");
254        }
255    }
256    let mut root_coords = BTreeMap::<usize, [i32; 2]>::new();
257    let mut root_names = BTreeMap::<usize, String>::new();
258    for (i, point) in points.iter().enumerate() {
259        let root = dsu.find(i);
260        let source_name = text(point.field("id")?)?;
261        if let Some(existing) = root_coords.insert(root, coords[i]) {
262            if existing != coords[i] {
263                return Err("coincident SolidCraft point coordinates differ");
264            }
265        }
266        root_names
267            .entry(root)
268            .and_modify(|name| {
269                if source_name < name {
270                    *name = source_name.to_owned();
271                }
272            })
273            .or_insert_with(|| source_name.to_owned());
274    }
275    let mut graph = BTreeMap::<usize, Vec<(usize, usize)>>::new();
276    let mut unique_edges = BTreeSet::new();
277    for (ci, c) in curves.iter().enumerate() {
278        let a = dsu.find(c.a);
279        let b = dsu.find(c.b);
280        if a == b || !unique_edges.insert((a.min(b), a.max(b))) {
281            return Err("degenerate or duplicate SolidCraft sketch edge");
282        }
283        graph.entry(a).or_default().push((b, ci));
284        graph.entry(b).or_default().push((a, ci));
285    }
286    if graph.values().any(|neighbors| neighbors.len() != 2) {
287        return Err("SolidCraft sketch curves must form closed loops");
288    }
289    let mut used_edges = BTreeSet::<usize>::new();
290    let mut loops = Vec::new();
291    while used_edges.len() < curves.len() {
292        let start = *graph
293            .iter()
294            .filter(|(_, edges)| edges.iter().any(|(_, e)| !used_edges.contains(e)))
295            .map(|(v, _)| v)
296            .min_by_key(|&&root| root_names[&root].clone())
297            .ok_or("SolidCraft sketch cycle extraction failed")?;
298        let mut cycle = Vec::new();
299        let mut edge_set = BTreeSet::new();
300        let mut previous = None;
301        let mut current = start;
302        loop {
303            cycle.push(current);
304            let next = graph[&current]
305                .iter()
306                .filter(|(_, e)| Some(*e) != previous && !used_edges.contains(e))
307                .min_by_key(|(v, e)| (root_names[v].clone(), curves[*e].id.clone()))
308                .copied()
309                .ok_or("open SolidCraft sketch cycle")?;
310            used_edges.insert(next.1);
311            edge_set.insert(curves[next.1].id.clone());
312            previous = Some(next.1);
313            current = next.0;
314            if current == start {
315                break;
316            }
317            if cycle.len() > curves.len() {
318                return Err("invalid SolidCraft sketch cycle");
319            }
320        }
321        let vertices = cycle
322            .iter()
323            .map(|root| {
324                Ok(Vertex {
325                    id: native_vertex_id(&root_names[root])?,
326                    xy_mm: root_coords[root],
327                })
328            })
329            .collect::<Result<Vec<_>, &'static str>>()?;
330        if vertices.len() < 3 {
331            return Err("SolidCraft sketch loop has fewer than three vertices");
332        }
333        let area2 = signed_area2(&vertices);
334        if area2 == 0 {
335            return Err("zero-area SolidCraft sketch loop");
336        }
337        loops.push(ProfileLoop {
338            vertices,
339            curve_ids: edge_set,
340        });
341    }
342    if loops.len() > 5 {
343        return Err("SolidCraft sketch has too many closed loops");
344    }
345    let outer_candidates = (0..loops.len())
346        .filter(|&i| {
347            (0..loops.len()).all(|j| {
348                i == j
349                    || loops[j]
350                        .vertices
351                        .iter()
352                        .all(|v| point_inside(v.xy_mm, &loops[i].vertices))
353            })
354        })
355        .collect::<Vec<_>>();
356    if outer_candidates.len() != 1 {
357        return Err("SolidCraft sketch must contain one outer loop and strict holes");
358    }
359    let outer_i = outer_candidates[0];
360    if loops.iter().enumerate().any(|(i, l)| {
361        i != outer_i
362            && l.vertices
363                .iter()
364                .any(|v| !point_inside(v.xy_mm, &loops[outer_i].vertices))
365    }) {
366        return Err("SolidCraft hole lies outside outer loop or touches it");
367    }
368    let mut ordered = Vec::with_capacity(loops.len());
369    ordered.push(loops.remove(outer_i));
370    loops.sort_by_key(|l| {
371        l.vertices
372            .iter()
373            .map(|v| v.id.clone())
374            .min()
375            .unwrap_or_default()
376    });
377    ordered.extend(loops);
378    let _ = sketch_id;
379    Ok(ordered)
380}

382fn validate_extrude(
383    extrude: &Json,
384    sketch_id: usize,
385    loops: &[ProfileLoop],
386) -> Result<i32, &'static str> {
387    extrude.keys(&[
388        "body_names",
389        "extent",
390        "id",
391        "name",
392        "operation",
393        "param_names",
394        "profiles",
395        "sketch",
396        "suppressed",
397        "type",
398    ])?;
399    let _ = feature_id(extrude)?;
400    if extrude.field("name")?.text()?.is_empty() || text(extrude.field("type")?)? != "extrude" {
401        return Err("invalid SolidCraft extrusion feature");
402    }
403    boolean(extrude.field("suppressed")?, false)?;
404    let reference: usize = extrude.field("sketch")?.number()?;
405    if reference != sketch_id {
406        return Err("SolidCraft extrusion references another sketch");
407    }
408    if text(extrude.field("operation")?)? != "new_body" {
409        return Err("SolidCraft extrusion must create a new body");
410    }
411    let bodies = extrude.field("body_names")?.items()?;
412    if bodies.len() != 1 || bodies[0].text()?.is_empty() {
413        return Err("SolidCraft extrusion must target one new body");
414    }
415    let params = extrude.field("param_names")?.items()?;
416    let mut param_names = BTreeSet::new();
417    if params.is_empty()
418        || params.iter().any(|p| {
419            p.text().map_or(true, |name| {
420                name.is_empty() || !param_names.insert(name.to_owned())
421            })
422        })
423    {
424        return Err("invalid SolidCraft extrusion parameter names");
425    }
426    let extent = extrude.field("extent")?;
427    extent.keys(&["convention", "direction", "distance"])?;
428    if text(extent.field("convention")?)? != "total_distance"
429        || text(extent.field("direction")?)? != "positive"
430    {
431        return Err("unsupported SolidCraft extrusion extent");
432    }
433    let height = exact_i32(extent.field("distance")?, true)?;
434    if !(1..=64).contains(&height) {
435        return Err("SolidCraft extrusion height outside native range");
436    }
437    let profile = extrude.field("profiles")?;
438    if loops.len() == 1 && profile.keys(&["type"]).is_ok() && text(profile.field("type")?)? == "all"
439    {
440        return Ok(height);
441    }
442    profile.keys(&["loops", "type"])?;
443    if text(profile.field("type")?)? != "curves" {
444        return Err("ambiguous SolidCraft extrusion profile selection");
445    }
446    let selected = profile.field("loops")?.items()?;
447    if selected.len() != 1 {
448        return Err("SolidCraft extrusion must select one material profile");
449    }
450    let ids = selected[0]
451        .items()?
452        .iter()
453        .map(|id| id.text().map(str::to_owned))
454        .collect::<Result<BTreeSet<_>, _>>()?;
455    if ids.len() != selected[0].items()?.len() || ids != loops[0].curve_ids {
456        return Err("SolidCraft extrusion selection is not the outer profile");
457    }
458    Ok(height)
459}

461/// Import an extracted `solidcraft/66` sketch/extrude history as an audited native prism.
462pub fn import_polygon_timeline(script: &Json) -> Result<Json, &'static str> {
463    let (mut loops, height) = extrusion_profiles(script, false)?;
464    let outer = loops.remove(0);
465    let recipe = if loops.is_empty() {
466        if outer.vertices.len() > 24 {
467            return Err("SolidCraft simple polygon exceeds native G17 limit");
468        }
469        object([
470            ("family", Json::String(FAMILY.into())),
471            (
472                "vertices",
473                Json::Array(
474                    outer
475                        .vertices
476                        .into_iter()
477                        .map(|v| {
478                            object([
479                                ("id", Json::String(v.id)),
480                                (
481                                    "xy_mm",
482                                    Json::Array(
483                                        v.xy_mm
484                                            .into_iter()
485                                            .map(|x| Json::Number(x.to_string()))
486                                            .collect(),
487                                    ),
488                                ),
489                            ])
490                        })
491                        .collect(),
492                ),
493            ),
494            ("height_mm", Json::Number(height.to_string())),
495            (
496                "top_shift_mm",
497                Json::Array(vec![Json::Number("0".into()), Json::Number("0".into())]),
498            ),
499        ])
500    } else {
501        let holes = loops
502            .into_iter()
503            .enumerate()
504            .map(|(i, l)| {
505                object([
506                    ("id", Json::String(format!("hole{i}"))),
507                    (
508                        "vertices",
509                        Json::Array(
510                            l.vertices
511                                .into_iter()
512                                .map(|v| {
513                                    object([
514                                        ("id", Json::String(v.id)),
515                                        (
516                                            "xy_mm",
517                                            Json::Array(
518                                                v.xy_mm
519                                                    .into_iter()
520                                                    .map(|x| Json::Number(x.to_string()))
521                                                    .collect(),
522                                            ),
523                                        ),
524                                    ])
525                                })
526                                .collect(),
527                        ),
528                    ),
529                ])
530            })
531            .collect();
532        object([
533            ("family", Json::String(PERFORATED.into())),
534            (
535                "outer",
536                object([
537                    ("id", Json::String("outer".into())),
538                    (
539                        "vertices",
540                        Json::Array(
541                            outer
542                                .vertices
543                                .into_iter()
544                                .map(|v| {
545                                    object([
546                                        ("id", Json::String(v.id)),
547                                        (
548                                            "xy_mm",
549                                            Json::Array(
550                                                v.xy_mm
551                                                    .into_iter()
552                                                    .map(|x| Json::Number(x.to_string()))
553                                                    .collect(),
554                                            ),
555                                        ),
556                                    ])
557                                })
558                                .collect(),
559                        ),
560                    ),
561                ]),
562            ),
563            ("holes", Json::Array(holes)),
564            ("height_mm", Json::Number(height.to_string())),
565            (
566                "top_shift_mm",
567                Json::Array(vec![Json::Number("0".into()), Json::Number("0".into())]),
568            ),
569        ])
570    };
571    polygon_json::build(&recipe)
572}

574fn check_preview_coordinates(feature: &Json) -> Result<(), &'static str> {
575    for point in feature.field("sketch")?.field("points")?.items()? {
576        for value in point.field("pos")?.items()? {
577            if !(-4096..=4096).contains(&exact_i32(value, false)?) {
578                return Err("sketch preview exceeds planner coordinate range");
579            }
580        }
581    }
582    Ok(())
583}

585pub(crate) fn preview_contours(feature: &Json) -> Result<Vec<Vec<[i32; 2]>>, &'static str> {

593fn extrusion_profiles(
594    script: &Json,
595    preview: bool,
596) -> Result<(Vec<ProfileLoop>, i32), &'static str> {
597    script.keys(&["about", "commands"])?;
598    let about = script.field("about")?;
599    about.keys(&["design", "format", "replay", "version"])?;
600    if text(about.field("format")?)? != "solidcraft/66"
601        || text(about.field("version")?)? != "0.48.1"
602        || text(about.field("design")?)?.is_empty()
603        || text(about.field("replay")?)?.is_empty()
604    {
605        return Err("unsupported SolidCraft timeline metadata");
606    }
607    let commands = script.field("commands")?.items()?;
608    let first = commands.first().ok_or("empty SolidCraft timeline")?;
609    let sketch = feature(first, "sketch")?;
610    let sketch_id = feature_id(sketch)?;
611    if preview {
612        check_preview_coordinates(sketch)?;
613    }
614    let loops = parse_loops(sketch)?;
615    let extrude = match commands {
616        [_, extrusion] => feature(extrusion, "extrude")?,
617        [_, a, b] if preview => {
618            let (overlay, extrusion, prefix) = if feature(a, "sketch").is_ok() {
619                (a, b, vec![first.clone()])
620            } else {
621                (b, a, vec![first.clone(), a.clone()])
622            };
623            let prefix = object([("commands", Json::Array(prefix))]);
624            let expected = crate::solidcraft_preview::descriptor(
625                &prefix,
626                sketch_id.try_into().map_err(|_| "invalid sketch id")?,
627            )?;
628            if feature(overlay, "sketch")? != expected.field("feature")? {
629                return Err("additional sketch differs from the generated construction preview");
630            }
631            feature(extrusion, "extrude")?
632        }
633        _ => return Err("SolidCraft polygon import requires a sketch and extrusion"),
634    };
635    let height = validate_extrude(extrude, sketch_id, &loops)?;
636    Ok((loops, height))
637}
638
639/// Extrude the same planar layout shown by the native sketch preview.
640pub fn extrude_preview(input: &Json) -> Result<Json, &'static str> {
641    let script = crate::solidcraft_preview::snapshot_script(input)?;
642    let (profiles, height) = extrusion_profiles(script, true)?;
643    extrude_profiles(profiles, height)
644}
645
646/// Create a new native body directly from a selected sketch without needing a host extrusion.
647pub fn extrude_source_sketch(
648    input: &Json,
649    source_id: u32,
650    height: i32,
651) -> Result<Json, &'static str> {
652    if !(1..=64).contains(&height) {
653        return Err("SolidCraft extrusion height outside native range");
654    }
655    let script = crate::solidcraft_preview::snapshot_script(input)?;
656    let source = crate::solidcraft_preview::source_feature(script, source_id)?.0;
657    check_preview_coordinates(source)?;
658    extrude_profiles(parse_loops(source)?, height)
659}
660
661fn extrude_profiles(profiles: Vec<ProfileLoop>, height: i32) -> Result<Json, &'static str> {
662    let n = |i: i32| Json::Number(i.to_string());
663    let mut contours = profiles.into_iter().enumerate().map(|(i, profile)| {
664        object([
665            ("id", Json::String(format!("loop{i}"))),
666            (
667                "vertices",
668                Json::Array(
669                    profile
670                        .vertices
671                        .iter()
672                        .map(|v| Json::Array(v.xy_mm.map(n).to_vec()))
673                        .collect(),
674                ),
675            ),
676        ])
677    });
678    let outer = contours.next().ok_or("empty extrusion profile")?;
679    let mut cuts = Vec::new();
680    for contour in contours {
681        let Json::Object(mut values) = contour else {
682            unreachable!()
683        };
684        values.insert("z_interval_mm".into(), Json::Array(vec![n(0), n(height)]));
685        cuts.push(Json::Object(values));
686    }
687    crate::interchange::build(&object([
688        ("family", Json::String("sketch_layer_partition_v1".into())),
689        ("outer", outer),
690        ("cuts", Json::Array(cuts)),
691        ("height_mm", n(height)),
692    ]))
693}
