1//! Independent exact-template audit for polygonal through bores with a blind
2//! counterbore at one end.
3//!
4//! The proof is deliberately limited to the G21 guarded polygon partition and
5//! the fixed concentric circular-sector template. G21 certifies the exterior
6//! and guard-to-counterbore-radius partition; this module independently
7//! checks every replacement annulus, its exact circular boundaries, positive
8//! continuous planar Jacobian, incidence and face provenance.
9

49fn gcd(mut a: u128, mut b: u128) -> u128 {
50    while b != 0 {
51        (a, b) = (b, a % b);
52    }
53    a
54}

56fn primitive(controls: &[[i128; 4]]) -> Result<Vec<[i128; 4]>, &'static str> {
57    let common = controls
58        .iter()
59        .flatten()
60        .fold(0_u128, |g, x| gcd(g, x.unsigned_abs()));
61    if common == 0 {
62        return Err("counterbore_zero_homogeneous_net");
63    }
64    let common = i128::try_from(common).map_err(|_| "counterbore_integer_overflow")?;
65    Ok(controls.iter().map(|p| p.map(|x| x / common)).collect())
66}
67
68fn primitive_block(block: &BracketBlock) -> Result<Vec<[i128; 4]>, &'static str> {
69    if block.controls.is_empty() || block.controls.iter().any(|p| p[3] <= 0) {
70        return Err("counterbore_nonpositive_weight");
71    }
72    primitive(
73        &block
74            .controls
75            .iter()
76            .map(|p| p.map(i128::from))
77            .collect::<Vec<_>>(),
78    )
79}

81fn expected_block(id: String, controls: Vec<[i128; 4]>) -> Result<BracketBlock, &'static str> {
82    let controls = primitive(&controls)?;
83    Ok(BracketBlock {
84        id,
85        degrees: [2, 1, 1],
86        controls: controls
87            .into_iter()
88            .map(|p| {
89                Ok([
90                    i64::try_from(p[0]).map_err(|_| "counterbore_control_overflow")?,
91                    i64::try_from(p[1]).map_err(|_| "counterbore_control_overflow")?,
92                    i64::try_from(p[2]).map_err(|_| "counterbore_control_overflow")?,
93                    i64::try_from(p[3]).map_err(|_| "counterbore_control_overflow")?,
94                ])
95            })
96            .collect::<Result<Vec<_>, _>>()?,
97    })
98}

100fn rotate(point: [i128; 3], quadrant: usize) -> [i128; 3] {
101    let [x, y, w] = point;
102    match quadrant {
103        0 => [x, y, w],
104        1 => [-x, y, w],
105        2 => [-x, -y, w],
106        _ => [x, -y, w],
107    }
108}
109
110fn source_index(i: usize, quadrant: usize) -> usize {
111    if quadrant % 2 == 1 {
112        2 - i
113    } else {
114        i
115    }
116}

118fn circle_profile(radius: i32) -> [[i128; 3]; 3] {
119    let r = i128::from(radius);
120    [[2 * r, 0, 2], [2 * r, 2 * r, 2], [0, 4 * r, 4]]
121}
122
123fn restrict_half(a: [i128; 4], b: [i128; 4], c: [i128; 4], half: usize, i: usize) -> [i128; 4] {
124    match (half, i) {
125        (0, 0) => a.map(|x| 4 * x),
126        (0, 1) => std::array::from_fn(|k| 2 * (a[k] + b[k])),
127        (0, 2) | (1, 0) => std::array::from_fn(|k| a[k] + 2 * b[k] + c[k]),
128        (1, 1) => std::array::from_fn(|k| 2 * (b[k] + c[k])),
129        (1, 2) => c.map(|x| 4 * x),
130        _ => unreachable!("validated half-sector index"),
131    }
132}

134/// Reconstruct a pilot-only annulus R→r, including the exact axial shear.
135fn expected_pilot_sector(
136    bore: &Counterbore,
137    recipe: &CounterborePrism,
138    z0: i32,
139    z1: i32,
140    quadrant: usize,
141    half: usize,
142) -> Result<BracketBlock, &'static str> {
143    let [cx, cy] = bore.center_mm.map(i128::from);
144    let h = i128::from(recipe.height_mm);
145    let profiles = [
146        circle_profile(bore.counterbore_radius_mm),
147        circle_profile(bore.radius_mm),
148    ];
149    let mut net = [[[0_i128; 4]; 2]; 3]; // [u][v][homogeneous 4-vector before axial restriction]
150    let mut controls = Vec::with_capacity(12);
151    for (u, row) in net.iter_mut().enumerate() {
152        let source = source_index(u, quadrant);
153        for (v, point) in row.iter_mut().enumerate() {
154            let [rx, ry, w] = rotate(profiles[v][source], quadrant);
155            let x = cx * w + rx;
156            let y = cy * w + ry;
157            *point = [x, y, w, w];
158        }
159    }
160    // Assemble all homogeneous axial pairs first, then restrict each v/w line
161    // in u. The base planar coordinates are unchanged by the axial shear.
162    let mut full = [[[[0_i128; 4]; 2]; 2]; 3];
163    for (source_row, row) in net.iter().zip(full.iter_mut()) {
164        for (source, pair) in source_row.iter().zip(row.iter_mut()) {
165            let [x, y, w, _] = *source;
166            for (k, z) in [z0, z1].into_iter().enumerate() {
167                pair[k] = [
168                    h * x + i128::from(recipe.top_shift_mm[0]) * i128::from(z) * w,
169                    h * y + i128::from(recipe.top_shift_mm[1]) * i128::from(z) * w,
170                    h * i128::from(z) * w,
171                    h * w,
172                ];
173            }
174        }
175    }
176    let mut restricted = [[[[0_i128; 4]; 2]; 2]; 3];
177    for (u, row) in restricted.iter_mut().enumerate() {
178        for (v, pair) in row.iter_mut().enumerate() {
179            for (w, out) in pair.iter_mut().enumerate() {
180                *out = restrict_half(full[0][v][w], full[1][v][w], full[2][v][w], half, u);
181            }
182        }
183    }
184    for row in restricted {
185        for pair in row {
186            controls.extend(pair);
187        }
188    }
189    expected_block(
190        format!(
191            "bore_{}_pilot_sector_{}_half_{half}_z{z0}_{z1}",
192            bore.id, QUADRANTS[quadrant]
193        ),
194        controls,
195    )
196}

198fn planar_lower_bound(block: &BracketBlock) -> Result<f64, &'static str> {
199    let [du, dv, dw] = block.degrees;
200    if [du, dv, dw] != [2, 1, 1] {
201        return Err("counterbore_unexpected_ring_degrees");
202    }
203    let controls = (0..=du)
204        .flat_map(|u| (0..=dv).map(move |v| block.controls[(u * 2 + v) * 2]))
205        .map(|p| [p[0], p[1], p[3]])
206        .collect::<Vec<_>>();
207    certify_planar_refined([du, dv], &controls).map_err(|e| match e {
208        PlanarError::Invalid => "counterbore_nonpositive_jacobian",
209        PlanarError::Indeterminate => "counterbore_indeterminate_jacobian",
210    })
211}

213fn layer_bounds(recipe: &CounterborePrism) -> BTreeSet<i32> {
214    let mut events = BTreeSet::from([0, recipe.height_mm]);
215    for bore in &recipe.bores {
216        events.insert(match bore.opening {
217            CounterboreOpening::Start => bore.depth_mm,
218            CounterboreOpening::End => recipe.height_mm - bore.depth_mm,
219        });
220    }
221    events
222}

224fn g21_comparator(recipe: &CounterborePrism, use_counter_radii: bool) -> IntervalBorePrism {
225    let bores = recipe
226        .bores
227        .iter()
228        .map(|bore| {
229            let (radius, interval) = if use_counter_radii {
230                let interval = match bore.opening {
231                    CounterboreOpening::Start => [0, bore.depth_mm],
232                    CounterboreOpening::End => [recipe.height_mm - bore.depth_mm, recipe.height_mm],
233                };
234                (bore.counterbore_radius_mm, interval)
235            } else {
236                (bore.radius_mm, [0, recipe.height_mm])
237            };
238            IntervalBore {
239                bore: CircularPolygonBore {
240                    id: bore.id.clone(),
241                    center_mm: bore.center_mm,
242                    radius_mm: radius,
243                    guard_mm: bore.guard_mm,
244                },
245                z_interval_mm: interval,
246            }
247        })
248        .collect();
249    IntervalBorePrism {
250        outer: recipe.outer.clone(),
251        bores,
252        height_mm: recipe.height_mm,
253        top_shift_mm: recipe.top_shift_mm,
254    }
255}

257fn mapped_counter_ring_id(reference_id: &str, bores: &[Counterbore]) -> Option<(String, String)> {
258    // IDs may themselves contain `_sector_`; match the longest complete
259    // `bore_{id}_sector_` prefix, as the constructor does.
260    bores
261        .iter()
262        .filter_map(|bore| {
263            let prefix = format!("bore_{}_sector_", bore.id);
264            reference_id.strip_prefix(&prefix).map(|suffix| {
265                (
266                    bore.id.len(),
267                    format!("bore_{}_counter_sector_{suffix}", bore.id),
268                    bore.id.clone(),
269                )
270            })
271        })
272        .max_by_key(|(length, _, _)| *length)
273        .map(|(_, mapped, id)| (mapped, id))
274}

276/// Independently audit a fixed-template polygonal counterbore model.
277pub fn audit_counterbores(
278    recipe: &CounterborePrism,
279    blocks: &[BracketBlock],
280) -> Result<CounterboreCertificate, &'static str> {
281    if !(1..=4).contains(&recipe.bores.len()) || recipe.height_mm <= 1 {
282        return Err("counterbore_recipe_range");
283    }
284    let ordered = {
285        let mut v = recipe.bores.clone();
286        v.sort_by(|a, b| a.id.cmp(&b.id));
287        v
288    };
289    let events = layer_bounds(recipe).into_iter().collect::<Vec<_>>();
290    if events.len() < 3 {
291        return Err("counterbore_missing_shoulder_event");
292    }
293
294    // The G21 counter-radius comparator is itself independently audited. Its
295    // whole-disk fills are replaced only on material slabs by the annulus
296    // certified below; its outer partition and G→R ring are reused exactly.
297    let outer_recipe = g21_comparator(recipe, true);
298    let outer_blocks = build_interval_bore_prism(&outer_recipe)?;
299    let outer_certificate = audit_interval_bore_prism(&outer_recipe, &outer_blocks)?;
300    let pilot_recipe = g21_comparator(recipe, false);
301    let pilot_blocks = build_interval_bore_prism(&pilot_recipe)?;
302    audit_interval_bore_prism(&pilot_recipe, &pilot_blocks)?;
303
304    let mut actual = BTreeMap::new();
305    for block in blocks {
306        if actual.insert(block.id.as_str(), block).is_some() {
307            return Err("counterbore_duplicate_block");
308        }
309        let _ = primitive_block(block)?;
310    }
311    let mut expected_roles = BTreeMap::<String, ExpectedRole>::new();
312    let mut min_det = f64::INFINITY;
313    let mut expected_count = 0usize;
314    for pair in events.windows(2) {
315        let [z0, z1] = [pair[0], pair[1]];
316        for reference in outer_blocks
317            .iter()
318            .filter(|block| block.id.ends_with(&format!("_z{z0}_{z1}")))
319        {
320            let ring = mapped_counter_ring_id(&reference.id, &ordered);
321            let is_counter_ring = ring.is_some();
322            let (id, role) = if let Some((mapped, bore_id)) = ring {
323                let bore = ordered
324                    .iter()
325                    .find(|b| b.id == bore_id)
326                    .ok_or("counterbore_unknown_counter_ring")?;
327                let active = match bore.opening {
328                    CounterboreOpening::Start => z0 < bore.depth_mm,
329                    CounterboreOpening::End => z1 > recipe.height_mm - bore.depth_mm,
330                };
331                (
332                    mapped,
333                    ExpectedRole::CounterRing {
334                        bore: bore_id,
335                        active,
336                        z0,
337                        z1,
338                    },
339                )
340            } else {
341                (reference.id.clone(), ExpectedRole::Outer)
342            };
343            if is_counter_ring || reference.id.starts_with("tri/") {
344                let a = actual
345                    .get(id.as_str())
346                    .ok_or("counterbore_missing_reused_block")?;
347                if a.degrees != reference.degrees
348                    || primitive_block(a)? != primitive_block(reference)?
349                {
350                    return Err("counterbore_altered_reused_g21_block");
351                }
352                expected_roles.insert(id, role);
353                expected_count += 1;
354                if is_counter_ring {
355                    let det = planar_lower_bound(reference)? * f64::from(z1 - z0);
356                    if !det.is_finite() || det <= 0.0 {
357                        return Err("counterbore_nonpositive_jacobian");
358                    }
359                    min_det = min_det.min(det);
360                }
361            }
362        }
363        {
364            for bore in &ordered {
365                let is_pilot_layer = match bore.opening {
366                    CounterboreOpening::Start => z0 >= bore.depth_mm,
367                    CounterboreOpening::End => z1 <= recipe.height_mm - bore.depth_mm,
368                };
369                if !is_pilot_layer {
370                    continue;
371                }
372                for q in 0..4 {
373                    for half in 0..2 {
374                        let expected = expected_pilot_sector(bore, recipe, z0, z1, q, half)?;
375                        let actual_block = actual
376                            .get(expected.id.as_str())
377                            .ok_or("counterbore_missing_pilot_sector")?;
378                        if actual_block.degrees != expected.degrees
379                            || primitive_block(actual_block)? != primitive_block(&expected)?
380                        {
381                            return Err("counterbore_altered_pilot_sector");
382                        }
383                        let det = planar_lower_bound(&expected)? * f64::from(z1 - z0);
384                        if !det.is_finite() || det <= 0.0 {
385                            return Err("counterbore_nonpositive_jacobian");
386                        }
387                        min_det = min_det.min(det);
388                        expected_roles.insert(
389                            expected.id.clone(),
390                            ExpectedRole::PilotRing {
391                                bore: bore.id.clone(),
392                                z0,
393                                z1,
394                            },
395                        );
396                        expected_count += 1;
397                    }
398                }
399            }
400        }
401    }
402    if expected_count != blocks.len() || actual.keys().any(|id| !expected_roles.contains_key(*id)) {
403        return Err("counterbore_block_inventory_mismatch");
404    }
405
406    // The exact R and r circle profiles were certified by the two G21 audits;
407    // the independently reconstructed pilot net matches the fixed conic
408    // profiles coefficient-for-coefficient after a positive common gauge.
409    let incidence = enumerate_faces(blocks)?;
410    let mut g21_tags = BTreeMap::<(String, usize, bool), String>::new();
411    for (side, label) in &outer_certificate.certificate.design_faces {
412        g21_tags.insert((side.block.clone(), side.axis, side.upper), label.clone());
413    }
414    let mut design_faces = Vec::new();
415    for side in &incidence.exterior {
416        let role = expected_roles
417            .get(&side.block)
418            .ok_or("counterbore_unknown_boundary")?;
419        let label = match role {
420            ExpectedRole::Outer => g21_tags
421                .get(&(side.block.clone(), side.axis, side.upper))
422                .cloned()
423                .ok_or("counterbore_unclassified_outer_face")?,
424            ExpectedRole::CounterRing {
425                bore,
426                active,
427                z0,
428                z1,
429            } => {
430                if side.axis == 1 && side.upper {
431                    if !active {
432                        return Err("counterbore_exposed_internal_radius");
433                    }
434                    format!("bore/{bore}/counterbore_wall")
435                } else if side.axis == 2 && !side.upper && *z0 == 0 {
436                    "start".to_owned()
437                } else if side.axis == 2 && side.upper && *z1 == recipe.height_mm {
438                    "end".to_owned()
439                } else {
440                    return Err("counterbore_unexpected_counter_ring_boundary");
441                }
442            }
443            ExpectedRole::PilotRing { bore, z0, z1 } => {
444                if side.axis == 1 && side.upper {
445                    format!("bore/{bore}/pilot_wall")
446                } else if side.axis == 2 {
447                    let shoulder = ordered
448                        .iter()
449                        .find(|b| b.id == *bore)
450                        .ok_or("counterbore_unknown_bore")?;
451                    let z = match shoulder.opening {
452                        CounterboreOpening::Start => shoulder.depth_mm,
453                        CounterboreOpening::End => recipe.height_mm - shoulder.depth_mm,
454                    };
455                    if side.upper && *z1 == z || !side.upper && *z0 == z {
456                        format!("bore/{bore}/shoulder")
457                    } else if !side.upper && *z0 == 0 {
458                        "start".to_owned()
459                    } else if side.upper && *z1 == recipe.height_mm {
460                        "end".to_owned()
461                    } else {
462                        return Err("counterbore_unexpected_pilot_cap");
463                    }
464                } else {
465                    return Err("counterbore_unexpected_pilot_boundary");
466                }
467            }
468        };
469        design_faces.push((side.clone(), label));
470    }
471    design_faces.sort_by(|a, b| a.0.cmp(&b.0));
472    for bore in &ordered {
473        let count = |feature: &str| {
474            design_faces
475                .iter()
476                .filter(|(_, tag)| tag == &format!("bore/{}/{feature}", bore.id))
477                .count()
478        };
479        let pilot_layers = events
480            .windows(2)
481            .filter(|pair| match bore.opening {
482                CounterboreOpening::Start => pair[0] >= bore.depth_mm,
483                CounterboreOpening::End => pair[1] <= recipe.height_mm - bore.depth_mm,
484            })
485            .count();
486        let counter_layers = events
487            .windows(2)
488            .filter(|pair| match bore.opening {
489                CounterboreOpening::Start => pair[1] <= bore.depth_mm,
490                CounterboreOpening::End => pair[0] >= recipe.height_mm - bore.depth_mm,
491            })
492            .count();
493        if count("pilot_wall") != 8 * pilot_layers
494            || count("counterbore_wall") != 8 * counter_layers
495            || count("shoulder") != 8
496        {
497            return Err("counterbore_face_provenance_inventory");
498        }
499    }
500    let outer_area_twice_mm2 = outer_certificate.outer_area_twice_mm2;
501    let removed = ordered.iter().try_fold(0_i64, |sum, bore| {
502        let r = i64::from(bore.radius_mm);
503        let big = i64::from(bore.counterbore_radius_mm);
504        let h = i64::from(recipe.height_mm);
505        let d = i64::from(bore.depth_mm);
506        let r2 = r.checked_mul(r).ok_or("counterbore_volume_overflow")?;
507        let big2 = big.checked_mul(big).ok_or("counterbore_volume_overflow")?;
508        let value = r2
509            .checked_mul(h)
510            .and_then(|x| {
511                big2.checked_sub(r2)
512                    .and_then(|z| z.checked_mul(d))
513                    .and_then(|y| x.checked_add(y))
514            })
515            .ok_or("counterbore_volume_overflow")?;
516        sum.checked_add(value).ok_or("counterbore_volume_overflow")
517    })?;
518    min_det = min_det.min(outer_certificate.certificate.determinant_lower_bound_mm3);
519    let certificate = BracketCertificate {
520        incidence,
521        determinant_lower_bound_mm3: min_det,
522        radial_error_upper_mm: 0.0,
523        design_faces,
524    };
525    Ok(CounterboreCertificate {
526        certificate,
527        outer_area_twice_mm2,
528        removed_pi_volume_coefficient_mm3: removed,
529        enclosed_voids: 0,
530        through_bores: ordered.len(),
531    })
