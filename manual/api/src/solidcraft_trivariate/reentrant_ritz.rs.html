1//! Scalar harmonic Ritz benchmark on the three extruded charts of an L section.
2//! Displacements are independent of height. The outer trace is mass-projected;
3//! stiffness integration is exact for the piecewise bicubic polynomial basis.
4use crate::{
5    bezier::basis,
6    constrained_elasticity::solve_constrained,
7    constraint_system::{resolve, Ratio},
8    elasticity::{gauss, Stiffness},
9    json::{object, Json},
10    thb_reentrant_edge::{IntegrationLeaf, ReentrantHierarchy},
11};
12use std::collections::BTreeMap;
13type Result<T> = std::result::Result<T, &'static str>;
14type Rows = Vec<BTreeMap<usize, f64>>;
15fn scalar(x: impl ToString) -> Json {
16    Json::Number(x.to_string())
17}
18fn chart(p: usize, [u, v]: [f64; 2]) -> [f64; 2] {
19    match p {
20        0 => [-u, -v],
21        1 => [v, -u],
22        2 => [-v, u],
23        _ => unreachable!(),
24    }
25}
26fn exact([x, y]: [f64; 2]) -> (f64, [f64; 2]) {
27    let theta = (-y).atan2(x).rem_euclid(std::f64::consts::TAU);
28    let r = x.hypot(y);
29    let k = 2. / 3.;
30    let (st, ct) = (k * theta).sin_cos();
31    let scale = k * r.powf(k - 1.);
32    (
33        r.powf(k) * st,
34        [
35            scale * (st * x / r + ct * y / r),
36            scale * (st * y / r - ct * x / r),
37        ],
38    )
39}
40fn widths(e: &IntegrationLeaf) -> [f64; 2] {
41    e.uv.map(|[a, b]| b - a)
42}
43fn xy(e: &IntegrationLeaf, t: [f64; 2]) -> [f64; 2] {
44    chart(
45        e.patch,
46        std::array::from_fn(|k| e.uv[k][0] + widths(e)[k] * t[k]),
47    )
48}
49/// Exterior section boundary. The two interior chart seams are eliminated in the basis.
50fn boundary(e: &IntegrationLeaf) -> Vec<(usize, usize)> {
51    let mut faces = Vec::new();
52    for a in 0..2 {
53        if e.uv[a][1] == 1. {
54            faces.push((a, 1))
55        }
56        if e.uv[a][0] == 0. && ((e.patch == 1 && a == 0) || (e.patch == 2 && a == 1)) {
57            faces.push((a, 0))
58        }
59    }
60    faces
61}
62/// Apply the kernel PCG solver in one active scalar component. The independent
63/// affine lift contributes the known nonzero Dirichlet values before reduction.
64fn solve_scalar(rows: &Rows, rhs: &[f64], active: &[usize]) -> Result<(Vec<f64>, f64)> {
65    let n = active.len();
66    let mut index = vec![usize::MAX; rows.len()];
67    for (j, &i) in active.iter().enumerate() {
68        index[i] = j;
69    }
70    let mut k = Stiffness {
71        offsets: vec![0],
72        columns: Vec::new(),
73        values: Vec::new(),
74        integrated_volume: 0.,
75    };
76    let mut f = vec![0.; 3 * n];
77    for (j, &i) in active.iter().enumerate() {
78        f[3 * j] = rhs[i];
79        for (&t, &v) in &rows[i] {
80            if index[t] != usize::MAX {
81                let mut m = [[0.; 3]; 3];
82                m[0][0] = v;
83                k.values.push(m);
84                k.columns.push(index[t]);
85            }
86        }
87        k.offsets.push(k.columns.len());
88    }
89    let free = resolve(n, &[], &[])?;
90    let fixed = resolve(
91        n,
92        &[],
93        &(0..n).map(|i| (i, Ratio::zero())).collect::<Vec<_>>(),
94    )?;
95    let solution = solve_constrained(&k, [&free, &fixed, &fixed], &f, 1e-12, 20000)?;
96    Ok((
97        solution
98            .displacement_mm
99            .as_chunks::<3>()
100            .0
101            .iter()
102            .map(|r| r[0])
103            .collect(),
104        solution.relative_residual,
105    ))
106}
107fn project_trace(leaves: &[IntegrationLeaf], count: usize) -> Result<(Vec<f64>, Vec<usize>)> {
108    let mut mass = vec![BTreeMap::new(); count];
109    let mut load = vec![0.; count];
110    let q = gauss(16)?;
111    for e in leaves {
112        for (a, b) in boundary(e) {
113            let t_axis = 1 - a;
114            for &(t, w) in &q {
115                let mut loc = [t; 2];
116                loc[a] = b as f64;
117                let (bern, _) = basis(3, t);
118                let wt = w * widths(e)[t_axis];
119                let value = exact(xy(e, loc)).0;
120                let mut trace = vec![0.; e.dofs.len()];
121                for (j, &nj) in bern.iter().enumerate() {
122                    let ij = if a == 0 { 12 * b + j } else { 4 * j + 3 * b };
123                    for (v, c) in trace.iter_mut().zip(&e.bernstein[ij]) {
124                        *v += nj * c;
125                    }
126                }
127                for (i, &id) in e.dofs.iter().enumerate().filter(|(i, _)| trace[*i] != 0.) {
128                    load[id] += wt * trace[i] * value;
129                    for (j, &jd) in e.dofs.iter().enumerate().filter(|(j, _)| trace[*j] != 0.) {
130                        *mass[id].entry(jd).or_default() += wt * trace[i] * trace[j];
131                    }
132                }
133            }
134        }
135    }
136    let active: Vec<_> = (0..count).filter(|&i| !mass[i].is_empty()).collect();
137    let (trace, residual) = solve_scalar(&mass, &load, &active)?;
138    if residual > 1e-10 {
139        return Err("boundary projection residual too large");
140    }
141    let mut field = vec![0.; count];
142    for (&i, x) in active.iter().zip(trace) {
143        field[i] = x;
144    }
145    Ok((field, (0..count).filter(|&i| mass[i].is_empty()).collect()))
146}
147/// Tensor Bernstein stiffness on the unit square. All cells are square in
148/// physical coordinates, so the Jacobian cancels the two gradient scales.
149fn stiffness(leaves: &[IntegrationLeaf], count: usize) -> Result<Rows> {
150    let q = gauss(4)?;
151    let mut k = [[0.; 16]; 16];
152    for &(u, wu) in &q {
153        for &(v, wv) in &q {
154            let (bu, du) = basis(3, u);
155            let (bv, dv) = basis(3, v);
156            let grads: [[f64; 2]; 16] =
157                std::array::from_fn(|i| [du[i / 4] * bv[i % 4], bu[i / 4] * dv[i % 4]]);
158            for (i, a) in grads.iter().enumerate() {
159                for (j, b) in grads.iter().enumerate() {
160                    k[i][j] += wu * wv * (a[0] * b[0] + a[1] * b[1]);
161                }
162            }
163        }
164    }
165    let mut rows = vec![BTreeMap::new(); count];
166    for e in leaves {
167        let [hx, hy] = widths(e);
168        if hx <= 0. || hx != hy {
169            return Err("expected positive dyadic square");
170        }
171        let mut temp = vec![vec![0.; e.dofs.len()]; 16];
172        for (i, t) in temp.iter_mut().enumerate() {
173            for (j, ej) in e.bernstein.iter().enumerate() {
174                for (a, &c) in t.iter_mut().zip(ej) {
175                    *a += k[i][j] * c;
176                }
177            }
178        }
179        for (j, ej) in e.bernstein.iter().enumerate() {
180            for (&i, &c) in e.dofs.iter().zip(ej).filter(|(_, v)| **v != 0.) {
181                for (&t, &x) in e.dofs.iter().zip(&temp[j]) {
182                    *rows[i].entry(t).or_default() += c * x;
183                }
184            }
185        }
186    }
187    Ok(rows)
188}
189fn value(c: &[f64; 16], loc: [f64; 2]) -> (f64, [f64; 2]) {
190    let (bu, du) = basis(3, loc[0]);
191    let (bv, dv) = basis(3, loc[1]);
192    let mut f = 0.;
193    let mut grad = [0.; 2];
194    for (i, x) in c.iter().enumerate() {
195        f += x * bu[i / 4] * bv[i % 4];
196        grad[0] += x * du[i / 4] * bv[i % 4];
197        grad[1] += x * bu[i / 4] * dv[i % 4];
198    }
199    (f, grad)
200}
201fn integrate(leaves: &[IntegrationLeaf], fields: &[[f64; 16]], order: usize) -> Result<[f64; 3]> {
202    let q = gauss(order)?;
203    let mut e2 = 0.;
204    let mut g2 = 0.;
205    let mut u2 = 0.;
206    let mut du2 = 0.;
207    let mut be = 0.;
208    let mut bn = 0.;
209    for (e, c) in leaves.iter().zip(fields) {
210        let [hx, hy] = widths(e);
211        let mut point = |t: [f64; 2], weight: f64| {
212            let (u, g) = exact(xy(e, t));
213            let (uh, d) = value(c, t);
214            let gh = chart(e.patch, [d[0] / hx, d[1] / hy]);
215            e2 += weight * ((g[0] - gh[0]).powi(2) + (g[1] - gh[1]).powi(2));
216            g2 += weight * (g[0] * g[0] + g[1] * g[1]);
217            u2 += weight * (u - uh).powi(2);
218            du2 += weight * u * u;
219        };
220        let touches = e.uv.iter().all(|i| i[0] == 0.);
221        for &(s, ws) in &q {
222            for &(t, wt) in &q {
223                let w = ws * wt * hx * hy;
224                if touches {
225                    // Split the corner square into two triangles. Duffy radius
226                    // r=s³ resolves the fractional power in the energy integrand.
227                    let r = s.powi(3);
228                    let jac = 3. * s * s * r;
229                    point([r, r * t], w * jac);
230                    point([r * t, r], w * jac);
231                } else {
232                    point([s, t], w);
233                }
234            }
235        }
236        for (a, b) in boundary(e) {
237            for &(s, w) in &q {
238                let mut t = [s; 2];
239                t[a] = b as f64;
240                let u = exact(xy(e, t)).0;
241                let uh = value(c, t).0;
242                be += w * widths(e)[1 - a] * (u - uh).powi(2);
243                bn += w * widths(e)[1 - a] * u * u;
244            }
245        }
246    }
247    Ok([(e2 / g2).sqrt(), (u2 / du2).sqrt(), (be / bn).sqrt()])
248}
249/// Solve the planar scalar Ritz problem and integrate the error directly on
250/// its physical cells. The non-polynomial trace is projected independently at
251/// each level. Area, dimensionality and the complete quadrature history are reported.
252pub fn solve(space: &ReentrantHierarchy) -> Result<Json> {
253    let leaves = space.integration_leaves()?;
254    let count = space.active_scalar_dofs;
255    let (mut field, free) = project_trace(&leaves, count)?;
256    let k = stiffness(&leaves, count)?;
257    let rhs = k
258        .iter()
259        .map(|r| -r.iter().map(|(&j, &c)| field[j] * c).sum::<f64>())
260        .collect::<Vec<_>>();
261    let (solution, residual) = solve_scalar(&k, &rhs, &free)?;
262    for (&i, x) in free.iter().zip(solution) {
263        field[i] = x
264    }
265    let fields = leaves
266        .iter()
267        .map(|e| {
268            std::array::from_fn(|j| {
269                e.dofs
270                    .iter()
271                    .zip(&e.bernstein[j])
272                    .map(|(&i, v)| field[i] * v)
273                    .sum()
274            })
275        })
276        .collect::<Vec<_>>();
277    let mut history = Vec::new();
278    let mut previous: Option<[f64; 2]> = None;
279    let mut stable = false;
280    let mut norms = [0.; 3];
281    for q in [8, 12, 16, 24, 32] {
282        norms = integrate(&leaves, &fields, q)?;
283        if let Some([h, l]) = previous {
284            stable = (norms[0] / h - 1_f64).abs() <= 2e-3 && (norms[1] / l - 1_f64).abs() <= 5e-4;
285        }
286        previous = Some([norms[0], norms[1]]);
287        history.push(object([
288            ("order", scalar(q)),
289            ("relative_H1_seminorm", scalar(norms[0])),
290            ("relative_L2", scalar(norms[1])),
291            ("relative_boundary_L2", scalar(norms[2])),
292        ]));
293    }
294    let area = leaves
295        .iter()
296        .map(|e| widths(e).iter().product::<f64>())
297        .sum::<f64>();
298    if (area - 3.).abs() > 1e-13 || residual > 1e-10 || !stable {
299        return Err("singular benchmark failed area, residual or quadrature acceptance");
300    }
301    Ok(object([
302        ("active_scalar_dofs", scalar(count)),
303        ("free_scalar_dofs", scalar(free.len())),
304        ("maximum_level", scalar(space.maximum_level)),
305        ("leaves", scalar(leaves.len())),
306        ("area", scalar(area)),
307        ("relative_free_residual", scalar(residual)),
308        ("relative_H1_seminorm", scalar(norms[0])),
309        ("relative_L2", scalar(norms[1])),
310        ("relative_boundary_L2", scalar(norms[2])),
311        ("quadrature_history", Json::Array(history)),
312    ]))
313}

319    fn polar_branch_matches_both_notch_sides_and_all_chart_gradients() {
320        assert!(exact([0.4, 0.]).0.abs() < 1e-14);
321        assert!(exact([0., 0.4]).0.abs() < 1e-14);
322        for p in 0..3 {
323            let x = chart(p, [0.23, 0.47]);
324            let (_, g) = exact(x);
325            for a in 0..2 {
326                let mut lo = x;
327                let mut hi = x;
328                lo[a] -= 1e-6;
329                hi[a] += 1e-6;

331            }
