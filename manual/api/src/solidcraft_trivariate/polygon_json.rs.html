1//! Versioned polygon extrusion exchange.
2use crate::{
3    bracket::BracketBlock,
4    bracket_json::{decode_blocks, design, encode, topology},
5    json::{object, Json},
6    perforated_prism::{build_perforated_prism, LoopProfile, PerforatedPrism},
7    perforated_prism_audit::audit_perforated_prism,
8    polygon_prism::{build_polygon_prism, PolygonPrism, Vertex},
9    polygon_prism_audit::{audit_polygon_prism, PolygonCertificate},
10};
11pub(crate) const FAMILY: &str = "polygon_prism_v1";
12pub(crate) const PERFORATED: &str = "polygonal_perforated_prism_v1";
13fn n(v: impl ToString) -> Json {
14    Json::Number(v.to_string())
15}
16pub(super) fn pair(v: &Json) -> Result<[i32; 2], &'static str> {
17    let [a, b] = v.items()? else {
18        return Err("expected two polygon coordinates");
19    };
20    Ok([a.number()?, b.number()?])
21}
22fn vertices(j: &Json, max: usize) -> Result<Vec<Vertex>, &'static str> {
23    let items = j.items()?;
24    if !(3..=max).contains(&items.len()) {
25        return Err("polygon_prism_range");
26    }
27    let mut vertices = Vec::with_capacity(items.len());
28    for v in items {
29        v.keys(&["id", "xy_mm"])?;
30        vertices.push(Vertex {
31            id: v.field("id")?.text()?.to_owned(),
32            xy_mm: pair(v.field("xy_mm")?)?,
33        });
34    }
35    Ok(vertices)
36}
37pub(super) fn profile(j: &Json) -> Result<LoopProfile, &'static str> {
38    j.keys(&["id", "vertices"])?;
39    Ok(LoopProfile {
40        id: j.field("id")?.text()?.into(),
41        vertices: vertices(j.field("vertices")?, 32)?,
42    })
43}
44enum Prism {
45    Simple(PolygonPrism),
46    Perforated(PerforatedPrism),
47}
48impl Prism {
49    fn parse(j: &Json, family: &str) -> Result<Self, &'static str> {
50        let height_mm = j.field("height_mm")?.number()?;
51        let top_shift_mm = pair(j.field("top_shift_mm")?)?;
52        match family {
53            FAMILY => Ok(Self::Simple(PolygonPrism {
54                vertices: vertices(j.field("vertices")?, 24)?,
55                height_mm,
56                top_shift_mm,
57            })),
58            PERFORATED => {
59                let holes = j.field("holes")?.items()?;
60                if holes.len() > 4 {
61                    return Err("too many polygon holes");
62                }
63                Ok(Self::Perforated(PerforatedPrism {
64                    outer: profile(j.field("outer")?)?,
65                    holes: holes.iter().map(profile).collect::<Result<_, _>>()?,
66                    height_mm,
67                    top_shift_mm,
68                }))
69            }
70            _ => Err("unknown polygon extrusion family"),
71        }
72    }
73    fn build(&self) -> Result<Vec<BracketBlock>, &'static str> {
74        match self {
75            Self::Simple(r) => build_polygon_prism(r),
76            Self::Perforated(r) => build_perforated_prism(r),
77        }
78    }
79    fn audit(&self, blocks: &[BracketBlock]) -> Result<PolygonCertificate, &'static str> {
80        match self {
81            Self::Simple(r) => audit_polygon_prism(r, blocks),
82            Self::Perforated(r) => audit_perforated_prism(r, blocks),
83        }
84    }
85}
86fn keys(family: &str) -> Result<&'static [&'static str], &'static str> {
87    match family {
88        FAMILY => Ok(&["vertices", "height_mm", "top_shift_mm"]),
89        PERFORATED => Ok(&["outer", "holes", "height_mm", "top_shift_mm"]),
90        _ => Err("unknown polygon extrusion family"),
91    }
92}
93pub(crate) fn build(j: &Json) -> Result<Json, &'static str> {
94    let family = j.field("family")?.text()?;
95    let fields = keys(family)?;
96    j.keys(&[&["family"][..], fields].concat())?;
97    let r = Prism::parse(j, family)?;
98    let blocks = r.build()?;
99    let audit = r.audit(&blocks)?;
100    let mut model = encode(family, blocks, &audit.certificate);
101    if let Json::Object(map) = &mut model {
102        for &key in fields {
103            map.insert(key.into(), j.field(key)?.clone());
104        }
105    }
106    Ok(model)
107}
108pub(crate) fn audit(j: &Json) -> Result<Json, &'static str> {
109    let family = j.field("schema")?.text()?;
110    j.keys(
111        &[
112            &["schema", "blocks", "topology", "design_faces"][..],
113            keys(family)?,
114        ]
115        .concat(),
116    )?;
117    let r = Prism::parse(j, family)?;
118    let blocks = decode_blocks(j)?;
119    let result = r.audit(&blocks)?;
120    let method = match r {
121        Prism::Simple(_) => "certified_polygon_prism",
122        Prism::Perforated(_) => "certified_perforated_polygon_prism",
123    };
124    let c = &result.certificate;
125    if j.field("topology")? != &topology(&c.incidence) || j.field("design_faces")? != &design(c) {
126        return Err("polygon_topology_or_provenance_mismatch");
127    }
128    Ok(object([
129        ("status", Json::String("valid".into())),
130        ("method", Json::String(method.into())),
131        ("blocks", n(blocks.len())),
132        ("shared_faces", n(c.incidence.interfaces.len())),
133        ("exterior_sides", n(c.incidence.exterior.len())),
134        (
135            "determinant_lower_bound_mm3",
136            n(c.determinant_lower_bound_mm3),
137        ),
138        (
139            "projected_area_twice_mm2",
140            n(result.projected_area_twice_mm2),
141        ),
142        ("volume_sixths_mm3", n(result.volume_sixths_mm3)),
143    ]))
144}
