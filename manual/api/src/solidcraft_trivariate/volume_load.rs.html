10pub(crate) struct Density {
11    constant: Vector,
12    gradient: [Vector; 3],
13}
14
15impl Density {
16    pub(crate) fn parse(spec: &Json) -> Result<Self, &'static str> {
17        spec.keys(&["constant_n_per_mm3", "gradient_n_per_mm4"])?;
18        let [a, b, c] = spec.field("gradient_n_per_mm4")?.items()? else {
19            return Err("expected three body force gradient rows");
20        };
21        Ok(Self {
22            constant: vector(spec.field("constant_n_per_mm3")?)?,
23            gradient: [vector(a)?, vector(b)?, vector(c)?],
24        })
25    }
26
27    fn variation(&self, x: Vector) -> Vector {
28        self.gradient
29            .map(|r| r.into_iter().zip(x).map(|(a, b)| a * b).sum())
30    }
31
32    pub(crate) fn integrate(
33        &self,
34        space: &AnalysisSpace,
35        order: usize,
36        velocity: Option<&Velocity>,
37    ) -> Result<VolumeLoad, &'static str> {
38        let q = gauss(order)?;
39        let size = space
40            .ndof
41            .checked_mul(3)
42            .ok_or("body load dimension overflow")?;
43        let mut out = VolumeLoad {
44            force: vec![0.; size],
45            force_rate: velocity.map(|_| vec![0.; size]),
46            volume: 0.,
47            scale: 0.,
48            moment_scale: 0.,
49            resultant: [0.; 3],
50            moment: [0.; 3],
51        };
52        for block in &space.blocks {
53            if block.dofs.len() != 27 || block.dofs.iter().any(|&id| id >= space.ndof) {
54                return Err("invalid body load block DOFs");
55            }
56            for &(u, a) in &q {
57                for &(v, b) in &q {
58                    for &(w, c) in &q {
59                        let sample = basis_geometry(block, [u, v, w])?;
60                        let measure = sample.determinant * a * b * c;
61                        let x = sample.position;
62                        let linear = self.variation(x);
63                        let density: Vector = std::array::from_fn(|i| self.constant[i] + linear[i]);
64                        let df = density.map(|f| f * measure);
65                        out.volume += measure;
66                        out.scale += df[0].hypot(df[1]).hypot(df[2]);
67                        let moment: Vector = std::array::from_fn(|i| {
68                            let j = (i + 1) % 3;
69                            let k = (i + 2) % 3;
70                            x[j] * df[k] - x[k] * df[j]
71                        });
72                        out.moment_scale += moment[0].hypot(moment[1]).hypot(moment[2]);
73                        for i in 0..3 {
74                            out.resultant[i] += df[i];
75                            out.moment[i] += moment[i];
76                        }
77                        let distribute = |nodal: &mut [f64], load: Vector| {
78                            for (&id, shape) in block.dofs.iter().zip(sample.values) {
79                                for (i, f) in load.iter().enumerate() {
80                                    nodal[3 * id + i] += shape * f;
81                                }
82                            }
83                        };
84                        distribute(&mut out.force, df);
85                        if let (Some(flow), Some(rate)) = (velocity, &mut out.force_rate) {
86                            let gradient_rate = self.variation(flow.at(x));
87                            let div = flow.divergence(x);
88                            distribute(
89                                rate,
90                                std::array::from_fn(|i| {
91                                    measure * (gradient_rate[i] + div * density[i])
92                                }),
93                            );
94                        }
95                    }
96                }
97            }
98        }
99        if !out.scale.is_finite()
100            || !out.moment_scale.is_finite()
101            || !out.volume.is_finite()
102            || out
103                .force
104                .iter()
105                .chain(out.force_rate.iter().flatten())
106                .chain(&out.resultant)
107                .chain(&out.moment)
108                .any(|x| !x.is_finite())
109        {
110            return Err("nonfinite integrated body load");
111        }
112        Ok(out)
113    }
114}

126/// Adjacent global tensor Gauss rules for the volume right-hand side.
127pub(crate) struct BodyQuadrature {
128    min_order: usize,
129    max_order: usize,
130    relative_tolerance: f64,
131    max_points: usize,
132}
133
134impl BodyQuadrature {
135    pub(crate) fn parse(spec: &Json) -> Result<Self, &'static str> {
136        spec.keys(&["min_order", "max_order", "relative_tolerance", "max_points"])?;
137        let config = Self {
138            min_order: spec.field("min_order")?.number()?,
139            max_order: spec.field("max_order")?.number()?,
140            relative_tolerance: spec.field("relative_tolerance")?.number()?,
141            max_points: spec.field("max_points")?.number()?,
142        };
143        if !(2..=16).contains(&config.min_order)
144            || config.max_order > 16
145            || config.min_order >= config.max_order
146            || !config.relative_tolerance.is_finite()
147            || config.relative_tolerance <= 0.
148            || config.relative_tolerance > 1.
149            || config.max_points == 0
150        {
151            return Err("invalid body quadrature configuration");
152        }
153        Ok(config)
154    }
155
156    pub(crate) fn integrate(
157        &self,
158        density: &Density,
159        space: &AnalysisSpace,
160        velocity: Option<&Velocity>,
161    ) -> Result<(VolumeLoad, Json), &'static str> {
162        use crate::json::object;
163        let n = |v: usize| Json::Number(v.to_string());
164        let mut points = 0usize;
165        let mut levels = Vec::new();
166        let mut previous: Option<VolumeLoad> = None;
167        let mut order = self.min_order;
168        loop {
169            let sample_count = space
170                .blocks
171                .len()
172                .checked_mul(order.pow(3))
173                .ok_or("body quadrature count overflow")?;
174            points = points
175                .checked_add(sample_count)
176                .ok_or("body quadrature count overflow")?;
177            if points > self.max_points {
178                return Err("body quadrature point budget exceeded");
179            }
180            let integrated = density.integrate(space, order, velocity)?;
181            let estimate = previous
182                .as_ref()
183                .map(|old| body_discrepancy(old, &integrated))
184                .transpose()?;
185            levels.push(object([
186                ("order", n(order)),
187                ("points", n(sample_count)),
188                (
189                    "relative_discrepancies",
190                    estimate.map_or(Json::Null, |v| {
191                        Json::Array(v.map(|x| Json::Number(x.to_string())).to_vec())
192                    }),
193                ),
194            ]));
195            if estimate.is_some_and(|values| values.iter().all(|v| *v <= self.relative_tolerance)) {
196                return Ok((
197                    integrated,
198                    object([
199                        ("selected_order", n(order)),
200                        ("points", n(points)),
201                        (
202                            "metrics",
203                            Json::Array(
204                                ["volume", "nodal_force", "resultant", "moment"]
205                                    .map(|v| Json::String(v.into()))
206                                    .to_vec(),
207                            ),
208                        ),
209                        ("history", Json::Array(levels)),
210                    ]),
211                ));
212            }
213            if order == self.max_order {
214                return Err("body quadrature tolerance not met within order budget");
215            }
216            previous = Some(integrated);
217            order = (order + 2).min(self.max_order);
218        }
219    }
220}

222fn body_discrepancy(old: &VolumeLoad, new: &VolumeLoad) -> Result<[f64; 4], &'static str> {
223    fn relative(a: &[f64], b: &[f64], scale: f64) -> f64 {
224        let delta = a.iter().zip(b).fold(0_f64, |s, (x, y)| s.hypot(x - y));
225        let norm = b.iter().fold(0_f64, |s, x| s.hypot(*x));
226        let denominator = if norm > 1e-12 * scale { norm } else { scale };
227        if denominator == 0. && delta == 0. {
228            0.
229        } else {
230            delta / denominator
231        }
232    }
233    let values = [
234        (old.volume - new.volume).abs() / new.volume,
235        relative(&old.force, &new.force, new.scale),
236        relative(&old.resultant, &new.resultant, new.scale),
237        relative(&old.moment, &new.moment, new.moment_scale),
238    ];
239    if values.iter().any(|x| !x.is_finite()) {
240        return Err("invalid body quadrature estimate");
241    }
242    Ok(values)
243}
