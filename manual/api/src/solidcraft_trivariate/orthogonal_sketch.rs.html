74fn half(a: i32, b: i32) -> Ratio {
75    Ratio::new(i128::from(a) + i128::from(b), 2).expect("dyadic coordinate")
76}
77fn in_base(j: &OrthogonalSketch, p: [Ratio; 3]) -> Result<bool, &'static str> {
78    Ok(p[2].numerator() > 0
79        && p[2].checked_sub(ratio(j.height))?.numerator() < 0
80        && inside(&j.outer, [p[0], p[1]])?)
81}
82fn cell_at(j: &OrthogonalSketch, b: &Bounds) -> Result<bool, &'static str> {
83    in_base(j, b.map(|t| half(t[0], t[1])))
84}
85fn enclosed(c: &Bounds, b: &Bounds) -> bool {
86    (0..3).all(|k| b[k][0] <= c[k][0] && c[k][1] <= b[k][1])
87}
88fn overlap(a: &Bounds, b: &Bounds) -> bool {
89    (0..3).all(|k| a[k][0].max(b[k][0]) < a[k][1].min(b[k][1]))
90}
91fn straight(id: String, c: Bounds) -> BracketBlock {
92    let mut controls = Vec::with_capacity(8);
93    for x in c[0] {
94        for y in c[1] {
95            for z in c[2] {
96                controls.push([i64::from(x), i64::from(y), i64::from(z), 1]);
97            }
98        }
99    }
100    BracketBlock {
101        id,
102        degrees: [1; 3],
103        controls,
104    }
105}
106pub(crate) fn corner<'a>(j: &OrthogonalSketch, f: &'a Fillet) -> Result<Corner<'a>, &'static str> {
107    if !j.outer.vertices.contains(&f.vertex) || !(1..=1024).contains(&f.radius) {
108        return Err("invalid_sketch_fillet");
109    }
110    let mut notch = Vec::new();
111    for x in [-1, 1] {
112        for y in [-1, 1] {
113            let p = [
114                half(f.vertex[0], f.vertex[0] + x),
115                half(f.vertex[1], f.vertex[1] + y),
116            ];
117            if !inside(&j.outer, p)? {
118                notch.push([x, y]);
119            }
120        }
121    }
122    let &[sign] = notch.as_slice() else {
123        return Err("fillet_requires_a_reentrant_right_angle");
124    };
125    let thickness = std::array::from_fn::<_, 2, _>(|k| {
126        j.outer
127            .vertices
128            .iter()
129            .map(|v| (f.vertex[k] - v[k]) * sign[k])
130            .filter(|d| *d > 0)
131            .min()
132            .unwrap_or(0)
133    });
134    if thickness[0] <= 0 || thickness[1] <= 0 {
135        return Err("fillet_corner_frame_requires_positive_clearance");
136    }
137    let size_xy = [
138        thickness[0]
139            .checked_add(f.radius)
140            .ok_or("fillet_corner_frame_overflow")?,
141        thickness[1]
142            .checked_add(f.radius)
143            .ok_or("fillet_corner_frame_overflow")?,
144    ];
145    let origin = std::array::from_fn(|k| f.vertex[k] - thickness[k] * sign[k]);
146    let bounds = std::array::from_fn(|k| {
147        if k == 2 {
148            [0, j.height]
149        } else {
150            let h = origin[k] + sign[k] * size_xy[k];
151            [origin[k].min(h), origin[k].max(h)]
152        }
153    });
154    Ok(Corner {
155        f,
156        origin,
157        sign,
158        size_xy,
159        bounds,
160    })
161}
162fn tube<'a>(
163    j: &OrthogonalSketch,
164    b: &'a Bore,
165    base: &[BTreeSet<i32>; 3],
166    guard: i32,
167) -> Result<Tube<'a>, &'static str> {
168    if !in_base(j, b.center.map(ratio))? {
169        return Err("bore_seed_outside_material");
170    }
171    let levels: Vec<_> = base[b.axis].iter().copied().collect();
172    let mut runs: Vec<[i32; 2]> = Vec::new();
173    for w in levels.windows(2) {
174        let mut p = b.center.map(ratio);
175        p[b.axis] = half(w[0], w[1]);
176        if in_base(j, p)? {
177            if let Some(last) = runs.last_mut() {
178                if last[1] == w[0] {
179                    last[1] = w[1];
180                    continue;
181                }
182            }
183            runs.push([w[0], w[1]]);
184        }
185    }
186    let run = runs
187        .into_iter()
188        .find(|w| w[0] < b.center[b.axis] && b.center[b.axis] < w[1])
189        .ok_or("ambiguous_bore_axis_interval")?;
190    let bounds = std::array::from_fn(|k| {
191        if k == b.axis {
192            run
193        } else {
194            [b.center[k] - guard, b.center[k] + guard]
195        }
196    });
197    Ok(Tube { b, bounds, guard })
198}
199pub(crate) fn transform_fillet(
200    block: &mut BracketBlock,
201    c: &Corner<'_>,
202) -> Result<(), &'static str> {
203    for p in &mut block.controls {
204        for k in 0..2 {
205            p[k] = i64::try_from(
206                i128::from(c.origin[k]) * i128::from(p[3])
207                    + i128::from(c.sign[k]) * i128::from(p[k]),
208            )
209            .map_err(|_| "fillet_transform_overflow")?;
210        }
211    }
212    if c.sign[0] * c.sign[1] < 0 {
213        let stride = (block.degrees[1] + 1) * (block.degrees[2] + 1);
214        for i in 0..stride {
215            block.controls.swap(i, block.degrees[0] * stride + i);
216        }
217    }
218    Ok(())
219}
220fn side_points(block: &BracketBlock, axis: usize, upper: bool) -> Vec<[i64; 4]> {
221    let d = block.degrees;
222    block
223        .controls
224        .iter()
225        .enumerate()
226        .filter_map(|(i, &h)| {
227            let uvw = [
228                i / ((d[1] + 1) * (d[2] + 1)),
229                i / (d[2] + 1) % (d[1] + 1),
230                i % (d[2] + 1),
231            ];
232            (uvw[axis] == usize::from(upper) * d[axis]).then_some(h)
233        })
234        .collect()
235}
236fn planar_label(j: &OrthogonalSketch, points: &[[i64; 4]]) -> Option<String> {
237    let coord =
238        |p: &[i64; 4], k: usize, x: i32| i128::from(p[k]) - i128::from(x) * i128::from(p[3]);
239    for (z, name) in [(0, "start"), (j.height, "end")] {
240        if points.iter().all(|p| coord(p, 2, z) == 0) {
241            return Some(name.into());
242        }
243    }
244    for (a, b) in edges(&j.outer) {
245        let k = usize::from(a[1] == b[1]);
246        let along = 1 - k;
247        if a[k] == b[k]
248            && points.iter().all(|p| {
249                coord(p, k, a[k]) == 0
250                    && coord(p, along, a[along].min(b[along])) >= 0
251                    && coord(p, along, a[along].max(b[along])) <= 0
252            })
253        {
254            return Some(format!("outer/{}", j.outer.id));
255        }
256    }
257    None
258}
259/// Five unisolvent evaluations prove the quartic circle identity for each
260/// quadratic homogeneous curve. Identical transverse coefficients on the two
261/// axial ends extend the identity through the entire straight sweep.
262pub(crate) fn certify_circular_side(
263    block: &BracketBlock,
264    axis: usize,
265    center: [i32; 3],
266    radius: i32,
267) -> Result<(), &'static str> {
268    if block.degrees != [2, 1, 1] {
269        return Err("unexpected_circular_feature_degrees");
270    }
271    let a = (axis + 1) % 3;
272    let b = (axis + 2) % 3;
273    let mut net = [[0_i128; 3]; 3];
274    for (i, out) in net.iter_mut().enumerate() {
275        let p = block.controls[i * 4 + 2];
276        let q = block.controls[i * 4 + 3];
277        if [p[a], p[b], p[3]] != [q[a], q[b], q[3]] {
278            return Err("circular_feature_axial_mismatch");
279        }
280        *out = [
281            i128::from(p[a]) - i128::from(center[a]) * i128::from(p[3]),
282            i128::from(p[b]) - i128::from(center[b]) * i128::from(p[3]),
283            i128::from(radius) * i128::from(p[3]),
284        ];
285    }
286    for j in 0..=4i128 {
287        let basis = [(4 - j).pow(2), 2 * j * (4 - j), j.pow(2)];
288        let v =
289            std::array::from_fn::<_, 3, _>(|k| (0..3).map(|i| basis[i] * net[i][k]).sum::<i128>());
290        let sq = |x: i128| {
291            x.checked_mul(x)
292                .ok_or("circular_feature_certificate_overflow")
293        };
294        if sq(v[0])?.checked_add(sq(v[1])?) != Some(sq(v[2])?) {
295            return Err("circular_feature_polynomial_identity_failed");
296        }
297    }
298    Ok(())
299}
300/// For a rectangular prism the four outer edges can directly bound the four
301/// annular charts. No interior guard or continuation plane is needed.
302fn compact_plate(j: &OrthogonalSketch) -> Result<Option<SweepPartition>, &'static str> {
303    let [bore] = j.bores.as_slice() else {
304        return Ok(None);
305    };
306    if !j.fillets.is_empty() || j.outer.vertices.len() != 4 {
307        return Ok(None);
308    }
309    // Validation has already proved a simple orthogonal quadrilateral.
310    let bounds: Bounds = std::array::from_fn(|k| {
311        if k == 2 {
312            return [0, j.height];
313        }
314        j.outer
315            .vertices
316            .iter()
317            .fold([i32::MAX, i32::MIN], |range, p| {
318                [range[0].min(p[k]), range[1].max(p[k])]
319            })
320    });
321    if bore.axis >= 3
322        || !(1..=1024).contains(&bore.radius)
323        || bore.center.iter().any(|v| !(-4096..=4096).contains(v))
324        || bounds.iter().enumerate().any(|(k, &[lo, hi])| {
325            let margin = if k == bore.axis { 0 } else { bore.radius };
326            bore.center[k] - margin <= lo || bore.center[k] + margin >= hi
327        })
328    {
329        return Ok(None);
330    }
331    let blocks = crate::compact_wing::build(bore.axis, bounds, bore.center, bore.radius, &bore.id)?;
332    let mut physical = BTreeMap::new();
333    for b in &blocks {
334        certify_circular_side(b, bore.axis, bore.center, bore.radius)?;
335        physical.insert((b.id.clone(), 1, true), format!("bore/{}/wall", bore.id));
336    }
337    let thickness = bounds[bore.axis][1] - bounds[bore.axis][0];
338    let twice_constant = i128::from(area2(&j.outer)) * i128::from(j.height);
339    let four_pi = -4 * i128::from(bore.radius).pow(2) * i128::from(thickness);
340    finish_partition(
341        j,
342        blocks,
343        physical,
344        twice_constant,
345        four_pi,
346        bounds.map(|pair| pair.to_vec()),
347        vec![(bore.id.clone(), bounds)],
348    )
349    .map(Some)
350}

357pub(crate) fn build_with_strategy(
358    j: &OrthogonalSketch,
359    prefer_compact: bool,
360    compare_cuts: bool,
361) -> Result<SweepPartition, &'static str> {
362    validate(&SketchLayers {
363        outer: j.outer.clone(),
364        height: j.height,
365        shift: [0, 0],
366        strategy: crate::sketch_layers::PlannerStrategy::Automatic,
367        cuts: Vec::new(),
368        guides: Vec::new(),
369        force_triangulation: false,
370        operations: None,
371    })?;
372    if edges(&j.outer).any(|(a, b)| a[0] != b[0] && a[1] != b[1])
373        || j.bores.len() > 8
374        || j.fillets.len() > 4
375    {
376        return Err("unsupported_orthogonal_sketch");
377    }
378    let mut names = BTreeSet::new();
379    names.insert(&j.outer.id);
380    for id in j
381        .fillets
382        .iter()
383        .map(|v| &v.id)
384        .chain(j.bores.iter().map(|v| &v.id))
385    {
386        if !valid_name(id) || !names.insert(id) {
387            return Err("invalid_sketch_feature_identity");
388        }
389    }
390    if let Some(p) = crate::steinmetz_partition::try_build(j)? {
391        return Ok(p);
392    }
393    // Validate every bore before selecting a common guard using its neighbours.
394    if j.bores.iter().any(|b| {
395        b.axis >= 3
396            || !(1..=1024).contains(&b.radius)
397            || b.center.iter().any(|v| !(-4096..=4096).contains(v))
398    }) {
399        return Err("invalid_sketch_bore");
400    }
401    let prior = build_local_or_grid(j, prefer_compact);
402    if !compare_cuts {
403        return prior;
404    }
405    let propagated = match crate::compact_orthogonal::try_propagated(j) {
406        Ok(None) => return prior,
407        Ok(Some(p)) => Ok(p),
408        Err(reason) => Err(reason),
409    };
410    let block_count = |p: &SweepPartition| p.blocks.len();
411    let prior_count = prior.as_ref().map(block_count).map_err(|e| *e);
412    let propagated_count = propagated.as_ref().map(block_count).map_err(|e| *e);
413    let selected = match (prior_count, propagated_count) {
414        (Err(_), Err(reason)) => return Err(reason),
415        (_, Err(_)) => false,
416        (Err(_), Ok(_)) => true,
417        (Ok(a), Ok(b)) => b < a,
418    };
419    let mut best = if selected { propagated? } else { prior? };
420    best.cut_comparison = Some(CutComparison {
421        prior: prior_count,
422        propagated: propagated_count,
423        selected,
424    });
425    Ok(best)
426}

428fn build_local_or_grid(
429    j: &OrthogonalSketch,
430    prefer_compact: bool,
431) -> Result<SweepPartition, &'static str> {
432    let mut fallback_reason: Option<String> = None;
433    if prefer_compact {
434        let unequal_fillet = if j.fillets.len() == 1 {
435            corner(j, &j.fillets[0])
436                .ok()
437                .is_some_and(|c| c.size_xy[0] != c.size_xy[1])
438        } else {
439            false
440        };
441        if unequal_fillet {
442            fallback_reason = Some("compact_bracket_unequal_leg_thickness".into());
443        } else if j.fillets.is_empty() {
444            let plate_failure = match compact_plate(j) {
445                Ok(Some(mut candidate)) => {
446                    candidate.compact = true;
447                    return Ok(candidate);
448                }
449                Ok(None) => "compact_plate_outside_supported_class".to_owned(),
450                Err(error) => format!("compact_plate_error:{error}"),
451            };
452            match crate::compact_hosts::try_plate(j) {
453                Ok(Some(mut candidate)) => {
454                    candidate.compact = true;
455                    candidate.fallback_reason =
456                        Some(format!("{plate_failure};compact_local_host_selected"));
457                    return Ok(candidate);
458                }
459                Ok(None) => {
460                    fallback_reason = Some("compact_local_host_outside_supported_class".into());
461                }
462                Err(error) => {
463                    fallback_reason = Some(format!("compact_local_host_error:{error}"));
464                }
465            }
466        } else {
467            match crate::compact_orthogonal::try_build(j) {
468                Ok(Some(mut candidate)) => {
469                    candidate.compact = true;
470                    return Ok(candidate);
471                }
472                Ok(None) => {
473                    fallback_reason = Some("compact_bracket_outside_supported_class".into());
474                }
475                Err(error) => {
476                    fallback_reason = Some(format!("compact_bracket_error:{error}"));
477                }
478            }
479        }
480    }
481    let mut levels: [BTreeSet<i32>; 3] = std::array::from_fn(|k| {
482        if k == 2 {
483            BTreeSet::from([0, j.height])
484        } else {
485            j.outer.vertices.iter().map(|p| p[k]).collect()
486        }
487    });
488    let tubes = j
489        .bores
490        .iter()
491        .map(|b| {
492            // Perpendicular bores at a shared axial level use one common
493            // guard so their transverse subdivisions agree exactly.
494            let guard = j
495                .bores
496                .iter()
497                .filter(|other| {
498                    b.axis < 2
499                        && other.axis < 2
500                        && other.center[2] == b.center[2]
501                        && other.axis != b.axis
502                })
503                .map(|other| other.radius)
504                .chain(std::iter::once(b.radius))
505                .max()
506                .ok_or("empty_sketch_bore_set")?
507                .checked_add(1)
508                .ok_or("invalid_sketch_bore")?;
509            tube(j, b, &levels, guard)
510        })
511        .collect::<Result<Vec<_>, _>>()?;
512    let corners = j
513        .fillets
514        .iter()
515        .map(|f| corner(j, f))
516        .collect::<Result<Vec<_>, _>>()?;
517    let guards: Vec<_> = tubes
518        .iter()
519        .map(|t| (t.b.id.clone(), t.bounds))
520        .chain(corners.iter().map(|c| (c.f.id.clone(), c.bounds)))
521        .collect();
522    for (idx, (_, g)) in guards.iter().enumerate() {
523        if guards[..idx].iter().any(|(_, h)| overlap(g, h)) {
524            return Err("interacting_feature_guards_require_a_transition_partition");
525        }
526        for k in 0..3 {
527            levels[k].extend(g[k]);
528        }
529    }
530    for t in &tubes {
531        for (k, axis) in levels.iter_mut().enumerate() {
532            if k != t.b.axis {
533                axis.insert(t.b.center[k]);
534            }
535        }
536    }
537    let planes = levels.map(|s| s.into_iter().collect::<Vec<_>>());
538    for t in &tubes {
539        for (k, axis) in planes.iter().enumerate() {
540            if k != t.b.axis
541                && axis
542                    .iter()
543                    .any(|&x| x > t.bounds[k][0] && x < t.bounds[k][1] && x != t.b.center[k])
544            {
545                return Err("bore_guard_needs_additional_trace_subdivision");
546            }
547        }
548    }
549    let mut blocks = Vec::new();
550    let mut physical = BTreeMap::new();
551    for x in planes[0].windows(2) {
552        for y in planes[1].windows(2) {
553            for z in planes[2].windows(2) {
554                let cell = [[x[0], x[1]], [y[0], y[1]], [z[0], z[1]]];
555                let material = cell_at(j, &cell)?;
556                if tubes.iter().any(|t| enclosed(&cell, &t.bounds)) {
557                    if !material {
558                        return Err("bore_guard_leaves_sketch_material");
559                    }
560                    continue;
561                }
562                if let Some(c) = corners.iter().find(|c| enclosed(&cell, &c.bounds)) {
563                    let notch = (0..2)
564                        .all(|k| (cell[k][0] + cell[k][1] - 2 * c.f.vertex[k]) * c.sign[k] > 0);
565                    if material == notch {
566                        return Err("fillet_guard_is_not_a_simple_corner");
567                    }
568                    continue;
569                }
570                if material {
571                    blocks.push(straight(format!("cell_{}_{}_{}", x[0], y[0], z[0]), cell));
572                }
573            }
574        }
575    }
576    let mut twice_constant = i128::from(area2(&j.outer)) * i128::from(j.height);
577    let mut four_pi = 0i128;
578    for t in tubes {
579        for w in planes[t.b.axis]
580            .windows(2)
581            .filter(|w| w[0] >= t.bounds[t.b.axis][0] && w[1] <= t.bounds[t.b.axis][1])
582        {
583            let id = format!("bore_{}_{}_{}", t.b.id, w[0], w[1]);
584            for block in crate::sketch_bore_chart::bore_charts(
585                t.b.axis,
586                t.b.center,
587                [w[0], w[1]],
588                t.guard,
589                t.b.radius,
590                &id,
591            )? {
592                certify_circular_side(&block, t.b.axis, t.b.center, t.b.radius)?;
593                physical.insert((block.id.clone(), 1, true), format!("bore/{}/wall", t.b.id));
594                blocks.push(block);
595            }
596        }
597        four_pi -= 4
598            * i128::from(t.b.radius).pow(2)
599            * i128::from(t.bounds[t.b.axis][1] - t.bounds[t.b.axis][0]);
600    }
601    for c in corners {
602        let vol = i128::from(c.f.radius).pow(2) * i128::from(j.height);
603        twice_constant += 2 * vol;
604        four_pi -= vol;
605        for w in planes[2].windows(2) {
606            let id = format!("fillet_{}_{}_{}", c.f.id, w[0], w[1]);
607            for mut block in crate::sketch_fillet_chart::fillet_charts_rect(
608                [0, 0],
609                c.size_xy,
610                c.f.radius,
611                [w[0], w[1]],
612                &id,
613            )? {
614                transform_fillet(&mut block, &c)?;
615                let center = [
616                    c.f.vertex[0] + c.sign[0] * c.f.radius,
617                    c.f.vertex[1] + c.sign[1] * c.f.radius,
618                    0,
619                ];
620                certify_circular_side(&block, 2, center, c.f.radius)?;
621                physical.insert((block.id.clone(), 1, true), format!("fillet/{}", c.f.id));
622                blocks.push(block);
623            }
624        }
625    }
626    let mut partition =
627        finish_partition(j, blocks, physical, twice_constant, four_pi, planes, guards)?;
628    partition.fallback_reason = fallback_reason;
629    Ok(partition)
630}
631
632pub(crate) fn finish_partition(
633    j: &OrthogonalSketch,
634    mut blocks: Vec<BracketBlock>,
635    mut physical: BTreeMap<(String, usize, bool), String>,
636    twice_constant: i128,
637    four_pi: i128,
638    planes: [Vec<i32>; 3],
639    guards: Vec<(String, Bounds)>,
640) -> Result<SweepPartition, &'static str> {
641    blocks.sort_by(|a, b| a.id.cmp(&b.id));
642    if blocks.is_empty() || blocks.len() > 10000 {
643        return Err("orthogonal_sketch_block_budget");
644    }
645    let incidence = enumerate_faces(&blocks)?;
646    let index: BTreeMap<_, _> = blocks.iter().map(|b| (&b.id, b)).collect();
647    let mut tags = Vec::new();
648    for side in &incidence.exterior {
649        let b = index[&side.block];
650        let tag = physical
651            .remove(&(side.block.clone(), side.axis, side.upper))
652            .or_else(|| planar_label(j, &side_points(b, side.axis, side.upper)))
653            .ok_or("uncoupled_orthogonal_sketch_interface")?;
654        tags.push((side.clone(), tag));
655    }
656    if !physical.is_empty() {
657        return Err("physical_feature_face_buried");
658    }
659    let mut det = f64::INFINITY;
660    for b in &blocks {
661        let c = certify_spatial(b.degrees, &b.controls, CertificateBudget::default());
662        if c.status != CertificateStatus::Valid {
663            return Err("orthogonal_sketch_spatial_certificate_incomplete");
664        }
665        det = det.min(
666            c.determinant_lower_bound_mm3
667                .ok_or("orthogonal_sketch_missing_bound")?,
668        );
669    }
670    Ok(SweepPartition {
671        cut_comparison: None,
672        cut_rounds: Vec::new(),
673        cut_sources: Vec::new(),
674        compact: false,
675        fallback_reason: None,
676        blocks,
677        certificate: BracketCertificate {
678            incidence,
679            determinant_lower_bound_mm3: det,
680            design_faces: tags,
681            radial_error_upper_mm: 0.,
682        },
683        twice_constant: i64::try_from(twice_constant).map_err(|_| "sketch_volume_range")?,
684        junction: None,
685        four_pi: i64::try_from(four_pi).map_err(|_| "sketch_volume_range")?,
686        planes,
687        guards,
688    })
689}
