1//! Display samples of audited geometry and its one-sided chart distortion.
2use crate::{
3    field_vtu::{array, row, write_cells},
4    interchange,
5    json::Json,
6    quality, RationalBezierMap,
7};
8use std::fmt::Write;

10fn map(block: &Json, affine_pair: bool) -> Result<RationalBezierMap, &'static str> {
11    let degrees = if affine_pair {
12        [1, 1, 1]
13    } else {
14        let [u, v, w] = block.field("degrees")?.items()? else {
15            return Err("invalid geometry chart degrees");
16        };
17        [u.number()?, v.number()?, w.number()?]
18    };
19    let mut controls = Vec::new();
20    for point in block.field("controls")?.items()? {
21        let [x, y, z, w] = point.items()? else {
22            return Err("invalid homogeneous chart point");
23        };
24        controls.push([x.number()?, y.number()?, z.number()?, w.number()?]);
25    }
26    RationalBezierMap::new(degrees, controls).map_err(|_| "invalid or unsupported geometry chart")
27}

29/// Each VTK piece retains the native block index. Point diagnostics are samples
30/// of the original rational chart, not quality bounds for the linear display cells.
31pub fn export_geometry_vtu(model: &Json, n: usize) -> Result<String, &'static str> {
32    if !(1..=32).contains(&n) {
33        return Err("invalid geometry subdivision count");
34    }
35    let blocks = model.field("blocks")?.items()?;
36    if blocks.is_empty() || blocks.len() > 1_000_000 / (n + 1).pow(3) {
37        return Err("geometry sample budget exceeded");
38    }
39    let audit = interchange::audit(model)?;
40    if audit.field("status")?.text()? != "valid" {
41        return Err("geometry export requires a valid audit");
42    }
43    let affine_pair = model.field("schema")?.text()? == "affine_pair_v1";
44    let mut out=String::from("<?xml version=\"1.0\"?>\n<VTKFile type=\"UnstructuredGrid\" version=\"0.1\" byte_order=\"LittleEndian\">\n<UnstructuredGrid>\n");
45    for (id, block) in blocks.iter().enumerate() {
46        let chart = map(block, affine_pair)?;
47        let [mut points, mut params, mut det, mut stretches, mut condition, mut scaled] =
48            std::array::from_fn(|_| String::new());
49        for i in 0..=n {
50            for j in 0..=n {
51                for k in 0..=n {
52                    let t = [i, j, k].map(|i| i as f64 / n as f64);
53                    let sample = chart
54                        .evaluate(t)
55                        .map_err(|_| "geometry sample outside numerical range")?;
56                    let jac = sample.jacobian_columns;
57                    let q = quality::at_jacobian(jac)?;
58                    let [a, b, c] = jac;
59                    let determinant = a[0] * (b[1] * c[2] - b[2] * c[1])
60                        + a[1] * (b[2] * c[0] - b[0] * c[2])
61                        + a[2] * (b[0] * c[1] - b[1] * c[0]);
62                    if determinant <= 0. {
63                        return Err("nonpositive sampled geometry determinant");
64                    }
65                    row(&mut points, sample.position)?;
66                    row(&mut params, t)?;
67                    row(&mut det, [determinant])?;
68                    row(&mut stretches, q.singular_values)?;
69                    row(&mut condition, [q.condition_number])?;
70                    row(&mut scaled, [q.scaled_determinant])?;
71                }
72            }
73        }
74        writeln!(
75            out,
76            r#"<Piece NumberOfPoints="{}" NumberOfCells="{}">"#,
77            (n + 1).pow(3),
78            n.pow(3)
79        )
80        .unwrap();
81        out.push_str("<Points>\n");
82        array(&mut out, "position_mm", "Float64", 3, &points);
83        out.push_str("</Points>\n<PointData Scalars=\"condition_number\">\n");
84        for (name, components, values) in [
85            ("uvw", 3, &params),
86            ("jacobian_determinant_mm3", 1, &det),
87            ("singular_values_mm", 3, &stretches),
88            ("condition_number", 1, &condition),
89            ("scaled_jacobian", 1, &scaled),
90        ] {
91            array(&mut out, name, "Float64", components, values);
92        }
93        out.push_str("</PointData>\n<CellData Scalars=\"geometry_block_index\">\n");
94        array(
95            &mut out,
96            "geometry_block_index",
97            "Int32",
98            1,
99            &format!("{id}\n").repeat(n.pow(3)),
100        );
101        out.push_str("</CellData>\n");
102        write_cells(&mut out, n);
103    }
104    out.push_str("</UnstructuredGrid>\n</VTKFile>\n");
105    Ok(out)
106}
