42/// Parse the strict, versioned list of successive active-leaf refinements.
43pub fn parse_refinement_plan(plan: &Json) -> Result<RefinementPlan, &'static str> {
44    plan.keys(&["schema", "max_depth", "max_cells", "steps"])?;
45    if plan.field("schema")?.text()? != "adaptive_refinement_plan_v1" {
46        return Err("unsupported adaptive refinement plan schema");
47    }
48    let max_depth = plan.field("max_depth")?.number::<u8>()?;
49    let max_cells = plan.field("max_cells")?.number::<usize>()?;
50    if max_depth > 20 || max_cells == 0 {
51        return Err("invalid adaptive refinement budget");
52    }
53    let mut steps = Vec::new();
54    for step in plan.field("steps")?.items()? {
55        step.keys(&["refine"])?;
56        let mut leaves = Vec::new();
57        let mut unique = BTreeSet::new();
58        for leaf in step.field("refine")?.items()? {
59            leaf.keys(&["root", "level", "index"])?;
60            let root = leaf.field("root")?.text()?.to_owned();
61            let level = leaf.field("level")?.number::<u8>()?;
62            let index: [u32; 3] = leaf
63                .field("index")?
64                .items()?
65                .iter()
66                .map(Json::number)
67                .collect::<Result<Vec<_>, _>>()?
68                .try_into()
69                .map_err(|_| "expected three dyadic indices")?;
70            if root.is_empty() || level > max_depth || index.iter().any(|&i| i >= (1_u32 << level))
71            {
72                return Err("invalid adaptive leaf request");
73            }
74            let key = LeafKey { root, level, index };
75            if !unique.insert(key.clone()) {
76                return Err("duplicate leaf in one refinement step");
77            }
78            leaves.push(key);
79        }
80        if leaves.is_empty() {
81            return Err("empty adaptive refinement step");
82        }
83        leaves.sort();
84        steps.push(leaves);
85    }
86    Ok(RefinementPlan {
87        max_depth,
88        max_cells,
89        steps,
90    })
91}

106fn side_ref_json(adaptive: &AdaptiveSpace, side: &LeafSide) -> Result<Json, &'static str> {
107    Ok(object([
108        (
109            "block",
110            text(
111                adaptive
112                    .leaf_blocks
113                    .get(&side.leaf)
114                    .ok_or("leaf block is missing")?,
115            ),
116        ),
117        ("side", text(&side_name(&side.root_side))),
118        ("root_block", text(&side.root_side.block)),
119        ("root_side", text(&side_name(&side.root_side))),

123fn face_json(
124    adaptive: &AdaptiveSpace,
125    patch: &FacePatch,
126    denominator: u32,
127) -> Result<Json, &'static str> {
128    let o = patch.orientation;
129    Ok(object([
130        ("first", side_ref_json(adaptive, &patch.first)?),
131        ("second", side_ref_json(adaptive, &patch.second)?),
132        ("first_level", number(patch.first.leaf.level)),
133        ("second_level", number(patch.second.leaf.level)),
134        (
135            "level_difference_first_minus_second",
136            number(i16::from(patch.first.leaf.level) - i16::from(patch.second.leaf.level)),
137        ),
138        (
139            "two_to_one_balanced",
140            Json::Bool(patch.first.leaf.level.abs_diff(patch.second.leaf.level) <= 1),
141        ),
142        ("master_rect", rect_json(patch.master_rect)),
143        ("first_subrect", rect_json(patch.first_subrect)),
144        ("second_subrect", rect_json(patch.second_subrect)),
145        ("grid_denominator", number(denominator)),
146        (
147            "orientation",
148            object([
149                ("swap", Json::Bool(o.swap)),
150                ("flip_first", Json::Bool(o.flip_first)),
151                ("flip_second", Json::Bool(o.flip_second)),
152                ("maps", text("second_trace_into_first")),
153            ]),
154        ),
155    ]))
156}
157fn exterior_json(
158    adaptive: &AdaptiveSpace,
159    patch: &ExteriorPatch,
160    denominator: u32,
161    feature: Option<&str>,
162) -> Result<Json, &'static str> {
163    let mut fields = match side_ref_json(adaptive, &patch.side)? {
164        Json::Object(fields) => fields,
165        _ => unreachable!(),
166    };
167    fields.insert("root_rect".into(), rect_json(patch.root_rect));
168    fields.insert("grid_denominator".into(), number(denominator));
169    if let Some(tag) = feature {
170        fields.insert("feature".into(), text(tag));
171    }
172    Ok(Json::Object(fields))
173}
174pub(crate) fn lift_json(lift: &Lift) -> Json {
175    object([
176        ("free_roots", numbers(&lift.free_roots)),
177        (
178            "master_index_convention",
179            text("zero_based_position_in_free_roots"),
180        ),
181        (
182            "lines",
183            arr(&lift.lines, |line| {
184                object([
185                    ("dof", number(line.dof)),
186                    (
187                        "terms",
188                        arr(&line.terms, |(master, ratio)| {
189                            numbers([
190                                master.to_string(),
191                                ratio.numerator().to_string(),
192                                ratio.denominator().to_string(),
193                            ])
194                        }),
195                    ),
196                    (
197                        "offset",
198                        numbers([line.offset.numerator(), line.offset.denominator()]),
199                    ),
200                ])
201            }),
202        ),
203    ])
204}
205fn boundary_faces(
206    adaptive: &AdaptiveSpace,
207    partition: &AdaptivePartition,
208    tags: &BTreeMap<BlockSide, String>,
209    wanted: &str,
210) -> Result<Vec<Json>, &'static str> {
211    let denom = partition.grid_denominator();
212    partition
213        .exterior_patches()
214        .iter()
215        .filter(|p| tags.get(&p.side.root_side).is_some_and(|f| f == wanted))
216        .map(|p| exterior_json(adaptive, p, denom, Some(wanted)))
217        .collect()
218}
219fn adaptive_reduction(
220    space: &AnalysisSpace,
221    adaptive: &AdaptiveSpace,
222) -> Result<(Json, Json, Json), &'static str> {
223    let y50 = space
224        .boundary_dofs
225        .get("y50")
226        .ok_or("adaptive space has no y50 selection")?;
227    if y50.is_empty() {
228        return Err("empty adaptive y50 selection");
229    }
230    let prescribed = y50.iter().map(|&d| (d, Ratio::zero())).collect::<Vec<_>>();
231    let mounted = resolve(space.ndof, &adaptive.constraints, &prescribed)?;
232    let nfull = 3 * space.ndof;
233    let nfree = mounted.free_roots.len();
234    let mut fixed = Vec::with_capacity(3 * y50.len());
235    for &d in y50 {
236        for c in 0..3 {
237            fixed.push(3 * d + c);
238        }
239    }
240    let free_vector = mounted
241        .free_roots
242        .iter()
243        .flat_map(|&d| (0..3).map(move |c| 3 * d + c))
244        .collect::<Vec<_>>();
245    let mut entries = Vec::<[String; 4]>::new();
246    for line in &mounted.lines {
247        if line.offset != Ratio::zero() {
248            return Err("adaptive y50 reduction has nonzero prescribed offset");
249        }
250        for &(master, r) in &line.terms {
251            for c in 0..3 {
252                entries.push([
253                    (3 * line.dof + c).to_string(),
254                    (c * nfree + master).to_string(),
255                    r.numerator().to_string(),
256                    r.denominator().to_string(),
257                ]);
258            }
259        }
260    }
261    let clamping = object([
262        ("feature", text("y50")),
263        ("scalar_dofs", numbers(y50)),
264        ("vector_dofs", numbers(&fixed)),
265        (
266            "component_order",
267            Json::Array(["ux", "uy", "uz"].into_iter().map(text).collect()),
268        ),
269        ("value_mm", numbers([0, 0, 0])),
270    ]);
271    let reduction = object([
272        ("full_vector_dofs", number(nfull)),
273        ("reduced_vector_dofs", number(3 * nfree)),
274        ("ordering", text("scalar_node_major_xyz")),
275        ("reduced_ordering", text("component_major_free_roots")),
276        ("free_master_scalar_dofs", numbers(&mounted.free_roots)),
277        ("fixed_vector_dofs", numbers(&fixed)),
278        ("free_vector_dofs", numbers(&free_vector)),
279        (
280            "C",
281            object([
282                ("shape", numbers([nfull, 3 * nfree])),
283                (
284                    "representation",
285                    text("sparse_exact_scalar_lift_xyz_interleaved_rows_component_major_columns"),
286                ),
287                (
288                    "unit_entries_row_column_numerator_denominator",
289                    arr(entries, numbers),
290                ),
291                ("g_is_zero", Json::Bool(true)),
292            ]),
293        ),
294        ("clamped_nodes", clamping.clone()),
295    ]);
296    Ok((clamping, reduction, lift_json(&adaptive.lift)))
297}

299fn leaf_domain(leaf: &LeafKey) -> Json {
300    let scale = 1_u32 << leaf.level;
301    object([
302        ("numerator", arr(leaf.index, |i| numbers([i, i + 1]))),
303        ("denominator", number(scale)),
304    ])
305}
306fn design_face_json(
307    adaptive: &AdaptiveSpace,
308    patch: &ExteriorPatch,
309    feature: &str,
310    denom: u32,
311) -> Result<Json, &'static str> {
312    let side = side_ref_json(adaptive, &patch.side)?;
313    let Json::Object(mut fields) = side else {
314        unreachable!()
315    };
316    fields.insert("feature".into(), text(feature));
317    fields.insert("root_rect".into(), rect_json(patch.root_rect));
318    fields.insert("grid_denominator".into(), number(denom));
319    Ok(Json::Object(fields))

321fn adaptive_mesh_json(
322    adaptive: &AdaptiveSpace,
323    partition: &AdaptivePartition,
324    certificate: &BracketCertificate,
325    tags: &BTreeMap<BlockSide, String>,
326    plan: &RefinementPlan,
327) -> Result<(Json, Vec<Json>), &'static str> {
328    let oriented_complex = adaptive_complex::build_oriented_complex(adaptive, partition)?;
329    let report = partition.snapshot()?;
330    let denom = partition.grid_denominator();
331    let leaves = report
332        .leaves
333        .iter()
334        .map(|leaf| {
335            let id = adaptive
336                .leaf_blocks
337                .get(leaf)
338                .ok_or("adaptive leaf has no block")?;
339            Ok(object([
340                ("root", text(&leaf.root)),
341                ("level", number(leaf.level)),
342                ("index", numbers(leaf.index)),
343                ("block_id", text(id)),
344                ("domain", leaf_domain(leaf)),
345                (
346                    "determinant_lower_bound_mm3",
347                    number(
348                        certificate.determinant_lower_bound_mm3
349                            * 2.0_f64.powi(-3 * i32::from(leaf.level)),
350                    ),
351                ),
352            ]))
353        })
354        .collect::<Result<Vec<_>, &'static str>>()?;
355    let interfaces = report
356        .interfaces
357        .iter()
358        .map(|face| face_json(adaptive, face, denom))
359        .collect::<Result<Vec<_>, _>>()?;
360    let exterior = report
361        .exterior
362        .iter()
363        .map(|face| {
364            exterior_json(
365                adaptive,
366                face,
367                denom,
368                tags.get(&face.side.root_side).map(String::as_str),
369            )
370        })
371        .collect::<Result<Vec<_>, _>>()?;
372    let design = report
373        .exterior
374        .iter()
375        .filter_map(|face| {
376            tags.get(&face.side.root_side)
377                .map(|tag| design_face_json(adaptive, face, tag, denom))
378        })
379        .collect::<Result<Vec<_>, _>>()?;
380    let steps = plan
381        .steps
382        .iter()
383        .map(|step| object([("refine", arr(step, leaf_json))]))
384        .collect::<Vec<_>>();
385    let mesh = object([
386        ("schema", text("adaptive_q2_mesh_v1")),
387        ("max_depth", number(partition.max_depth())),
388        ("max_cells", number(partition.max_cells())),
389        ("grid_denominator", number(denom)),
390        ("refinement_step_count", number(plan.steps.len())),
391        (
392            "refinement_plan",
393            object([
394                ("schema", text("adaptive_refinement_plan_v1")),
395                ("max_depth", number(plan.max_depth)),
396                ("max_cells", number(plan.max_cells)),
397                ("steps", Json::Array(steps)),
398            ]),
399        ),
400        ("leaves", Json::Array(leaves)),
401        ("interfaces", Json::Array(interfaces)),
402        ("exterior_patches", Json::Array(exterior)),
403        ("oriented_complex", oriented_complex.to_json()),
404        ("exact_lift", lift_json(&adaptive.lift)),
405        (
406            "geometry_relation",
407            text(
408                "each leaf is the exact homogeneous dyadic restriction of its certified root chart",
409            ),
410        ),
411        ("design_faces", Json::Array(design.clone())),
412        (
413            "root_determinant_lower_bound_mm3",
414            number(certificate.determinant_lower_bound_mm3),
415        ),
416    ]);
417    Ok((mesh, design))
418}

420fn replace_object_field(json: &mut Json, key: &str, value: Json) -> Result<(), &'static str> {
421    let Json::Object(fields) = json else {
422        return Err("expected package object");
423    };
424    fields.insert(key.to_owned(), value);
425    Ok(())
426}
427fn get_object_mut<'a>(
428    json: &'a mut Json,
429    key: &str,
430) -> Result<&'a mut BTreeMap<String, Json>, &'static str> {
431    let Json::Object(fields) = json else {
432        return Err("expected package object");
433    };
434    match fields.get_mut(key) {
435        Some(Json::Object(value)) => Ok(value),
436        _ => Err("expected nested package object"),
437    }
438}

440/// Export X1's package schema with an exact A1/A2 adaptive mesh extension.
441pub fn export_adaptive_package(
442    model: &Json,
443    plan_json: &Json,
444    order: usize,
445) -> Result<Json, &'static str> {
446    let plan = parse_refinement_plan(plan_json)?;
447    let mut package = solver_package::export_analysis(model, order)?;
448    let schema = model.field("schema")?.text()?;
449    if schema != "filleted_bracket_v1" && schema != "parameterized_bracket_v1" {
450        return Err("adaptive export supports fixed and parameterized brackets");
451    }
452    let blocks = decode_blocks(model)?;
453    let cert = if schema == "filleted_bracket_v1" {
454        bracket_audit::audit_bracket(&blocks)?
455    } else {
456        let [rx, ry] = model.field("radii_sixteenths_mm")?.items()? else {
457            return Err("invalid bracket radii");
458        };
459        bracket_parameters_audit::audit_bracket_parameters(rx.number()?, ry.number()?, &blocks)?
460            .certificate
461    };
462    let parent = if schema == "filleted_bracket_v1" {
463        analysis_space::build_analysis_space(&blocks)?
464    } else {
465        let [rx, ry] = model.field("radii_sixteenths_mm")?.items()? else {
466            return Err("invalid bracket radii");
467        };
468        analysis_space::parameterized_bracket_space(rx.number()?, ry.number()?, &blocks)?
469    };
470    let roots = parent
471        .blocks
472        .iter()
473        .map(|b| b.id.clone())
474        .collect::<Vec<_>>();
475    let mut partition =
476        AdaptivePartition::new(&roots, &cert.incidence, plan.max_depth, plan.max_cells)?;
477    for step in &plan.steps {
478        partition.refine(step)?;
479    }
480    let adaptive = adaptive_space::build_adaptive_space(&parent, &partition)?;
481    let space = &adaptive.space;
482    let tags = cert
483        .design_faces
484        .iter()
485        .cloned()
486        .collect::<BTreeMap<_, _>>();
487    let (mesh, design) = adaptive_mesh_json(&adaptive, &partition, &cert, &tags, &plan)?;
488    let exterior = partition
489        .exterior_patches()
490        .iter()
491        .map(|p| BlockSide {
492            block: adaptive.leaf_blocks[&p.side.leaf].clone(),
493            axis: p.side.root_side.axis,
494            upper: p.side.root_side.upper,
495        })
496        .collect::<BTreeSet<_>>();
497    let patches = partition
498        .exterior_patches()
499        .iter()
500        .filter(|p| tags.get(&p.side.root_side).is_some_and(|f| f == "x60"))
501        .map(|p| {
502            LoadPatch::whole(BlockSide {
503                block: adaptive.leaf_blocks[&p.side.leaf].clone(),
504                axis: p.side.root_side.axis,
505                upper: p.side.root_side.upper,
506            })
507        })
508        .collect::<Vec<_>>();
509    let load = integrate_traction(space, &exterior, &patches, order, |_, _| {
510        [0., -1000. / 320., 0.]
511    })?;
512    if (load.area_mm2 - 320.).abs() > 1e-8
513        || (load.resultant_n[1] + 1000.).abs() > 1e-8
514        || load.resultant_n[0].abs() > 1e-10
515        || load.resultant_n[2].abs() > 1e-10
516    {
517        return Err("adaptive x60 load does not preserve the certified area and resultant");
518    }
519    let volume = solver_package::integrate_volume(space, order)?;
520    let (clamping, reduction, _) = adaptive_reduction(space, &adaptive)?;
521    let skeleton = encode_skeleton(space)?;
522    let y_faces = boundary_faces(&adaptive, &partition, &tags, "y50")?;
523    let x_faces = boundary_faces(&adaptive, &partition, &tags, "x60")?;
524    let Json::Object(fields) = &mut package else {
525        return Err("solver package is not an object");
526    };
527    fields.insert(
528        "schema".into(),
529        text("solidcraft_adaptive_solver_package_v1"),
530    );
531    fields.insert("package_extension".into(), text("adaptive_q2_v1"));
532    for (key, source_key) in [
533        ("derivatives", "source_derivatives"),
534        ("topology", "source_topology"),
535        ("certificate", "source_certificate"),
536    ] {
537        let value = fields.remove(key).ok_or("missing source package data")?;
538        fields.insert(source_key.into(), value);
539    }
540    fields.insert("analysis_derivatives".into(), object([
541        ("status",text("unavailable")),
542        ("scope",text("source_derivatives refers to root charts and the conforming root space only. Adaptive leaf and constraint derivatives have not been exported.")),
543    ]));
544    fields.insert("skeleton_scope".into(),text("Unsplit edges of individual leaves, identified by shared Q2 node indices. For hanging-face subrectangles use adaptive_mesh.interfaces."));
545    fields.insert("adaptive_mesh".into(), mesh);
546    fields.insert("design_faces".into(), Json::Array(design));
547    fields.insert("skeleton".into(), skeleton);
548    let analysis = get_object_mut(&mut package, "analysis_space")?;
549    analysis.insert("scalar_ndof".into(), number(space.ndof));
550    analysis.insert("vector_ndof".into(), number(3 * space.ndof));
551    analysis.insert(
552        "blocks".into(),
553        arr(&space.blocks, solver_package::analysis_block_json),
554    );
555    analysis.insert(
556        "boundary_dofs".into(),
557        Json::Object(
558            space
559                .boundary_dofs
560                .iter()
561                .map(|(k, v)| (k.clone(), numbers(v)))
562                .collect(),
563        ),
564    );
565    let selections = get_object_mut(&mut package, "selections")?;
566    if let Some(Json::Object(y50)) = selections.get_mut("y50") {
567        y50.insert("faces".into(), Json::Array(y_faces));
568        y50.insert("scalar_dofs".into(), numbers(&space.boundary_dofs["y50"]));
569    }
570    if let Some(Json::Object(x60)) = selections.get_mut("x60") {
571        x60.insert("faces".into(), Json::Array(x_faces));
572        x60.insert("area_mm2".into(), number(load.area_mm2));
573        x60.insert("resultant_n".into(), numbers(load.resultant_n));
574    }
575    let load_object = get_object_mut(&mut package, "load")?;
576    load_object.insert("area_mm2".into(), number(load.area_mm2));
577    load_object.insert("resultant_n".into(), numbers(load.resultant_n));
578    load_object.insert("nodal_rhs_n".into(), numbers(&load.nodal_force_n));
579    replace_object_field(&mut package, "clamping", clamping)?;
580    replace_object_field(&mut package, "reduction", reduction)?;
581    let measures = get_object_mut(&mut package, "measures")?;
582    measures.insert("volume_quadrature_mm3".into(), number(volume));
583    measures.insert("x60_area_mm2".into(), number(load.area_mm2));
584    Ok(package)
585}
