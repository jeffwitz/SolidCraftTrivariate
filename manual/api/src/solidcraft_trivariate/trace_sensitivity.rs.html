1//! First derivatives of normalized rational field transfer on an audited subface.
2
3use crate::face_constraints::{nested_constraints, FaceNet, Span};
4use crate::incidence::Orientation;
5
6#[derive(Debug, Clone, PartialEq)]
7pub struct CoefficientJet {
8    pub master: usize,
9    pub value: f64,
10    pub derivative: f64,
11}
12
13/// Differentiate the scalar transfer for a fixed orientation and degree pair.
14///
15/// Weight derivatives use the same homogeneous scale as their respective net.
16/// An independent time-dependent global rescaling of either net leaves field
17/// transfer invariant. Position-jet compatibility and differentiability of the
18/// selected charts are separate preconditions for a moving geometry problem.
19pub fn nested_weight_jets(
20    master: &FaceNet,
21    master_dw: &[f64],
22    slave: &FaceNet,
23    slave_dw: &[f64],
24    spans: [Span; 2],
25    orientation: Orientation,
26) -> Result<Vec<Vec<CoefficientJet>>, &'static str> {
27    let rows = nested_constraints(master, slave, spans, orientation)?;
28    if master_dw.len() != master.controls.len() || slave_dw.len() != slave.controls.len() {
29        return Err("weight derivative count mismatch");
30    }
31    if master_dw.iter().chain(slave_dw).any(|v| !v.is_finite()) {
32        return Err("nonfinite weight derivative");
33    }
34    let log_derivative: Vec<_> = master_dw
35        .iter()
36        .zip(&master.controls)
37        .map(|(dw, p)| dw / p[3] as f64)
38        .collect();
39    let mut gauge = None;
40    let mut result = Vec::with_capacity(rows.len());
41    for (index, row) in rows.into_iter().enumerate() {
42        let value = |numerator: i128| numerator as f64 / row.denominator as f64;
43        let mean: f64 = row
44            .terms
45            .iter()
46            .map(|&(j, n)| value(n) * log_derivative[j])
47            .sum();
48        let observed = slave_dw[index] / slave.controls[index][3] as f64;
49        let scale_derivative = observed - mean;
50        // The only admissible difference is a single logarithmic scale rate.
51        // This numerical check bounds roundoff of at most nine weighted terms.
52        let expected = *gauge.get_or_insert(scale_derivative);
53        let max_rate = row
54            .terms
55            .iter()
56            .map(|&(j, _)| log_derivative[j].abs())
57            .fold(observed.abs().max(expected.abs()).max(1.), f64::max);
58        if !scale_derivative.is_finite()
59            || (scale_derivative - expected).abs() > 64. * f64::EPSILON * max_rate
60        {
61            return Err("incompatible restricted weight derivatives");
62        }
63        let mut terms = Vec::with_capacity(row.terms.len());
64        for (master, numerator) in row.terms {
65            let coefficient = value(numerator);
66            let derivative = coefficient * (log_derivative[master] - mean);
67            if !derivative.is_finite() {
68                return Err("nonfinite field coefficient derivative");
69            }
70            terms.push(CoefficientJet {
71                master,
72                value: coefficient,
73                derivative,
74            });
75        }
76        result.push(terms);
77    }
78    Ok(result)
79}
