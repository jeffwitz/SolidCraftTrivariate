14/// Construction contract selected for every transaction in a session.
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
16pub enum TravelGeometry {
17    #[default]
18    ExactMicrometreCuts,
19    BoundedRationalArcs,
20}
21
22#[derive(Clone, Copy, Debug, PartialEq)]
23pub enum TravelFidelity {
24    ExactCircularArcs,
25    CertifiedApproximation {
26        radial_upper_mm: f64,
27        hausdorff_upper_mm: f64,
28    },
29}
30
31#[derive(Debug)]
32pub struct TravelState {
33    pub center_x_um: i32,
34    pub physical: TravellingBoreReport,
35    pub fidelity: TravelFidelity,
36    pub blocks: Vec<BracketBlock>,
37    pub certificate: BracketCertificate,
38    pub selections: BTreeMap<String, Vec<BlockSide>>,
39}
40
41#[derive(Debug, PartialEq, Eq)]
42pub struct TravelChange {
43    /// Critical center positions along the directed straight translation.
44    /// The departure point is excluded and the arrival point included.
45    pub crossed_centers_um: Vec<i32>,
46    pub created_features: Vec<String>,
47    pub lost_features: Vec<String>,
48    /// Surviving design identities whose supporting block sides have changed.
49    pub remapped_features: Vec<String>,
50}
51
52#[derive(Default)]
53pub struct TravellingBoreSession {
54    geometry: TravelGeometry,
55    current: Option<TravelState>,
56}
57
58impl TravellingBoreSession {
59    pub fn with_geometry(geometry: TravelGeometry) -> Self {
60        Self {
61            geometry,
62            current: None,
63        }
64    }
65
66    pub fn current(&self) -> Option<&TravelState> {
67        self.current.as_ref()
68    }
69
70    /// A missing required selection or an invalid new geometry aborts the
71    /// transaction. Feature names refer to physical design identities.
72    pub fn apply(
73        &mut self,
74        center_x_um: i32,
75        required_features: &[&str],
76    ) -> Result<TravelChange, &'static str> {
77        let physical = analyze_travelling_bore(i64::from(center_x_um))?;
78        let (blocks, certificate, fidelity) = match self.geometry {
79            TravelGeometry::ExactMicrometreCuts => {
80                let blocks = travelling_bore_blocks(center_x_um)?;
81                let cert = audit_travelling_bore(center_x_um, &blocks)?;
82                (blocks, cert, TravelFidelity::ExactCircularArcs)
83            }
84            TravelGeometry::BoundedRationalArcs => {
85                let blocks = travelling_bore_bounded_blocks(center_x_um)?;
86                let result = audit_travelling_bore_bounded(center_x_um, &blocks)?;
87                let fidelity = TravelFidelity::CertifiedApproximation {
88                    radial_upper_mm: result.radial_error_upper_mm,
89                    hausdorff_upper_mm: result.hausdorff_error_upper_mm,
90                };
91                (blocks, result.certificate, fidelity)
92            }
93        };
94        let mut selections: BTreeMap<String, Vec<BlockSide>> = BTreeMap::new();
95        for (side, feature) in &certificate.design_faces {
96            selections
97                .entry(feature.clone())
98                .or_default()
99                .push(side.clone());
100        }
101        if required_features
102            .iter()
103            .any(|f| !selections.contains_key(*f))
104        {
105            return Err("travelling_bore_required_selection_absent");
106        }
107        let mut crossed_centers_um = Vec::new();
108        let new_names: BTreeSet<_> = selections.keys().cloned().collect();
109        let mut old_names = BTreeSet::new();
110        let mut remapped_features = Vec::new();
111        if let Some(old) = &self.current {
112            old_names.extend(old.selections.keys().cloned());
113            for feature in new_names.intersection(&old_names) {
114                if old.selections[feature] != selections[feature] {
115                    remapped_features.push(feature.clone());
116                }
117            }
118            for candidate in [-4000, -2000, 0, 8000, 10000, 12000] {
119                if (old.center_x_um < candidate && candidate <= center_x_um)
120                    || (center_x_um <= candidate && candidate < old.center_x_um)
121                {
122                    crossed_centers_um.push(candidate);
123                }
124            }
125            if center_x_um < old.center_x_um {
126                crossed_centers_um.reverse();
127            }
128        }
129        let change = TravelChange {
130            crossed_centers_um,
131            created_features: new_names.difference(&old_names).cloned().collect(),
132            lost_features: old_names.difference(&new_names).cloned().collect(),
133            remapped_features,
134        };
135        self.current = Some(TravelState {
136            center_x_um,
137            physical,
138            fidelity,
139            blocks,
140            certificate,
141            selections,
142        });
143        Ok(change)
144    }
145}

147#[cfg(test)]
148mod tests {
149    use super::*;

151    #[test]
152    fn tangency_and_unrepresentable_intersection_preserve_the_accepted_state() {
153        let mut session = TravellingBoreSession::default();
154        session.apply(3000, &["bore_wall"]).unwrap();
155        let change = session.apply(-800, &["bore_wall"]).unwrap();
156        assert_eq!(change.crossed_centers_um, [0]);
157        let old = session.current().unwrap();
158        let blocks = old.blocks.clone();
159        let certificate = old.certificate.clone();
160        let selections = old.selections.clone();
161        for refused in [0, -100] {
162            assert!(session.apply(refused, &["bore_wall"]).is_err());
163            let state = session.current().unwrap();
164            assert_eq!(state.center_x_um, -800);
165            assert_eq!(state.blocks, blocks);
166            assert_eq!(state.certificate, certificate);
167            assert_eq!(state.selections, selections);
168        }
169    }
170
171    #[test]
172    fn bounded_edits_preserve_fidelity_and_selections_across_rejected_events() {
173        let mut session = TravellingBoreSession::with_geometry(TravelGeometry::BoundedRationalArcs);
174        session.apply(-100, &["bore_wall"]).unwrap();
175        assert!(session.apply(0, &[]).is_err());
176        assert_eq!(session.current().unwrap().center_x_um, -100);
177        session.apply(11900, &["bore_wall"]).unwrap();
178        let saved = session.current().unwrap().fidelity;
179        let TravelFidelity::CertifiedApproximation {
180            hausdorff_upper_mm, ..
181        } = saved
182        else {
183            panic!("missing bounded fidelity")
184        };
185        assert!((0.0..=1e-8).contains(&hausdorff_upper_mm));
186        assert!(session.apply(13000, &["bore_wall"]).is_err());
187        assert_eq!(session.current().unwrap().fidelity, saved);
188        let mut cold = TravellingBoreSession::with_geometry(TravelGeometry::BoundedRationalArcs);
189        cold.apply(11900, &[]).unwrap();
190        let (a, b) = (session.current().unwrap(), cold.current().unwrap());
191        assert_eq!(a.blocks, b.blocks);
192        assert_eq!(a.selections, b.selections);
193        assert_eq!(a.certificate, b.certificate);
194        assert_eq!(a.fidelity, b.fidelity);
195    }
196
197    #[test]
198    fn crossing_the_edge_reconciles_physical_selections_with_cold_replay() {
199        let mut session = TravellingBoreSession::default();
200        session.apply(11920, &["bore_wall"]).unwrap();
201        assert_eq!(
202            session.apply(13000, &["bore_wall"]),
203            Err("travelling_bore_required_selection_absent")
204        );
205        assert_eq!(session.current().unwrap().center_x_um, 11920);
206        let outside = session.apply(13000, &[]).unwrap();
207        assert_eq!(outside.crossed_centers_um, [12000]);
208        assert!(outside.lost_features.contains(&"bore_wall".to_string()));
209        let reverse = session.apply(-800, &["bore_wall"]).unwrap();
210        assert_eq!(reverse.crossed_centers_um, [12000, 10000, 8000, 0]);
211        assert!(reverse.created_features.contains(&"bore_wall".to_string()));
212        let mut cold = TravellingBoreSession::default();
213        cold.apply(-800, &["bore_wall"]).unwrap();
214        let (a, b) = (session.current().unwrap(), cold.current().unwrap());
215        assert_eq!(a.blocks, b.blocks);
216        assert_eq!(a.selections, b.selections);
217        assert_eq!(a.certificate, b.certificate);
218    }
