17fn orient(a: [i32; 2], b: [i32; 2], c: [i32; 2]) -> i64 {
18    let cross = |u: usize, v: usize| i64::from(b[u] - a[u]) * i64::from(c[v] - a[v]);
19    cross(0, 1) - cross(1, 0)
20}
21fn on_segment(a: [i32; 2], b: [i32; 2], p: [i32; 2]) -> bool {
22    orient(a, b, p) == 0 && (0..2).all(|d| (a[d].min(b[d])..=a[d].max(b[d])).contains(&p[d]))
23}
24fn touch(a: [i32; 2], b: [i32; 2], c: [i32; 2], d: [i32; 2]) -> bool {
25    let (ac, ad, ca, cb) = (
26        orient(a, b, c),
27        orient(a, b, d),
28        orient(c, d, a),
29        orient(c, d, b),
30    );
31    (ac.signum() * ad.signum() < 0 && ca.signum() * cb.signum() < 0)
32        || on_segment(a, b, c)
33        || on_segment(a, b, d)
34        || on_segment(c, d, a)
35        || on_segment(c, d, b)
36}

38fn simple_contour(recipe: &PolygonPrism) -> Result<Vec<&Vertex>, &'static str> {
39    let mut polygon = recipe.vertices.iter().collect::<Vec<_>>();
40    let n = polygon.len();
41    if !(3..=24).contains(&n)
42        || !(1..=64).contains(&recipe.height_mm)
43        || recipe.top_shift_mm.iter().any(|v| !(-32..=32).contains(v))
44    {
45        return Err("polygon_prism_range");
46    }
47    let mut ids = BTreeSet::new();
48    let mut points = BTreeSet::new();
49    for v in &polygon {
50        if v.id.is_empty()
51            || v.id.len() > 24
52            || !v.id.bytes().all(|c| c.is_ascii_alphanumeric() || c == b'_')
53            || !ids.insert(&v.id)
54        {
55            return Err("polygon_vertex_id");
56        }
57        if v.xy_mm.iter().any(|x| !(-64..=64).contains(x)) || !points.insert(v.xy_mm) {
58            return Err("polygon_vertex_coordinates");
59        }
60    }
61    let mut area = 0;
62    for i in 0..n {
63        let a = polygon[i].xy_mm;
64        let b = polygon[(i + 1) % n].xy_mm;
65        let c = polygon[(i + 2) % n].xy_mm;
66        if orient(a, b, c) == 0 {
67            return Err("polygon_collinear_adjacent_edges");
68        }
69        area += orient([0, 0], a, b);
70        for j in i + 1..n {
71            if j == i + 1 || (i == 0 && j == n - 1) {
72                continue;
73            }
74            if touch(a, b, polygon[j].xy_mm, polygon[(j + 1) % n].xy_mm) {
75                return Err("polygon_nonadjacent_edge_contact");
76            }
77        }
78    }
79    if area == 0 {
80        return Err("polygon_zero_area");
81    }
82    if area < 0 {
83        polygon.reverse();
84    }
85    let first = polygon
86        .iter()
87        .enumerate()
88        .min_by_key(|(_, v)| &v.id)
89        .unwrap()
90        .0;
91    polygon.rotate_left(first);
92    Ok(polygon)
93}

95fn ear(polygon: &[&Vertex], i: usize) -> bool {
96    let n = polygon.len();
97    let (prev, next) = ((i + n - 1) % n, (i + 1) % n);
98    let (a, b, c) = (polygon[prev].xy_mm, polygon[i].xy_mm, polygon[next].xy_mm);
99    orient(a, b, c) > 0
100        && polygon.iter().enumerate().all(|(j, p)| {
101            [prev, i, next].contains(&j)
102                || orient(a, b, p.xy_mm) < 0
103                || orient(b, c, p.xy_mm) < 0
104                || orient(c, a, p.xy_mm) < 0
105        })
106}

108pub(crate) fn append_triangle(
109    mut t: [&Vertex; 3],
110    recipe: &PolygonPrism,
111    blocks: &mut Vec<BracketBlock>,
112) {
113    let first = (0..3).min_by_key(|&i| &t[i].id).unwrap();
114    t.rotate_left(first);
115    let centroid: [i64; 2] =
116        std::array::from_fn(|d| 2 * t.iter().map(|v| i64::from(v.xy_mm[d])).sum::<i64>());
117    for i in 0..3 {
118        let a = t[i];
119        let next = t[(i + 1) % 3];
120        let prev = t[(i + 2) % 3];
121        let midpoint = |b: &Vertex| std::array::from_fn(|d| 3 * i64::from(a.xy_mm[d] + b.xy_mm[d]));
122        let mut controls = Vec::with_capacity(8);
123        for [x, y] in [
124            a.xy_mm.map(|v| 6 * i64::from(v)),
125            midpoint(prev),
126            midpoint(next),
127            centroid,
128        ] {
129            controls.push([x, y, 0, 6]);
130            controls.push([
131                x + 6 * i64::from(recipe.top_shift_mm[0]),
132                y + 6 * i64::from(recipe.top_shift_mm[1]),
133                6 * i64::from(recipe.height_mm),
134                6,
135            ]);
136        }
137        blocks.push(BracketBlock {
138            id: format!("tri/{}/{}/{}/{}", t[0].id, t[1].id, t[2].id, a.id),
139            degrees: [1, 1, 1],
140            controls,
141        });
142    }
143}

145/// Triangulate by exact ears in a deterministic contour order, then split each triangle into three quads.
146pub fn build_polygon_prism(recipe: &PolygonPrism) -> Result<Vec<BracketBlock>, &'static str> {
147    let mut contour = simple_contour(recipe)?;
148    let mut blocks = Vec::with_capacity(3 * (contour.len() - 2));
149    while contour.len() > 3 {
150        let n = contour.len();
151        let i = (0..n)
152            .filter(|&i| ear(&contour, i))
153            .min_by_key(|&i| &contour[i].id)
154            .ok_or("polygon_no_valid_ear")?;
155        append_triangle(
156            [contour[(i + n - 1) % n], contour[i], contour[(i + 1) % n]],
157            recipe,
158            &mut blocks,
159        );
160        contour.remove(i);
161    }
162    append_triangle([contour[0], contour[1], contour[2]], recipe, &mut blocks);
163    blocks.sort_by(|a, b| a.id.cmp(&b.id));
164    Ok(blocks)
165}
