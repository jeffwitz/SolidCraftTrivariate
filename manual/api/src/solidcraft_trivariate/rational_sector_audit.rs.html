1//! An exact certificate for the stored homogeneous nets of a rational angular sector.
2//!
3//! At three angular nodes and both radial and axial ends, check all four
4//! homogeneous polynomials against the analytic polar chart, up to one positive
5//! factor per chart. This tensor grid is unisolvent for degree (2,1,1).
6//! The weight is 1+s², the polar angle is 2 atan(s), and radius decreases
7//! linearly. These identities on the closed cube establish an inverse and
8//! show that distinct chart interiors have disjoint angular intervals.
9
10use crate::{
11    bracket::BracketBlock, bracket_audit::BracketCertificate, incidence::enumerate_faces,
12    rational_sector::RationalSector,
13};
14
15pub fn audit_rational_sector(
16    recipe: &RationalSector,
17    blocks: &[BracketBlock],
18) -> Result<BracketCertificate, &'static str> {
19    let [n, d] = recipe.half_angle_tangent.map(i64::from);
20    let [ri, ro] = recipe.radii_mm.map(i64::from);
21    let h = i64::from(recipe.height_mm);
22    if !(1..=256).contains(&n)
23        || !(1..=256).contains(&d)
24        || 4 * n < d
25        || n > 4 * d
26        || !(1..ro).contains(&ri)
27        || ro > 64
28        || !(1..=64).contains(&h)
29    {
30        return Err("sector_parameters_outside_certificate");
31    }
32    // Compare against 1+sqrt(2) without rounding the breakpoint.
33    let split = n > d && (n - d) * (n - d) > 2 * d * d;
34    let charts: &[(&str, i64, i64)] = if split {
35        &[("first", 0, 1), ("second", 2, 1)]
36    } else {
37        &[("whole", 0, 2)]
38    };
39    if blocks.len() != charts.len() {
40        return Err("sector_chart_inventory");
41    }
42    let q = i128::from(4 * d);
43    for &(id, first, stride) in charts {
44        let block = blocks
45            .iter()
46            .find(|b| b.id == id)
47            .ok_or("sector_chart_identity")?;
48        if block.degrees != [2, 1, 1]
49            || block.controls.len() != 12
50            || block.controls.iter().any(|p| p[3] <= 0)
51        {
52            return Err("sector_control_net_dimensions_or_weight");
53        }
54        let w0 = i128::from(block.controls[0][3]);
55        let s0 = i128::from(first * n);
56        let w_ref = q * q + s0 * s0;
57        for (j, blend) in [[4_i128, 0, 0], [1, 2, 1], [0, 0, 4]].iter().enumerate() {
58            let s = i128::from((first + j as i64 * stride) * n);
59            let weight = q * q + s * s;
60            for (v, radius) in [ro, ri].into_iter().enumerate() {
61                let r = i128::from(radius);
62                for (w, z) in [0, h].into_iter().enumerate() {
63                    let wanted = [
64                        r * (q * q - s * s),
65                        2 * r * q * s,
66                        i128::from(z) * weight,
67                        weight,
68                    ];
69                    for (c, ideal) in wanted.into_iter().enumerate() {
70                        let value: i128 = blend
71                            .iter()
72                            .enumerate()
73                            .map(|(i, &a)| a * i128::from(block.controls[4 * i + 2 * v + w][c]))
74                            .sum();
75                        if value * w_ref != 4 * w0 * ideal {
76                            return Err("sector_homogeneous_polynomial_identity");
77                        }
78                    }
79                }
80            }
81        }
82    }
83    let incidence = enumerate_faces(blocks)?;
84    if incidence.interfaces.len() + 1 != charts.len()
85        || incidence.exterior.len() != 4 * charts.len() + 2
86    {
87        return Err("sector_chart_incidence");
88    }
89    let mut design_faces = Vec::new();
90    for side in &incidence.exterior {
91        let tag = match (side.axis, side.upper) {
92            (0, false) if side.block != "second" => "angle_start",
93            (0, true) if side.block != "first" => "angle_end",
94            (1, false) => "outer",
95            (1, true) => "inner",
96            (2, false) => "start",
97            (2, true) => "end",
98            _ => return Err("sector_external_side"),
99        };
100        design_faces.push((side.clone(), tag.to_owned()));
101    }
102    // Integer products in this bounded domain are exactly representable in binary64.
103    let numerator = 2 * n * d * ri * (ro - ri) * h;
104    let denominator = charts.len() as i64 * (d * d + n * n);
105    Ok(BracketCertificate {
106        incidence,
107        design_faces,
108        determinant_lower_bound_mm3: (numerator as f64 / denominator as f64).next_down(),
109        radial_error_upper_mm: 0.,
110    })
111}
