35fn checked_div(a: Ratio, b: Ratio) -> Result<Ratio> {
36    let numerator = a
37        .numerator()
38        .checked_mul(b.denominator())
39        .ok_or("spline support rational numerator overflow")?;
40    let denominator = a
41        .denominator()
42        .checked_mul(b.numerator())
43        .ok_or("spline support rational denominator overflow")?;
44    Ratio::new(numerator, denominator)
45}

47fn parse_curve(curve: &Json) -> Result<CurveData> {
48    curve.keys(&["degree", "knots", "controls"])?;
49    let degree = curve.field("degree")?.number::<usize>()?;
50    if !(1..=3).contains(&degree) {
51        return Err("spline parent degree must be in 1..=3");
52    }
53    let knots = curve
54        .field("knots")?
55        .items()?
56        .iter()
57        .map(Json::number::<i64>)
58        .collect::<Result<Vec<_>>>()?;
59    let controls = curve
60        .field("controls")?
61        .items()?
62        .iter()
63        .map(|point| {
64            let [x, y, w] = point.items()? else {
65                return Err("spline parent expects homogeneous curve triples");
66            };
67            Ok([x.number::<i64>()?, y.number::<i64>()?, w.number::<i64>()?])
68        })
69        .collect::<Result<Vec<_>>>()?;
70    if controls.len() < degree + 1 || knots.len() != controls.len() + degree + 1 {
71        return Err("spline parent knot and control counts do not match");
72    }
73    if controls.iter().any(|point| point[2] <= 0) {
74        return Err("spline parent weights must be strictly positive");
75    }
76    if knots.windows(2).any(|pair| pair[0] > pair[1]) {
77        return Err("spline parent knots must be nondecreasing");
78    }
79    if knots[..=degree].iter().any(|&knot| knot != knots[degree])
80        || knots[controls.len()..]
81            .iter()
82            .any(|&knot| knot != knots[controls.len()])
83        || knots[degree] >= knots[controls.len()]
84    {
85        return Err("spline parent source must be clamped with a nonempty domain");
86    }
87    Ok(CurveData {
88        degree,
89        knots,
90        controls,
91    })
92}

105fn hpoint(point: [i64; 3]) -> HCurvePoint {
106    point.map(|value| Ratio::new(i128::from(value), 1).expect("integer ratio"))
107}

109fn blend(a: HCurvePoint, b: HCurvePoint, alpha: Ratio) -> Result<HCurvePoint> {
110    let complement = Ratio::one().checked_sub(alpha)?;
111    let mut result = [Ratio::zero(); 3];
112    for component in 0..3 {
113        result[component] = a[component]
114            .checked_mul(alpha)?
115            .checked_add(b[component].checked_mul(complement)?)?;
116    }
117    Ok(result)
118}

120fn insert_knot(
121    degree: usize,
122    knots: &[i64],
123    controls: &[HCurvePoint],
124    value: i64,
125) -> Result<(Vec<i64>, Vec<HCurvePoint>)> {
126    let last = controls
127        .len()
128        .checked_sub(1)
129        .ok_or("empty spline parent curve")?;
130    let multiplicity = knots.iter().filter(|&&knot| knot == value).count();
131    if multiplicity >= degree {
132        return Err("spline parent knot insertion reached full multiplicity");
133    }
134    let span = (degree..=last)
135        .find(|&index| knots[index] <= value && value < knots[index + 1])
136        .ok_or("spline parent insertion knot is outside the active domain")?;
137    let mut next = vec![[Ratio::zero(); 3]; controls.len() + 1];
138    let unchanged = span - degree + 1;
139    next[..unchanged].copy_from_slice(&controls[..unchanged]);
140    let blend_start = span - degree + 1;
141    let blend_end = span - multiplicity;
142    if blend_start <= blend_end {
143        for index in blend_start..=blend_end {
144            let denominator = i128::from(knots[index + degree]) - i128::from(knots[index]);
145            let numerator = i128::from(value) - i128::from(knots[index]);
146            if denominator <= 0 || numerator < 0 || numerator > denominator {
147                return Err("invalid spline parent knot insertion interval");
148            }
149            next[index] = blend(
150                controls[index],
151                controls[index - 1],
152                ratio(numerator, denominator)?,
153            )?;
154        }
155    }
156    let shifted = span - multiplicity + 1;
157    next[shifted..last + 2].copy_from_slice(&controls[shifted - 1..=last]);
158    let mut next_knots = knots.to_vec();
159    next_knots.insert(span + 1, value);
160    Ok((next_knots, next))
161}

163/// Extract Bézier controls while retaining the original homogeneous gauge.
164/// The public curve extractor remains the canonical checked implementation;
165/// this companion retains its exact rational scale for lineage comparison.
166fn extract_parent_gauge(curve: &CurveData) -> Result<Vec<([i64; 2], Vec<HCurvePoint>)>> {
167    // Reuse the native extractor as the input validity gate.
168    extract_bezier(curve.degree, &curve.knots, &curve.controls)?;
169    let mut knots = curve.knots.clone();
170    let mut controls = curve
171        .controls
172        .iter()
173        .copied()
174        .map(hpoint)
175        .collect::<Vec<_>>();
176    let start = knots[curve.degree];
177    let end = knots[curve.controls.len()];
178    let internal = knots[curve.degree + 1..curve.controls.len()]
179        .iter()
180        .copied()
181        .filter(|&knot| start < knot && knot < end)
182        .collect::<std::collections::BTreeSet<_>>();
183    for knot in internal {
184        while knots.iter().filter(|&&entry| entry == knot).count() < curve.degree {
185            (knots, controls) = insert_knot(curve.degree, &knots, &controls, knot)?;
186        }
187    }
188    let last = controls.len() - 1;
189    let mut spans = Vec::new();
190    for index in curve.degree..=last {
191        if knots[index] < knots[index + 1] {
192            let first = index - curve.degree;
193            spans.push((
194                [knots[index], knots[index + 1]],
195                controls[first..=index].to_vec(),
196            ));
197        }
198    }
199    Ok(spans)
200}

202/// The affine local-to-parent coordinate map for one source-cell restriction.
203#[derive(Debug, Clone, PartialEq, Eq)]
204pub struct ParametricRestriction {
205    pub parent_id: String,
206    /// Increasing source intervals, retained in the source's actual units.
207    pub intervals: [[i64; 2]; 3],
208    /// Local coordinate axis `a` maps to source axis `axes[a]`.
209    pub axes: [usize; 3],
210    /// Reversal flags are applied to local coordinates before interval mapping.
211    pub reversals: [bool; 3],
212}
213
214impl ParametricRestriction {
215    /// Map local unit-cube coordinates into the parent's actual parameter domain.
216    pub fn map_parameters(&self, local: [Ratio; 3]) -> Result<[Ratio; 3]> {
217        let mut permutation = self.axes;
218        permutation.sort();
219        if permutation != [0, 1, 2] {
220            return Err("spline parent restriction axes must be a permutation");
221        }
222        let mut source = [Ratio::zero(); 3];
223        for (local_axis, local_parameter) in local.iter().enumerate() {
224            let parent_axis = self.axes[local_axis];
225            let [lo, hi] = self.intervals[parent_axis];
226            if lo >= hi {
227                return Err("spline parent restriction interval must be increasing");
228            }
229            let parameter = if self.reversals[local_axis] {
230                Ratio::one().checked_sub(*local_parameter)?
231            } else {
232                *local_parameter
233            };
234            let length = i128::from(hi) - i128::from(lo);
235            source[parent_axis] =
236                ratio(i128::from(lo), 1)?.checked_add(parameter.checked_mul(ratio(length, 1)?)?)?;
237        }
238        Ok(source)
239    }
240
241    /// Return `d(source_axis)/d(local_axis)`, including orientation and scale.
242    pub fn derivative_scale(&self, local_axis: usize) -> Result<Ratio> {
243        if local_axis >= 3 {
244            return Err("spline parent restriction axis is outside 0..3");
245        }
246        let mut permutation = self.axes;
247        permutation.sort();
248        if permutation != [0, 1, 2] {
249            return Err("spline parent restriction axes must be a permutation");
250        }
251        let parent_axis = self.axes[local_axis];
252        let [lo, hi] = self.intervals[parent_axis];
253        if lo >= hi {
254            return Err("spline parent restriction interval must be increasing");
255        }
256        let length = i128::from(hi) - i128::from(lo);
257        ratio(
258            if self.reversals[local_axis] {
259                -length
260            } else {
261                length
262            },
263            1,
264        )
265    }
266}

319fn fingerprint(recipe: &Json) -> String {
320    // This is a deterministic lineage token, not a cryptographic integrity hash.
321    let mut hash = 0xcbf29ce484222325_u64;
322    for byte in recipe.to_string().bytes() {
323        hash ^= u64::from(byte);
324        hash = hash.wrapping_mul(0x100000001b3);
325    }
326    format!("fnv1a64:{hash:016x}")
327}

329fn same_projective_vector(actual: &[i64; 4], expected: [Ratio; 4], gauge: Ratio) -> Result<bool> {
330    for component in 0..4 {
331        let scaled = expected[component].checked_mul(gauge)?;
332        if scaled != Ratio::new(i128::from(actual[component]), 1)? {
333            return Ok(false);
334        }
335    }
336    Ok(true)
337}

339fn source_restrictions(
340    model: &Json,
341    lower: &CurveData,
342    upper: &CurveData,
343    parent_id: &str,
344    height: i32,
345) -> Result<Vec<SourceSpan>> {
346    let lower_spans = extract_parent_gauge(lower)?;
347    let upper_spans = extract_parent_gauge(upper)?;
348    if lower_spans.len() != upper_spans.len() {
349        return Err("spline strip source span counts differ");
350    }
351    for ((lower_interval, _), (upper_interval, _)) in lower_spans.iter().zip(&upper_spans) {
352        if lower_interval != upper_interval {
353            return Err("spline strip source intervals differ");
354        }
355    }
356    let intervals = model.field("span_intervals")?.items()?;
357    let blocks = bracket_json::decode_blocks(model)?;
358    if intervals.len() != lower_spans.len() || blocks.len() != lower_spans.len() {
359        return Err("spline strip block and source span counts differ");
360    }
361    let mut result = Vec::with_capacity(blocks.len());
362    for (index, (interval, block)) in intervals.iter().zip(&blocks).enumerate() {
363        let [a, b] = interval.items()? else {
364            return Err("spline strip interval must be a pair");
365        };
366        let a = a.number::<i64>()?;
367        let b = b.number::<i64>()?;
368        let reversed = a > b;
369        let source_interval = if reversed { [b, a] } else { [a, b] };
370        let lower_span = lower_spans
371            .iter()
372            .find(|span| span.0 == source_interval)
373            .ok_or("spline strip source interval lineage mismatch")?;
374        let upper_span = upper_spans
375            .iter()
376            .find(|span| span.0 == source_interval)
377            .ok_or("spline strip source interval lineage mismatch")?;
378        if lower_span.0 != source_interval || upper_span.0 != source_interval {
379            return Err("spline strip source interval lineage mismatch");
380        }
381        let degree = lower.degree;
382        if block.degrees != [degree, 1, 1] || block.controls.len() != (degree + 1) * 4 {
383            return Err("spline strip source block degree or control count mismatch");
384        }
385        let lower_controls: Vec<_> = if reversed {
386            lower_span.1.iter().rev().copied().collect()
387        } else {
388            lower_span.1.clone()
389        };
390        let upper_controls: Vec<_> = if reversed {
391            upper_span.1.iter().rev().copied().collect()
392        } else {
393            upper_span.1.clone()
394        };
395        let gauge = checked_div(
396            Ratio::new(i128::from(block.controls[0][3]), 1)?,
397            lower_controls[0][2],
398        )?;
399        if gauge.numerator() <= 0 {
400            return Err("spline strip source gauge must be positive");
401        }
402        for u in 0..=degree {
403            for v in 0..=1 {
404                let curve_control = if v == 0 {
405                    lower_controls[u]
406                } else {
407                    upper_controls[u]
408                };
409                for w in 0..=1 {
410                    let actual = &block.controls[(u * 2 + v) * 2 + w];
411                    let expected = [
412                        curve_control[0],
413                        curve_control[1],
414                        curve_control[2].checked_mul(Ratio::new(
415                            i128::from(height)
416                                .checked_mul(
417                                    i128::try_from(w).map_err(|_| "invalid extrusion layer")?,
418                                )
419                                .ok_or("spline parent homogeneous z overflow")?,
420                            1,
421                        )?)?,
422                        curve_control[2],
423                    ];
424                    if !same_projective_vector(actual, expected, gauge)? {
425                        return Err("spline strip block is not the exact parent restriction");
426                    }
427                }
428            }
429        }
430        let source_block_id = block.id.clone();
431        result.push(SourceSpan {
432            source_block_id: source_block_id.clone(),
433            restriction: ParametricRestriction {
434                parent_id: parent_id.to_owned(),
435                intervals: [source_interval, [0, 1], [0, 1]],
436                axes: [0, 1, 2],
437                reversals: [reversed, false, false],
438            },
439            gauge: [gauge.numerator(), gauge.denominator()],
440            generating_operation: "nurbs_strip_extrusion".into(),
441            event_provenance: format!("source knot span {}..{}; model span {}", a, b, index),
442        });
443    }
444    Ok(result)
445}

447fn continuity_evidence(curve: &CurveData) -> Vec<ContinuityEvidence> {
448    let mut knots = std::collections::BTreeSet::new();
449    for &knot in &curve.knots[curve.degree + 1..curve.controls.len()] {
450        if curve.knots[curve.degree] < knot && knot < curve.knots[curve.controls.len()] {
451            knots.insert(knot);
452        }
453    }
454    knots
455        .into_iter()
456        .map(|knot| {
457            let multiplicity = curve.knots.iter().filter(|&&entry| entry == knot).count();
458            ContinuityEvidence {
459                interface: format!("source-knot:{knot}"),
460                coordinate_frame: "parent parameter coordinates in source knot units".into(),
461                direction: 0,
462                knot,
463                multiplicity,
464                claimed_order: curve.degree.saturating_sub(multiplicity),
465                method: "exact clamped B-spline knot-multiplicity continuity".into(),
466                assumptions: vec![
467                    "source homogeneous controls are unchanged".into(),
468                    "all source weights are strictly positive".into(),
469                    "both strip boundaries share the same knot vector and weights".into(),
470                ],
471                status: "verified_from_source_support".into(),
472            }
473        })
474        .collect()
475}

477/// Recover a complete tensor parent from an audited NURBS strip model.
478///
479/// The model is first rebuilt from its retained recipe. The recovery then
480/// verifies source compatibility and every exact homogeneous block restriction.
481pub fn recover_nurbs_strip_parent(model: &Json) -> Result<TensorSupport> {
482    let schema = model.field("schema")?.text()?;
483    if schema != SOURCE_FAMILY {
484        return Err("unsupported source family for spline parent recovery");
485    }
486    let recipe = recipe_from_model(model)?;
487    if crate::nurbs_sketch_json::build(&recipe)? != *model {
488        return Err("spline parent source model does not replay exactly");
489    }
490    let lower = curve_json(recipe.field("lower")?)?;
491    let upper = curve_json(recipe.field("upper")?)?;
492    if lower.degree != upper.degree || lower.knots != upper.knots {
493        return Err("spline parent boundary curves need matching degrees and knots");
494    }
495    if lower.controls.len() != upper.controls.len()
496        || lower
497            .controls
498            .iter()
499            .zip(&upper.controls)
500            .any(|(a, b)| a[0] != b[0] || a[2] != b[2])
501    {
502        return Err("spline parent boundaries need a common x map and weights");
503    }
504    let height = recipe.field("height_mm")?.number::<i32>()?;
505    if height <= 0 {
506        return Err("spline parent extrusion height must be positive");
507    }
508    let u_count = lower.controls.len();
509    let degrees = [lower.degree, 1, 1];
510    let control_shape = [u_count, 2, 2];
511    let domain = [
512        [lower.knots[lower.degree], lower.knots[u_count]],
513        [0, 1],
514        [0, 1],
515    ];
516    let fingerprint = fingerprint(&recipe);
517    let id = format!("nurbs-sketch-strip-parent-v1:{fingerprint}");
518    let mut homogeneous_controls = Vec::with_capacity(u_count * 4);
519    for u in 0..u_count {
520        for v in 0..2 {
521            let point = if v == 0 {
522                lower.controls[u]
523            } else {
524                upper.controls[u]
525            };
526            for w in 0..2 {
527                homogeneous_controls.push([
528                    i128::from(point[0]),
529                    i128::from(point[1]),
530                    i128::from(point[2])
531                        .checked_mul(i128::from(height) * i128::from(w))
532                        .ok_or("spline parent homogeneous z overflow")?,
533                    i128::from(point[2]),
534                ]);
535            }
536        }
537    }
538    let source_spans = source_restrictions(model, &lower, &upper, &id, height)?;
539    Ok(TensorSupport {
540        id,
541        source_revision: fingerprint,
542        degrees,
543        knots: [lower.knots.clone(), vec![0, 0, 1, 1], vec![0, 0, 1, 1]],
544        domain,
545        control_shape,
546        homogeneous_controls,
547        source_spans,
548        continuity_evidence: continuity_evidence(&lower),
549        generating_recipe: recipe,
550        certificate_references: vec![
551            "nurbs_sketch_strip_v1 exact strip separation and positive Jacobian audit".into(),
552            "exact knot insertion and homogeneous restriction comparison".into(),
553        ],
554    })
555}

557/// A complete single affine cell is already a tensor parent on a unit cube.
558/// Verify the affine parallelepiped on every control. Positive orientation
559/// and equality with the complete material come from the native model audit.
560pub fn recover_complete_parent(model: &Json) -> Result<TensorSupport> {
561    if model.field("schema")?.text()? == SOURCE_FAMILY {
562        return recover_nurbs_strip_parent(model);
563    }
564    let audit = crate::interchange::audit(model)?;
565    if audit.field("status")?.text()? != "valid" {
566        return Err("source geometry is not valid");
567    }
568    let cells = bracket_json::decode_blocks(model)?;
569    let [cell] = cells.as_slice() else {
570        return Err("unsupported: no complete parent witness for this material");
571    };
572    if cell.degrees != [1; 3] || cell.controls.len() != 8 {
573        return Err("unsupported: single cell is outside affine parent recovery");
574    }
575    let net: Vec<_> = cell.controls.iter().map(|p| p.map(i128::from)).collect();
576    let w = net[0][3];
577    if w <= 0 || net.iter().any(|h| h[3] != w) {
578        return Err("unsupported: affine parent requires constant positive weights");
579    }
580    for (i, h) in net.iter().enumerate() {
581        for c in 0..3 {
582            let affine = [4usize, 2, 1]
583                .iter()
584                .filter(|&&k| i & k != 0)
585                .fold(net[0][c], |x, &k| x + net[k][c] - net[0][c]);
586            if affine != h[c] {
587                return Err("unsupported: trilinear control net lacks affine parent witness");
588            }
589        }
590    }
591    let edge = |i: usize, j: usize| Ratio::new(net[i][j] - net[0][j], w);
592    let mut det = Ratio::zero();
593    for (i, j, k) in [(0, 1, 2), (1, 2, 0), (2, 0, 1)] {
594        let minor = edge(2, j)?
595            .checked_mul(edge(1, k)?)?
596            .checked_sub(edge(2, k)?.checked_mul(edge(1, j)?)?)?;
597        det = det.checked_add(edge(4, i)?.checked_mul(minor)?)?;
598    }
599    if det.numerator() <= 0 {
600        return Err("unsupported: affine parent lacks a positive orientation");
601    }
602    let revision = fingerprint(model);
603    let id = format!("affine-cell-parent-v1:{revision}");
604    let intervals = [[0, 1]; 3];
605    Ok(TensorSupport {
606        id: id.clone(),
607        source_revision: revision,
608        degrees: [1; 3],
609        knots: std::array::from_fn(|_| vec![0, 0, 1, 1]),
610        domain: intervals,
611        control_shape: [2; 3],
612        homogeneous_controls: net,
613        source_spans: vec![SourceSpan {
614            source_block_id: cell.id.clone(),
615            restriction: ParametricRestriction {
616                parent_id: id,
617                intervals,
618                axes: [0, 1, 2],
619                reversals: [false; 3],
620            },
621            gauge: [1, 1],
622            generating_operation: "single audited affine source cell".into(),
623            event_provenance: cell.id.clone(),

633pub(crate) fn initial_mesh_admission(
634    support: &TensorSupport,
635) -> std::result::Result<(), &'static str> {
636    for (a, knots) in support.knots.iter().enumerate() {
637        let [lo, hi] = support.domain[a];
638        let mut breaks = vec![lo];
639        for &k in knots.iter().filter(|&&k| lo < k && k < hi) {
640            if breaks.last() == Some(&k) {
641                return Err("repeated source interior knot");
642            }
643            breaks.push(k);
644        }
645        breaks.push(hi);
646        if breaks.len() > 2 && support.degrees[a] < 3 {
647            return Err("degree elevation has no proof of cubic simple-knot membership");
648        }
649        let widths: Vec<_> = breaks
650            .windows(2)
651            .map(|v| i128::from(v[1]) - i128::from(v[0]))
652            .collect();
653        if widths.iter().any(|&w| w != widths[0]) {
654            return Err("source knot intervals are nonuniform under affine scaling");
655        }
656    }
657    Ok(())
658}

662        ("denominator", string("1")),
663    ])

798fn known_curve_sources(model: &Json, schema: &str) -> Result<Vec<Json>> {
799    if schema != "nurbs_cross_bores_v1" {
800        return Ok(Vec::new());
801    }
802    let profiles = model.field("profiles")?;
803    ["x_min", "x_max", "z_min", "z_max"]
804        .into_iter()
805        .map(|name| {
806            let curve = profiles.field(name)?;
807            let degree = curve.field("degree")?.number::<usize>()?;
808            let knots = curve
809                .field("knots")?
810                .items()?
811                .iter()
812                .map(Json::number::<i64>)
813                .collect::<Result<Vec<_>>>()?;
814            let controls = curve.field("controls")?.items()?.len();
815            let start = knots[degree];
816            let end = knots[controls];
817            let spans = (degree..controls)
818                .filter(|&index| knots[index] < knots[index + 1])
819                .map(|index| [knots[index], knots[index + 1]])
820                .collect::<Vec<_>>();
821            Ok(object([
822                ("source_id", string(name)),
823                ("degree", number(degree)),
824                ("knots", arr(knots, number)),
825                ("domain", arr([start, end], number)),
826                (
827                    "source_span_intervals",
828                    arr(spans, |pair| arr(pair, number)),
829                ),
830                (
831                    "role",
832                    string("generating sketch curve; not a complete 3D material parent"),
833                ),
834            ]))
835        })
836        .collect()
837}

839/// Produce a read-only conversion-eligibility report while preserving source validity.
840///
841/// A valid source model can be outside the first single-parent T-spline profile.
842/// Every exact shared-face incidence is listed, and higher continuity is never
843/// inferred from geometric coincidence alone.
844pub fn eligibility_report(model: &Json) -> Result<Json> {
845    let audit = crate::interchange::audit(model)?;
846    let schema = model.field("schema")?.text()?;
847    let single = model.field("blocks")?.items()?.len() == 1;
848    let candidate = if schema == SOURCE_FAMILY {
849        Some(recover_nurbs_strip_parent(model)?)
850    } else if single {
851        recover_complete_parent(model).ok()
852    } else {
853        None
854    };
855    if let Some(support) = candidate {
856        let mesh = initial_mesh_admission(&support);
857        let reason = mesh
858            .err()
859            .unwrap_or("one exact complete parent qualifies for the initial uniform tensor mesh");
860        return Ok(object([
861            ("schema", string("spline_eligibility_report_v1")),
862            ("profile", string("structured_tricubic_m2_v1")),
863            ("source_schema", string(schema)),
864            ("source_validity", audit.field("status")?.clone()),
865            (
866                "conversion_status",
867                string(if mesh.is_ok() {
868                    "eligible_single_complete_parent"
869                } else {
870                    "unsupported_profile"
871                }),
872            ),
873            ("parent_recovery", string("verified")),
874            ("parent_support", serialize_support(&support)),
875            (
876                "retained_joins",
877                arr(
878                    support
879                        .continuity_evidence
880                        .iter()
881                        .filter(|e| e.claimed_order == 0),
882                    |e| {
883                        object([
884                            ("source_interface", string(&e.interface)),
885                            ("basis", string("source_knot_multiplicity_retains_c0")),
886                        ])
887                    },
888                ),
889            ),
890            ("reason", string(reason)),
891        ]));
892    }
893    let interfaces = model
894        .field("topology")?
895        .field("interfaces")?
896        .items()?
897        .iter()
898        .map(exact_interface_record)
899        .collect::<Result<Vec<_>>>()?;
900    let known_sources = known_curve_sources(model, schema)?;
901    let common_parent_obstruction = if matches!(
902        schema,
903        "box_cylinder_boolean_v1" | "box_cylinder_boolean_suite_v1"
904    ) {
905        let stock = model.field("stock")?.field("max_mm")?.items()?;
906        let stock_min = model.field("stock")?.field("min_mm")?.items()?;
907        let tool = model.field("tool")?;
908        let [z_min, z_max] = tool.field("interval_mm")?.items()? else {
909            return Err("cylinder interval must have two endpoints");
910        };
911        let [_, _, stock_z_max] = stock else {
912            return Err("box max point must have three coordinates");
913        };
914        let stock_z_min = stock_min[2].number::<i64>()?;
915        let radius = tool.field("radius_mm")?.number::<i64>()?;
916        let [center_x, center_y] = tool.field("center_mm")?.items()? else {
917            return Err("cylinder center must have two coordinates");
918        };
919        let center_x = center_x.number::<i64>()?;
920        let center_y = center_y.number::<i64>()?;
921        let recognized_tool = tool.field("kind")?.text()? == "cylinder"
922            && tool.field("axis")?.text()? == "z"
923            && matches!(
924                model.field("operation")?.text()?,
925                "difference" | "a_minus_b"
926            );
927        let contained_in_xy = stock_min[0].number::<i64>()? < center_x - radius
928            && center_x + radius < stock[0].number::<i64>()?
929            && stock_min[1].number::<i64>()? < center_y - radius
930            && center_y + radius < stock[1].number::<i64>()?;
931        let through = z_min.number::<i64>()? < stock_z_min
932            && z_max.number::<i64>()? > stock_z_max.number::<i64>()?
933            && radius > 0
934            && recognized_tool
935            && contained_in_xy;
936        if through {
937            object([
938                ("kind", string("exact_through_tunnel_topology")),
939                (
940                    "evidence",
941                    string(
942                        "positive-radius cylindrical tool strictly crosses both stock end planes",
943                    ),
944                ),
945                (
946                    "consequence",
947                    string(
948                        "material domain is not homeomorphic to one injective tensor-product cube",
949                    ),
950                ),
951            ])
952        } else {
953            object([
954                ("kind", string("not_demonstrated")),
955                (
956                    "evidence",
957                    string("through-tunnel preconditions were not all established"),
958                ),
959                (
960                    "consequence",
961                    string("common-parent topology remains unclassified"),
962                ),
963            ])
964        }
965    } else if schema == "nurbs_cross_bores_v1" {
966        object([
967            ("kind", string("multiple_generating_supports")),
968            (
969                "evidence",
970                string("four source profiles generate a multi-cell cross construction"),
971            ),
972            (
973                "consequence",
974                string("no exact complete 3D tensor parent restriction is recorded for every material block"),
975            ),
976        ])
977    } else {
978        object([
979            ("kind", string("no_complete_parent_witness")),
980            (
981                "evidence",
982                string("no complete source restriction proof in this recovery family"),
983            ),
984            (
985                "consequence",
986                string("global topology and mathematical representability remain unclassified"),
987            ),
988        ])
989    };
990    Ok(object([
991        ("schema", string("spline_eligibility_report_v1")),
992        ("profile", string("structured_tricubic_m2_v1")),
993        ("source_schema", string(schema)),
994        ("source_validity", audit.field("status")?.clone()),
995        ("conversion_status", string("unsupported_profile")),
996        ("source_model_preserved", Json::Bool(true)),
997        ("known_generating_sources", Json::Array(known_sources)),
998        (
999            "known_source_restriction_status",
1000            string("recipe-level supports retained; complete per-block 3D parent restrictions unavailable"),
1001        ),
1002        ("retained_joins", Json::Array(interfaces)),
1003        ("regional_source_columns", recover_translation_columns(model, 2)?),
1004        ("common_parent_obstruction", common_parent_obstruction),
1005        (
1006            "reason",
1007            string("the first profile requires one regular injective parent covering the entire material body"),
1008        ),
1009    ]))
1010}

1012/// Recover exact translation columns for blocks extruded along any parametric axis.
1013///
1014/// For a degree-one local axis, equal endpoint weights and the identity
1015/// `H_1 = H_0 + t W` on every transverse control prove `X(q)=X(0)+q t` over
1016/// the complete block. The returned source net is the exact `q=0` restriction.
1017pub fn recover_translation_columns(model: &Json, local_axis: usize) -> Result<Json> {
1018    if local_axis >= 3 {
1019        return Err("translation-column axis is outside 0..3");
1020    }
1021    let blocks = bracket_json::decode_blocks(model)?;
1022    let mut reports = Vec::with_capacity(blocks.len());
1023    let mut groups: std::collections::BTreeMap<[Ratio; 3], Vec<String>> =
1024        std::collections::BTreeMap::new();
1025    for block in blocks {
1026        let shape = block.degrees.map(|degree| degree + 1);
1027        if block.degrees[local_axis] != 1 || shape.iter().product::<usize>() != block.controls.len()
1028        {
1029            reports.push(object([
1030                ("block_id", string(block.id)),
1031                ("status", string("unsupported_column_axis")),
1032                (
1033                    "reason",
1034                    string("selected parametric axis is not a valid degree-one extrusion"),
1035                ),
1036            ]));
1037            continue;
1038        }
1039        let mut transverse_shape = shape;
1040        transverse_shape[local_axis] = 1;
1041        let mut translation: Option<[Ratio; 3]> = None;
1042        let mut base_controls = Vec::new();
1043        let mut control_indices = Vec::new();
1044        let mut rejection = None;
1045        'transverse: for i in 0..transverse_shape[0] {
1046            for j in 0..transverse_shape[1] {
1047                for k in 0..transverse_shape[2] {
1048                    let low_index = [i, j, k];
1049                    let mut high_index = low_index;
1050                    high_index[local_axis] = 1;
1051                    let flat = |[a, b, c]: [usize; 3]| (a * shape[1] + b) * shape[2] + c;
1052                    let low_flat = flat(low_index);
1053                    let high_flat = flat(high_index);
1054                    let low = block.controls[low_flat];
1055                    let high = block.controls[high_flat];
1056                    if low[3] <= 0 || high[3] != low[3] {
1057                        rejection = Some("endpoint weights are not identical and positive");
1058                        break 'transverse;
1059                    }
1060                    let candidate: [Result<Ratio>; 3] = std::array::from_fn(|component| {
1061                        Ratio::new(
1062                            i128::from(high[component]) - i128::from(low[component]),
1063                            i128::from(low[3]),
1064                        )
1065                    });
1066                    let candidate = [candidate[0]?, candidate[1]?, candidate[2]?];
1067                    if translation.is_some_and(|previous| previous != candidate) {
1068                        rejection =
1069                            Some("homogeneous endpoint differences are not one translation vector");
1070                        break 'transverse;
1071                    }
1072                    translation = Some(candidate);
1073                    base_controls.push(low);
1074                    control_indices.push(low_flat);
1075                }
1076            }
1077        }
1078        if let Some(reason) = rejection {
1079            reports.push(object([
1080                ("block_id", string(block.id)),
1081                ("status", string("not_a_translation_column")),
1082                ("parametric_axis", number(local_axis)),
1083                ("reason", string(reason)),
1084            ]));
1085            continue;
1086        }
1087        let translation = translation.ok_or("empty translation-column control net")?;
1088        groups
1089            .entry(translation)
1090            .or_default()
1091            .push(block.id.clone());
1092        reports.push(object([
1093            ("block_id", string(block.id)),
1094            ("status", string("certified_exact_prismatic_column")),
1095            ("parametric_axis", number(local_axis)),
1096            ("degrees", arr(block.degrees, number)),
1097            ("source_column_shape", arr(transverse_shape, number)),
1098            ("source_control_indices", arr(control_indices, number)),
1099            (
1100                "source_homogeneous_controls",
1101                arr(base_controls, |control| arr(control, number)),
1102            ),
1103            (
1104                "translation_vector_mm",
1105                arr(translation, |component| {
1106                    object([
1107                        ("numerator", string(component.numerator().to_string())),
1108                        ("denominator", string(component.denominator().to_string())),
1109                    ])
1110                }),
1111            ),
1112            (
1113                "proof",
1114                string("all transverse controls satisfy equal positive weights and H1-H0=t*W"),
1115            ),
1116        ]));
1117    }
1118    let grouped = groups
1119        .into_iter()
1120        .map(|(translation, block_ids)| {
1121            object([
1122                (
1123                    "translation_vector_mm",
1124                    arr(translation, |component| {
1125                        object([
1126                            ("numerator", string(component.numerator().to_string())),
1127                            ("denominator", string(component.denominator().to_string())),
1128                        ])
1129                    }),
1130                ),
1131                ("block_ids", arr(block_ids, string)),
1132                (
1133                    "role",
1134                    string("parallel translation direction; distinct source columns remain distinct geometry"),
1135                ),
1136            ])
1137        })
1138        .collect();
1139    Ok(object([
1140        ("schema", string("exact_translation_columns_v1")),
1141        ("parametric_axis", number(local_axis)),
1142        ("blocks", Json::Array(reports)),
1143        ("parallel_column_groups", Json::Array(grouped)),
1144    ]))
1145}
