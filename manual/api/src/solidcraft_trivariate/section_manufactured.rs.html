13type V = [f64; 3];
14type M = [V; 3];
15fn n(v: impl ToString) -> Json {
16    Json::Number(v.to_string())
17}
18fn dot(a: V, b: V) -> f64 {
19    a.into_iter().zip(b).map(|(x, y)| x * y).sum()
20}
21fn cross(a: V, b: V) -> V {
22    [
23        a[1] * b[2] - a[2] * b[1],
24        a[2] * b[0] - a[0] * b[2],
25        a[0] * b[1] - a[1] * b[0],
26    ]
27}
28enum Profile {
29    Constant,
30    Power {
31        degree: i32,
32        origin: f64,
33        width: f64,
34    },
35    Gaussian {
36        center: V,
37        lengths: V,
38    },
39    AffinePower {
40        coefficients: V,
41        offset: f64,
42        scale: f64,
43        degree: i32,
44    },
45}
46impl Profile {
47    fn jets(&self, x: V) -> (f64, V, M) {
48        match self {
49            Self::Constant => (1., [0.; 3], [[0.; 3]; 3]),
50            Self::Power {
51                degree,
52                origin,
53                width,
54            } => {
55                let t = (x[0] - origin) / width;
56                let mut h = [[0.; 3]; 3];
57                h[0][0] = f64::from(degree * (degree - 1)) * t.powi(degree - 2) / width.powi(2);
58                (
59                    t.powi(*degree),
60                    [f64::from(*degree) * t.powi(degree - 1) / width, 0., 0.],
61                    h,
62                )
63            }
64            Self::Gaussian { center, lengths } => {
65                let t: V = std::array::from_fn(|i| (x[i] - center[i]) / lengths[i]);
66                let f = (-0.5 * dot(t, t)).exp();
67                let a: V = std::array::from_fn(|i| -t[i] / lengths[i]);
68                let d = a.map(|v| v * f);
69                let h = std::array::from_fn(|i| {
70                    std::array::from_fn(|j| {
71                        f * (a[i] * a[j] - if i == j { 1. / lengths[i].powi(2) } else { 0. })
72                    })
73                });
74                (f, d, h)
75            }
76            Self::AffinePower {
77                coefficients,
78                offset,
79                scale,
80                degree,
81            } => {
82                let q = (dot(*coefficients, x) + offset) / scale;
83                let factor = coefficients.map(|v| v / scale);
84                let f = q.powi(*degree);
85                let d = factor.map(|v| f64::from(*degree) * q.powi(*degree - 1) * v);
86                let h = std::array::from_fn(|i| {
87                    std::array::from_fn(|j| {
88                        if *degree < 2 {
89                            0.
90                        } else {
91                            f64::from(*degree * (*degree - 1))
92                                * q.powi(*degree - 2)
93                                * factor[i]
94                                * factor[j]
95                        }
96                    })
97                });
98                (f, d, h)
99            }
100        }
101    }
102}
103struct Exact {
104    profile: Profile,
105    axial_sine: bool,
106    z0: f64,
107    height: f64,
108    lambda: f64,
109    mu: f64,
110}
111impl Exact {
112    fn axial(&self, z: f64) -> (f64, f64, f64) {
113        if self.axial_sine {
114            let wave = std::f64::consts::PI / self.height;
115            let phase = wave * (z - self.z0);
116            (phase.sin(), wave * phase.cos(), -wave * wave * phase.sin())
117        } else {
118            ((z - self.z0) / self.height, 1. / self.height, 0.)
119        }
120    }
121    fn field(&self, x: V) -> (V, M) {
122        let (f, df, _) = self.profile.jets(x);
123        let (h, dh, _) = self.axial(x[2]);
124        let mut gradient = df.map(|a| a * h);
125        gradient[2] += f * dh;
126        ([0., 0., f * h], [[0.; 3], [0.; 3], gradient])
127    }
128    fn stress(&self, g: M) -> M {
129        let div = g[0][0] + g[1][1] + g[2][2];
130        std::array::from_fn(|i| {
131            std::array::from_fn(|j| {
132                self.mu * (g[i][j] + g[j][i]) + if i == j { self.lambda * div } else { 0. }
133            })
134        })
135    }
136    fn body(&self, x: V) -> V {
137        let (f, df, d2f) = self.profile.jets(x);
138        let (h, dh, d2h) = self.axial(x[2]);
139        let laplace = h * (d2f[0][0] + d2f[1][1] + d2f[2][2]) + 2. * df[2] * dh + f * d2h;
140        std::array::from_fn(|i| {
141            let mixed = h * d2f[i][2] + dh * df[i] + if i == 2 { df[2] * dh + f * d2h } else { 0. };
142            -(self.lambda + self.mu) * mixed - if i == 2 { self.mu * laplace } else { 0. }
143        })
144    }
145}

147/// E = 2.6 MPa, nu = 0.3. Prescribe zero displacement on the lowest exterior
148/// face(s), use -div(sigma) in the volume and sigma*n over the full remaining
149/// boundary, including pocket floors. The auxiliary bottom traction is also
150/// integrated for the independent unrestricted virtual-work check.
151pub fn solve(model: &Json, space: &InheritedSpace, job: &Json) -> Result<Json, &'static str> {
152    let z0 = space
153        .cells
154        .iter()
155        .map(|c| c.z_bounds[0])
156        .fold(f64::INFINITY, f64::min);
157    let z1 = space
158        .cells
159        .iter()
160        .map(|c| c.z_bounds[1])
161        .fold(f64::NEG_INFINITY, f64::max);
162    let faces = enumerate_faces(&decode_blocks(model)?)?.exterior;
163    let cells: BTreeMap<_, _> = space.cells.iter().map(|c| (c.id.as_str(), c)).collect();
164    let bottom: Vec<_> = faces
165        .iter()
166        .filter(|s| s.axis == 2 && !s.upper && cells[s.block.as_str()].z_bounds[0] == z0)
167        .cloned()
168        .collect();
169    let essential = space.zero_face_lift(&bottom)?;
170    let indices: BTreeMap<_, _> = space
171        .cells
172        .iter()
173        .enumerate()
174        .map(|(i, c)| (c.id.as_str(), i))
175        .collect();
176    let boundary: Vec<_> = faces
177        .iter()
178        .map(|s| (indices[s.block.as_str()], s.axis, s.upper))
179        .collect();
180    let domain = IntegrationDomain {
181        scalar_dofs: space.ndof,
182        cell_count: space.cells.len(),
183        z_bounds: [z0, z1],
184        boundary: &boundary,
185        essential: &essential,
186    };
187    let mut result = solve_sampled(&domain, job, |i, p| space.cells[i].sample(p))?;
188    if let Json::Object(fields) = &mut result {
189        for (key, value) in [
190            ("inherited_joins", space.common_parent_faces),
191            ("physical_c1_joins", space.transverse_faces),
192            ("remaining_c0_joins", space.retained_faces),
193        ] {
194            fields.insert(key.into(), n(value));
195        }
196    }
197    Ok(result)
198}

200pub struct IntegrationDomain<'a> {
201    pub scalar_dofs: usize,
202    pub cell_count: usize,
203    pub z_bounds: [f64; 2],
204    /// (leaf index, parametric axis, high side), excluding internal faces.
205    pub boundary: &'a [(usize, usize, bool)],
206    pub essential: &'a crate::constraint_system::Lift,
207}

209pub fn solve_sampled<F>(
210    domain: &IntegrationDomain<'_>,
211    job: &Json,
212    sample: F,
213) -> Result<Json, &'static str>
214where
215    F: Fn(
216        usize,
217        V,
218    ) -> Result<
219        (
220            crate::extracted_space::ElementGeometrySample,
221            crate::extracted_space::ElementBasisSample,
222        ),
223        &'static str,
224    >,
225{
226    let [z0, z1] = domain.z_bounds;
227    if !z0.is_finite()
228        || !z1.is_finite()
229        || z0 >= z1
230        || domain.boundary.is_empty()
231        || domain
232            .boundary
233            .iter()
234            .any(|&(i, a, _)| i >= domain.cell_count || a >= 3)
235    {
236        return Err("invalid manufactured integration domain");
237    }
238    let schema = job.field("schema")?.text()?;
239    let profile = match schema {
240        "section_axial_sine_mms_v1" => {
241            job.keys(&["schema", "quadrature"])?;
242            Profile::Constant
243        }
244        "section_quadratic_mms_v1" | "section_cubic_mms_v1" | "section_quintic_mms_v1" => {
245            job.keys(&["schema", "quadrature", "x_origin_mm", "x_scale_mm"])?;
246            let origin = job.field("x_origin_mm")?.number::<f64>()?;
247            let width = job.field("x_scale_mm")?.number::<f64>()?;
248            if !origin.is_finite() || !width.is_finite() || width <= 0. {
249                return Err("invalid manufactured polynomial scale");
250            }
251            Profile::Power {
252                degree: match schema {
253                    "section_cubic_mms_v1" => 3,
254                    "section_quintic_mms_v1" => 5,
255                    _ => 2,
256                },
257                origin,
258                width,
259            }
260        }
261        "local_gaussian_mms_v1" => {
262            job.keys(&["schema", "quadrature", "center_mm", "lengths_mm"])?;
263            let vector = |key: &str| -> Result<V, &'static str> {
264                let v = job.field(key)?.items()?;
265                if v.len() != 3 {
266                    return Err("Gaussian load expects three components");
267                }
268                Ok([v[0].number()?, v[1].number()?, v[2].number()?])
269            };
270            let center = vector("center_mm")?;
271            let lengths = vector("lengths_mm")?;
272            if center.iter().any(|x| !x.is_finite())
273                || lengths.iter().any(|x| !x.is_finite() || *x <= 0.)
274            {
275                return Err("invalid Gaussian load parameters");
276            }
277            Profile::Gaussian { center, lengths }
278        }
279        "section_affine_power_mms_v1" => {
280            job.keys(&[
281                "schema",
282                "quadrature",
283                "coefficients",
284                "offset",
285                "scale",
286                "degree",
287            ])?;
288            let coefficients = job.field("coefficients")?.items()?;
289            if coefficients.len() != 3 {
290                return Err("affine power expects three coefficients");
291            }
292            let coefficients: V = [
293                coefficients[0].number::<f64>()?,
294                coefficients[1].number()?,
295                coefficients[2].number()?,
296            ];
297            let offset = job.field("offset")?.number::<f64>()?;
298            let scale = job.field("scale")?.number::<f64>()?;
299            let degree = job.field("degree")?.number::<i32>()?;
300            if coefficients.iter().any(|x| !x.is_finite())
301                || !offset.is_finite()
302                || !scale.is_finite()
303                || scale <= 0.
304                || !(1..=5).contains(&degree)
305            {
306                return Err("invalid affine power parameters");
307            }
308            Profile::AffinePower {
309                coefficients,
310                offset,
311                scale,
312                degree,
313            }
314        }
315        _ => return Err("unknown section verification job"),
316    };
317    let order = job.field("quadrature")?.number()?;
318    let exact = Exact {
319        profile,
320        axial_sine: schema == "section_axial_sine_mms_v1",
321        z0,
322        height: z1 - z0,
323        lambda: 1.5,
324        mu: 1.,
325    };
326    let start = std::time::Instant::now();
327    let mut assembly = assemble_sampled(
328        domain.scalar_dofs,
329        domain.cell_count,
330        exact.lambda,
331        exact.mu,
332        order,
333        &sample,
334        |x| exact.body(x),
335    )?;
336    let q = gauss(order)?;
337    let mut boundary_resultants = [[0.; 3]; 4]; // bottom, top, walls, pocket floor
338    for &(leaf, axis, upper) in domain.boundary {
339        let [a, b] = face_axes(axis);
340        for &(u, wu) in &q {
341            for &(v, wv) in &q {
342                let mut pt = [0.; 3];
343                pt[a] = u;
344                pt[b] = v;
345                pt[axis] = usize::from(upper) as f64;
346                let (g, basis) = sample(leaf, pt)?;
347                let j = g.jacobian_columns;
348                let area = cross(j[(axis + 1) % 3], j[(axis + 2) % 3]);
349                let sigma = exact.stress(exact.field(g.position).1);
350                let factor = wu * wv * if upper { 1. } else { -1. };
351                let group = if axis != 2 {
352                    2
353                } else if (g.position[2] - z0).abs() < 1e-9 {
354                    0
355                } else if (g.position[2] - z1).abs() < 1e-9 {
356                    1
357                } else {
358                    3
359                };
360                for (&id, &r) in basis.dofs.iter().zip(&basis.values) {
361                    for (i, &row) in sigma.iter().enumerate() {
362                        assembly.force[3 * id + i] += r * dot(row, area) * factor;
363                    }
364                }
365                for (i, row) in sigma.iter().enumerate() {
366                    boundary_resultants[group][i] += dot(*row, area) * factor;
367                }
368            }
369        }
370    }
371    let assembly_ms = start.elapsed().as_secs_f64() * 1000.;
372    let start = std::time::Instant::now();
373    let solution = solve_constrained(
374        &assembly.stiffness,
375        [domain.essential; 3],
376        &assembly.force,
377        1e-11,
378        12000,
379    )?;
380    let solve_ms = start.elapsed().as_secs_f64() * 1000.;
381    let mut weak = vec![0.; 3 * domain.scalar_dofs];
382    let mut integrated_uz = 0.;
383    let mut totals = [0f64; 6];
384    let mut body_resultant = [0.; 3];
385    let mut body_moment = [0.; 3];
386    for leaf in 0..domain.cell_count {
387        for &(u, wu) in &q {
388            for &(v, wv) in &q {
389                for &(w, ww) in &q {
390                    let (g, b) = sample(leaf, [u, v, w])?;
391                    let (x, d) = exact.field(g.position);
392                    let body = exact.body(g.position);
393                    let mut uh = [0.; 3];
394                    let mut du = [[0.; 3]; 3];
395                    let sig = exact.stress(d);
396                    let weight = g.determinant * wu * wv * ww;
397                    let moment = cross(g.position, body);
398                    for i in 0..3 {
399                        body_resultant[i] += weight * body[i];
400                        body_moment[i] += weight * moment[i];
401                    }
402                    for (k, &id) in b.dofs.iter().enumerate() {
403                        for i in 0..3 {
404                            let coeff = solution.displacement_mm[3 * id + i];
405                            uh[i] += b.values[k] * coeff;
406                            for (j, entry) in du[i].iter_mut().enumerate() {
407                                *entry += b.physical_gradients[k][j] * coeff;
408                            }
409                            weak[3 * id + i] += weight * dot(sig[i], b.physical_gradients[k]);
410                        }
411                    }
412                    integrated_uz += weight * uh[2];
413                    let e: V = std::array::from_fn(|i| uh[i] - x[i]);
414                    let de: M =
415                        std::array::from_fn(|i| std::array::from_fn(|j| du[i][j] - d[i][j]));
416                    let ds = exact.stress(de);
417                    totals[0] += weight * dot(e, e);
418                    totals[1] += weight * dot(x, x);
419                    for k in 0..3 {
420                        totals[2] += weight * dot(ds[k], de[k]);
421                        totals[3] += weight * dot(sig[k], d[k]);
422                        totals[4] += weight * dot(ds[k], ds[k]);
423                        totals[5] += weight * dot(sig[k], sig[k]);
424                    }
425                }
426            }
427        }
428    }
429    let weak_defect = (weak
430        .iter()
431        .zip(&assembly.force)
432        .map(|(a, b)| (a - b).powi(2))
433        .sum::<f64>()
434        / weak.iter().map(|x| x * x).sum::<f64>())
435    .sqrt();
436    if totals.iter().any(|v| !v.is_finite())
437        || !weak_defect.is_finite()
438        || !integrated_uz.is_finite()
439        || [1, 3, 5].iter().any(|&i| totals[i] <= 0.)
440    {
441        return Err("indeterminate: manufactured physical norm is nonfinite or has zero reference");
442    }
443    let ratio = |i: usize| (totals[i] / totals[i + 1]).max(0.).sqrt();
444    let resultant =
445        |group: usize| Json::Array(boundary_resultants[group].iter().map(|x| n(*x)).collect());
446    let body_vec = |values: &[f64; 3]| Json::Array(values.iter().map(|x| n(*x)).collect());
447    Ok(object([
448        (
449            "schema",
450            Json::String("section_manufactured_result_v1".into()),
451        ),
452        ("manufactured_field", Json::String(schema.to_string())),
453        ("quadrature", n(order)),
454        ("scalar_dofs", n(domain.scalar_dofs)),
455        ("free_vector_dofs", n(3 * domain.essential.free_roots.len())),
456        (
457            "mean_displacement_z_mm",
458            n(integrated_uz / assembly.stiffness.integrated_volume),
459        ),
460        ("volume_mm3", n(assembly.stiffness.integrated_volume)),
461        ("quadrature_points", n(assembly.quadrature_points)),
462        ("block_nnz", n(assembly.stiffness.columns.len())),
463        ("assembly_ms", n(assembly_ms)),
464        ("solve_ms", n(solve_ms)),
465        ("virtual_work_relative_defect", n(weak_defect)),
466        ("relative_residual", n(solution.relative_residual)),
467        ("energy_N_mm", n(solution.strain_energy_n_mm)),
468        ("relative_displacement_L2", n(ratio(0))),
469        ("relative_energy_norm", n(ratio(2))),
470        ("relative_stress_L2", n(ratio(4))),
471        ("bottom_traction_resultant_N", resultant(0)),
472        ("top_traction_resultant_N", resultant(1)),
473        ("wall_traction_resultant_N", resultant(2)),
474        ("floor_traction_resultant_N", resultant(3)),
475        ("body_force_resultant_N", body_vec(&body_resultant)),
476        ("body_force_moment_N_mm", body_vec(&body_moment)),
477    ]))
478}

484    fn gaussian_body_matches_independent_stress_divergence() {
485        let exact = Exact {
486            profile: Profile::Gaussian {
487                center: [1.7, 2., 2.5],
488                lengths: [0.4, 0.35, 0.55],
489            },
490            axial_sine: false,
491            z0: 0.,
492            height: 5.,
493            lambda: 1.5,
494            mu: 1.,
495        };
496        let x = [1.9, 2.1, 2.4];
497        check_body(&exact, x);
498        let cube = Exact {
499            profile: Profile::Gaussian {
500                center: [0.31, 0.57, 0.68],
501                lengths: [0.30, 0.41, 0.36],
502            },
503            height: 1.,
504            ..exact
505        };
506        check_body(&cube, [0.24, 0.49, 0.76]);
507        let pocket = Exact {
508            profile: Profile::Power {
509                degree: 5,
510                origin: -10.,
511                width: 70.,
512            },
513            height: 20.,
514            ..cube
515        };
516        check_body(&pocket, [17., 12., 7.]);
517        let curved_pocket = Exact {
518            profile: Profile::AffinePower {
519                coefficients: [1., 2., 0.],
520                offset: 10.,
521                scale: 150.,
522                degree: 5,
523            },
524            height: 20.,
525            ..pocket
526        };
527        check_body(&curved_pocket, [17., 12., 7.]);
528    }

530    fn check_body(exact: &Exact, x: V) {
531        let body = exact.body(x);
532        let step = 1e-5;
533        let mut numerical = [0.; 3];
534        for j in 0..3 {
535            let mut a = x;
536            let mut b = x;
537            a[j] -= step;
538            b[j] += step;
539            let stress_a = exact.stress(exact.field(a).1);
540            let stress_b = exact.stress(exact.field(b).1);
541            for i in 0..3 {
542                numerical[i] -= (stress_b[i][j] - stress_a[i][j]) / (2. * step);
543            }
544        }
545        for (expected, found) in body.into_iter().zip(numerical) {
546            assert!((expected - found).abs() < 1e-7);
547        }
548    }
